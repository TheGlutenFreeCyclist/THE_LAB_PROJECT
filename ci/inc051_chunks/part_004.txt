def _v106_session_is_hard(sess):
    s=sess if isinstance(sess,dict) else {}
    return bool(
        s.get("hard_spacing_relevant") or s.get("provisional_quality") or
        str(s.get("quality_relevance") or "").upper()=="MEANINGFUL" or

def _v106_recommendation_contract(sessions):
    rows=[x for x in (sessions or []) if isinstance(x,dict)]
    hard_indexes=[i for i,x in enumerate(rows) if _v106_session_is_hard(x)]
    easy_indexes=[i for i,x in enumerate(rows) if str(x.get("intensity_class") or "").lower() in {"recovery","endurance"} and not _v106_session_is_hard(x)]
    plan_ids=sorted({str(x.get("nova_canonical_plan_id")) for x in rows if x.get("nova_canonical_plan_id")})
    workout_ids=[str(x.get("nova_workout_id")) for x in rows if x.get("nova_workout_id")]

        "schema":"V4.8.85-R109-1",
        "source":"FINAL_CANONICAL_NEXT_SESSIONS",
        "session_count":len(rows),
        "all_easy":bool(rows) and len(easy_indexes)==len(rows),

        "hard_session_count":len(hard_indexes),
        "canonical_plan_id":plan_ids[0] if len(plan_ids)==1 else None,
        "canonical_plan_identity_pass":len(plan_ids)<=1,
        "workout_ids":workout_ids,
        "session_signatures":[_v87_session_signature(x) for x in rows],

def _v106_recommendation_projection(sessions):
    rows=[x for x in (sessions or []) if isinstance(x,dict)]
    if not rows:
        return "No training session is currently prescribed in this Snapshot."
    contract=_v106_recommendation_contract(rows)
    first=rows[0]
    title=str(first.get("title") or "next session").strip()
    dur=str(first.get("duration") or "").strip()
    main=str(first.get("main_set") or "").strip()
    why=str(first.get("why") or "").strip()
    slot=str(first.get("slot") or "").strip()
    intent_notice=str(first.get("athlete_intent_notice") or "").strip()
    lead=(f"{slot}: {title}" if slot else f"Your next session is {title}")+(f" ({dur})" if dur else "")+"."
    parts=[]
    if intent_notice:
        parts.extend(_v4844_sentence_list(intent_notice)[:2])
    parts.append(lead)
    if main:
        parts.extend(_v4844_sentence_list(main)[:2])
    if why:
        parts.extend(_v4844_sentence_list(why)[:1])
    hard_i=contract.get("first_hard_index")
    if hard_i is not None and hard_i>0:
        hard=rows[hard_i]
        hslot=str(hard.get("slot") or "").strip()

    elif len(rows)>1:
        parts.append("Follow the remaining listed sessions exactly as shown; this recommendation does not create any additional workout or intensity.")
    return " ".join(_v4844_dedupe_sentences(parts))[:1500].strip()

def _v110_recommendation_commentary_guard(text):
    """Keep AI language strictly explanatory: no new workout mechanics or scheduling."""
    sentences=_v4844_sentence_list(str(text or "").strip())
    kept=[]
    forbidden=re.compile(
        r"(?:\d|\b(?:x|×)\b|\b(?:w|watts?|min(?:ute)?s?|sec(?:ond)?s?|hour|hours|%|ftp|vo2|threshold|tempo|sprint)\b|"
        r"\b(?:do|perform|add|remove|replace|instead|move|advance|delay|skip|shorten|extend|increase|decrease|harder|easier|"
        r"fai|aggiungi|togli|sostituisci|sposta|anticipa|ritarda|salta|accorcia|allunga|aumenta|riduci)\b)", re.I
    )
    for sentence in sentences:

        if len(kept)>=3:

def _v163_recent_progress_badges(snapshot, limit=3):
    """Presentation-only recent earned progress for the Roadmap.

    This projection never changes coaching state. It reads already-established
    evidence from the current Snapshot, prefers one meaningful earned item per
    event/date so the Roadmap does not duplicate the full trophy cabinet, and
    returns at most ``limit`` compact badges for UI rendering.
    """
    if not isinstance(snapshot, dict):
        return []
    try:
        limit = max(0, min(int(limit), 3))
    except Exception:
        limit = 3
    if limit <= 0:
        return []

    road = snapshot.get("adaptive_roadmap") if isinstance(snapshot.get("adaptive_roadmap"), dict) else {}
    qs = snapshot.get("question_state") if isinstance(snapshot.get("question_state"), dict) else {}
    pe = snapshot.get("performance_evidence") if isinstance(snapshot.get("performance_evidence"), dict) else {}
    current_goal = str(road.get("goal_key") or qs.get("goal_key") or "").strip().upper()
    candidates = []

    def clean_date(value):
        raw = str(value or "").strip()
        if re.match(r"^\d{4}-\d{2}-\d{2}$", raw):
            return raw
        return None

    def duration_label(seconds):
        try:
            sec = int(round(float(seconds)))
        except Exception:
            return None
        if sec <= 0:
            return None
        if sec % 3600 == 0:
            hours = sec // 3600
            return f"{hours}-hour"
        if sec % 60 == 0:
            mins = sec // 60
            return f"{mins}-min"
        return f"{sec}-sec"

    def add(date_value, label, icon="🏅", priority=0, key=None, detail=None, activity_id=None):
        earned_at = clean_date(date_value)
        label = str(label or "").strip()
        if not earned_at or not label:
            return
        candidates.append({
            "earned_at": earned_at,
            "label": label,
            "icon": icon,
            "priority": int(priority or 0),
            "key": str(key or label).strip().lower(),
            "detail": str(detail or "").strip() or None,
            "activity_id": str(activity_id or "").strip() or None,
        })

    # Direct, date-proven performance achievements for the active goal.
    for event in pe.get("events") or []:
        if not isinstance(event, dict):
            continue
        event_goal = str(event.get("goal_key") or "").strip().upper()
        if current_goal and event_goal and event_goal != current_goal:
            continue
        if current_goal and event.get("goal_relevant") is False:
            continue
        earned_at = clean_date(event.get("date"))
        if not earned_at:
            continue
        kind = str(event.get("kind") or "").strip().upper()
        try:
            delta = float(event.get("delta_w")) if event.get("delta_w") is not None else None
        except Exception:
            delta = None
        try:
            watts = int(round(float(event.get("watts")))) if event.get("watts") is not None else None
        except Exception:
            watts = None
        dlabel = duration_label(event.get("secs"))
        activity_id = event.get("activity_id")

        if kind == "VALIDATION" and str(event.get("outcome") or "").upper() == "VALIDATED_GAIN":
            if dlabel:
                gain = f" · +{int(round(delta))} W" if delta is not None and delta > 0 else ""
                detail = f"{watts} W" if watts is not None else None
                add(earned_at, f"{dlabel} power improved{gain}", "🏅", 120,
                    key=f"validation:{dlabel}", detail=detail, activity_id=activity_id)
            continue

        try:
            tier = int(event.get("tier") or 0)
        except Exception:
            tier = 0
        recognized_pb = (
            kind == "PB"
            and (event.get("the_lab_recognized") is True or event.get("quality_linked") is True or tier >= 2)
            and (event.get("timing_proven") is not False)
        )
        if recognized_pb and dlabel and delta is not None and delta > 0:
            detail = f"{watts} W" if watts is not None else None
            add(earned_at, f"{dlabel} power improved · +{int(round(delta))} W", "🏅", 100,
                key=f"pb:{dlabel}", detail=detail, activity_id=activity_id)

    # Evidence-gated dimensions that are actually completed/closed.
    dimension_labels = {
        "FRESH_CAPACITY": ("Fresh capacity validated", "✅"),
        "REPEATABILITY": ("Repeatability consolidated", "✅"),
        "SPECIFIC_DOSE": ("Specific dose established", "✅"),
        "RECOVERY_UNDER_LOAD": ("Recovery under load established", "✅"),
        "DURABILITY": ("Durability consolidated", "✅"),
        "RACE_TRANSFER": ("Race transfer demonstrated", "✅"),
        "VARIABLE_LOAD": ("Variable-load control established", "✅"),
    }
    dimensions = qs.get("dimensions") if isinstance(qs.get("dimensions"), dict) else {}
    for code, item in dimensions.items():
        if not isinstance(item, dict) or not item.get("exit_satisfied"):
            continue
        earned_at = clean_date(item.get("last_evidence_date"))
        if not earned_at:
            continue
        code_u = str(code or "").strip().upper()
        label, icon = dimension_labels.get(code_u, (code_u.replace("_", " ").title() + " established", "✅"))
        # Concrete same-day validated power is more informative than the generic
        # fresh-capacity closure; keep the latter only as a fallback candidate.
        priority = 70 if code_u == "FRESH_CAPACITY" else 110
        add(earned_at, label, icon, priority, key=f"dimension:{code_u}")

    if not candidates:
        return []

    # Exact semantic duplicates collapse first, newest/strongest evidence wins.
    candidates.sort(key=lambda x: (x["earned_at"], x["priority"]), reverse=True)
    unique = []
    seen_keys = set()
    for item in candidates:
        if item["key"] in seen_keys:
            continue
        seen_keys.add(item["key"])
        unique.append(item)

    # Avoid turning the Roadmap into a second trophy cabinet: first pass keeps
    # only the strongest earned item for each date/event. If fewer than three
    # distinct dates exist, a second pass may fill the remaining compact slots.
    selected = []
    seen_dates = set()
    for item in unique:
        if item["earned_at"] in seen_dates:
            continue
        selected.append(item)
        seen_dates.add(item["earned_at"])
        if len(selected) >= limit:
            break
    if len(selected) < limit:
        selected_ids = {id(x) for x in selected}
        for item in unique:
            if id(item) in selected_ids:
                continue
            selected.append(item)
            if len(selected) >= limit:
                break

    for item in selected:
        try:
            dt = datetime.strptime(item["earned_at"], "%Y-%m-%d")
            item["date_display"] = dt.strftime("%d %b")
        except Exception:
            item["date_display"] = item["earned_at"]
        item.pop("priority", None)
        item.pop("key", None)
        item.pop("activity_id", None)
    return selected[:limit]

def _v4846_sanitize_legacy_ledger_prescription(ledger):

            f = float(value)

        if not isinstance(entry, dict):
            return entry

        if not main_set:
            return entry
        status = str(entry.get("status") or "").upper()
        qrel = str(entry.get("quality_relevance") or "").upper()

        if not eligible:

        interval = profile.get("interval_minutes")
        lo = profile.get("power_low")
        hi = profile.get("power_high")
        if interval is not None:
            entry["interval_minutes"] = _clean_number(interval)
        if lo is not None and hi is not None:
            entry["prescribed_power_low"] = _clean_number(lo)
            entry["prescribed_power_high"] = _clean_number(hi)
        elif _v124_strip_non_prescriptive_power_context(main_set) != main_set:
            # Historical R123 reports could accidentally serialize a quoted
            # curve estimate as an executable target. Do not retain those

            entry["prescribed_power_low"] = None
            entry["prescribed_power_high"] = None
            entry["quality_target_ratio"] = None
            entry["reference_power_cleanup"] = "R124_NONPRESCRIPTIVE_CONTEXT_ONLY"
        return entry
    anchor = out.get("primary_anchor") if isinstance(out.get("primary_anchor"), dict) else None
    if anchor:
        out["primary_anchor"] = _sanitize_entry(anchor, force=True)
    planned = []
    for item in out.get("planned_sessions") or []:
        planned.append(_sanitize_entry(item, force=False) if isinstance(item, dict) else item)

        out["planned_sessions"] = planned

def _credential_fernet():
    if Fernet is None:
        raise RuntimeError("cryptography is required to protect athlete credentials")
    material = CREDENTIAL_ENCRYPTION_KEY or SECRET_KEY
    if not material or material == "change-this-key":
        raise RuntimeError("A strong credential encryption key is required")
    digest = hashlib.sha256(("THE-LAB-CREDENTIALS|" + material).encode("utf-8")).digest()
    return Fernet(base64.urlsafe_b64encode(digest))
def _encrypt_credential(value):
    value = (value or "").strip()
    if not value:
        raise ValueError("API credential is required")
    return _credential_fernet().encrypt(value.encode("utf-8")).decode("ascii")
def _decrypt_credential(value):
    if not value:
        return ""
    try:
        return _credential_fernet().decrypt(str(value).encode("ascii")).decode("utf-8")
    except InvalidToken as exc:
        raise RuntimeError("Stored credential cannot be decrypted; re-save the API key") from exc

def _ai_model_standard_price(provider, model):
    provider_key = str(provider or "").strip().upper()
    model_key = str(model or "").strip().lower()
    catalog = _AI_STANDARD_TEXT_PRICING_USD_PER_MTOK.get(provider_key) or {}
    if not model_key or not catalog:
        return None
    for known_model in sorted(catalog, key=len, reverse=True):
        if model_key == known_model or model_key.startswith(known_model + "-"):
            input_rate, output_rate = catalog[known_model]
            return {
                "provider": provider_key, "matched_model": known_model,
                "input_per_mtok": float(input_rate), "output_per_mtok": float(output_rate),
                "catalog_date": _AI_PRICE_CATALOG_DATE,
            }
    return None
def _estimate_ai_call_cost(provider, model, input_tokens, output_tokens):
    price = _ai_model_standard_price(provider, model)
    if not price:
        return None
    try:
        in_tok = int(input_tokens) if input_tokens is not None else None
        out_tok = int(output_tokens) if output_tokens is not None else None
    except (TypeError, ValueError):
        return None
    if in_tok is None and out_tok is None:
        return None
    in_tok = max(0, in_tok or 0)
    out_tok = max(0, out_tok or 0)
    input_usd = (in_tok / 1_000_000.0) * price["input_per_mtok"]
    output_usd = (out_tok / 1_000_000.0) * price["output_per_mtok"]
    total_usd = input_usd + output_usd
    usd_to_eur, fx_source = _ai_cost_usd_to_eur_rate()
    input_eur = input_usd * usd_to_eur
    output_eur = output_usd * usd_to_eur
    total_eur = total_usd * usd_to_eur
    input_rate_eur = price["input_per_mtok"] * usd_to_eur
    output_rate_eur = price["output_per_mtok"] * usd_to_eur
    return {
        **price,
        "input_tokens": in_tok, "output_tokens": out_tok,
        "input_usd": input_usd, "output_usd": output_usd, "total_usd": total_usd,
        "input_eur": input_eur, "output_eur": output_eur, "total_eur": total_eur,
        "display_currency": _AI_COST_DISPLAY_CURRENCY,
        "usd_to_eur_rate": usd_to_eur, "fx_source": fx_source,
        "fx_reference_date": _AI_COST_FX_REFERENCE_DATE,
        "total_label": f"€{total_eur:.4f}",
        "input_label": f"€{input_eur:.4f}", "output_label": f"€{output_eur:.4f}",
        "rate_label": f"€{input_rate_eur:.4g}/M in · €{output_rate_eur:.4g}/M out",
        "fx_label": f"USD→EUR {usd_to_eur:.4f} · {fx_source}",
        "basis": "standard uncached paid text list price converted from USD to EUR",
    }

def _openai_output_text(payload):
    if isinstance(payload, dict) and isinstance(payload.get("output_text"), str):
        return payload.get("output_text", "").strip()
    chunks = []
    for item in (payload.get("output") or []) if isinstance(payload, dict) else []:
        if not isinstance(item, dict):
            continue
        for part in item.get("content") or []:
            if not isinstance(part, dict):
                continue
            text = part.get("text")
            if isinstance(text, str) and text:
                chunks.append(text)
    return "".join(chunks).strip()
def _gemini_output_text(payload):
    chunks = []
    for candidate in (payload.get("candidates") or []) if isinstance(payload, dict) else []:
        if not isinstance(candidate, dict):
            continue
        content = candidate.get("content") or {}
        if not isinstance(content, dict):
            continue
        for part in content.get("parts") or []:
            if not isinstance(part, dict):
                continue
            value = part.get("text")
            if isinstance(value, str) and value:
                chunks.append(value)
    return "".join(chunks).strip()

def _ai_usage_token_counts(provider, payload):
    provider = str(provider or "").upper()
    if provider == "GEMINI":
        usage = payload.get("usageMetadata") if isinstance(payload, dict) else None
        if not isinstance(usage, dict):
            return None, None, None
        input_tokens = usage.get("promptTokenCount")
        output_tokens = usage.get("candidatesTokenCount")
        total_tokens = usage.get("totalTokenCount")
    else:
        usage = payload.get("usage") if isinstance(payload, dict) else None
        if not isinstance(usage, dict):
            return None, None, None
    if provider == "ANTHROPIC":
        input_tokens = usage.get("input_tokens")
        output_tokens = usage.get("output_tokens")
        total_tokens = None
        try:
            if input_tokens is not None or output_tokens is not None:
                total_tokens = int(input_tokens or 0) + int(output_tokens or 0)
        except (TypeError, ValueError):
            total_tokens = None
    elif provider != "GEMINI":
        input_tokens = usage.get("input_tokens")
        output_tokens = usage.get("output_tokens")
        total_tokens = usage.get("total_tokens")
        if total_tokens is None:
            try:
                if input_tokens is not None or output_tokens is not None:
                    total_tokens = int(input_tokens or 0) + int(output_tokens or 0)
            except (TypeError, ValueError):
                total_tokens = None
    def _safe_int(value):
        try:
            return int(value) if value is not None else None
        except (TypeError, ValueError):
            return None
    return _safe_int(input_tokens), _safe_int(output_tokens), _safe_int(total_tokens)

def _reserve_ai_usage(user_id, call_type, provider, model, credential_source, counts_toward_limit=True, enforce_limit=True):
    owner_id = _data_owner_user_id(user_id)
    user = _get_user_by_id(owner_id) or {}
    local_date = _ai_usage_local_date(owner_id)
    is_athlete = user.get("role") != "admin"
    source = str(credential_source or "").strip().lower()
    owner_sources = ("platform_managed", "the_lab_standard", "managed_fallback_standard")
    limited_lane = source in owner_sources
    allowance = _owner_ai_allowance(owner_id, user=user)
    should_enforce = bool(enforce_limit and counts_toward_limit and is_athlete and limited_lane)
    effective_counts = bool(counts_toward_limit and is_athlete and limited_lane)
    event_id = str(uuid.uuid4())
    conn = _db_connect()
    try:
        cur = conn.cursor()
        if _db_is_postgres():
            cur.execute(_db_sql("SELECT id FROM users WHERE id=? FOR UPDATE"), (owner_id,))
        else:
            cur.execute("BEGIN IMMEDIATE")
        if should_enforce:
            owner_placeholders = ",".join("?" for _ in owner_sources)
            owner_sql = (
                "SELECT COUNT(*) FROM ai_usage_events WHERE user_id=? AND local_date=? "
                "AND counts_toward_limit=1 AND credential_source IN (" + owner_placeholders + ")"
            )
            cur.execute(_db_sql(owner_sql), (owner_id, local_date, *owner_sources))
            owner_used = int(cur.fetchone()[0])
            total_limit = int(allowance.get("total_limit") or 0)
            if total_limit <= 0 or owner_used >= total_limit:
                conn.rollback()
                audit_event(
                    "AI_DAILY_LIMIT_BLOCKED", actor_user_id=owner_id, target_user_id=owner_id,
                    details={"local_date": local_date, "lane": source, "limit": total_limit, "call_type": str(call_type or "AI")[:40], "scope": "ATHLETE_TOTAL"},
                )
                raise AIUsageLimitError(
                    f"Daily THE LAB owner-funded AI allowance reached ({total_limit} calls). "
                    "Try again tomorrow; saved data and your latest Snapshot remain available."
                )
            if source == "platform_managed":
                managed_limit = int(allowance.get("managed_limit") or 0)
                cur.execute(
                    _db_sql("SELECT COUNT(*) FROM ai_usage_events WHERE user_id=? AND local_date=? AND counts_toward_limit=1 AND credential_source='platform_managed'"),
                    (owner_id, local_date),
                )
                managed_used = int(cur.fetchone()[0])
                if managed_used >= managed_limit:
                    conn.rollback()
                    if int(allowance.get("standard_limit") or 0) > 0 and allowance.get("standard_ready"):
                        raise AIManagedFallbackRequired()
                    audit_event(
                        "AI_DAILY_LIMIT_BLOCKED", actor_user_id=owner_id, target_user_id=owner_id,
                        details={"local_date": local_date, "lane": "MANAGED", "limit": managed_limit, "call_type": str(call_type or "AI")[:40], "scope": "MANAGED_SUBCAP"},
                    )
                    raise AIUsageLimitError(
                        f"Daily THE LAB owner-funded AI allowance reached ({total_limit} calls). "
                        "Try again tomorrow; saved data and your latest Snapshot remain available."
                    )
            elif source in {"the_lab_standard", "managed_fallback_standard"}:
                standard_limit = int(allowance.get("standard_limit") or 0)
                cur.execute(

                    (owner_id, local_date),
                )

                    audit_event(
                        "AI_DAILY_LIMIT_BLOCKED", actor_user_id=owner_id, target_user_id=owner_id,

                    )
                    raise AIUsageLimitError(
                        f"Daily THE LAB owner-funded AI allowance reached ({total_limit} calls). "
                        "Try again tomorrow; saved data and your latest Snapshot remain available."
                    )
        cur.execute(
            _db_sql("INSERT INTO ai_usage_events (id, user_id, created_at_utc, local_date, call_type, provider, model, credential_source, counts_toward_limit, status) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, 'started')"),

        )

    finally:
        conn.close()

def _ai_output_truncation_reason(provider, payload):
    provider = str(provider or "").upper()
    payload = payload or {}
    if provider == "ANTHROPIC":
        reason = str(payload.get("stop_reason") or "").lower()
        return "MAX_OUTPUT_TOKENS" if reason == "max_tokens" else None
    if provider == "GEMINI":
        for cand in payload.get("candidates") or []:
            reason = str((cand or {}).get("finishReason") or "").upper()
            if reason in {"MAX_TOKENS", "MAX_OUTPUT_TOKENS"}:
                return "MAX_OUTPUT_TOKENS"
        return None
    if provider == "OPENAI":
        status = str(payload.get("status") or "").lower()
        reason = str(((payload.get("incomplete_details") or {}).get("reason") or "")).lower()
        if status == "incomplete" and reason in {"max_output_tokens", "max_tokens"}:
            return "MAX_OUTPUT_TOKENS"
    return None

def _call_ai_provider(provider, api_key, model, prompt, max_tokens=1200, *, usage_kind="AI", user_id=None, credential_source=None, counts_toward_limit=True, enforce_limit=True, prompt_profile=None, return_event_id=False):
    call_kind=str(usage_kind or "AI").strip().upper(); prompt_chars=len(str(prompt or ""))
    if "SNAPSHOT" in call_kind:
        if prompt_chars>AI_SNAPSHOT_PROMPT_CHAR_BUDGET:
            raise ValueError(f"AI Snapshot prompt budget exceeded before provider call ({prompt_chars}>{AI_SNAPSHOT_PROMPT_CHAR_BUDGET} chars). Historical context is not silently dropped.")
        max_tokens=min(int(max_tokens),AI_SNAPSHOT_MAX_OUTPUT_TOKENS)
    elif call_kind=="COACH_CHAT":
        if prompt_chars>AI_CHAT_PROMPT_CHAR_BUDGET:
            raise ValueError(f"AI Coach prompt budget exceeded before provider call ({prompt_chars}>{AI_CHAT_PROMPT_CHAR_BUDGET} chars).")
        max_tokens=min(int(max_tokens),AI_CHAT_MAX_OUTPUT_TOKENS)
    provider = str(provider or "").strip().upper()
    api_key = (api_key or "").strip()
    model = (model or "").strip()
    credential_source = str(credential_source or "").strip()
    if provider not in _AI_PROVIDER_LABELS:
        raise ValueError("Unsupported AI provider")
    if not api_key:
        raise ValueError("AI API key is required")
    if not model:
        model = _AI_DEFAULT_MODELS[provider]
    try:
        event_id = _reserve_ai_usage(
            user_id, usage_kind, provider, model, credential_source,
            counts_toward_limit=counts_toward_limit, enforce_limit=enforce_limit,
        )
    except AIManagedFallbackRequired:
        standard = _standard_ai_runtime_config()
        if not standard.get("configured"):
            owner_id = _data_owner_user_id(user_id)
            local_date = _ai_usage_local_date(owner_id)
            audit_event(
                "AI_MANAGED_FALLBACK_UNAVAILABLE", actor_user_id=owner_id, target_user_id=owner_id,
                details={"local_date": local_date, "managed_limit": MANAGED_AI_DAILY_CALL_LIMIT, "call_type": str(usage_kind or "AI")[:40]},
            )
            raise AIUsageLimitError(
                f"THE LAB Managed has used its {MANAGED_AI_DAILY_CALL_LIMIT} premium calls today and the Standard Gemini fallback is not configured. "
                "Try again tomorrow; saved data and your latest Snapshot remain available."
            )
        provider = standard["provider"]
        model = standard["model"]
        api_key = standard["api_key"]
        credential_source = "managed_fallback_standard"
        event_id = _reserve_ai_usage(
            user_id, usage_kind, provider, model, credential_source,
            counts_toward_limit=counts_toward_limit, enforce_limit=enforce_limit,
        )
    try:
        safe_profile = {}
        for key, value in (prompt_profile or {}).items():
            if isinstance(key, str) and isinstance(value, (int, float)):
                safe_profile[key[:80]] = int(value)
        safe_profile["final_prompt_chars"] = len(str(prompt or ""))
        _db_execute(
            "UPDATE ai_usage_events SET prompt_chars=?, prompt_profile_json=? WHERE id=?",
            (safe_profile["final_prompt_chars"], json.dumps(safe_profile, sort_keys=True, separators=(",", ":")), event_id),
        )
    except Exception:
        pass
    resp = None
    payload = None
    try:
        if provider == "ANTHROPIC":
            resp = requests.post(
                "https://api.anthropic.com/v1/messages",
                headers={"x-api-key": api_key, "anthropic-version": "2023-06-01", "content-type": "application/json"},
                json={"model": model, "max_tokens": int(max_tokens), "messages": [{"role": "user", "content": prompt}]},
                timeout=60,
            )
            try:
                payload = resp.json()
            except Exception:
                payload = {}
            resp.raise_for_status()
            text = "".join(block.get("text", "") for block in payload.get("content", []) if isinstance(block, dict)).strip()
        elif provider == "GEMINI":
            resp = requests.post(
                f"https://generativelanguage.googleapis.com/v1beta/models/{model}:generateContent",
                headers={"x-goog-api-key": api_key, "Content-Type": "application/json"},
                json={
                    "contents": [{"role": "user", "parts": [{"text": prompt}]}],
                    "generationConfig": {"maxOutputTokens": int(max_tokens)},
                },
                timeout=60,
            )
            try:
                payload = resp.json()
            except Exception:
                payload = {}
            resp.raise_for_status()
            text = _gemini_output_text(payload)
        else:
            resp = requests.post(
                "https://api.openai.com/v1/responses",
                headers={"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"},
                json={"model": model, "input": prompt, "max_output_tokens": int(max_tokens)},
                timeout=60,
            )
            try:
                payload = resp.json()
            except Exception:
                payload = {}
            resp.raise_for_status()
            text = _openai_output_text(payload)
        trunc_reason = _ai_output_truncation_reason(provider, payload)
        if trunc_reason:
            _finish_ai_usage(event_id, provider, payload, status="truncated", http_status=getattr(resp, "status_code", None), error_type=trunc_reason)
            raise AIProviderTransportError("TRUNCATED")
        if not text:
            raise AIProviderTransportError("EMPTY_RESPONSE")
        provider_status = "provider_ok" if call_kind == "SNAPSHOT" else "success"
        _finish_ai_usage(event_id, provider, payload, status=provider_status, http_status=getattr(resp, "status_code", None))
        return (text, event_id) if return_event_id else text
    except Exception as exc:
        if not (isinstance(exc, AIProviderTransportError) and str(getattr(exc, "kind", "")).upper() == "TRUNCATED"):
            _finish_ai_usage(
                event_id, provider, payload, status="error", http_status=getattr(resp, "status_code", None), error_type=exc.__class__.__name__,
            )
        if isinstance(exc, requests.Timeout):
            raise AIProviderTransportError("TIMEOUT") from exc
        if isinstance(exc, requests.ConnectionError):
            raise AIProviderTransportError("CONNECTION") from exc
        raise
def _safe_external_network_error(exc, context="AI"):
    label = str(context or "AI").strip() or "AI"
    kind = str(getattr(exc, "kind", "") or "").upper()
    if kind == "TIMEOUT" or isinstance(exc, requests.Timeout):
        return f"{label} timed out before the request completed. Your saved data and latest Snapshot remain safe. Please try again."
    if kind == "CONNECTION" or isinstance(exc, requests.ConnectionError):
        return f"{label} could not be reached. Your saved data and latest Snapshot remain safe. Check connectivity and try again."
    if kind == "EMPTY_RESPONSE":
        return f"{label} returned no usable response. Your saved data and latest Snapshot remain safe. Please try again."
    if kind == "TRUNCATED":
        return f"{label} reached its output limit before completing a valid Snapshot. The incomplete result was not saved and THE LAB did not retry automatically, so there was no second paid call."
    return f"{label} is temporarily unavailable. Your saved data and latest Snapshot remain safe. Please try again."
def _safe_external_http_error(exc):
    response = getattr(exc, "response", None)
    status = getattr(response, "status_code", None)
    url = str(getattr(response, "url", "") or "")
    code = f"HTTP {status}" if status else "HTTP error"
    if "generativelanguage.googleapis.com" in url:
        if status == 404:
            return (
                "Google Gemini could not serve the configured model (HTTP 404). "
                "THE LAB Standard/QA uses Gemini 3.5 Flash in V4.8.33; if this persists, "
                "check the Standard Gemini project/key access in Google AI Studio."
            )
        return f"Google Gemini request failed ({code}). Your saved Snapshot remains available."
    if "intervals.icu" in url:
        return f"Intervals.icu request failed ({code}). Check the athlete connection and try again."

def _validated_timezone(value):
    value = (value or "").strip()
    if not value:
        raise ValueError("Choose your timezone")
    try:
        ZoneInfo(value)
    except Exception as exc:
        raise ValueError("Timezone is not a valid IANA timezone, e.g. Europe/Rome") from exc
    return value
def _number_field(form, name, minimum=None, maximum=None, integer=False, required=False):
    raw = (form.get(name) or "").strip()
    if raw == "":
        if required:
            raise ValueError(f"{name.replace('_', ' ').title()} is required")
        return None
    try:
        value = int(raw) if integer else float(raw)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"{name.replace('_', ' ').title()} must be numeric") from exc
    if minimum is not None and value < minimum:
        raise ValueError(f"{name.replace('_', ' ').title()} is below the allowed range")
    if maximum is not None and value > maximum:
        raise ValueError(f"{name.replace('_', ' ').title()} is above the allowed range")
    return value
def _validated_birth_date(value, required=True):
    raw = str(value or "").strip()
    if not raw:
        if required:
            raise ValueError("Date of birth is required")
        return None
    try:
        parsed = date.fromisoformat(raw)
    except Exception as exc:
        raise ValueError("Date of birth must be a valid calendar date") from exc
    today = date.today()
    if parsed >= today:
        raise ValueError("Date of birth must be in the past")
    if parsed.year < 1900:
        raise ValueError("Date of birth is outside the supported range")
    return parsed.isoformat()
def _age_years_from_birth_date(value, on_date=None):
    raw = str(value or "").strip()
    if not raw:
        return None
    try:
        born = date.fromisoformat(raw)
    except Exception:
        return None
    ref = on_date if isinstance(on_date, date) else date.today()
    if born >= ref:
        return None
    return int(ref.year - born.year - ((ref.month, ref.day) < (born.month, born.day)))

def _utc_parse(value):
    if not value:
        return None
    try:
        return datetime.fromisoformat(str(value).replace("Z", "+00:00")).replace(tzinfo=None)

        return None
def _security_hmac(value):
    key = (SECRET_KEY or "change-this-key").encode("utf-8")
    return hmac.new(key, str(value).encode("utf-8"), hashlib.sha256).hexdigest()

def _parse_admin_date(value, field_label="Date"):
    raw = (value or "").strip()
    if not raw:
        return None
    for fmt in ("%d/%m/%Y", "%Y-%m-%d"):
        try:
            return datetime.strptime(raw, fmt).date()
        except ValueError:
            pass
    raise ValueError(f"{field_label} must use DD/MM/YYYY (for example 30/09/2026).")

def _authenticate_user(username, password):
    user = _get_user_by_username(username)
    now = datetime.utcnow()
    if not user:
        check_password_hash(_DUMMY_PASSWORD_HASH, password or "")
        audit_event("LOGIN_FAILED_UNKNOWN_USER", details={"username": str(username)[:80]})
        return None, "Incorrect username or password."
    locked_until = _utc_parse(user.get("locked_until_utc"))
    if locked_until and locked_until > now:
        audit_event("LOGIN_BLOCKED_LOCKOUT", target_user_id=user["id"])
        return None, "Incorrect username or password."
    ok = check_password_hash(user.get("password_hash") or _DUMMY_PASSWORD_HASH, password or "")
    if not ok:
        failures = int(user.get("failed_login_count") or 0) + 1
        locked_text = None
        if failures >= MAX_LOGIN_FAILURES:
            locked_text = (now + timedelta(minutes=LOGIN_LOCKOUT_MINUTES)).replace(microsecond=0).isoformat() + "Z"
            failures = 0
        _db_execute(
            "UPDATE users SET failed_login_count=?, locked_until_utc=?, updated_at_utc=? WHERE id=?",
            (failures, locked_text, _now_utc_text(), user["id"]),
        )
        audit_event("LOGIN_FAILED", target_user_id=user["id"], details={"lockout_started": bool(locked_text)})
        return None, "Incorrect username or password."
    if str(user.get("status") or "").lower() != "active":
        audit_event("LOGIN_BLOCKED_STATUS", target_user_id=user["id"], details={"status": user.get("status")})
        return None, "This account is not currently authorized."
    if user.get("role") != "admin" and not _active_license_for_user(user["id"]):
        audit_event("LOGIN_BLOCKED_LICENSE", target_user_id=user["id"])
        return None, "This account does not have an active THE LAB entitlement."
    now_text = _now_utc_text()
    _db_execute("UPDATE users SET failed_login_count=0, locked_until_utc=NULL, last_login_at_utc=?, updated_at_utc=? WHERE id=?", (now_text, now_text, user["id"]))
    user = _get_user_by_id(user["id"])
    audit_event("LOGIN_SUCCESS", actor_user_id=user["id"], target_user_id=user["id"])
    return user, None
def _validate_new_account(username, email, password):
    username = (username or "").strip()
    email = (email or "").strip()
    if not _USERNAME_RE.fullmatch(username):
        raise ValueError("Username must be 3–40 characters using letters, numbers, dot, dash or underscore.")
    if email and (len(email) > 254 or "@" not in email or email.startswith("@") or email.endswith("@")):
        raise ValueError("Enter a valid email address.")
    if len(password or "") < 12:
        raise ValueError("Password must be at least 12 characters long.")
    if _get_user_by_username(username):
        raise ValueError("That username is already in use.")
    if email:
        rows = _db_execute("SELECT id FROM users WHERE email_key=? LIMIT 1", (_normalize_email(email),), fetch=True)
        if rows:
            raise ValueError("That email is already linked to an account.")
    return username, email

def redeem_access_code(raw_code, username, email, password, birth_date):
    username, email = _validate_new_account(username, email, password)
    birth_date = _validated_birth_date(birth_date, required=True)
    code_hash = _security_hmac((raw_code or "").strip().upper())
    if not raw_code:
        raise ValueError("Enter your THE LAB access code.")
    conn = _db_connect()
    try:
        cur = conn.cursor()
        if _db_is_postgres():
            cur.execute(_db_sql("SELECT * FROM access_codes WHERE code_hash=? FOR UPDATE"), (code_hash,))
        else:
            cur.execute("BEGIN IMMEDIATE")
            cur.execute(_db_sql("SELECT * FROM access_codes WHERE code_hash=?"), (code_hash,))
        raw = cur.fetchone()
        if not raw:
            raise ValueError("That access code is invalid or no longer available.")
        if _db_is_postgres():
            cols = [d[0] for d in cur.description]
            code = dict(zip(cols, raw))
        else:
            code = dict(raw)
        if str(code.get("status") or "") != "unused":
            raise ValueError("That access code is invalid or no longer available.")
        expiry = _utc_parse(code.get("expires_at_utc"))
        if expiry is not None and expiry < datetime.utcnow():
            cur.execute(_db_sql("UPDATE access_codes SET status='expired' WHERE id=?"), (code["id"],))
            conn.commit()
            raise ValueError("That access code is invalid or no longer available.")
        user_id = str(uuid.uuid4())
        license_id = str(uuid.uuid4())
        now_text = _now_utc_text()
        cur.execute(
            _db_sql("INSERT INTO users (id, username, username_key, email, email_key, birth_date, password_hash, role, status, created_at_utc, updated_at_utc, failed_login_count) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)") ,
            (user_id, username, _normalize_username(username), email or None, _normalize_email(email) if email else None, birth_date, generate_password_hash(password), "athlete", "active", now_text, now_text, 0),
        )
        cur.execute(
            _db_sql("INSERT INTO licenses (id, user_id, license_type, status, created_at_utc, activated_at_utc, expires_at_utc, revoked_at_utc, created_by_user_id, note) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)") ,
            (license_id, user_id, code.get("license_type") or "BETA", "active", now_text, now_text, code.get("expires_at_utc"), None, code.get("created_by_user_id"), code.get("note")),
        )
        cur.execute(
            _db_sql("UPDATE access_codes SET status='redeemed', redeemed_at_utc=?, redeemed_by_user_id=? WHERE id=?"),
            (now_text, user_id, code["id"]),
        )
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()
    audit_event("ACCESS_CODE_REDEEMED", actor_user_id=user_id, target_user_id=user_id, details={"prefix": (raw_code or "")[:10]})
    return _get_user_by_id(user_id)

def set_user_access_state(user_id, action, actor_user_id):
    user = _get_user_by_id(user_id)
    if not user or user.get("role") == "admin":
        raise ValueError("Administrator access cannot be changed from this control.")
    now_text = _now_utc_text()
    action = (action or "").strip().lower()
    if action == "suspend":
        _db_execute("UPDATE users SET status='suspended', updated_at_utc=? WHERE id=?", (now_text, user_id))
        revoke_all_user_sessions(user_id, actor_user_id)
    elif action == "reactivate":
        _db_execute("UPDATE users SET status='active', admin_hidden=0, updated_at_utc=? WHERE id=?", (now_text, user_id))
        rows = _db_execute("SELECT id FROM licenses WHERE user_id=? ORDER BY created_at_utc DESC LIMIT 1", (user_id,), fetch=True)
        if rows:
            _db_execute("UPDATE licenses SET status='active', revoked_at_utc=NULL WHERE id=?", (rows[0]["id"],))
    elif action == "revoke":
        _db_execute("UPDATE users SET status='revoked', updated_at_utc=? WHERE id=?", (now_text, user_id))
        _db_execute("UPDATE licenses SET status='revoked', revoked_at_utc=? WHERE user_id=? AND status='active'", (now_text, user_id))
        revoke_all_user_sessions(user_id, actor_user_id)
    elif action == "logout_all":
        revoke_all_user_sessions(user_id, actor_user_id)
    else:
        raise ValueError("Unknown access action.")
    audit_event("USER_ACCESS_" + action.upper(), actor_user_id=actor_user_id, target_user_id=user_id)

def update_user_license_expiry(user_id, action, expires_on, actor_user_id):
    user = _get_user_by_id(user_id)
    if not user or user.get("role") == "admin":
        raise ValueError("Administrator licensing cannot be changed from this control.")
    lic = _latest_license_for_user(user_id)
    if not lic:
        raise ValueError("This athlete does not have a license to renew.")
    action = (action or "set").strip().lower()
    old_expiry = lic.get("expires_at_utc")
    today = datetime.utcnow().date()
    if action == "no_expiry":
        new_expiry = None
    elif action in {"add_30", "add_90"}:
        days = 30 if action == "add_30" else 90
        current_day = _stored_expiry_to_date(old_expiry)
        base_day = current_day if current_day and current_day > today else today
        new_expiry = _inclusive_expiry_to_utc(base_day + timedelta(days=days))
    elif action == "set":
        selected = _parse_admin_date(expires_on, "License expiry")
        if selected is None:
            raise ValueError("Enter the new expiry as DD/MM/YYYY, or use No expiry.")
        if selected < today:
            raise ValueError("License expiry cannot be earlier than today.")
        new_expiry = _inclusive_expiry_to_utc(selected)
    else:
        raise ValueError("Unknown license expiry action.")
    _db_execute("UPDATE licenses SET expires_at_utc=? WHERE id=?", (new_expiry, lic["id"]))
    audit_event(
        "LICENSE_EXPIRY_UPDATED", actor_user_id=actor_user_id, target_user_id=user_id,
        details={"action": action, "old_expiry": old_expiry, "new_expiry": new_expiry, "license_id": lic.get("id")},
    )
    return _latest_license_for_user(user_id)

def _admin_last_snapshot_ai_map():
    # Postgres/psycopg2 treats literal % characters as parameter-format markers
    # when execute() receives a params tuple, even an empty one. Keep LIKE data
    # in the bound parameter so /admin cannot crash with "tuple index out of range".
    rows = _db_execute(
        "SELECT user_id, created_at_utc, status, error_type, call_type FROM ai_usage_events "
        "WHERE UPPER(call_type) LIKE ? ORDER BY created_at_utc DESC LIMIT 500",
        ("%SNAPSHOT%",), fetch=True,
    )
    out = {}
    for row in rows:
        user_id = str(row.get("user_id") or "")
        if user_id and user_id not in out:
            out[user_id] = row
    return out

def format_ui_date(value):
    if value in (None, ""):
        return "—"
    try:
        return date.fromisoformat(str(value)[:10]).strftime("%d/%m/%Y")
    except (ValueError, TypeError):
        return str(value)
def format_ui_time(value):
    if value in (None, ""):
        return "—"
    raw = str(value).strip()
    for fmt in ("%H:%M", "%H:%M:%S"):
        try:
            return datetime.strptime(raw, fmt).strftime("%I:%M %p")
        except ValueError:
            pass
    try:
        return datetime.fromisoformat(raw.replace("Z", "+00:00")).strftime("%I:%M %p")
    except (ValueError, TypeError):
        return raw

def _race_duration_display(low, high, unknown="Duration uncertain"):
    low = _nutrition_number(low)
    high = _nutrition_number(high)
    if low is None and high is None:
        return unknown
    if low is None:
        low = high
    if high is None:
        high = low
    low, high = min(low, high), max(low, high)
    def _fmt(v):
        v = float(v)
        if v < 60:
            return f"{int(round(v))} min"
        h = v / 60.0
        if abs(h - round(h)) < 0.03:
            return f"{int(round(h))} h"
        if v % 60 == 0:
            return f"{int(v // 60)} h"
        hours = int(v // 60)
        mins = int(round(v - hours * 60))
        return f"{hours} h {mins} min" if hours else f"{mins} min"
    if abs(high - low) < 0.5:
        return f"~{_fmt(low)} expected"
    return f"~{_fmt(low)}–{_fmt(high)} expected"
def _planning_duration_value(value):
    if value in (None, ""):
        return None
    try:
        v = float(value)
    except (TypeError, ValueError) as exc:
        raise ValueError("Expected duration must be numeric minutes") from exc
    if not 10 <= v <= 1440:
        raise ValueError("Expected duration must be between 10 and 1440 minutes")
    return round(v, 1)

def _planning_race_fields(kind, training_environment=None, training_demand=None, expected_duration_min=None, expected_duration_max=None):
    kind = str(kind or "").upper()
    env = str(training_environment or "").strip().upper() or None
    demand = str(training_demand or "").strip().upper() or None
    dmin = _planning_duration_value(expected_duration_min)
    dmax = _planning_duration_value(expected_duration_max)
    if kind in ("TRAINING", "RACE"):
        if env not in ("INDOOR", "OUTDOOR"):
            if kind == "RACE":
                raise ValueError("Choose whether the race is indoor / virtual or outdoor")
            env = "OUTDOOR"
    else:
        env = None
    if kind == "TRAINING":
        demand = _training_profile_key(demand)
        duration = dmin if dmin is not None else dmax
        dmin = dmax = duration
    elif kind == "RACE":
        demand = None
        if dmin is None and dmax is not None:
            dmin = dmax
        if dmax is None and dmin is not None:
            dmax = dmin
        if dmin is not None and dmax is not None and dmin > dmax:
            dmin, dmax = dmax, dmin
    else:
        demand = None
        dmin = dmax = None
    return env, demand, dmin, dmax

def load_planning_items(active_only=True, limit=100, owner_user_id=None):
    owner_id = _data_owner_user_id(owner_user_id)
    where = "AND active=1" if active_only else ""
    rows = _data_execute(owner_id,
        f"SELECT * FROM planning_items WHERE user_id=? {where} ORDER BY start_date ASC, created_at_utc DESC LIMIT ?",
        (owner_id, int(limit)), fetch=True,
    )
    for row in rows:
        row.setdefault("expected_duration_min", None)
        row.setdefault("expected_duration_max", None)
        note = str(row.get("note") or "")
        if row.get("kind") == "TRAINING" and note.startswith("NOVA_POSTPONE_V1:"):
            try:
                binding = json.loads(note[len("NOVA_POSTPONE_V1:"):])
                if (isinstance(binding, dict) and binding.get("workout_id")
                        and binding.get("original_plan_id") and binding.get("original_date")
                        and str(row.get("start_date") or "") == str(binding.get("target_date") or "")):
                    row["nova_postponement"] = binding
                    row["note"] = "Existing NOVA workout moved by athlete; identity preserved."
            except (ValueError, TypeError):
                pass
    return rows
def _activity_context_from_list(activities, now=None, source="Intervals.icu", context_date=None):
    now = now or get_rome_now()
    today_iso = now.date().isoformat()
    context_date = _safe_iso_date(context_date, today_iso)
    day_acts = [a for a in (activities or []) if str(a.get("start_date_local") or "")[:10] == context_date]
    parsed = []
    for a in day_acts:
        dt = parse_local_activity_datetime(a.get("start_date_local"))
        parsed.append((a, dt))
    am_done = any(dt is not None and dt.hour < 12 for _, dt in parsed)
    pm_done = any(dt is not None and dt.hour >= 17 for _, dt in parsed)
    day_load = sum(float(a.get("icu_training_load") or 0) for a, _ in parsed if isinstance(a.get("icu_training_load"), (int, float)))
    if context_date < today_iso:
        phase = "RETRO · FULL DAY"
    elif day_acts:
        n = len(day_acts)
        phase = f"POST-TRAINING · {n} SESSION{'S' if n != 1 else ''}"
    else:
        phase = "PRE-TRAINING"
    return {
        "phase": phase,
        "session_count": len(day_acts),
        "today_load": round(day_load, 1),
        "am_done": am_done,
        "pm_done": pm_done,
        "context_source": source,
        "context_date": context_date,
    }
def fetch_activity_context_for_date(context_date=None, now=None):
    now = now or get_rome_now()
    today_iso = now.date().isoformat()
    target = _safe_iso_date(context_date, today_iso)
    if target > today_iso:
        raise ValueError("Daily Log cannot be dated in the future")
    retrospective = target < today_iso
    suffix = " · retrospective" if retrospective else ""
    try:
        runtime = _resolve_intervals_runtime()
    except RuntimeError:
        return _activity_context_from_list([], now=now, source="Clock only" + suffix, context_date=target)
    fields = "id,name,type,start_date_local,moving_time,elapsed_time,icu_training_load"
    url = (
        f"https://intervals.icu/api/v1/athlete/{runtime['athlete_id']}/activities"
        f"?oldest={target}&newest={target}&fields={fields}"
    )
    try:
        response = requests.get(url, headers=get_intervals_headers(runtime), timeout=12)
        response.raise_for_status()
        return _activity_context_from_list(response.json(), now=now, source="Intervals.icu" + suffix, context_date=target)
    except Exception:
        return _activity_context_from_list([], now=now, source="Clock only · Intervals unavailable" + suffix, context_date=target)

def _v4886_encode_execution_activity_key(activity_id=None, activity_date=None, activity_name=None):
    payload = {"id": str(activity_id or "").strip() or None, "date": str(activity_date or "")[:10] or None, "name": str(activity_name or "").strip()[:220] or None}
    if not payload["id"] and not (payload["date"] and payload["name"]):
        return ""
    raw = json.dumps(payload, ensure_ascii=False, separators=(",", ":")).encode("utf-8")
    return base64.urlsafe_b64encode(raw).decode("ascii").rstrip("=")
def _v4886_decode_execution_activity_key(value):
    token = str(value or "").strip()
    if not token:
        return {"id": None, "date": None, "name": None}
    try:
        raw = base64.urlsafe_b64decode(token + "=" * (-len(token) % 4))
        payload = json.loads(raw.decode("utf-8"))
    except Exception as exc:
        raise ValueError("Choose a valid completed activity for execution feedback") from exc
    if not isinstance(payload, dict):
        raise ValueError("Choose a valid completed activity for execution feedback")
    activity_id = str(payload.get("id") or "").strip()[:160] or None
    activity_date = _safe_iso_date(payload.get("date"))
    activity_name = str(payload.get("name") or "").strip()[:220] or None
    if not activity_id and not (activity_date and activity_name):
        raise ValueError("Choose a valid completed activity for execution feedback")
    return {"id": activity_id, "date": activity_date, "name": activity_name}
def _v4886_execution_activity_options(owner_user_id=None, limit=16, lookback_days=21):
    owner_id = _data_owner_user_id(owner_user_id)
    options = []
    seen = set()
    def append_activity(activity, source="INTERVALS_LIVE", assume_cycling=False):
        if not isinstance(activity, dict) or (not assume_cycling and not _v4887_is_cycling_activity(activity)):
            return
        activity_id = str(activity.get("id") or activity.get("activity_id") or "").strip() or None
        activity_date = str(activity.get("start_date_local") or activity.get("date") or "")[:10] or None
        activity_name = str(activity.get("name") or "").strip()[:220] or None
        activity_time = str(activity.get("start_date_local") or "")[11:16] or str(activity.get("time") or "").strip()[:8] or None
        if not activity_date or not activity_name:
            return
        dedupe = activity_id or (activity_date, activity_name, activity_time or "")
        if dedupe in seen:
            return
        seen.add(dedupe)
        key = _v4886_encode_execution_activity_key(activity_id, activity_date, activity_name)
        if not key:
            return
        family = str(activity.get("family") or classify_training_family(activity) or "")
        options.append({
            "key": key, "id": activity_id, "date": activity_date, "name": activity_name,
            "time": activity_time, "family": family, "hard": bool(_v4895_activity_quality_truth(activity, family).get("quality_relevant")),
            "source": source,
            "label": f"{activity_date} · {activity_time or ''} · {activity_name}".replace(" ·  · ", " · "),
        })
    try:
        runtime = _resolve_intervals_runtime(owner_id)
        newest = get_rome_now().date()
        oldest = newest - timedelta(days=max(1, int(lookback_days or 14)))
        fields = "id,name,type,sub_type,start_date_local,trainer,icu_training_load,icu_intensity"
        url = (
            f"https://intervals.icu/api/v1/athlete/{runtime['athlete_id']}/activities"
            f"?oldest={oldest.isoformat()}&newest={newest.isoformat()}&fields={fields}"
        )
        response = requests.get(url, headers=get_intervals_headers(runtime), timeout=8)
        response.raise_for_status()
        live = response.json()
        if isinstance(live, list):
            for activity in live:
                append_activity(activity, "INTERVALS_LIVE")
    except Exception:
        try:
            report = load_latest_report(include_payload=True, owner_user_id=owner_id)
        except Exception:
            report = None
        data = (report or {}).get("data") if isinstance(report, dict) else None
        blocks = list((data or {}).get("previous_training_blocks") or []) if isinstance(data, dict) else []
        for block in blocks:
            append_activity(block, "STORED_SNAPSHOT_FALLBACK", assume_cycling=True)
    options.sort(
        key=lambda x: (str(x.get("date") or ""), str(x.get("time") or ""), bool(x.get("hard"))),
        reverse=True,
    )
    return options[:max(1, min(16, int(limit or 16)))]
def save_daily_log(feeling, note="", status="NORMAL", restriction_type="NONE", duration_days=None, until_date=None, activity_context=None, log_date=None, fueling_gph=None, fueling_tolerance=None, execution_posture=None, execution_terrain=None, execution_activity_id=None, execution_activity_date=None, execution_activity_name=None, session_execution_score=None, post_session_state=None, session_rpe=None, owner_user_id=None):
    owner_id = _data_owner_user_id(owner_user_id)
    now = get_rome_now()
    today_iso = now.date().isoformat()
    log_date = _safe_iso_date(log_date, today_iso)
    if log_date > today_iso:
        raise ValueError("Daily Log cannot be dated in the future")
    feeling = int(feeling)
    if not 1 <= feeling <= 10:
        raise ValueError("Feeling must be between 1 and 10")
    status = (status or "NORMAL").strip().upper()
    if status not in ("NORMAL", "REST", "SICK", "INJURED", "UNAVAILABLE"):
        status = "NORMAL"
    restriction_type = (restriction_type or "NONE").strip().upper()
    if restriction_type not in ("NONE", "NO_TRAINING", "NO_INTENSITY"):
        restriction_type = "NONE"
    if status in ("SICK", "REST", "UNAVAILABLE") and restriction_type == "NONE":
        restriction_type = "NO_TRAINING"
    if status == "INJURED" and restriction_type == "NONE":
        restriction_type = "NO_INTENSITY"
    end_date = None
    try:
        days = int(duration_days) if duration_days not in (None, "") else None
    except (TypeError, ValueError):
        days = None
    if days and days > 0:
        end_date = (date.fromisoformat(log_date) + timedelta(days=days - 1)).isoformat()
    else:
        end_date = _safe_iso_date(until_date)
    if (status != "NORMAL" or restriction_type != "NONE") and not end_date:
        end_date = log_date
    if end_date and end_date < log_date:
        end_date = log_date
    feedback_gph = None
    if fueling_gph not in (None, ""):
        try:
            feedback_gph = float(fueling_gph)
        except (TypeError, ValueError) as exc:
            raise ValueError("Fueling feedback must be numeric g CHO/h") from exc
        if not 0 <= feedback_gph <= 180:
            raise ValueError("Fueling feedback must be between 0 and 180 g CHO/h")
