# R196.8.6 — evidence-anchored, DOSE-AWARE exploratory sustained power.
# A qualified 20–29-minute effort is a performance observation; its FTP_UP is
# an ESTIMATE, and neither establishes the repeatability of a particular set.
# The previous blanket 79–92% of estimated FTP routinely produced tempo-like
# ranges even for POWER_20 threshold development (e.g. 2x15 minutes).
# Instead, constrain an exploration to the RECORDED test watts and adjust for
# total demanding minutes, single repetition length and test recency. No
# fixed target, no pass/fail on watts, and NO bypass of recovery/spacing gates.
R19686_SCHEMA = "V4.9.97-R196.8.6-DOSE-AWARE-SUSTAINED-EXPLORATION-1"
R19686_SOURCE = "OBSERVED_20_29_MIN_TEST_DOSE_AWARE_EXPLORATION"
_r19685_band_before = _v1943_band


def _v1943_band(template, evidence_ledger=None, ftp_anchor=None, as_of_date=None, dimension=None):
    band = _r19685_band_before(template, evidence_ledger=evidence_ledger,
                               ftp_anchor=ftp_anchor, as_of_date=as_of_date,
                               dimension=dimension)
    # Preserve every genuine repeat-work or valid-FTP contract, and do not
    # overrule failed comparable-recovery evidence or unsupported structures.
    if (not isinstance(band, dict)
            or band.get('state') != 'NO_DEFENSIBLE_POWER_TARGET'
            or band.get('reason') != 'NO_RECENT_COMPARABLE_COMPLETED_REPS_OR_VALID_FTP'):
        return band
    t = template if isinstance(template, dict) else {}
    ftp = ftp_anchor if isinstance(ftp_anchor, dict) else {}
    goal = str(t.get('goal') or '').upper()
    work = _v1943_finite(t.get('work_s'), 480, 1200)
    reps = _v1943_finite(t.get('reps'), 2, 4)
    recovery = _v1943_finite(t.get('recovery_s'), 120, 900)
    seconds = _v1943_finite(ftp.get('effort_secs'), 1200, 1799)
    estimate = _v1943_finite(ftp.get('anchor_watts'), 80, 600)
    observed = _v1943_finite(ftp.get('effort_watts'), 80, 1200)
    if (goal not in {'POWER_20', 'TIME_TRIAL'}
            or str(t.get('pattern') or '').upper() != 'SUSTAINED'
            or work is None or reps is None or recovery is None
            or seconds is None or estimate is None or observed is None
            or not float(reps).is_integer()
            or ftp.get('available') is not True
            or ftp.get('evidence_schema') != R1966_SCHEMA
            or str(ftp.get('status') or '').upper() != 'TEST ESTIMATE'
            or ftp.get('evidence_grade') not in {None, '20_MIN_PROVISIONAL_ESTIMATE'}
            or ftp.get('prescription_watts') is not None):
        return band
    # A badly inconsistent FTP_UP signal cannot calibrate even an exploration.
    # The directly recorded test effort, NOT the estimated FTP or rolling eFTP,
    # remains the physical ceiling for provisional repeat-work guidance.
    if not 0.90 <= estimate / observed <= 1.08:
        return band
    total_work = float(reps) * work
    dose_ratio = total_work / seconds
    rep_ratio = work / seconds
    # Extrapolating a single 20-minute effort into e.g. 2x20 is not a modest
    # starting dose. Let the library try shorter, evidenced structures instead.
    if work > seconds or dose_ratio > 1.65:
        return band
    try:
        tested_on = date.fromisoformat(str(ftp.get('anchor_date') or '')[:10])
        planned_on = date.fromisoformat(str(as_of_date)[:10]) if as_of_date else get_rome_now().date()
        age = (planned_on - tested_on).days
    except (ValueError, TypeError, AttributeError):
        return band
    if not 0 <= age <= FTP_ANCHOR_VALID_DAYS:
        return band

    # At ~same total work as the original test, explore near demonstrated
    # sustained intensity; increasing total work or rep length costs power.
    # This is a transparent *coaching hypothesis*, not a physiological test
    # curve, completed-repetition proof or precision-matched power target.
    upper_fraction = (1.01
                      - 0.11 * max(0.0, dose_ratio - 0.90)
                      - 0.04 * max(0.0, rep_ratio - 0.55))
    upper_fraction = min(1.0, max(0.91, upper_fraction))
    if str(dimension or '').upper() in {'DURABILITY', 'RACE_TRANSFER'}:
        upper_fraction -= 0.012
    if age > 84:
        upper_fraction -= 0.015
    elif age > 42:
        upper_fraction -= 0.008
    lower_fraction = upper_fraction - 0.045

    # 5 W roundoff tolerance only; no boost above the observed tested power.
    ceiling = int(round(observed / 5.0) * 5)
    low = int(round(observed * lower_fraction / 5.0) * 5)
    high = min(ceiling, int(round(observed * upper_fraction / 5.0) * 5))
    if not 80 <= low < high <= 600:
        return band
    return {
        'schema': R1943_SCHEMA, 'state': 'PRESCRIBED',
        'evidence_schema': R19686_SCHEMA, 'source': R19686_SOURCE,
        'confidence': 'LOW', 'exploratory': True, 'non_punitive': True,
        'scoring_policy': 'STRUCTURE_ONLY_POWER_DESCRIPTIVE',
        'power_target_mode': 'EXPLORATORY_RANGE', 'target_watts': None,
        'low_watts': low, 'high_watts': high, 'reps': int(reps),
        'work_seconds': int(work), 'recovery_seconds': int(recovery),
        'total_work_seconds': int(total_work),
        'test_duration_seconds': int(seconds),
        'test_date': tested_on.isoformat(), 'test_age_days': age,
        'observed_test_effort_watts': int(round(observed)),
        'ftp_test_estimate_watts': int(round(estimate)),
        'test_to_work_dose_ratio': round(dose_ratio, 3),
        'repeatability_state': 'NOT_VERIFIED_FOR_THIS_WORKOUT',
        'repeatability_reason': 'SINGLE_TEST_NOT_COMPLETED_REPEATS',
        'observed_single_best_is_repeatability_proof': False,
        'observed_repeated_work_is_direct_evidence': False,
        'reason': 'DOSE_AWARE_EXPLORATION_FROM_RECORDED_TEST_NOT_VERIFIED_REPEATS',
    }

def _v19693_established_weekday_evidence(cadence, weekday):
    """Avoid turning a short recent dip into erased established training history.

    A sample can support a *timing preference*, never readiness or eligibility.
    Count independent observed days, not hard-start occurrences or smoothing
    priors. Missing/ambiguous data stays unknown, not automatically positive.
    """
    context = (cadence or {}).get("weekly_session_role") or {}
    counts = context.get("weekday_counts") or {}
    row = counts.get(weekday, counts.get(str(weekday), {})) or {}
    if not isinstance(row, dict):
        return None
    try:
        training, quality = int(row.get("training_days")), int(row.get("quality_days"))
    except (ValueError, TypeError):
        return None
    if training < 4 or quality < 2 or quality > training:
        return None
    if context.get("shift_detected") and weekday in set(context.get("shift_weekdays") or []):
        return None
    rate = quality / training
    # Established-day pattern can support an early candidate even if the
    # 21-day blended rate crosses an arbitrary 0.5 classification boundary.
    return {"training_days": training, "quality_days": quality,
            "observed_quality_rate": round(rate, 3)} if rate >= 0.5 else None


def _v19693_comparable_quality_time(row, cadence, preferred_row):
    dt = row.get("dt"); later = preferred_row.get("dt")
    if not isinstance(dt, datetime) or not isinstance(later, datetime):
        return False
    if (dt.hour < 12) != (later.hour < 12):
        return False
    habitual = str((cadence or {}).get("hard_daypart") or "").upper()
    confidence = str((cadence or {}).get("time_confidence") or "").upper()
    if habitual in {"AM", "PM"} and confidence in {"MODERATE", "HIGH"}:
        if ("AM" if dt.hour < 12 else "PM") != habitual:
            return False
    minute = (cadence or {}).get("hard_start_minute")
    if minute is not None and confidence in {"MODERATE", "HIGH"}:
        try:
            habitual_min = float(minute)
            earlier_delta = abs((dt.hour * 60 + dt.minute) - habitual_min)
            later_delta = abs((later.hour * 60 + later.minute) - habitual_min)
            if earlier_delta > later_delta + 90:
                return False
        except (TypeError, ValueError):
            return False
    return True


def _v117_select_quality_window(candidates, last_end, spacing, cadence, extra_delay, second_hard=False):
    result = _r19693_quality_window_before(candidates, last_end, spacing, cadence,
        extra_delay, second_hard=second_hard)
    if not isinstance(result, tuple) or len(result) != 4:
        return result
    earliest, preferred, reason, chosen = result
    if not isinstance(earliest, datetime) or not isinstance(preferred, datetime) or earliest >= preferred:
        return result
    try:
        if float(extra_delay) != 0.0:  # no recovered-early override with a penalty
            return result
        lower = last_end + timedelta(hours=float((spacing or {}).get("earliest_hours") or 36)) if last_end else None
        preferred_min = last_end + timedelta(hours=float((spacing or {}).get("preferred_hours") or 48)) if last_end else None
        valid = sorted([r for r in (candidates or []) if isinstance(r, dict)
                        and isinstance(r.get("dt"), datetime)
                        and (lower is None or r["dt"] >= lower)], key=lambda r: r["dt"])
        if not valid or any(r.get("explicit") for r in valid):
            return result  # explicit athlete/Calendar intent stays authoritative
        if chosen is None or chosen.get("explicit"):
            return result
        for row in valid:
            t = row["dt"]
            if t >= preferred:
                break
            if preferred_min is not None and t < preferred_min:
                continue  # not yet at the native preferred physiological spacing
            if not _v19693_comparable_quality_time(row, cadence, chosen):
                continue
            support = _v19693_established_weekday_evidence(cadence, t.weekday())
            if not support:
                continue
            # The native filter has already removed blocked calendar/note/race
            # days and enforced recovery. No new slot is invented.
            return earliest, t, "R19693_ESTABLISHED_EARLIER_ELIGIBLE_QUALITY", row
    except (TypeError, ValueError, OverflowError, KeyError):
        return result
    return result

def _v19693_race_purpose_candidates(goal, dimension, slot, evidence_ledger):
    duration_hint = _rhythm_num((slot or {}).get("duration_hint_min")) or _parse_duration_minutes(
        (slot or {}).get("duration_hint_label"))

    for e in events:
        try:
            d = date.fromisoformat(str(e.get("date") or "")[:10])
            latest_day = max(latest_day, d) if latest_day else d
        except (ValueError, TypeError):
            pass
    ranked = []
    for template in _NOVA_WORKOUT_LIBRARY:
        if (str(template.get("goal") or "").upper() != goal
                or dimension not in set(template.get("dimensions") or ())

        if duration_hint is not None and est > float(duration_hint) + 15:
            continue
        sig = _v109_template_signature(template)
        # Canonical identical previously executed structure is not a new lever.
        duplicate = False

            if _v109_event_signature(evt) != sig:

                continue

                break

def _v19693_strong_transfer_background(evidence_ledger):

            continue

        except (TypeError, ValueError):
            continue

def _v196933_raw_attested(prescription, index, compiled_session, roadmap):
    """Recheck the original scientific source, not just a plausible label."""
    p = prescription if isinstance(prescription,dict) else {}
    rows = p.get('raw_sessions')
    if (type(index) is not int or not isinstance(rows,list) or not 0<=index<len(rows)
        or not isinstance(rows[index],dict)
        or not isinstance(roadmap,dict)):
        return False
    # R196.0 checks the watts/intervals but does not itself require the native
    # activity identifier. Require an identifiable completed-source anchor.
    science=roadmap.get('scientific_progression')
    if not isinstance(science,dict):return False
    dose=science.get('dose_contract')
    if not isinstance(dose,dict):return False
    reference=dose.get('reference')
    selection=dose.get('reference_selection') or {}
    prescription_dose=dose.get('prescription')
    if (not isinstance(reference,dict) or not isinstance(selection,dict)
        or not isinstance(prescription_dose,dict)):
        return False
    # R196.9.55: R196.9.24 stores the verified completed-activity ID in
    # reference_selection, while R195.9.reference retains the measured dose.
    # Accept either authentic producer shape; still reject absent/conflicting IDs.
    source_id=str(reference.get('activity_id') or selection.get('activity_id') or '').strip()
    if (not source_id or (selection.get('activity_id') and
        str(selection.get('activity_id'))!=source_id)):
        return False
    proven=_v1960_coupled_power_band(rows[index],compiled_session,roadmap)
    band=(compiled_session or {}).get('power_prescription') or {}
    return bool(isinstance(proven,dict) and proven.get('state')=='PRESCRIBED'
        and proven.get('source')=='OBSERVED_COUPLED_RECOVERY_DOSE'
        and proven.get('low_watts')==band.get('low_watts')==compiled_session.get('prescribed_power_low')
        and proven.get('high_watts')==band.get('high_watts')==compiled_session.get('prescribed_power_high')
        and all(proven.get(k)==band.get(k) for k in
              ('reps','work_seconds','recovery_seconds','recovery_watts_low','recovery_watts_high')))

def _v19696_strip_conflicting_power_headers(main_set):
    """Remove every stale composer prefix; never drop the mechanical main set."""
    raw=str(main_set or '').strip()
    # R196.9.7: compiler re-entry must not stack its own canonical range.
    # Only erase a byte-identifiable prefix authored by the R196.9.6 compiler;
    # do not touch observational watts, pacing cues or arbitrary free text.
    prefix=r'^(?:EXPLORATORY )?WORK RANGE:\s*\d+(?:\s*[–-]\s*\d+)?\s*W\s+(?:exploratory work range; not a pass/fail target|validated prescription range)\.\s*(?:These watts are an opening suggestion, not an upper limit; increase if sustainably comfortable\.\s*)?'
    while re.match(prefix,raw,flags=re.I):
        raw=re.sub(prefix,'',raw,count=1,flags=re.I).strip()
    raw=re.sub(r'EXPLORATORY POWER TARGET:.*?assess structure and recovery separately\.\s*',
               '',raw,flags=re.I|re.S)
    raw=re.sub(r'WORK TARGET:.*?\)\.\s*','',raw,flags=re.I|re.S)
    return raw.strip()

def _v19697_workout_core_seconds(t):
    """Compute immutable library on/off time, including the final float."""
    try:
        pat = str(t.get('pattern') or '').upper()
        if pat in {'MICRO_30_15', 'SPRINT_CLUSTER'}:
            sets = int(t['sets']); reps = int(t['reps_per_set'])
            work = float(t['work_s']); flt = float(t['float_s'])
            set_rest = float(t['set_recovery_s']) if sets > 1 else 0.0
            if sets < 1 or reps < 1 or min(work,flt) <= 0 or set_rest < 0:
                return None
            core = sets * reps * (work + flt) + (sets - 1) * set_rest
            return (core, sets * reps * work, sets * reps * flt + (sets - 1) * set_rest)
        reps = int(t['reps']); work = float(t['work_s'])
        rest = float(t['recovery_s']) if reps > 1 else 0.0
        if reps < 1 or work <= 0 or rest < 0:
            return None
        core = reps * work + (reps - 1) * rest
        return (core, reps * work, (reps - 1) * rest)
    except (KeyError, ValueError, TypeError, OverflowError):
        return None


def _v19697_time_label(seconds):
    seconds = round(float(seconds))
    minutes, remainder = divmod(seconds, 60)
    if not remainder: return f'{minutes} min'
    return f'{minutes} min {remainder} s' if minutes else f'{remainder} s'

    explicit_cool = re.search(r'\bCool[- ]?down:\s*(\d+(?:\.\d+)?)\s*min\b', text, re.I)
    warm_seconds = (prescribed_preload[0] if prescribed_preload else
                    round(float(explicit_warm.group(1)) * 60) if explicit_warm else
                    (15 if str(t.get('pattern') or '').upper() in {'SPRINT','SPRINT_CLUSTER','SHORT_POWER'} else 12) * 60)
    cool_seconds = round(float(explicit_cool.group(1)) * 60) if explicit_cool else 8 * 60
    required = round(core[0]) + warm_seconds + cool_seconds
    audit.update(state='ASSESSED', declared_seconds=total_seconds,
                 core_seconds=round(core[0]), quality_work_seconds=round(core[1]),
                 recovery_seconds=round(core[2]), preparation_seconds=warm_seconds,
                 cooldown_seconds=cool_seconds, min_required_seconds=required)
    if min(warm_seconds, cool_seconds) < 0 or required > total_seconds:
        audit.update(state='INFEASIBLE_ENVELOPE',
                     issues=['FULL_WORKOUT_COMPONENTS_EXCEED_DECLARED_DURATION'])
        return s, audit
    fill_seconds = total_seconds - required
    # This is a purpose gate, NOT a new safety/quality decision: an abnormally
    # large empty envelope cannot become hours of 'free training' without an
    # independent durability/aerobic purpose.
    if fill_seconds > 75 * 60 and dimension not in {'DURABILITY','RACE_TRANSFER'}:
        audit.update(state='UNASSESSED_LONG_FILL',
                     issues=['AEROBIC_FILL_REQUIRES_PURPOSE_AT_THIS_LENGTH'])
        return s, audit
    blocks = [
        {'type':'PREPARATION', 'duration_seconds':warm_seconds,
         'duration_min':round(warm_seconds / 60, 2),
         'intensity':'AS_ORIGINALLY_PRESCRIBED' if prescribed_preload else 'Z1_TO_LOW_Z2'},
        {'type':'CANONICAL_INTERVALS', 'duration_seconds':round(core[0]),
         'duration_min':round(core[0] / 60, 2),
         'quality_work_seconds':round(core[1]),'recovery_seconds':round(core[2]),
         'intensity':'CANONICAL_LIBRARY_UNCHANGED'},
    ]
    if fill_seconds:
        blocks.append({'type':'AEROBIC_FILL','duration_seconds':fill_seconds,
                       'duration_min':round(fill_seconds / 60, 2),'intensity':'Z1_TO_LOW_Z2'})
    blocks.append({'type':'COOLDOWN','duration_seconds':cool_seconds,
                   'duration_min':round(cool_seconds / 60, 2),'intensity':'Z1'})
    assert sum(b['duration_seconds'] for b in blocks) == total_seconds
    if prescribed_preload:
        warm_label = prescribed_preload[1]
    else:
        warm_label = f'{_v19697_time_label(warm_seconds)} progressive Z1–low-Z2 warm-up'
    parts = [warm_label,
             (f'{_v19697_time_label(core[0])} for the EXACT original canonical intervals '
              f'({_v19697_time_label(core[1])} work + {_v19697_time_label(core[2])} float / recovery)')]
    if fill_seconds:
        parts.append(f'{_v19697_time_label(fill_seconds)} conversational Z1–low-Z2 aerobic completion; NO extra hard reps')
    parts.append(f'{_v19697_time_label(cool_seconds)} very easy Z1 cool-down')

                + '; '.join(f'{i+1}) {v}' for i,v in enumerate(parts)) + '.')
    # Idempotence and controlled replacement of the older generic finish only.
    marker = 'SESSION TIMELINE ('
    older = text.split(marker, 1)[0].rstrip() if marker in text else text
    older = re.sub(r'\s*Finish with easy spinning to complete [^.]+, without adding an unplanned hard finisher\.\s*',
                   ' ', older, flags=re.I).strip()

                       if s.get('is_planned_training') is True else None)

        athlete_minutes = _v19699_number(s.get('duration_hint_min'), 1, 360)
    if athlete_minutes is not None and total_seconds / 60.0 > athlete_minutes + .01:
        over = total_seconds / 60.0 - athlete_minutes
        main += (f' TIME TRANSPARENCY: this complete session is {over:.1f} min over the '
                 f'{athlete_minutes:g}-min athlete slot. Keep the prescribed MAIN SET '
                 'and between-rep recoveries unchanged. If needed, shorten only '
                 'optional warm-up/cool-down or aerobic fill, never a required '
                 '30-min durability preload or the safe post-quality cooldown. '
                 'Do not perform an incomplete main set.')
    comp = {
        'schema': R19697_SCHEMA, 'state':'COMPLETE', 'total_duration':s.get('duration'),
        'total_seconds':total_seconds, 'total_min_low':round(total_seconds / 60, 2),

        'primary_duration_min':round(core[1] / 60, 2),
        'primary_work_seconds':round(core[1]),
        'interval_recovery_seconds':round(core[2]),
        'aerobic_fill_min_low':round(fill_seconds / 60, 2),
        'aerobic_fill_min_high':round(fill_seconds / 60, 2),

    if s.get('session_composition') not in (None, {}) and (s.get('session_composition') or {}).get('schema') != R19697_SCHEMA:
        audit.update(state='EXISTING_COMPOSITION_PRESERVED',issues=['OTHER_AUTHORITY_OWNS_COMPOSITION'])
        return s, audit

    out.pop('r19697_timing_warning',None)  # a corrected envelope clears its old warning
    # A previously issued R195.9 dose is ledger/fingerprint-sensitive: its
    # complete original instructions are already present in main_set.  Add
    # composition metadata for the time authority without altering the issued
    # text or the canonical plan identity on recompilation.
    out['main_set'] = text if wid == _R196933_VIRTUAL_ID else main
    out['session_composition'] = comp
    out['aerobic_fill_state'] = 'PLANNED_LOW_Z2' if fill_seconds else 'NOT_REQUIRED'
    out['aerobic_fill_minutes'] = _v19697_time_label(fill_seconds) if fill_seconds else None
    changed = any(out.get(key) != s.get(key) for key in
                  ('main_set','session_composition','aerobic_fill_state',
                   'aerobic_fill_minutes','r19697_timing_warning'))
    audit.update(state='COMPLETE', modified=changed, aerobic_fill_seconds=fill_seconds,
                 accounted_seconds=sum(b['duration_seconds'] for b in blocks))
    return out, audit


def _v19697_complete_session_envelopes(p):
    """Project into FINAL compiled sessions; NEVER select or promote a HARD."""
    if not isinstance(p, dict): return p
    out = dict(p)
    original_rows = list(out.get('sessions') or [])
    rows=[]; checks=[]; modified=[]
    if str(out.get('status') or '').upper() in {'BLOCKED','HOLD','DEFERRED'} or out.get('hard_prescription_available') is False:
        out['r19697_composition_audit']={'schema':R19697_SCHEMA,'checked':[],
            'modified_indices':[], 'state':'UPSTREAM_VETO_PRESERVED'}
        return out
    for i, row in enumerate(original_rows):
        updated, audit = _v19697_session_envelope(row)
        if isinstance(row,dict) and audit['state'] in {
            'INFEASIBLE_ENVELOPE','UNASSESSED_DURATION',
            'UNASSESSED_EXISTING_PRELOAD','UNASSESSED_LONG_FILL',
            'UNASSESSED_UNUSUAL_ENVELOPE'}:
            if audit['state']=='INFEASIBLE_ENVELOPE':
                warning=(f"Workout timing is INFEASIBLE as written: {_v19697_time_label(audit['min_required_seconds'])} "
                         f"are needed for the planned mechanics, preparation and cooldown, "
                         f"but only {_v19697_time_label(audit['declared_seconds'])} are declared. "
                         "Do not execute this incomplete version; it needs a new feasible prescription.")
            else:
                warning=('Workout timing is not fully verified for this duration. '
                         'Keep the existing quality decision unchanged, but do not treat '
                         'this as a complete ready-to-execute timeline.')
            updated=dict(row,r19697_timing_warning=warning)
            audit['modified']=updated!=row
        rows.append(updated)
        if audit['state'] != 'NOT_APPLICABLE':
            checks.append(dict(audit,index=i))
        if audit['modified']:modified.append(i)
    out['r19697_composition_audit'] = {
        'schema':R19697_SCHEMA, 'checked':checks, 'modified_indices':modified,
        'rule':'Time must be accounted for without adding work intervals or changing authorization; unfillable envelopes are reported, not disguised.'}
    if modified:
        out['sessions']=rows
        _v145_rehash_plan(out)
        if isinstance(out.get('single_plan_authority'),dict):
            out['single_plan_authority']=dict(out['single_plan_authority'],plan_id=out['canonical_plan_id'])
    return out

def compile_nova_prescription(prescription, coach_clock, training_definitions=None,
        ftp_anchor=None, recent_activities=None, undefined_training_intent=None,
        power_model=None, microcycle_ledger=None, coaching_contract=None,
        adaptive_roadmap=None, strength_pattern=None, training_rhythm=None,
        execution_model=None):
    out=_r19696_compile_before(prescription,coach_clock,training_definitions=training_definitions,
        ftp_anchor=ftp_anchor,recent_activities=recent_activities,
        undefined_training_intent=undefined_training_intent,power_model=power_model,
        microcycle_ledger=microcycle_ledger,coaching_contract=coaching_contract,
        adaptive_roadmap=adaptive_roadmap,strength_pattern=strength_pattern,
        training_rhythm=training_rhythm,execution_model=execution_model)
    if not isinstance(out,dict):return out
    rows=list(out.get('sessions') or [])
    checked=[];blocked=[];normalized=[]
    for i,original in enumerate(rows):
        if not isinstance(original,dict):continue
        if str(original.get('stimulus_architecture_state') or '') != 'NOVA_CANONICAL_LIBRARY':continue
        # Check the actual HARD label too. A broken upstream flag must not
        # silently remove an otherwise hard canonical workout from this gate.
        hard=bool(original.get('hard_spacing_relevant') or original.get('provisional_quality')
            or str(original.get('intensity') or '').upper().startswith('HARD')
            or str(original.get('intensity_class') or '').lower() in {'vo2','threshold','tempo','sprint','high','hard'})
        if not hard:continue
        sid=str(original.get('nova_workout_id') or '')
        spec=_v19696_execution_contract(original)
        if sid == _R196933_VIRTUAL_ID and spec.get('ready'):
            if not _v196933_raw_attested(prescription,i,original,adaptive_roadmap):
                spec=dict(spec,ready=False,
                    issues=list(spec.get('issues') or [])+['ORIGINAL_COUPLED_DOSE_ATTESTATION_FAILED'])
        checked.append({'index':i,'workout_id':sid,'goal':spec.get('goal'),
                        'pattern':spec.get('pattern'),'result':'PASS' if spec['ready'] else 'BLOCKED',
                        'issues':list(spec['issues'])})
        if not spec['ready']:
            s=dict(original)
            # Never disguise a failed *technical* compilation as the coach's
            # decision to prescribe an ordinary easy ride. The unissued hard
            # contract must be visible and explicitly NOT executable.
            s.update({'title':'Quality not executable · incomplete contract',
                      'intensity':'BLOCKED','intensity_class':'recovery',
                      'hard_spacing_relevant':False,'provisional_quality':False,
                      'stimulus_architecture_state':'NOT_HARD','stimulus_role':None,
                      'stimulus_domain':None,'nova_workout_id':None,
                      'nova_contract_id':None,'quality_relevance':'NOT_HARD',
                      'prescribed_power_low':None,'prescribed_power_high':None,
                      'power_band':None,'quality_target_ratio':None,
                      'main_set':'No hard workout has been issued for this slot: the selected canonical mechanics are incomplete. Follow only the separately issued easy-day guidance.',
                      'why':'The selected hard dose needs a complete work/recovery/power contract before it can be executed. This is a technical compilation failure, not a coaching instruction to replace quality with easy work.',
                      'unissued_candidate_workout_id':sid,
                      'unissued_quality_reason':'R19696_INCOMPLETE_WORKOUT_CONTRACT'})
            rows[i]=s;blocked.append(i)
        elif (spec.get('pattern') in _R19696_NUMERIC_PATTERNS and
              (original.get('power_prescription') or {}).get('source')!='CANONICAL_SELF_PACED_QUALITY'):
            s=dict(original)
            raw=_v19696_strip_conflicting_power_headers(s.get('main_set'))
            band=s.get('power_prescription') or {}
            power=spec.get('power')
            prefix='EXPLORATORY WORK RANGE: ' if ('exploratory' in str(power).lower()) else 'WORK RANGE: '
            canonical=prefix+str(power)+'. '+raw
            if canonical!=s.get('main_set'):
                s['main_set']=canonical
                rows[i]=s;normalized.append(i)
    out['r19696_completeness_audit']={'schema':R19696_SCHEMA,'checked':checked,
        'blocked_indices':blocked,'normalized_indices':normalized,
        'rule':'Any selected canonical HARD needs an exact pattern, recovery and an attested numerical or effort-based dose; otherwise it is not issued.'}
    if blocked or normalized:
        out['sessions']=rows
        _v145_rehash_plan(out)
        if isinstance(out.get('single_plan_authority'),dict):
            out['single_plan_authority']['plan_id']=out['canonical_plan_id']
        out['hard_session_count']=sum(bool(x.get('hard_spacing_relevant')) and
                str(x.get('intensity_class') or '').lower() in {'tempo','threshold','vo2','sprint'}
                for x in rows if isinstance(x,dict))
        out['hard_prescription_available']=out['hard_session_count']>0
        if blocked and not out['hard_prescription_available']:
            out['execution_status']='BLOCKED_INCOMPLETE_WORKOUT'
            out['status']='BLOCKED'
            out['workout_id']=None;out['contract_id']=None
            if isinstance(out.get('single_plan_authority'),dict):
                out['single_plan_authority']['workout_id']=None
            out['fail_closed_reason']='R19696_INCOMPLETE_WORKOUT_CONTRACT'
    # Composition is a POST-COMPILE projection only. It never edits the
    # decision, template, gate, authorization, intensity or HARD availability.
    if callable(globals().get('_v19697_complete_session_envelopes')):
        out=_v19697_complete_session_envelopes(out)
    # R196.9.8: final low-intensity allocation decision.  It cannot touch the
    # already-selected HARD mechanics or upstream authorization.
    if callable(globals().get('_v19698_compose_decisions')):
        out=_v19698_compose_decisions(out,training_rhythm,execution_model)
    return out

# R196.9.8 — Evaluate the quality-vs-volume opportunity at the LAST projection
# boundary.  This module is deliberately NOT another Nova prescriber:
# upstream scientific dose, canonical identity, power, permission, HOLD and
# prescription continuity retain exclusive authority for HARD mechanics.
# A larger time window alone is NEVER evidence for extra intervals.
R19698_SCHEMA = 'V4.9.109-R196.9.8-DOSE-VOLUME-ALLOCATION-1'


def _v19698_candidate_dose_review(session, template, total_seconds, preparation_seconds, cooldown_seconds):
    """Catalog comparison for QA only; NOT a selectable/approved workout.

    Purpose and interval family must match. More work must actually fit the
    same available envelope, but this *cannot* verify athlete-specific dose.
    No candidate ID or dose suggestion appears in the athlete instruction.
    """
    goal = str(template.get('goal') or '').upper()
    pat = str(template.get('pattern') or '').upper()
    dim = str(session.get('nova_prescription_dimension') or '').upper()
    current = _v19697_workout_core_seconds(template)
    if not current or not dim:
        return {'state':'NO_COMPARABLE_AUTHORIZED_DOSE','candidates':[],
                'reason':'No purpose-compatible evidence authorizes additional HARD work.'}
    found=[]
    for alt in _R19696_LIBRARY_INDEX.values():
        if (str(alt.get('id') or '') == str(template.get('id') or '') or
            str(alt.get('goal') or '').upper() != goal or
            str(alt.get('pattern') or '').upper() != pat or
            dim not in set(alt.get('dimensions') or ()) or alt.get('adaptive_only')):
            continue
        mechanics = _v19697_workout_core_seconds(alt)
        if not mechanics or mechanics[1] <= current[1] or mechanics[0] <= current[0]:
            continue
        # Headroom estimate is mechanical only; existing canonical warm-up,
        # cooldown, watt evidence and recovery must be recalculated upstream.
        if mechanics[0] + preparation_seconds + cooldown_seconds > total_seconds:
            continue
        found.append((int(round(mechanics[1])), str(alt.get('id') or '')))
    found.sort()
    return {'state':'UNAUTHORIZED_CATALOG_ALTERNATIVES' if found else 'NO_FITTING_LIBRARY_ALTERNATIVE',
            'candidates':[x[1] for x in found],
            'reason':'Same-purpose/same-family mechanical alternatives are NOT an athlete-specific escalation clearance. Existing issued dose and continuity are authoritative.'}


def _v19698_extension_evidence(session, training_rhythm=None, execution_model=None):
    """Describe the source of time, WITHOUT turning history into a hard cap."""
    s=session if isinstance(session,dict) else {}
    declared=_parse_duration_minutes(s.get('duration'))
    hint=s.get('duration_hint_min')
    try:
        hint=float(hint)
        if not math.isfinite(hint) or hint<=0:hint=None
    except (TypeError,ValueError,OverflowError):hint=None
    planned=s.get('is_planned_training') is True
    cap=None
    if planned:
        try:
            cap=float(s.get('expected_duration_max'))
            if not math.isfinite(cap) or cap<=0:cap=None
        except (TypeError,ValueError,OverflowError):cap=None
    envelope=((execution_model or {}).get('session_envelope') or {}) if isinstance(execution_model,dict) else {}
    learned=bool(envelope.get('available') and envelope.get('state')=='ESTABLISHED' and
                 hint is not None and str(s.get('timing_confidence') or '').upper() in {'MODERATE','HIGH'})
    source = ('EXPLICIT_PLANNED_SLOT' if planned else
              'LEARNED_SINGLE_SESSION' if learned else
              'UPSTREAM_COMPILED_SESSION')
    conflict=(declared is not None and cap is not None and declared > cap + 15.0 + .01)
    return {'source':source, 'declared_minutes':declared, 'learned_session_hint_minutes':hint if learned else None,
            'explicit_slot_max_minutes':cap, 'declared_overrun_min':round(max(0.0,declared-cap),2) if declared is not None and cap is not None else None, 'tolerance_min':15.0, 'conflict':conflict,
            'note':'Learned duration is a probabilistic *single-session* reference, not a daily sum or an order to use every available minute.'}


def _v19698_allocate(session, training_rhythm=None, execution_model=None):
    """Final dose-vs-topper decision; never issues or mutates HARD work."""
    s=session if isinstance(session,dict) else {}
    wid=str(s.get('nova_workout_id') or '')
    t=_R19696_LIBRARY_INDEX.get(wid)
    result={'schema':R19698_SCHEMA,'workout_id':wid,'state':'NOT_APPLICABLE',
            'decision':'UNCHANGED', 'changed':False,'issues':[]}
    if not isinstance(t,dict) or str(t.get('pattern') or '').upper().startswith('ENDURANCE'):
        return s,result
    comp=s.get('session_composition') if isinstance(s.get('session_composition'),dict) else {}
    if comp.get('schema') != R19697_SCHEMA or comp.get('state') != 'COMPLETE':
        result.update(state='PREVIOUS_COMPOSER_NOT_COMPLETE',decision='NO_EXTENSION')
        return s,result
    if (str(s.get('restriction') or 'NONE').upper()!='NONE' or
            str(s.get('stimulus_architecture_state') or '')!='NOVA_CANONICAL_LIBRARY' or
            not s.get('hard_spacing_relevant') or
            not _v19696_execution_contract(s).get('ready')):
        result.update(state='UPSTREAM_SAFETY_UNCHANGED',decision='NO_EXTENSION')
        return s,result
    minutes=_v19698_extension_evidence(s,training_rhythm,execution_model)
    total=int(comp.get('total_seconds') or 0)
    fill=sum(int(x.get('duration_seconds') or 0) for x in comp.get('blocks') or []
             if x.get('type') in {'AEROBIC_FILL','AEROBIC_TOPPER'})
    dose=_v19698_candidate_dose_review(s,t,total,
        int(next((b.get('duration_seconds') or 0 for b in comp.get('blocks') or [] if b.get('type')=='PREPARATION'),0)),
        int(next((b.get('duration_seconds') or 0 for b in comp.get('blocks') or [] if b.get('type')=='COOLDOWN'),0)))
    result.update(time_evidence=minutes,dose_review=dose,
                  existing_quality_seconds=int(comp.get('primary_work_seconds') or 0),
                  remaining_seconds=fill)
    if (s.get('is_planned_training') is True and
            str(s.get('planned_demand') or '').upper() in {'RECOVERY','EASY','ENDURANCE','NORMAL'}):
        result.update(state='EXPLICIT_EASY_PLAN_CONFLICT',decision='NO_NEW_EXTENSION',
                      issues=['UPSTREAM_HARD_CONTRADICTS_EXPLICIT_EASY_PLAN'])
        warning=('The compiled HARD workout conflicts with the explicitly planned easy/endurance demand. '
                 'Do not execute this conflicting plan before Nova issues a consistent prescription.')
        revised=dict(s,r19697_timing_warning=warning)
        result['changed']=revised!=s
        return revised,result
    if minutes['conflict']:
        result.update(state='EXPLICIT_AVAILABILITY_CONFLICT',decision='NO_NEW_EXTENSION',
                      issues=['DECLARED_WORKOUT_LONGER_THAN_EXPLICIT_SLOT_MAX'])
        # Do not overwrite the working prescription or rewrite its identity;
        # display the actual inconsistency via the existing warning surface.
        warning=('Workout duration exceeds the explicit planned maximum. '
                 'Do not execute this timeline before the slot and prescription are reconciled.')
        out=dict(s,r19697_timing_warning=warning)
        result['changed']=out!=s
        return out,result
    if comp.get('r19698_topper') and (s.get('r19698_extension_decision') or {}).get('decision')=='AEROBIC_TOPPER':
        result.update(state='COMPLETE_ALREADY_APPLIED',decision='AEROBIC_TOPPER')
        return s,result
    if fill==0:
        result.update(state='COMPLETE_NO_REMAINDER',decision='KEEP_AUTHORIZED_QUALITY')
        return s,result
    # No athlete-independent 'X% of time must be HARD' rule.  Specific-purpose
    # work is immutable. Choose a low-intensity topper according to stimulus;
    # sprints/short-power must not accrue extra moderate intensity by default.
    sprint=str(t.get('pattern') or '').upper() in {'SPRINT','SPRINT_CLUSTER','SHORT_POWER'}
    topper={'state':'AUTHORIZED_WITHIN_PREVIOUS_ENVELOPE',
            'duration_seconds':fill,
            'role':'POST_QUALITY_AEROBIC_TOPPER',
            'intensity':'Z1' if sprint else 'Z1_TO_LOW_Z2',
            'rationale':('Protect neuromuscular execution and recovery: easy Z1 volume only.' if sprint else
                         'Develop low-intensity aerobic volume without an unapproved extra quality dose.'),
            'dose': 'UPSTREAM_LIBRARY_HARD_UNCHANGED'}
    text=str(s.get('main_set') or '')
    marker=' TOPPER DECISION ('
    clean=text.split(marker,1)[0].rstrip() if marker in text else text.rstrip()
    directive=(f' TOPPER DECISION ({_v19697_time_label(fill)}): '
               + ('easy Z1 only after the prescribed sprint/short-power intervals' if sprint else
                  'conversational Z1–low-Z2 after the prescribed intervals')
               + '; do not add further hard repetitions. '
               'The original quality dose was selected upstream and has not been increased.')
    new=dict(s)
    cc=dict(comp)
    blocks=[]
    for b in comp.get('blocks') or []:
        r=dict(b)
        if r.get('type')=='AEROBIC_FILL':
            r.update(type='AEROBIC_TOPPER',intensity=topper['intensity'],
                     purpose=topper['rationale'])
        blocks.append(r)
    cc['blocks']=blocks
    cc['schema']=R19697_SCHEMA  # same total-composition authority and schema
    cc['r19698_topper']=topper
    cc['policy']=str(comp.get('policy') or '')+' + PURPOSE_AWARE_AEROBIC_TOPPER_NO_EXTRA_HARD'
    new['main_set']=clean+directive
    new['session_composition']=cc
    new['aerobic_fill_state']='PURPOSE_AWARE_AEROBIC_TOPPER'
    new['r19698_extension_decision']={'schema':R19698_SCHEMA, 'decision':'AEROBIC_TOPPER',
                                      'topper':topper,'quality_dose':'UNCHANGED_UPSTREAM'}
    result.update(state='COMPLETE',decision='AEROBIC_TOPPER',topper=topper,changed=new!=s)
    return new,result

def _v19699_number(value, lower=0.0, upper=1e6):

        v = float(value)

    except (TypeError, ValueError, OverflowError):

def _v19699_structure_step(original, proposed):
    """A single monotonic, bounded progression lever; never change rest/pattern."""
    a, b = original or {}, proposed or {}
    if a.get('goal') != b.get('goal') or a.get('pattern') != b.get('pattern'):
        return False
    if str(a.get('work_domain') or 'GOAL_SPECIFIC').upper()!=str(b.get('work_domain') or 'GOAL_SPECIFIC').upper():
        return False
    pattern = str(a.get('pattern') or '').upper()
    if pattern.startswith('ENDURANCE'):
        return False
    # Matching interval, float and recovery is mandatory; unknown terms fail.
    for key in ('work_s', 'recovery_s', 'float_s', 'set_recovery_s'):
        if key in a or key in b:
            if _v19699_number(a.get(key)) != _v19699_number(b.get(key)):
                return False
    if pattern in {'MICRO_30_15', 'SPRINT_CLUSTER'}:
        old_sets, new_sets = a.get('sets'), b.get('sets')
        old_reps, new_reps = a.get('reps_per_set'), b.get('reps_per_set')
        if not all(isinstance(x, int) and x > 0 for x in (old_sets, new_sets, old_reps, new_reps)):
            return False
        return ((new_sets == old_sets + 1 and new_reps == old_reps) or
                (new_sets == old_sets and new_reps == old_reps + 1)) and (
                    new_sets * new_reps <= old_sets * old_reps * 1.25)
    if pattern not in {'STANDARD', 'SHORT_POWER', 'SPRINT', 'SUSTAINED', 'OVER_UNDER'}:
        return False
    n0, n1 = a.get('reps'), b.get('reps')
    return (isinstance(n0, int) and isinstance(n1, int) and n0 > 0
            and n1 == n0 + 1 and n1 <= n0 * 1.25)


def _v19699_comparable_events(original, evidence, as_of):
    """Two dated independent verified executions of the *exact* base protocol.

    A peak PB, a suggested workout, a neighboring interval length, or a strong
    but unstructured 5-minute event cannot certify a 30/15 or sprint cluster.
    """
    t = original or {}
    pattern = str(t.get('pattern') or '').upper()
    events = (((evidence or {}).get('repeatability') or {}).get('events') or [])
    try:
        day0 = date.fromisoformat(str(as_of)[:10])
    except (TypeError, ValueError):
        return []
    approved = {}
    for e in events:
        if not isinstance(e, dict) or e.get('strong') is not True:
            continue
        source = str(e.get('activity_truth_source') or e.get('source') or '').upper()
        if source not in {'DEVICE_LAPS', 'ICU_INTERVALS_CANONICAL',
                          'ORIGINAL_ACTIVITY_FIT_LAP', 'ORIGINAL_FIT_LAPS'}:
            continue
        try:
            d = date.fromisoformat(str(e.get('date') or '')[:10])
            age = (day0 - d).days
        except (ValueError, TypeError):
            continue
        if not 0 <= age <= 42:
            continue
        id_ = str(e.get('activity_id') or '')
        if not id_ or _v19699_number(e.get('interval_secs'), .1, 1800) is None:
            continue
        work = _v19699_number(t.get('work_s'), .1, 1800)
        if work is None or abs(float(e['interval_secs']) - work) > max(2.0, work * .02):
            continue
        if pattern in {'MICRO_30_15', 'SPRINT_CLUSTER', 'OVER_UNDER'}:
            if str(e.get('pattern') or '').upper() != pattern:
                continue
            keys = ('sets', 'reps_per_set', 'float_s', 'set_recovery_s') if pattern != 'OVER_UNDER' else ('recovery_secs',)
            expected = dict(t)
            if pattern == 'OVER_UNDER':
                expected['recovery_secs'] = t.get('recovery_s')
            if any(_v19699_number(e.get(k)) is None or
                   abs(float(e[k]) - float(expected.get(k) or 0)) > 2 for k in keys):
                continue
            if _v19699_number(e.get('reps'), 1, 500) is None:
                # Explicit documented micro-block total, not prose inferred.
                if _v19699_number(e.get('completed_reps'), 1, 500) is None:
                    continue
                executed = e['completed_reps']
            else:
                executed = e['reps']
            minimum = int(t.get('sets') or 0) * int(t.get('reps_per_set') or 0) if pattern != 'OVER_UNDER' else int(t.get('reps') or 0)
            if executed < minimum:
                continue
        else:
            if _v19699_number(e.get('reps'), 1, 100) is None or float(e['reps']) < (t.get('reps') or 0):
                continue
            prescribed_rest = _v19699_number(t.get('recovery_s'), 0, 3600)
            observed_rest = _v19699_number(e.get('recovery_secs'), 0, 3600)
            if prescribed_rest is None or observed_rest is None or abs(observed_rest - prescribed_rest) > max(3.0, prescribed_rest * .05):
                continue
        if pattern in {'SPRINT', 'SPRINT_CLUSTER', 'SHORT_POWER'} and str(
                e.get('execution_quality_state') or '').upper() != 'PASSED':
            continue
        if _v19699_number(e.get('cv_pct'), 0, 100) is None or float(e['cv_pct']) > 6:
            continue
        # One activity cannot become two independent exposures.
        approved[d.isoformat()] = id_
    return [{'date':k, 'activity_id':v} for k, v in sorted(approved.items())]


def _v19699_dose_decision(original, goal, dimension, slot, evidence, context):
    """Bounded evidence-based choice, never a replacement for Nova Decision."""
    t = original or {}

    audit = {'schema': R19699_SCHEMA, 'decision': 'KEEP_ORIGINAL',
             'initial_workout_id': t.get('id'), 'proposed_workout_id': t.get('id'),
             'eligible_alternatives': [], 'rejected_reasons': []}
    if not t or str(goal or '').upper() == 'ENDURANCE_BASE':
        audit['decision'] = 'UPSTREAM_VETO' if not t else 'KEEP_ORIGINAL'
        audit['rejected_reasons'].append('ENDURANCE_OR_NO_CANONICAL_HARD')
        return t, audit
    if (str(dec.get('action') or '').upper() != 'TRAIN' or
            dec.get('hard_session_allowed') is not True or
            str(dec.get('quality_window_state') or '').upper() in
                {'HOLD', 'CLOSED', 'RECOVERY HOLD', 'PROVISIONAL'} or
            str((slot or {}).get('quality_window_state') or 'OPEN').upper() in

                {'EASY', 'RECOVERY', 'ENDURANCE', 'NORMAL'})):
        audit['rejected_reasons'].append('UPSTREAM_HARD_NOT_AUTHORIZED')
        return t, audit
    if _v19699_locked(slot, context.get('ledger')):
        audit['rejected_reasons'].append('CONTINUITY_LOCK_OR_UNIDENTIFIED_SLOT')
        return t, audit
    minutes = _v19699_slot_limit(slot)
    if minutes is None:
        audit['rejected_reasons'].append('NO_TRUSTWORTHY_SINGLE_SLOT_ENVELOPE')
        return t, audit
    evidence_rows = _v19699_comparable_events(t, evidence, (slot or {}).get('date') or context.get('as_of'))
    audit['comparable_exposures'] = evidence_rows
    if len(evidence_rows) < 2:
        audit['rejected_reasons'].append('NO_TWO_INDEPENDENT_COMPARABLE_EXPOSURES')
        return t, audit
    current = _v19697_workout_core_seconds(t)
    if current is None:
        audit['rejected_reasons'].append('INVALID_ORIGINAL_MECHANICS')
        return t, audit
    # R196.9.7 uses 20/30min canonical preludes for these dimensions; otherwise
    # 12/15min preparation and 8min cooldown. Never trim the preload to fit HARD.
    pattern = str(t.get('pattern') or '').upper()

    candidates = []
    for alt in _NOVA_WORKOUT_LIBRARY:
        if (alt.get('goal') != goal or dimension not in set(alt.get('dimensions') or ())
                or alt.get('adaptive_only') or not _v19699_structure_step(t, alt)
                or alt.get('id') in _NOVA_QUALITY_CANDIDATE_EXCLUSIONS.get()):
            continue
        core = _v19697_workout_core_seconds(alt)
        if not core or core[1] <= current[1] or core[0] + (prep + 5) * 60 > (minutes + 15.0) * 60:
            continue
        if pattern in {'STANDARD', 'SUSTAINED', 'SHORT_POWER'}:
            band = _v1943_band(alt, evidence_ledger=evidence,
                              ftp_anchor=context.get('ftp_anchor'),
                              as_of_date=(slot or {}).get('date') or context.get('as_of'),

            if band.get('state') != 'PRESCRIBED' or band.get('source') != 'COMPARABLE_VERIFIED_REPEATED_WORK':
                continue

        candidates.append((core[1], core[0], str(alt.get('id') or ''), alt))
    candidates.sort(key=lambda row: (row[0], row[1], row[2]))
    audit['eligible_alternatives'] = [c[2] for c in candidates]
    if not candidates:
        audit['rejected_reasons'].append('NO_EVIDENCED_CANONICAL_STEP_FITTING_ENVELOPE')
        return t, audit
    candidate = dict(candidates[0][3])

                 selected_dose_reason='Two independent strong completed comparable exposures; same exact interval/rest architecture and single bounded progression lever; canonical duration fits within the athlete slot plus the disclosed 15-minute tolerance.')
    return candidate, audit

def _nova_quality_template_required_seconds(template, dimension):
    """Minimum duration from the same canonical interval core and preparation/cooldown as R196.9.7.

    For final compiled sessions R196.9.7 remains the authority, including any
    explicit preloads from the original session text.  This is a conservative
    preflight for catalogued mechanics, not an execution clearance.
    """
    core = _v19697_workout_core_seconds(template)
    if core is None:
        return None
    pat = str((template or {}).get('pattern') or '').upper()
    if pat.startswith('ENDURANCE'):
        return None
    dim = str(dimension or '').upper()
    prep = 30 if dim == 'DURABILITY' else (20 if dim == 'RACE_TRANSFER' else
           (15 if pat in {'SPRINT', 'SPRINT_CLUSTER', 'SHORT_POWER'} else 12))
    return round(core[0]) + prep * 60 + 8 * 60


def _nova_quality_slot_declared_seconds(slot):
    """The compiler receives slot duration; +15min feasibility cushions are NOT declared time."""
    s = slot if isinstance(slot, dict) else {}
    value = _rhythm_num(s.get('duration_hint_min'))
    if value is None:
        value = _parse_duration_minutes(s.get('duration_hint_label'))
    if value is None or not math.isfinite(float(value)) or float(value) <= 0:
        return None
    # Respect an explicit planned maximum even if a learned duration hint is longer.
    if s.get('is_planned_training') is True:
        cap = _rhythm_num(s.get('expected_duration_max'))
        if cap is not None and cap > 0:
            value = min(float(value),float(cap))
    return round(float(value) * 60)


def _v19699_select_workout_template(goal, dimension, evidence_ledger=None,
                                   training_definitions=None, slot=None):
    """Evaluate the original selector with canonical time before issuing its identity.

    Only when the first candidate is mechanically impossible in the declared
    slot do we ask the SAME upstream selector to re-rank the remaining catalog.
    All original scientific/repeat/FTP/consent gates still run for each try.
    Scoped context avoids cross-athlete concurrency or global library edits.
    """
    limit = _nova_quality_slot_declared_seconds(slot)
    if limit is None or str(goal or '').upper() == 'ENDURANCE_BASE':
        return _v19699_select_workout_template_once(goal, dimension,
            evidence_ledger=evidence_ledger, training_definitions=training_definitions, slot=slot)
    excluded = set(_NOVA_QUALITY_CANDIDATE_EXCLUSIONS.get())
    attempts = []
    eligible = sum(1 for t in _NOVA_WORKOUT_LIBRARY
        if t.get('goal') == str(goal or '').upper()
        and str(dimension or '').upper() in set(t.get('dimensions') or ())
        and not t.get('adaptive_only'))
    for _ in range(min(eligible + 1, 100)):
        token = _NOVA_QUALITY_CANDIDATE_EXCLUSIONS.set(frozenset(excluded))
        try:
            template, audit = _v19699_select_workout_template_once(goal, dimension,
                evidence_ledger=evidence_ledger, training_definitions=training_definitions, slot=slot)
        finally:
            _NOVA_QUALITY_CANDIDATE_EXCLUSIONS.reset(token)
        if not isinstance(template, dict):
            detail = dict(audit or {})
            if attempts:
                detail['canonical_timing_rejections'] = attempts
                detail['reason'] = 'NO_SCIENCE_APPROVED_TEMPLATE_FITS_DECLARED_TIME'
            return template, detail
        ident = str(template.get('id') or '')
        required = _nova_quality_template_required_seconds(template, dimension)
        if required is not None and required <= limit and ident not in excluded:
            detail = dict(audit or {})
            if attempts:
                detail['canonical_timing_rejections'] = attempts
                detail['selected_id'] = ident
                detail['reason'] = 'CANONICAL_TIMING_VALIDATED_AFTER_ORIGINAL_RANK_RETRY'
            return template, detail
        attempts.append({'workout_id': ident, 'required_seconds': required,
                         'declared_seconds': limit,
                         'reason': 'MINIMUM_CANONICAL_TIME_EXCEEDS_DECLARED' if required is not None and required > limit
                                   else 'UNASSESSED_OR_ALREADY_EXCLUDED'})
        if not ident or ident in excluded:
            break
        excluded.add(ident)
    return None, {'reason':'NO_SCIENCE_APPROVED_TEMPLATE_FITS_DECLARED_TIME',
        'candidate_count':eligible,'canonical_timing_rejections':attempts}

def _v19699_compile_nova_prescription(prescription, coach_clock,

        return out

            out.get('hard_prescription_available') is not True or

        row['main_set'] = main
        rows.append(row)
        modified = modified or row != old
    if modified:
        out['sessions'] = rows
        # Align R196.9.8 allocation audit with the upstream dose we actually emit.
        summary = dict(out.get('r19698_allocation_audit') or {})
        checks = []
        for record in summary.get('checked') or []:

def _v196910_raw_power_samples(payload):
    """Time-aligned, bounded watts from ICU raw streams. Missing != zero."""
    if not isinstance(payload, (list, dict)):
        return None
    if isinstance(payload, list):
        streams = {str(x.get('type') or '').lower(): x.get('data')
                   for x in payload if isinstance(x, dict)}
    else:
        streams = {str(k).lower(): v.get('data') if isinstance(v, dict) else v
                   for k, v in payload.items()}
    ts = streams.get('time')
    if ts is None: ts = streams.get('secs')
    watts = streams.get('watts')
    if ts is None or watts is None or not isinstance(ts, list) or not isinstance(watts, list):
        return None
    if len(ts) != len(watts) or len(ts) < 480 or len(ts) > 60000:
        return None
    readings = {}
    for t, w in zip(ts, watts):
        try:
            sec = float(t)
            val = float(w)
        except (TypeError, ValueError, OverflowError):
            continue
        if not (math.isfinite(sec) and math.isfinite(val) and 0 <= val <= 1600):
            continue
        if 0 <= sec <= 21600 and abs(sec - round(sec)) <= .55:
            readings[int(round(sec))] = val
    if not readings:
        return None
    duration = max(readings) + 1
    if duration > 21600 or len(readings) < 480:
        return None
    dense = [None] * duration
    for t, w in readings.items():
        dense[t] = w
    return dense


def _v196910_find_micro_30_15(samples):
    """Resolve the FULL measured 30/15 sets, without a hard-coded 10-rep window.

    Use the complete per-second time/watts streams delivered by Intervals.icu.
    Never use the workout name, Zwift laps or the requested prescription as an
    input to the detected repetition count. Each 45-s cycle needs an observed
    30-s work and 15-s float; at most isolated noisy/missing cycles are allowed.
    Two distinct maximal runs with the same measured length and a measured
    inter-set break are required. Unsupported/truncated evidence fails closed.
    """
    if not isinstance(samples, list) or not 900 <= len(samples) <= 21600:
        return None
    sums = [0.0]; counts = [0]
    for w in samples:
        ok = isinstance(w, (int, float)) and math.isfinite(w) and 0 <= w <= 1600
        counts.append(counts[-1] + int(ok))
        sums.append(sums[-1] + (float(w) if ok else 0.0))
    def avg(t, size):
        if t < 0 or t+size > len(samples):
            return None
        n = counts[t+size]-counts[t]
        if n < math.ceil(size*.85):
            return None
        return (sums[t+size]-sums[t])/n

    # Every possible second-phase is inspected. Splitting a continuous block
    # at 10 repetitions (as R196.9.10 did) is specifically prohibited.
    sets = []
    for phase in range(45):
        cycles = []
        for start in range(phase, len(samples)-45+1, 45):
            h, f = avg(start,30), avg(start+30,15)
            good = (h is not None and f is not None and h >= 85 and
                    h-f >= max(30.0, .16*h))
            cycles.append((start,h,f,good))
        # Recover ONE isolated low-contrast cycle between verified neighbours,
        # without extrapolating outside the measured first/last repetition.
        for i in range(1,len(cycles)-1):
            if (not cycles[i][3] and cycles[i-1][3] and cycles[i+1][3] and
                cycles[i][1] is not None and cycles[i][2] is not None):
                start,h,f,_ = cycles[i]
                cycles[i]=(start,h,f,'BRIDGED')
        i=0
        while i<len(cycles):
            if not cycles[i][3]:
                i+=1;continue
            j=i+1
            while j<len(cycles) and cycles[j][3]:
                j+=1
            part=cycles[i:j]
            n=len(part)
            if 8 <= n <= 24:
                hs=[r[1] for r in part];fs=[r[2] for r in part]
                passed=sum(r[3] is True for r in part)
                if (passed >= math.ceil(n*.85) and
                    statistics.median(hs)>=max(85,statistics.median(fs)*1.25) and
                    statistics.median(hs)-statistics.median(fs)>=35):
                    contrast=statistics.median(h-f for h,f in zip(hs,fs))
                    sets.append((part[0][0],part[-1][0]+45,n,passed,contrast,part))
            i=j
    # One complete group is not sufficient: two nonoverlapping sets need an
    # observed break, and equal repetitions, not an inferred title-based dose.
    if not sets:
        return None
    sets.sort(key=lambda r:(r[0],r[2]))
    best=None
    for a in sets:
        for b in sets:
            rest=b[0]-a[1]
            if not 150 <= rest <= 540 or a[2]!=b[2]:
                continue
            # Maximise VERIFIED repetitions first (not 10-window contrast).
            # Prefer a low-contrast break and stable measured phase afterwards.
            f=avg(a[1],min(90,rest))
            if f is None:
                continue
            hmed=statistics.median(x[1] for x in a[5]+b[5])
            if f > hmed*.82:
                continue
            key=(a[2]+b[2], a[3]+b[3], round(a[4]+b[4],1),
                 -abs(rest-300),-a[0])
            if best is None or key>best[0]:
                best=(key,a,b,rest)
    if best is None:
        return None
    _,first,second,between=best
    repetitions=first[2]
    seq=[];work=[]
    for set_no,grp in enumerate((first,second),1):
        for rep_idx,(start,h,f,passed) in enumerate(grp[5],1):
            wr={'type':'WORK','moving_time':30,'elapsed_time':30,
                'start_time':start,'end_time':start+30,'average_watts':round(h,2),
                'set':set_no,'rep_in_set':rep_idx,'_v196910_observed':True}
            fl={'type':'RECOVERY','moving_time':15,'elapsed_time':15,
                'start_time':start+30,'end_time':start+45,'average_watts':round(f,2),
                'set':set_no,'rep_in_set':rep_idx,'_v196910_observed':True}
            seq.extend((wr,fl));work.append(wr)
        if set_no==1:
            rest_w=avg(first[1],between)
            if rest_w is not None:
                seq.append({'type':'RECOVERY','moving_time':between,
                            'elapsed_time':between,'start_time':first[1],
                            'end_time':second[0],'average_watts':round(rest_w,2),
                            'between_sets':True,'_v196910_observed':True})
    seq.sort(key=lambda r:(r['start_time'],r['type']!='WORK'))
    for i,r in enumerate(seq):r['_truth_pos']=i
    return {'schema':R196910_SCHEMA,'status':'RESOLVED','source':'ICU_RAW_WATTS_30_15',
            'confidence':'HIGH','pattern':'MICRO_30_15','sets':2,
            'reps_per_set':repetitions,'work_s':30,'float_s':15,
            'set_recovery_s':between,'prescribed_set_recovery_s':None,
            'repeated_work_count':len(work),'interval_secs':30,
            'sequence':seq,'work_rows':work,'corrections':[],
            'recovery_median_secs':15,'raw_icu_work_count':0,
            'set_start_secs':[first[0],second[0]],
            'measured_work_watts':[r['average_watts'] for r in work],
            'coverage_rule':'EACH_30S_AND_15S_WINDOW_AT_LEAST_85_PERCENT_SAMPLES',
            'contrast_rule':'AT_LEAST_85_PERCENT_CONTRAST_IN_EACH_FULL_SET',
            'verified_set_pass_counts':[first[3],second[3]],

def _v196915_instruction_view(session, instruction=None):
    """Strip only known generator-owned implementation commentary, never data.

    Correct user instructions come from the structured composition blocks.
    Unrecognized prose is left alone rather than deleting safety information.
    """
    s=session if isinstance(session,dict) else {}
    original=str(s.get('main_set') if instruction is None else instruction or '')
    comp=s.get('session_composition') if isinstance(s.get('session_composition'),dict) else {}
    if comp.get('schema')!=R19697_SCHEMA or comp.get('state')!='COMPLETE':
        return original
    blocks=comp.get('blocks')
    if not isinstance(blocks,list) or not blocks:return original
    if 'SESSION TIMELINE (' not in original:return original
    # The renderer authored this exact marker; reconstruct from real timing
    # data, not guesses from the user or an imagined 'correct' duration.
    prefix=original.split('SESSION TIMELINE (',1)[0].rstrip()
    timeline=[]
    for b in blocks:
        if not isinstance(b,dict):continue
        name=str(b.get('type') or '')
        secs=b.get('duration_seconds')
        if not isinstance(secs,(int,float)) or secs<0: return original
        minutes=f'{int(secs // 60)}' if secs%60==0 else f'{secs/60:g}'
        if name=='PREPARATION': msg=f'{minutes} min warm-up, progressive and comfortable'
        elif name=='CANONICAL_INTERVALS':msg=f'{minutes} min for the prescribed work and recovery intervals'
        elif name in {'AEROBIC_FILL','AEROBIC_TOPPER'}:msg=f'{minutes} min easy aerobic riding; do not add extra hard repetitions'
        elif name=='COOLDOWN':msg=f'{minutes} min very easy cool-down'
        else:return original
        timeline.append(msg)
    if not timeline:return original
    if not isinstance(comp.get('total_seconds'),(int,float)) or sum(b.get('duration_seconds',0) for b in blocks if isinstance(b,dict))!=comp['total_seconds']:
        return original
    return (prefix+' Session plan: '+'; '.join(timeline)+'.').strip()

def _v196916_r27_mechanics(composition,workout_minutes,duration_display):
    """Verify existing R27 range-based benchmark, not its physiological dose."""
    c=composition if isinstance(composition,dict) else {}
    blocks=c.get('blocks')
    if (not isinstance(blocks,list) or not blocks or workout_minutes is None
        or c.get('total_duration')!=duration_display):return False
    def minutes(x):
        if isinstance(x,bool) or not isinstance(x,(int,float)):return None
        v=float(x)
        return v if math.isfinite(v) and v>=0 else None
    low=minutes(c.get('total_min_low'))
    high=minutes(c.get('total_min_high'))
    if low is None or high is None or low<=0 or low>high:return False
    if not low-0.05<=workout_minutes<=high+0.05:return False
    lows=highs=0.0;verified_primary=False;verified_warm=False;verified_cool=False;primary=0.0
    known={'WARMUP','PRE_TEST_EASY','PRIMARY_VALIDATION','IMMEDIATE_RECOVERY','AEROBIC_FILL','COOLDOWN'}
    for b in blocks:
        if not isinstance(b,dict):return False
        bl=minutes(b.get('duration_min') if 'duration_min' in b else b.get('duration_min_low'))
        bh=minutes(b.get('duration_min') if 'duration_min' in b else b.get('duration_min_high'))
        if bl is None or bh is None or bl>bh:return False
        lows+=bl;highs+=bh
        typ=str(b.get('type') or '').upper()
        if typ not in known:return False
        if typ=='PRIMARY_VALIDATION' and bl>0: verified_primary=True;primary+=bl
        elif typ=='WARMUP' and bl>0: verified_warm=True
        elif typ=='COOLDOWN' and bl>0: verified_cool=True
    recorded_primary=minutes(c.get('primary_duration_min'))
    return (verified_primary and verified_warm and verified_cool
            and recorded_primary is not None and abs(primary-recorded_primary)<=0.11
            and abs(lows-low)<=0.11 and abs(highs-high)<=0.11)


def _v196916_execution_reference_coverage(execution_model, already_fetched_activities):
    """Shadow audit of explicit activity references against fetched history.

    Never use summary-only activity data to infer completed intervals, wattage,
    stimulus family or posture. Do not alter execution_model or observation
    link_state; only report whether a later safe exact-ID enrichment is possible.
    Ref date/name mismatches remain unresolved for manual review.
    """
    observations=(execution_model or {}).get('execution_observations') or []
    activities=already_fetched_activities if isinstance(already_fetched_activities,list) else []
    by_id={}
    for a in activities:
        if not isinstance(a,dict):continue
        aid=str(a.get('id') or '').strip()
        if aid:by_id.setdefault(aid,[]).append(a)
    counts={'already_linked':0,'legacy_unlinked':0,'unresolved':0,
            'summary_id_unique_validated':0,'summary_id_missing':0,
            'summary_id_conflict':0,'summary_ref_mismatch':0}
    for obs in observations:
        if not isinstance(obs,dict):continue
        state=str(obs.get('link_state') or '')
        if state=='EXPLICIT_ACTIVITY_LINK':counts['already_linked']+=1;continue
        if state!='EXPLICIT_ACTIVITY_REFERENCE_UNRESOLVED':
            counts['legacy_unlinked']+=1;continue
        counts['unresolved']+=1
        ref=str(obs.get('activity_id') or '').strip()
        matches=by_id.get(ref,[]) if ref else []
        if not matches:counts['summary_id_missing']+=1;continue
        if len(matches)!=1:counts['summary_id_conflict']+=1;continue
        a=matches[0]
        expected_date=str(obs.get('activity_date') or '')[:10]
        native_date=str(a.get('start_date_local') or '')[:10]
        # Same exact ID and same local day if a day is declared. A missing
        # source date can never validate the claimed history link.
        if (not native_date or (expected_date and expected_date!=native_date)):
            counts['summary_ref_mismatch']+=1;continue
        counts['summary_id_unique_validated']+=1
    return {'schema':'R196916-EXACT-REFERENCE-COVERAGE-1','state':'OBSERVED',
            'counts':counts,'source':'ALREADY_FETCHED_CYCLING_HISTORY_SUMMARIES',
            'shadow_only':True,'extra_provider_calls':0,
            'domain_profile_modified':False,'verified_workout_detail':False,
            'policy':'Summary reference proof never becomes workout power or stimulus-domain evidence.'}
