def _human_bytes(value):

        text = re.sub(r"(\(\s*id\s*,\s*)user_id\s*,\s*", r"\1", text, count=1, flags=re.I)
        text = re.sub(r"(values\s*\(\s*\?\s*,\s*)\?\s*,\s*", r"\1", text, count=1, flags=re.I)
        if len(values) >= 2:
            values.pop(1)
    text = re.sub(r"\bsingleton_id\s*,\s*user_id\s*,", "singleton_id,", text, flags=re.I)
    text = re.sub(r"(\bSELECT\s+singleton_id\s*,\s*)user_id\s*,\s*", r"\1", text, count=1, flags=re.I)
    while True:
        m = re.search(r"(?P<prefix>\bWHERE\s+|\bAND\s+)user_id\s*=\s*\?", text, flags=re.I)
        if not m:
            break
        param_index = text[:m.start()].count("?")
        prefix = m.group("prefix").strip().upper()
        replacement = "WHERE 1=1" if prefix == "WHERE" else ""
        text = text[:m.start()] + replacement + text[m.end():]
        if 0 <= param_index < len(values):
            values.pop(param_index)
    text = re.sub(r"\s+", " ", text).strip()
    return text, tuple(values)

    init_memory_db()
    conn = _db_connect()
    try:
        cur = conn.cursor()
        cur.execute(_db_sql(sql), params)
        rows = []
        if fetch:
            raw = cur.fetchall()
            if _db_is_postgres():
                cols = [d[0] for d in cur.description]
                rows = [dict(zip(cols, row)) for row in raw]
            else:
                rows = [dict(row) for row in raw]
        conn.commit()

        conn.close()

def _v4840_easter_sunday(year):
    a = year % 19
    b, c = divmod(year, 100)
    d, e = divmod(b, 4)
    f = (b + 8) // 25
    g = (b - f + 1) // 3
    h = (19*a + b - d - g + 15) % 30
    i, k = divmod(c, 4)
    l = (32 + 2*e + 2*i - h - k) % 7
    m = (a + 11*h + 22*l) // 451
    month = (h + l - 7*m + 114) // 31
    day = ((h + l - 7*m + 114) % 31) + 1
    return date(year, month, day)
def _v4840_owner_long_day(day_value):

    if (day_value.month, day_value.day) in fixed:

    return day_value == (_v4840_easter_sunday(day_value.year) + timedelta(days=1))

def _v4844_dedupe_sentences(sentences):
    out = []
    seen_sets = []
    for sentence in sentences:
        tokens = {t for t in re.findall(r"[a-z0-9]+", sentence.lower()) if len(t) > 2}
        duplicate = False
        for prior in seen_sets:
            union = tokens | prior
            if union and len(tokens & prior) / len(union) >= 0.68:
                duplicate = True
                break
        if duplicate:
            continue
        out.append(sentence)
        seen_sets.append(tokens)
    return out
