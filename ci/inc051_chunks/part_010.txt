def build_aerobic_metabolic_range(ftp_anchor, aerobic_efficiency):
    """Estimated fat-oxidation-oriented endurance range; never a measured FatMax/crossover."""
    fa=ftp_anchor or {}; ae=aerobic_efficiency or {}
    ftp=_rhythm_num(fa.get("prescription_watts") if fa.get("prescription_watts") is not None else fa.get("anchor_watts"))
    if ftp is not None and not 80 <= ftp <= 600: ftp=None
    sessions=[x for x in (ae.get("latest_sessions") or []) if isinstance(x,dict) and _rhythm_num(x.get("watts")) is not None and (_rhythm_num(x.get("duration_min")) or 0)>=60]
    watts=[float(x.get("watts")) for x in sessions if 50 <= float(x.get("watts")) <= 500]
    lo=hi=None; source=""; status="LEARNING"; css="grey"
    if ftp is not None:
        prior_lo,prior_hi=.58*ftp,.70*ftp
        plausible=[w for w in watts if .50*ftp <= w <= .76*ftp]
        if plausible:
            med=statistics.median(plausible); obs_lo,obs_hi=.94*med,1.06*med
            lo=max(prior_lo,obs_lo); hi=min(prior_hi,obs_hi)
            if hi-lo < 10:
                mid=max(prior_lo,min(prior_hi,med)); lo=max(prior_lo,mid-7.5); hi=min(prior_hi,mid+7.5)
            source="PERSONAL STEADY Z2 + FTP CONTEXT"
            if len(plausible)>=3 and str(ae.get("confidence") or "").upper()=="HIGH": status,css="ESTABLISHED","cyan"
            elif len(plausible)>=2: status,css="PERSONALIZING","purple"
        else:
            lo,hi=prior_lo,prior_hi; source="FTP-ANCHORED ESTIMATE"
    elif watts:
        med=statistics.median(watts); lo,hi=.92*med,1.05*med; source="PERSONAL STEADY Z2 HISTORY"
        if len(watts)>=3: status,css="PERSONALIZING","purple"
    if lo is None or hi is None or hi<=lo:
        return {"available":False,"status":"LEARNING","status_class":"grey","method_note":"Insufficient power context for a responsible aerobic-metabolic estimate."}
    def r5(x): return int(5*round(float(x)/5.0))
    lo,hi=r5(lo),r5(hi)
    if hi<=lo: hi=lo+5
    context=str(ae.get("cohort_label") or "Steady endurance history")
    return {
        "available":True,"low_watts":lo,"high_watts":hi,"display":f"{lo}–{hi} W","status":status,"status_class":css,
        "context":context,"source":source,"ftp_context_watts":int(round(ftp)) if ftp is not None else None,
        "sample_count":len(watts),
        "method_note":"Estimated fat-oxidation-oriented aerobic range from power context and comparable steady endurance history. Heart rate is contextual only. This is not a measured FatMax or crossover threshold.",
        "long_ride_note":"For long endurance work, spend most steady aerobic time inside this range; start toward the lower half and do not force the upper edge late if durability, heat or fatigue deteriorates.",
    }

def _wellness_value_near(wellness, field, target_day, max_gap_days=4):
    best = None
    for w in wellness or []:
        value = w.get(field)
        if not isinstance(value, (int, float)):
            continue
        try:
            d = date.fromisoformat(str(w.get("id") or "")[:10])
        except Exception:
            continue
        gap = abs((d - target_day).days)
        if gap <= max_gap_days and (best is None or gap < best[0]):
            best = (gap, float(value))
    return best[1] if best else None
def build_training_state(season_activities, season_wellness, planning_context, metrics, feeling_trend=None, now=None):
    now = now or get_rome_now()
    today = now.date()
    planning_context = planning_context or {}
    health = planning_context.get("health_override")
    next_race = planning_context.get("next_race") or {}
    def load_between(start_day, end_day):
        total = 0.0
        count = 0
        for a in season_activities or []:
            raw = str(a.get("start_date_local") or "")[:10]
            try:
                d = date.fromisoformat(raw)
            except Exception:
                continue
            if start_day <= d <= end_day:
                load = a.get("icu_training_load")
                if isinstance(load, (int, float)):
                    total += max(0.0, float(load))
                count += 1
        return total, count
    load7, sessions7 = load_between(today - timedelta(days=6), today)
    prior21, prior_sessions = load_between(today - timedelta(days=27), today - timedelta(days=7))
    prior_weekly = prior21 / 3.0 if prior21 > 0 else None
    load_ratio = load7 / prior_weekly if prior_weekly and prior_weekly > 0 else None
    if load_ratio is not None:
        load_ratio = round(float(load_ratio), 3)
    latest_w = sorted(season_wellness or [], key=lambda x: str(x.get("id") or ""))[-1] if season_wellness else {}

    atl_now = latest_w.get("atl") if isinstance(latest_w.get("atl"), (int, float)) else None
    ctl_28 = _wellness_value_near(season_wellness, "ctl", today - timedelta(days=28))
    ctl_delta_28 = round(ctl_now - ctl_28, 1) if ctl_now is not None and ctl_28 is not None else None
    ramp = latest_w.get("rampRate") if isinstance(latest_w.get("rampRate"), (int, float)) else None
    atl_ctl = atl_now / ctl_now if atl_now is not None and ctl_now and ctl_now > 0 else None
    if atl_ctl is not None:
        atl_ctl = round(float(atl_ctl), 3)
    pb = metrics.get("personal_baselines") or {}

    except Exception:

    if health:
        state, css, icon = "HEALTH HOLD", "red", "🛡️"

        why = f"{health.get('kind', 'Health restriction')} is active through {health.get('until', 'the current restriction window')}; falling ATL or rising TSB is not treated as tapering or peaking."
    elif race_days is not None and race_days <= 2:
        state, css, icon = "RACE PREP", "green", "🏁"
        headline = "Preserve freshness and race specificity; there is no fitness to gain now."
        why = f"The priority race is {race_days} day(s) away. THE LAB prioritises readiness, fueling and low-cost activation over new training load."
    elif race_days is not None and 3 <= race_days <= 8:
        state, css, icon = "TAPER", "purple", "🎚️"
        headline = "Reduce accumulated fatigue while preserving the qualities already built."
        why = f"The priority race is {race_days} days away. This window is classified as taper even if a recent long ride temporarily keeps ATL high; health context would override this label if present."
    elif race_days is not None and 9 <= race_days <= 14:
        state, css, icon = "RACE PREP", "cyan", "🎯"
        headline = "Training is becoming race-specific while total fatigue must stay controlled."
        why = f"The priority race is {race_days} days away, so the final useful stimulus is balanced against the recovery time still available."
    elif len(recovery_flags) >= 2 or (atl_ctl is not None and atl_ctl >= 1.18) or (load_ratio is not None and load_ratio >= 1.35):

        if len(recovery_flags) >= 2:
            why = "Current recovery markers and recent training load together support an absorption-focused state."
        elif load_spike and acute_loaded:
            why = "Recent systemic load is well above the athlete's own baseline and acute load is elevated relative to chronic load, even if some current recovery markers are stable."
        elif load_spike:
            why = "Recent systemic load is unusually high versus the athlete's own baseline. Current recovery markers may still look stable; this RECOVERY state is load-driven rather than a claim of autonomic or sleep dysfunction."
        else:
            why = "Acute load is elevated relative to chronic load. This RECOVERY state is driven by training-load context, not by an assumption that every recovery marker is impaired."
    elif (ctl_delta_28 is not None and ctl_delta_28 >= 1.5 and (load_ratio is None or load_ratio >= 0.95)) or (ramp is not None and ramp >= 0.5):
        state, css, icon = "BUILD", "cyan", "🏗️"

    if race_days is not None: signals.append(f"Race {race_days}d")

    if atl_ctl is not None and atl_ctl >= 1.18: recovery_drivers.append("ACUTE_CHRONIC_LOAD")
    if load_ratio is not None and load_ratio >= 1.35: recovery_drivers.append("RECENT_LOAD_SPIKE")
    return {
        "state": state, "state_class": css, "icon": icon, "confidence": confidence,
        "headline": headline, "why": why, "signals": signals[:5], "recovery_drivers": recovery_drivers,
        "load_7d": round(load7), "prior_weekly_load": round(prior_weekly) if prior_weekly is not None else None,
        "load_ratio": round(load_ratio, 2) if load_ratio is not None else None,
        "ctl_delta_28": ctl_delta_28, "ramp_rate": round(ramp, 1) if ramp is not None else None,

def _positive_intensity_term(value, patterns):
    v = str(value or "").lower().replace("vo₂", "vo2")
    if not v:
        return False
    negative_cues = re.compile(
        r"\b(?:no|not|without|avoid|avoiding|never|resist|resisting|below|under|beneath|"
        r"non|senza|evita|evitare|sotto)\b"
    )
    clause_break = re.compile(r"[;,.]|\s+[–—]\s+|\n")
    for pattern in patterns:
        for match in re.finditer(pattern, v):
            prefix = v[max(0, match.start() - 72):match.start()]
            clause_prefix = clause_break.split(prefix)[-1]
            if negative_cues.search(clause_prefix):
                continue
            return True
    return False
def intensity_class(value):
    v = str(value or "").lower().replace("vo₂", "vo2")
    positive_vo2 = _positive_intensity_term(v, (r"\bvo2(?:max)?\b", r"\banaer(?:obic|obico|obica)?\b"))
    positive_threshold = _positive_intensity_term(v, (r"\bthreshold\b", r"\bsoglia\b"))
    positive_tempo = _positive_intensity_term(v, (r"\btempo\b", r"\bsweet\s*spot\b", r"\bsweetspot\b"))
    if positive_vo2:
        return "vo2"
    if positive_threshold:
        return "threshold"
    if positive_tempo:
        return "tempo"
    if any(k in v for k in ("recover", "recovery", "easy", "facile", "recupero")):
        return "recovery"
    return "endurance"
def _v4827_prescribed_power_range(*texts):
    reference_terms = ("pb", "personal best", "best", "record", "ftp anchor", "rolling ftp", "observed max")
    phase_terms = ("warm-up", "warmup", "warm up", "cool-down", "cooldown", "cool down", "recovery", "recover", "easy spin")
    range_re = re.compile(r"(?<!\d)(\d{2,3})\s*[–—-]\s*(\d{2,3})\s*w\b", re.I)
    single_re = re.compile(r"(?:target|at|@|around|circa|~)\s*(\d{2,3})\s*w\b", re.I)
    def local_prefix(text, pos):
        prefix = text[max(0, pos - 180):pos]
        cut = max(prefix.rfind("."), prefix.rfind(";"), prefix.rfind("\n"), prefix.rfind(":"))
        if cut >= 0:
            prefix = prefix[cut + 1:]
        return prefix.lower()
    def is_reference(text, match):
        ctx = text[max(0, match.start()-42):min(len(text), match.end()+42)].lower()
        if any(term in ctx for term in reference_terms):
            before = text[max(0, match.start()-42):match.start()].lower()
            if any(term in before for term in reference_terms):
                return True
        return False
    def is_phase_number(text, match):
        before = local_prefix(text, match.start())
        structured_target = bool(re.search(r"(?:\d+\s*[×x]\s*)?\d+(?:\.\d+)?\s*(?:min|m|sec|s)?[^.;:]{0,28}(?:@|\bat\s*)$", before, flags=re.I))
        return any(term in before for term in phase_terms) and not structured_target
    for text in texts:
        original = _v124_strip_non_prescriptive_power_context(text)
        if not original:
            continue
        profile = _v4845_primary_work_profile(original)
        if profile.get("power_low") is not None and profile.get("power_high") is not None:
            return (float(profile["power_low"]), float(profile["power_high"]))
        primary = _v4841_primary_prescription_text(original)
        variants = [primary]
        if primary.strip() != original.strip():
            variants.append(original)
        for s in variants:
            candidates = []
            for m in range_re.finditer(s):
                if is_reference(s, m) or is_phase_number(s, m):
                    continue
                lo, hi = float(m.group(1)), float(m.group(2))
                if 50 <= lo <= 800 and 50 <= hi <= 800:
                    candidates.append((m.start(), min(lo, hi), max(lo, hi)))
            for m in single_re.finditer(s):
                if is_reference(s, m) or is_phase_number(s, m):
                    continue
                w = float(m.group(1))
                if 50 <= w <= 800:
                    candidates.append((m.start(), w, w))
            if candidates:
                candidates.sort(key=lambda x: x[0])
                _, lo, hi = candidates[0]
                return (lo, hi)
    return (None, None)

    R94 counts explicit preload/lead-in, repeated work, between-rep recovery and

    cool-downs win; otherwise the deterministic floor is 5 min very easy.
    """
    text=re.sub(r"\s+"," ",str(main_set or "")).strip()

    if not rep_match:return base
    try:
        reps=int(rep_match.group(1));amount=float(rep_match.group(2));unit=str(rep_match.group(3) or "").lower()
    except Exception:return base

def _v90_session_duration_feasibility_guard(main_set, duration_display, slot=None, coaching_contract=None):
    profile=_v90_prescription_duration_floor(main_set)
    declared=_parse_duration_minutes(duration_display)
    out={**profile,"declared_min":declared,"state":"UNASSESSED","adjusted":False,"ceiling_min":None,"core_ceiling_min":None,"cooldown_extension_min":0.0}
    if not profile.get("available") or declared is None:
        return str(duration_display or ""),out,[]
    required=float(profile.get("min_required") or 0.0)
    if declared+0.01>=required:
        out["state"]="PASS"
        return str(duration_display),out,[]
    ceiling=None
    slot=slot or {}

    planned_min=_nutrition_number(slot.get("expected_duration_min")) if slot.get("is_planned_training") else None
    if planned_max is not None or planned_min is not None:
        ceiling=planned_max if planned_max is not None else planned_min
    elif (coaching_contract or {}).get("active"):
        try:d=date.fromisoformat(str(slot.get("date") or "")[:10])
        except Exception:d=get_rome_now().date()
        if not _v4840_owner_long_day(d):
            core_ceiling=float((coaching_contract or {}).get("weekday_max_minutes") or 65)
            cooldown_extension=min(float((coaching_contract or {}).get("post_quality_cooldown_extension_minutes") or 0),float(profile.get("cooldown_min") or 0))
            ceiling=core_ceiling+max(0.0,cooldown_extension)
            out["core_ceiling_min"]=core_ceiling
            out["cooldown_extension_min"]=round(max(0.0,cooldown_extension),2)
    out["ceiling_min"]=ceiling
    target=float(math.ceil(required-1e-9))
    if ceiling is None or target<=float(ceiling)+15.0+0.01:
        out.update({"state":"ADJUSTED","adjusted":True,"adjusted_min":target,
                    "time_tolerance_min":15.0,"planned_overrun_min":round(max(0.0,target-float(ceiling)),2) if ceiling is not None else None})

def _v4886_target_attainment_truth_guard(text):
    s = str(text or "")
    avg = re.search(r"\baverage\s+(?:of\s+)?(\d+(?:\.\d+)?)\s*W\b", s, flags=re.I)
    target = re.search(r"\btarget\s+of\s+holding\s+(\d+(?:\.\d+)?)\s*W\b", s, flags=re.I)
    if avg and target:
        try:
            avg_w = float(avg.group(1)); target_w = float(target.group(1))
        except Exception:
            return s
        if avg_w + 0.5 < target_w:
            s = re.sub(r"\balready\s+meets\s+your\s+stated\s+target\s+of\s+holding\s+" + re.escape(target.group(1)) + r"\s*W", f"is approaching your stated target of holding {target.group(1)} W", s, flags=re.I)
    return s

def compile_nova_prescription(prescription, coach_clock, training_definitions=None, ftp_anchor=None, recent_activities=None, undefined_training_intent=None, power_model=None, microcycle_ledger=None, coaching_contract=None, adaptive_roadmap=None, strength_pattern=None, training_rhythm=None, execution_model=None):
    """Compile R84 raw Nova mechanics through the same deterministic session guards used by Live."""
    p=dict(prescription or {})
    raw=list(p.get("raw_sessions") or [])
    sessions=normalize_next_sessions(
        raw, coach_clock, training_definitions=training_definitions, ftp_anchor=ftp_anchor,
        recent_activities=recent_activities, undefined_training_intent=undefined_training_intent, power_model=power_model,
        microcycle_ledger=microcycle_ledger, coaching_contract=coaching_contract, adaptive_roadmap=adaptive_roadmap,
        strength_pattern=strength_pattern, training_rhythm=training_rhythm, execution_model=execution_model,
    )
    # R109: seal Nova authority BEFORE the final safety/timing pass. That pass may
    # block execution, but it is not another coaching engine and may not replace
    # the canonical workout with a competing prescription.
    selected_idx=p.get("selected_slot_index")
    for i,sess in enumerate(sessions):
        if isinstance(sess,dict):
            sess["nova_prescription_authority"]=True
            sess["nova_prescription_dimension"]=p.get("dimension")
            if i==selected_idx and p.get("workout_id"):
                sess["nova_workout_id"]=p.get("workout_id")
                sess["nova_workout_library_version"]=((p.get("workout_library") or {}).get("version") or NOVA_WORKOUT_LIBRARY_VERSION)
    sessions,reasons=_v4897_compile_sessions(sessions,adaptive_roadmap)
    hard_count=sum(1 for x in sessions if isinstance(x,dict) and str(x.get("intensity_class") or "").lower() in {"tempo","threshold","vo2"})
    requested_hard=bool(p.get("hard_prescription_available"))
    p["sessions"]=sessions
    p["compile_reasons"]=list(reasons or [])
    p["compiled"]=True
    p["requested_hard_prescription_available"]=requested_hard
    p["hard_session_count"]=hard_count
    p["hard_prescription_available"]=bool(hard_count>0)
    if requested_hard and hard_count<=0:
        p["execution_status"]="BLOCKED_BY_FINAL_SAFETY_GATE"
        p["status"]="BLOCKED"
        if not p.get("fail_closed_reason"):p["fail_closed_reason"]="FINAL_SAFETY_GATE_BLOCKED_CANONICAL_WORKOUT"
    elif hard_count>0:
        p["execution_status"]="READY"
    else:
        p["execution_status"]="EASY_ONLY"
    canonical_payload=[{k:x.get(k) for k in ("date","start_minute","title","duration","intensity_class","main_set","nova_workout_id")} for x in sessions if isinstance(x,dict)]
    p["canonical_plan_id"]="NOVA-"+hashlib.sha256(json.dumps(canonical_payload,ensure_ascii=False,sort_keys=True,default=str).encode("utf-8")).hexdigest()[:16]
    for sess in sessions:
        if isinstance(sess,dict):
            sess["nova_canonical_plan_id"]=p["canonical_plan_id"]
    p["single_plan_authority"]={
        "plan_id":p["canonical_plan_id"],
        "workout_id":p.get("workout_id"),
        "library_version":((p.get("workout_library") or {}).get("version") or NOVA_WORKOUT_LIBRARY_VERSION),
        "rule":"All athlete-facing workout surfaces must project this canonical plan. Downstream code may block execution for safety/timing, but may not substitute another workout."
    }
    return p

def _nutrition_number(value):
    try:
        value = float(value)
    except (TypeError, ValueError):
        return None
    return value if math.isfinite(value) else None
def _activity_local_dt(activity):
    raw = str(activity.get("start_date_local") or "")
    if not raw:
        return None
    try:
        return datetime.fromisoformat(raw.replace("Z", "+00:00"))
    except ValueError:
        try:
            return datetime.strptime(raw[:16], "%Y-%m-%dT%H:%M")
        except ValueError:
            return None

            continue
        ingested = _nutrition_number(raw)
        secs = _nutrition_number(a.get("moving_time") or a.get("elapsed_time"))
        if ingested is None or ingested <= 0 or secs is None or secs < 30 * 60:
            continue
        hours = secs / 3600.0
        if hours <= 0:
            continue
        gph = ingested / hours
        if gph <= 0 or gph > 180:
            continue
        dt = _activity_local_dt(a)
        rows.append({
            "date": dt.date().isoformat() if dt else None,
            "name": str(a.get("name") or a.get("type") or "Activity")[:120],
            "gph": round(gph, 1),
            "duration_min": round(secs / 60.0, 1),
            "ingested_g": round(ingested, 1),
            "source": "INTERVALS_RECORDED_INTAKE",
        })
    rows.sort(key=lambda x: (x.get("date") or "", x.get("gph") or 0))
    return rows
def build_fueling_intelligence(activities=None, daily_logs=None, fueling_profile=None):
    fp = {**_GENERIC_FUELING_STARTER, **dict(fueling_profile or {})}
    knowledge = str(fp.get("fueling_knowledge") or "UNKNOWN").upper()
    declared = _nutrition_number(fp.get("practised_gph")) if knowledge == "KNOWN" else None
    gi = str(fp.get("gi_tolerance") or "UNKNOWN").upper()
    if gi not in {"GOOD", "MIXED", "POOR", "UNKNOWN"}:
        gi = "UNKNOWN"
    feedback = []
    for row in daily_logs or []:
        g = _nutrition_number(row.get("fueling_gph"))
        tol = str(row.get("fueling_tolerance") or "").strip().upper()
        if g is None or not 0 <= g <= 180 or tol not in {"GOOD", "MIXED", "POOR"}:
            continue
        feedback.append({"date": row.get("local_date"), "gph": round(g, 1), "tolerance": tol})
    feedback.sort(key=lambda x: x.get("date") or "")
    good = [x["gph"] for x in feedback if x["tolerance"] == "GOOD"]
    mixed = [x["gph"] for x in feedback if x["tolerance"] == "MIXED"]
    poor = [x["gph"] for x in feedback if x["tolerance"] == "POOR"]
    exposures = _fueling_activity_exposures(activities)
    exposure_values = [x["gph"] for x in exposures]
    observed_repeat = None
    if exposure_values:
        top = sorted(exposure_values, reverse=True)[:3]
        observed_repeat = round(statistics.median(top), 1)
    feedback_established = None
    for candidate in sorted(good, reverse=True):
        support = [x for x in good if abs(x - candidate) <= 10]
        if len(support) >= 2:
            feedback_established = round(statistics.median(support), 1)
            break
    declared_established = declared if declared is not None and gi == "GOOD" else None
    candidates = [x for x in (declared_established, feedback_established) if x is not None]
    established = round(max(candidates), 1) if candidates else None
    good_support_count = sum(1 for x in good if established is not None and abs(x - established) <= 10)
    if established is not None and good_support_count >= 3:
        confidence = "HIGH"
    elif established is not None:
        confidence = "MODERATE"
    else:
        confidence = "LEARNING"
    if established is not None:
        if established <= 0:
            next_practice = 30.0
            next_step = "Your current baseline is no carbohydrate during exercise. If the event demands it, first practise a modest intake in training and log GI response before using it in a race."
            detail = "Your current reported/tolerated in-ride carbohydrate baseline is 0 g/h. No positive race-fueling dose is established yet."
        elif established < 140:
            next_practice = min(140.0, round((established + 10.0) / 5.0) * 5.0)
            next_step = f"If long-event demands justify it, test ~{int(next_practice)} g/h in training before changing race-day intake."
            detail = f"~{int(round(established))} g CHO/h is currently established from athlete-reported tolerance and/or repeated good fueling feedback."
        else:
            next_practice = None
            next_step = "This is already a very high practised intake. THE LAB will not increase it automatically."
            detail = f"~{int(round(established))} g CHO/h is currently established from athlete-reported tolerance and/or repeated good fueling feedback."
    elif declared is not None and gi in {"MIXED", "POOR"}:
        next_practice = None
        detail = f"You report using ~{int(round(declared))} g CHO/h, but GI response is {gi.lower()}; THE LAB does not treat that dose as race-ready."
        next_step = "Stabilize a familiar lower intake in training and log GI response before progressing."
    elif observed_repeat is not None:
        next_practice = None
        detail = f"Recorded activity data shows high intake exposure around ~{int(round(observed_repeat))} g CHO/h, but GI tolerance has not been confirmed."
        next_step = "Add fueling feedback after training or confirm your practiced intake in Settings before THE LAB promotes it to a race target."
    else:
        next_practice = 30.0
        detail = "Fueling tolerance is still learning; no personal race CHO/h target is assumed."
        next_step = "Start by tracking a modest, familiar intake in suitable training and report GI response; race targets should follow successful practice."
    return {
        "status": "ESTABLISHED" if established is not None else "LEARNING",
        "confidence": confidence,
        "fueling_knowledge": knowledge,
        "declared_gph": round(declared, 1) if declared is not None else None,
        "declared_gi_tolerance": gi,
        "established_gph": established,
        "race_ready_gph": (established if established is not None and established > 0 else None),
        "next_practice_gph": next_practice,
        "feedback_count": len(feedback),
        "good_feedback_count": len(good),
        "good_support_count": good_support_count,
        "mixed_feedback_count": len(mixed),
        "poor_feedback_count": len(poor),
        "recorded_exposure_count": len(exposures),
        "observed_recorded_gph": observed_repeat,
        "recent_exposures": exposures[-6:],
        "detail": detail,
        "next_step": next_step,
    }

def assess_intervals_cho(activity):
    used = _nutrition_number(activity.get("carbs_used"))
    raw_ingested = _nutrition_number(activity.get("carbs_ingested"))
    intake_recorded = bool(raw_ingested is not None and raw_ingested > 0)
    ingested = max(0.0, raw_ingested) if intake_recorded else None
    kcal = _nutrition_number(activity.get("calories"))
    secs = _nutrition_number(activity.get("moving_time") or activity.get("elapsed_time")) or 0.0
    mins = secs / 60.0
    result = {
        "used_g": round(used, 1) if used is not None else None,
        "ingested_g": round(ingested, 1) if ingested is not None else None,
        "intake_recorded": intake_recorded,
        "intake_state": "RECORDED_POSITIVE" if intake_recorded else "UNTRACKED_OR_ZERO",
        "valid": False, "reason": "No Intervals CHO-use estimate", "duration_min": round(mins, 1),
    }
    if used is None or used < 0:
        return result
    if mins < 30 or mins > 90:
        result["reason"] = "Low confidence: Intervals CHO model is least reliable outside roughly 30–90 min"
        return result
    if kcal is not None and kcal > 0 and used * 4.0 > kcal * 1.10:
        result["reason"] = "Low confidence: estimated CHO energy exceeds total session calories"
        return result
    result["valid"] = True
    result["reason"] = "Accepted as a model estimate after duration/energy sanity checks"
    return result
def _v4864_post_workout_protein_target(weight_context, kind=None, duration_min=None):
    weight_context = weight_context or {}
    kind = str(kind or "").upper()
    long_or_race = bool(kind == "RACE" or (duration_min is not None and duration_min >= ATHLETE_FUELING_PROFILE["long_outdoor_min"]))
    gkg = POST_WORKOUT_PROTEIN_GKG_LONG if long_or_race else POST_WORKOUT_PROTEIN_GKG
    basis = "long/race recovery" if long_or_race else "demanding-session recovery"
    weight_kg = _nutrition_number(weight_context.get("kg"))
    if weight_kg and weight_kg > 0 and not weight_context.get("needs_update"):
        protein_g = int(round((weight_kg * gkg) / 5.0) * 5)
        protein_g = max(POST_WORKOUT_PROTEIN_MIN_G, min(POST_WORKOUT_PROTEIN_MAX_G, protein_g))
        return f"~{protein_g} g", f"~{gkg:.2f} g/kg · {basis} · current weight"
    return f"{POST_WORKOUT_PROTEIN_MIN_G}–{POST_WORKOUT_PROTEIN_MAX_G} g", f"0.25–0.40 g/kg evidence range · {basis} · current weight unavailable"
def _post_workout_fat_target():
    return "Normal meal", "No recovery-specific fat gram target is inferred; keep fat within your normal meal and daily intake."
def _v4864_completed_recovery_kind(activity, family=None, duration_min=None):
    a = activity or {}
    blob = " ".join(str(a.get(k) or "") for k in ("name", "type", "sub_type")).lower()
    if str(a.get("sub_type") or "").upper() == "RACE" or any(k in blob for k in ("race", "granfondo", "gran fondo", "criterium", "time trial")):
        return "RACE"
    f = str(family or classify_training_family(a) or "").lower()
    if any(k in f for k in ("vo2", "sprint", "neuromuscular", "high intensity")):
        return "VO2"
    if any(k in f for k in ("threshold", "tte", "over-under")):
        return "THRESHOLD"
    if any(k in f for k in ("tempo", "sweet spot")):
        return "TEMPO"
    intensity_factor = _v40_normalized_intensity_factor(a)
    if intensity_factor is not None and intensity_factor >= 0.90:
        return "VO2" if intensity_factor >= 1.0 else "THRESHOLD"
    if duration_min is not None and duration_min >= 90:
        return "ENDURANCE"
    return "TEMPO"
def _v4864_immediate_recovery_target(recent_activities, metabolic_context, now=None):
    now = now or get_rome_now()
    local_now = now.astimezone(_runtime_timezone()).replace(tzinfo=None) if now.tzinfo else now
    candidates = []
    for activity in recent_activities or []:
        end = _v4829_activity_end(activity)
        if end is None:
            continue
        if end.tzinfo:
            end = end.astimezone(_runtime_timezone()).replace(tzinfo=None)
        age_hours = (local_now - end).total_seconds() / 3600.0
        if age_hours < -0.25 or age_hours > 4.0:
            continue
        secs = _nutrition_number(activity.get("moving_time"))
        if secs is None:
            secs = _nutrition_number(activity.get("elapsed_time"))
        duration_min = (float(secs) / 60.0) if secs is not None and secs > 0 else None
        load = _nutrition_number(activity.get("icu_training_load"))
        quality = _v4829_is_quality_activity(activity)
        substantial = bool(quality or (duration_min is not None and duration_min >= 90) or (load is not None and load >= 60))
        if not substantial:
            continue
        candidates.append((end, activity, duration_min, load, quality, age_hours))
    if not candidates:
        return {"active": False}
    end, activity, duration_min, load, quality, age_hours = max(candidates, key=lambda x: x[0])
    family = classify_training_family(activity)
    kind = _v4864_completed_recovery_kind(activity, family=family, duration_min=duration_min)
    recovery = _post_workout_strategy(kind, metabolic_context, next_gap_hours=None, duration_min=duration_min)
    protein, protein_basis = _v4864_post_workout_protein_target((metabolic_context or {}).get("weight") or {}, kind=kind, duration_min=duration_min)
    fat, fat_basis = _post_workout_fat_target()
    return {
        "active": True,
        "activity_name": str(activity.get("name") or "Completed training")[:120],
        "activity_id": activity.get("id"),
        "family": family,
        "kind": kind,
        "duration_min": round(duration_min, 1) if duration_min is not None else None,
        "training_load": round(load, 1) if load is not None else None,
        "quality": bool(quality),
        "hours_since_end": round(max(0.0, age_hours), 1),
        "post_cho": recovery.get("post_cho"),
        "post_protein": protein,
        "post_protein_basis": protein_basis,
        "post_fat": fat,
        "post_fat_basis": fat_basis,
        "recovery_need": recovery.get("recovery_need"),
    }
def build_metabolic_context(recent_activities, season_wellness, planning_context, now=None, season_activities=None):
    now = now or get_rome_now()
    planning_context = planning_context or {}
    local_now = now.replace(tzinfo=None) if now.tzinfo else now
    since36 = local_now - timedelta(hours=36)
    since48 = local_now - timedelta(hours=48)
    acts36, acts48 = [], []
    for a in recent_activities or []:
        dt = _activity_local_dt(a)
        if not dt:
            continue
        if dt.tzinfo:
            dt = dt.astimezone(_runtime_timezone()).replace(tzinfo=None)
        if dt >= since48:
            acts48.append(a)
        if dt >= since36:
            acts36.append(a)
    kcal36 = round(sum(_nutrition_number(a.get("calories")) or 0 for a in acts36))
    kcal48 = round(sum(_nutrition_number(a.get("calories")) or 0 for a in acts48))
    load48 = round(sum(_nutrition_number(a.get("icu_training_load")) or 0 for a in acts48), 1)
    hard48 = sum(1 for a in acts48 if _v4895_activity_quality_truth(a, classify_training_family(a)).get("quality_relevant"))
    cho_checks = [assess_intervals_cho(a) for a in acts48]
    valid_cho = [c for c in cho_checks if c["valid"]]
    invalid_cho = [c for c in cho_checks if c.get("used_g") is not None and not c["valid"]]
    positive_intake = [c for c in valid_cho if c.get("intake_recorded") and c.get("ingested_g") is not None]
    used48 = round(sum(c["used_g"] or 0 for c in valid_cho), 1) if valid_cho else None
    ingested48 = round(sum(c["ingested_g"] or 0 for c in positive_intake), 1) if positive_intake else None
    intake_tracking_state = "PARTIAL_RECORDED" if positive_intake else "UNTRACKED"
    intake_tracking_detail = (
        f"At least {ingested48:g} g CHO was positively recorded during rides in the last 48 h; logging may still be partial."
        if ingested48 is not None else
        "During-ride carbohydrate intake is not reliably tracked in the last 48 h; missing/zero fields are treated as unknown."
    )
    exercise_gap = None
    cycling_acts48 = [a for a in acts48 if _v4887_is_cycling_activity(a)]
    context_acts48 = [a for a in acts48 if not _v4887_is_cycling_activity(a)]
    metabolic48 = [_v4894_session_metabolic_profile(a) for a in cycling_acts48]
    usable_metabolic48 = [p for p in metabolic48 if p.get("available")]
    glycogen_units_48h = sum(float(p.get("glycogen_units") or 0) for p in usable_metabolic48)
    context_load48 = sum(_nutrition_number(a.get("icu_training_load")) or 0 for a in context_acts48)
    context_minutes48 = sum((_nutrition_number(a.get("moving_time") or a.get("elapsed_time")) or 0) / 60.0 for a in context_acts48)
    if usable_metabolic48:
        if glycogen_units_48h >= 2.20 or (hard48 >= 2 and glycogen_units_48h >= 1.60):
            glycogen_pressure = "HIGH"
        elif glycogen_units_48h >= 0.90 or hard48 >= 1:
            glycogen_pressure = "MODERATE"
        else:
            glycogen_pressure = "LOW"
        if (context_load48 >= 120 or context_minutes48 >= 180) and glycogen_pressure != "HIGH":
            glycogen_pressure = "HIGH" if glycogen_pressure == "MODERATE" else "MODERATE"
            glycogen_pressure_source = "METABOLIC_DEMAND_MODEL + SYSTEMIC_CONTEXT_LOAD"
        elif (context_load48 >= 60 or context_minutes48 >= 90) and glycogen_pressure == "LOW":
            glycogen_pressure = "MODERATE"
            glycogen_pressure_source = "METABOLIC_DEMAND_MODEL + SYSTEMIC_CONTEXT_LOAD"
        else:
            glycogen_pressure_source = "METABOLIC_DEMAND_MODEL"
    else:
        fallback_score = (2 if load48 >= 140 else (1 if load48 >= 70 else 0)) + (2 if hard48 >= 2 else (1 if hard48 == 1 else 0))
        glycogen_pressure = "HIGH" if fallback_score >= 3 else ("MODERATE" if fallback_score >= 1 else "LOW")
        glycogen_pressure_source = "LOAD_FAMILY_FALLBACK"
    fueling_intelligence = build_fueling_intelligence(
        activities=(season_activities if season_activities is not None else recent_activities),
        daily_logs=load_daily_logs(240),
        fueling_profile=ATHLETE_FUELING_PROFILE.copy(),
    )
    weight = build_weight_context(season_wellness, now)
    race = planning_context.get("next_race") or None
    health = planning_context.get("health_override")
    race_low = _nutrition_number((race or {}).get("expected_duration_min"))
    race_high = _nutrition_number((race or {}).get("expected_duration_max"))
    if race_low is None and race_high is not None: race_low = race_high
    if race_high is None and race_low is not None: race_high = race_low
    if race_low is not None and race_high is not None and race_low > race_high:
        race_low, race_high = race_high, race_low
    race_env = str((race or {}).get("training_environment") or "UNKNOWN").upper()
    if race_env not in {"INDOOR", "OUTDOOR"}: race_env = "UNKNOWN"
    duration_display = _race_duration_display(race_low, race_high)
    crosses_90 = bool(race_low is not None and race_high is not None and race_low < 90 < race_high)
    loading_eligible = bool(race_low is not None and race_low >= 90)
    short_event = bool(race_high is not None and race_high <= 60)
    sub90_event = bool(race_high is not None and race_high <= 90)
    established = fueling_intelligence.get("race_ready_gph")
    if not race:
        race_state = "OFF"
        race_detail = "No upcoming race is registered in the active planning horizon."
    elif health or race.get("conflict"):
        race_state = "PAUSED"
        race_detail = "Health/restriction context outranks race-prep nutrition until the conflict is resolved."
    else:
        days = int(race.get("days_away") or 0)
        if short_event:
            race_state = "RACE DAY" if days <= 0 else "SHORT EVENT"
            race_detail = f"{race_env.title()} race · {duration_display}. Prioritize carbohydrate availability before the start; a per-hour race target is not automatically useful for a short event."
        elif sub90_event:
            race_state = "RACE DAY" if days <= 0 else "STANDARD PREP"
            race_detail = f"{race_env.title()} race · {duration_display}. Protect pre-race carbohydrate availability; multi-day carbohydrate loading is not automatically indicated from this duration."
        elif race_low is None or race_high is None:
            race_state = "RACE DAY" if days <= 0 else "DURATION NEEDED"
            race_detail = f"Race environment: {race_env}. Expected duration is still unknown, so THE LAB will not invent a carb-loading or race-fueling dose."
        elif crosses_90:
            race_state = "RACE DAY" if days <= 0 else "DURATION RANGE"
            race_detail = f"{race_env.title()} race · {duration_display}. The estimate crosses the ~90-minute boundary, so preparation stays conservative until the expected range is clearer."
        elif days <= 0:
            race_state = "RACE DAY"
            race_detail = f"{race_env.title()} race · {duration_display}. Use the athlete's established, practised fueling capability; do not test a higher intake for the first time today."
        elif days <= ATHLETE_FUELING_PROFILE["carb_loading_active_days"] and loading_eligible:
            race_state = "ACTIVE"
            race_detail = f"{race_env.title()} race · {duration_display}. Carbohydrate-loading relevance is high because the expected event duration is long enough to make glycogen availability important."
        elif days <= ATHLETE_FUELING_PROFILE["carb_loading_approaching_days"] and loading_eligible:
            race_state = "APPROACHING"
            race_detail = f"{race_env.title()} race · {duration_display}. Protect glycogen availability as the event approaches; any race intake remains limited by practiced tolerance."
        else:
            race_state = "NOT YET"
            race_detail = f"Race is {days} day(s) away · {race_env} · {duration_display}. Deliberate carbohydrate loading is not active yet."
    if short_event:
        race_fueling = "Pre-race availability first"
    elif established is not None:
        race_fueling = f"~{int(round(established))} g CHO/h · practised"
    else:
        observed = fueling_intelligence.get("observed_recorded_gph")
        race_fueling = (f"Learning · ~{int(round(observed))} g/h recorded, tolerance unconfirmed" if observed is not None else "Learning · no race target established")
    gkg_low = ATHLETE_FUELING_PROFILE["carb_loading_gkg_low"]
    gkg_high = ATHLETE_FUELING_PROFILE["carb_loading_gkg_high"]
    total_range = None
    if race_state == "ACTIVE" and weight.get("kg") and not weight.get("needs_update"):
        lo = int(round(weight["kg"] * gkg_low / 10.0) * 10)
        hi = int(round(weight["kg"] * gkg_high / 10.0) * 10)
        total_range = f"~{lo}–{hi} g CHO/day"
    metabolic_load = build_training_metabolic_load((season_activities if season_activities is not None else recent_activities), now=now)
    immediate_recovery_context = {
        "glycogen_pressure": glycogen_pressure,
        "weight": weight,
        "race_strategy": {"state": race_state},
    }
    immediate_recovery = _v4864_immediate_recovery_target(recent_activities, immediate_recovery_context, now=now)
    if immediate_recovery.get("active"):
        recovery_id = str(immediate_recovery.get("activity_id") or "")
        recovery_activity = next((a for a in (recent_activities or []) if str((a or {}).get("id") or "") == recovery_id), None)
        if recovery_activity is not None:
            recovery_profile = _v4894_session_metabolic_profile(recovery_activity)
            if recovery_profile.get("available"):
                immediate_recovery["metabolic_demand"] = recovery_profile.get("demand_label")
                immediate_recovery["glycogen_pressure"] = recovery_profile.get("glycogen_label")
                immediate_recovery["recovery_pressure"] = recovery_profile.get("recovery_label")
    return {
        "kcal_36h": kcal36, "kcal_48h": kcal48, "load_48h": load48,
        "hard_sessions_48h": hard48, "glycogen_pressure": glycogen_pressure,
        "glycogen_pressure_source": glycogen_pressure_source,
        "glycogen_proxy_48h": round(glycogen_units_48h, 3) if usable_metabolic48 else None,
        "systemic_context_load_48h": round(context_load48, 1),
        "systemic_context_minutes_48h": round(context_minutes48, 1),
        "training_cho_demand": glycogen_pressure,
        "cho_used_48h": used48, "cho_ingested_48h": ingested48,
        "cho_intake_tracking_state": intake_tracking_state,
        "cho_intake_tracking_detail": intake_tracking_detail,
        "cho_positive_recorded_count": len(positive_intake),
        "exercise_cho_gap_48h": exercise_gap, "cho_valid_count": len(valid_cho),
        "cho_low_confidence_count": len(invalid_cho), "weight": weight,
        "immediate_recovery": immediate_recovery,
        "fueling_intelligence": fueling_intelligence,
        "metabolic_load": metabolic_load,
        "race_strategy": {
            "state": race_state, "detail": race_detail,
            "title": race.get("title") if race else None,
            "date": race.get("start_date") if race else None,
            "days_away": race.get("days_away") if race else None,
            "environment": race_env if race else None,
            "expected_duration_min": race_low, "expected_duration_max": race_high,
            "duration_display": duration_display if race else None,
            "race_fueling": race_fueling,
            "loading_gkg": f"{gkg_low:.0f}–{gkg_high:.0f} g/kg/day reference" if race_state in ("ACTIVE", "APPROACHING") else None,
            "loading_total": total_range,
            "weight_update_required": bool(race_state == "ACTIVE" and weight.get("needs_update")),
        },
    }
def metabolic_context_text(context):

        kind = "RACE"
    elif restriction == "NO_INTENSITY" or profile == "RECOVERY" or intensity_cls == "recovery":

        kind = "TEMPO"

        kind = "THRESHOLD"
    elif profile in {"MIXED", "VO2", "SPRINTS"}:
        kind = "VO2"

        kind = "HEAT" if any(k in short_text for k in ("heat", "thermal", "hot room", "caldo")) else "ENDURANCE"
    elif intensity_cls == "tempo":

        kind = "VO2"
    else:

        demand = "LOW"

        demand = "LOW–MODERATE"

        demand = "MODERATE–HIGH"
    elif profile in {"THRESHOLD", "MIXED", "VO2", "SPRINTS"}:

        demand = {

        }[kind]

    if structured_env in ("INDOOR", "OUTDOOR"):

    return {"kind":kind,"demand":demand,"environment":environment,"planned_demand":profile}
def _parse_duration_minutes(value):
    if isinstance(value, (int, float)):
        v = _nutrition_number(value)
        return round(v, 1) if v and v > 0 else None
    text = str(value or "").strip().lower().replace(",", ".")
    if not text:
        return None
    hr_range = re.search(r"(\d+(?:\.\d+)?)\s*[–—-]\s*(\d+(?:\.\d+)?)\s*(?:hours?|hrs?|h|ore|ora)\b", text)
    if hr_range:
        return round(float(hr_range.group(2)) * 60.0, 1)
    min_range = re.search(r"(\d+(?:\.\d+)?)\s*[–—-]\s*(\d+(?:\.\d+)?)\s*(?:minutes?|mins?|min|m)\b", text)
    if min_range:
        return round(float(min_range.group(2)), 1)
    clock = re.search(r"(?<!\d)(\d{1,2}):(\d{2})(?!\d)", text)
    if clock:
        hours, minutes = int(clock.group(1)), int(clock.group(2))
        if minutes < 60:
            return round(hours * 60.0 + minutes, 1)
    hour = re.search(r"(\d+(?:\.\d+)?)\s*(?:hours?|hrs?|h|ore|ora)", text)
    minute = re.search(r"(\d+(?:\.\d+)?)\s*(?:minutes?|mins?|min|m)\b", text)
    if hour:
        total = float(hour.group(1)) * 60.0
        tail = text[hour.end():]
        compact_min = re.match(r"\s*(\d{1,2})(?:\s*(?:minutes?|mins?|min|m))?\b", tail)
        if compact_min:
            total += float(compact_min.group(1))
        elif minute and minute.start() >= hour.end():
            total += float(minute.group(1))
        return round(total, 1) if total > 0 else None
    if minute:
        return round(float(minute.group(1)), 1)
    return None
def _session_duration_range(session_item):
    item = session_item or {}
    low = _nutrition_number(item.get("duration_min") if item.get("duration_min") is not None else item.get("expected_duration_min"))
    high = _nutrition_number(item.get("duration_max") if item.get("duration_max") is not None else item.get("expected_duration_max"))
    if low is None and high is None:
        parsed = _parse_duration_minutes(item.get("duration"))
        return (parsed, parsed) if parsed is not None else (None, None)
    if low is None: low = high
    if high is None: high = low
    return (round(min(low, high), 1), round(max(low, high), 1))
def _session_duration_minutes(session_item):
    low, high = _session_duration_range(session_item)
    return high if high is not None else low
def _format_duration_hours(duration_min):
    if not duration_min:
        return None
    hours = duration_min / 60.0
    if abs(hours - round(hours)) < 0.05:
        return f"{int(round(hours))} h"
    return f"{hours:.1f} h"
def _fuel_total_display(low_gph, high_gph, duration_min):
    if duration_min is None or low_gph is None or high_gph is None or high_gph <= 0:
        return None
    hours = duration_min / 60.0
    low_total = int(round((low_gph * hours) / 10.0) * 10)
    high_total = int(round((high_gph * hours) / 10.0) * 10)
    dur = _format_duration_hours(duration_min)
    if low_gph == high_gph or low_total == high_total:
        return f"~{high_total} g CHO across {dur}"
    return f"~{low_total}–{high_total} g CHO across {dur}"
def _fuel_total_display_range(gph, duration_low, duration_high):
    if gph is None or duration_low is None or duration_high is None or gph <= 0:
        return None
    lo = int(round((gph * duration_low / 60.0) / 5.0) * 5)
    hi = int(round((gph * duration_high / 60.0) / 5.0) * 5)
    if lo == hi:
        return f"~{hi} g CHO across the expected race duration"
    return f"~{lo}–{hi} g CHO across the expected race-duration range"
def _race_intra_fueling_strategy(session_item, metabolic_context):
    item = session_item or {}
    ctx = metabolic_context or {}
    fi = ctx.get("fueling_intelligence") or {}
    low, high = _session_duration_range(item)
    established = _nutrition_number(fi.get("race_ready_gph"))
    observed = _nutrition_number(fi.get("observed_recorded_gph"))
    env = str(item.get("environment") or "UNKNOWN").upper()
    if high is None:
        return {
            "display": "Duration needed", "label": "RACE FUEL · ESTIMATE MISSING",
            "low_gph": None, "high_gph": None, "total": None, "duration_driver": "RACE DURATION UNKNOWN",
            "guidance": "Add a realistic expected duration range in Calendar. THE LAB will not invent a race-day carbohydrate dose from the RACE label alone.",
        }
    if high <= 60:
        return {
            "display": "Pre-race carbs first", "label": "SHORT RACE · NO FORCED HOURLY TARGET",
            "low_gph": None, "high_gph": None, "total": None, "duration_driver": "SHORT RACE",
            "guidance": f"{env.title()} race expected {_race_duration_display(low, high)}. Prioritize carbohydrate availability before the start; during-race intake is optional and should stay familiar rather than chasing a per-hour number.",
        }
    if established is None:
        evidence = f" Recorded intake around ~{int(round(observed))} g/h exists, but GI tolerance is unconfirmed." if observed is not None else ""
        return {
            "display": "Race target still learning", "label": "NO ESTABLISHED RACE CHO/H YET",
            "low_gph": None, "high_gph": None, "total": None, "duration_driver": "FUELING TOLERANCE LEARNING",
            "guidance": "Do not discover a higher carbohydrate dose on race day. Build and log fueling tolerance in training first." + evidence,
        }
    g = int(round(established))
    return {
        "display": f"~{g} g CHO/h", "label": "PRACTISED RACE TARGET",
        "low_gph": g, "high_gph": g,
        "total": _fuel_total_display_range(g, low, high),
        "duration_driver": "RACE DURATION + PRACTISED TOLERANCE",
        "guidance": (
            f"Use the practiced ~{g} g/h as the race-day ceiling/target, scaled across {_race_duration_display(low, high)}. "
            "If you want to move higher, test that step in training first rather than changing the race-day plan."
        ),
    }
def _training_fueling_bounds(low, high, metabolic_context):
    fi = (metabolic_context or {}).get("fueling_intelligence") or {}
    established = _nutrition_number(fi.get("established_gph"))
    low = float(low or 0)
    high = float(high or low)
    if established is None:
        starter = _nutrition_number(fi.get("next_practice_gph")) or 30.0
        cap = max(0.0, min(float(starter), high))
        if cap <= 0:
            return 0, 0, "LEARNING"
        return min(low, cap), cap, "LEARNING"
    cap = max(0.0, min(float(established), high))
    return min(low, cap), cap, "ESTABLISHED"
def _intra_fueling_strategy(kind, environment, planned_demand, duration_min, glycogen_pressure="LOW", metabolic_context=None, session_item=None):
    metabolic_context = metabolic_context or {}
    if kind == "RACE":
        return _race_intra_fueling_strategy(session_item or {}, metabolic_context)
    profile = _training_profile_key(planned_demand) if planned_demand else "COACH"
    explicit_profile = planned_demand not in (None, "", "COACH")
    inferred_hard = kind in ("TEMPO", "THRESHOLD", "VO2")
    if not explicit_profile and inferred_hard:
        profile = "TEMPO" if kind == "TEMPO" else ("THRESHOLD" if kind == "THRESHOLD" else "VO2")
    if environment == "INDOOR":
        hard_profile = profile in {"THRESHOLD", "MIXED", "VO2", "SPRINTS"}
        if hard_profile and ((duration_min is not None and duration_min >= 60) or str(glycogen_pressure).upper() == "HIGH"):
            if profile == "SPRINTS":
                lo0, hi0 = (0, 30) if (duration_min or 0) <= 75 else (20, 40)
            else:
                lo0, hi0 = (20, 40)
            lo, hi, cap_state = _training_fueling_bounds(lo0, hi0, metabolic_context)
            dose = f"~{int(round(hi))} g CHO/h" if abs(hi-lo) < 0.5 else (f"Optional · 0–{int(round(hi))} g CHO/h" if lo <= 0 else f"{int(round(lo))}–{int(round(hi))} g CHO/h")
            return {"display":dose + (" if pre-session fueling is incomplete" if lo > 0 else ""),
                    "label":("INDOOR QUALITY · FUELING LEARNING" if cap_state == "LEARNING" else "INDOOR QUALITY · CONDITIONAL"),
                    "low_gph":lo,"high_gph":hi,"total":_fuel_total_display(lo,hi,duration_min),
                    "duration_driver":"RIDE PROFILE + DURATION + GLYCOGEN CONTEXT + TOLERANCE"}
        g = ATHLETE_FUELING_PROFILE["indoor_default_g_per_hour"]
        return {"display":f"{g} g/h usual","label":"INDOOR BASELINE","low_gph":g,"high_gph":g,"total":None,"duration_driver":"ATHLETE BASELINE"}
    if environment != "OUTDOOR":
        return {"display":"No fixed target","label":"MATCH DURATION / DEMAND","low_gph":None,"high_gph":None,"total":None,
                "duration_driver":"UNKNOWN DURATION/ENVIRONMENT",
                "guidance":"THE LAB needs a usable duration and ride profile before assigning an hourly carbohydrate target."}
    if duration_min is None:
        return {"display":"No fixed target","label":"OUTDOOR RIDE · DURATION NEEDED","low_gph":None,"high_gph":None,"total":None,
                "duration_driver":"RIDE PROFILE + DURATION MISSING",
                "guidance":"Add the expected total duration for a more specific fueling estimate. THE LAB will not infer carbohydrate needs from OUTDOOR alone."}
    d = float(duration_min)
    label = None
    if profile == "RECOVERY":
        if d <= 120: lo0, hi0, label = 0, 20, "RECOVERY / SOCIAL · OPTIONAL"
        elif d <= 180: lo0, hi0, label = 0, 30, "LONGER EASY RIDE · OPTIONAL / CONTEXTUAL"
        else: lo0, hi0, label = 20, 40, "LONG EASY RIDE · DURATION MATTERS"
    elif profile == "ENDURANCE":
        if d <= 90: lo0, hi0, label = 0, 20, "ENDURANCE / Z2 · LOW FUELING DEMAND"
        elif d <= 120: lo0, hi0, label = 0, 30, "ENDURANCE / Z2 · OPTIONAL / CONTEXTUAL"
        elif d <= 180: lo0, hi0, label = 20, 40, "ENDURANCE / Z2 · DURATION + DEMAND"
        else: lo0, hi0, label = 40, 60, "LONG ENDURANCE / Z2 · DURATION + DEMAND"
    elif profile == "TEMPO":
        if d <= 60: lo0, hi0, label = 0, 30, "TEMPO / Z3 · LOW-MODERATE"
        elif d <= 120: lo0, hi0, label = 20, 40, "TEMPO / Z3 · DEMAND + DURATION"
        elif d <= 180: lo0, hi0, label = 30, 50, "TEMPO / Z3 · DEMAND + DURATION"
        else: lo0, hi0, label = 40, 60, "LONG TEMPO · DEMAND + DURATION"
    elif profile == "SPRINTS":
        if d <= 75: lo0, hi0, label = 0, 30, "SPRINTS / ANAEROBIC · CONTEXTUAL"
        elif d <= 120: lo0, hi0, label = 20, 40, "SPRINTS / ANAEROBIC · DEMAND + DURATION"
        else: lo0, hi0, label = 30, 50, "SPRINTS / ANAEROBIC · LONG SESSION"
    elif profile in {"THRESHOLD", "VO2"}:
        if d < 75: lo0, hi0 = 20, 40
        elif d <= 150: lo0, hi0 = 30, 60
        else: lo0, hi0 = 50, 80
        label = ("THRESHOLD / TTE" if profile == "THRESHOLD" else "VO2 / HARD REPEATS") + " · DEMAND + DURATION"
    elif profile == "MIXED":
        if d <= 90: lo0, hi0, label = 20, 40, "EASY + Z4+ EFFORTS · MIXED DEMAND"
        elif d <= 150: lo0, hi0, label = 30, 50, "EASY + Z4+ EFFORTS · MIXED DEMAND"
        else: lo0, hi0, label = 40, 60, "EASY + Z4+ EFFORTS · LONG MIXED RIDE"
    else:
        if inferred_hard:
            if d < 75: lo0, hi0 = 20, 40
            elif d <= 150: lo0, hi0 = 30, 60
            else: lo0, hi0 = 50, 80
            label = "OUTDOOR QUALITY · PRESCRIPTION-DERIVED"
        elif d <= 120: lo0, hi0, label = 0, 20, "OUTDOOR · PROFILE UNSURE · CONSERVATIVE"
        elif d <= 180: lo0, hi0, label = 0, 30, "OUTDOOR · PROFILE UNSURE · CONSERVATIVE"
        else: lo0, hi0, label = 20, 40, "LONG OUTDOOR · PROFILE UNSURE"
    lo, hi, cap_state = _training_fueling_bounds(lo0, hi0, metabolic_context)

    else: display = f"{int(round(lo))}–{int(round(hi))} g CHO/h"

def _session_clock(session_item):
    ds = str((session_item or {}).get("date") or "")[:10]
    try:
        d = date.fromisoformat(ds)
    except ValueError:
        return None
    start_minute = _rhythm_num((session_item or {}).get("start_minute"))
    if start_minute is None:
        return None
    start_minute = max(0, min(1439, int(round(start_minute))))
    hour, minute = divmod(start_minute, 60)
    return datetime.combine(d, datetime.min.time()).replace(hour=hour, minute=minute)
def _base_post_workout_range(kind, metabolic_context, next_gap_hours=None):
    base = {
        "RECOVERY": (0, 20), "ENDURANCE": (0, 30), "HEAT": (30, 40),
        "TEMPO": (40, 50), "THRESHOLD": (50, 60), "VO2": (50, 60), "RACE": (60, 90),
    }[kind]
    boost = 0
    pressure = str((metabolic_context or {}).get("glycogen_pressure") or "LOW")
    if pressure == "HIGH": boost += 15
    elif pressure == "MODERATE": boost += 5
    if next_gap_hours is not None:
        if next_gap_hours <= 10: boost += 10
        elif next_gap_hours <= 20: boost += 5
    race_state = str(((metabolic_context or {}).get("race_strategy") or {}).get("state") or "OFF")
    if race_state == "ACTIVE": boost += 20
    elif race_state == "APPROACHING": boost += 10
    caps = {"RECOVERY": 60, "ENDURANCE": 60, "HEAT": 70, "TEMPO": 80, "THRESHOLD": 90, "VO2": 90, "RACE": 100}
    lo = min(caps[kind], base[0] + boost)
    hi = min(caps[kind], base[1] + boost)
    lo = int(round(lo / 10.0) * 10)
    hi = int(round(hi / 10.0) * 10)
    if hi <= 20:
        return "0–20 g CHO", "LIGHT"
    if lo == hi:
        return f"~{hi} g CHO", "MODERATE" if hi < 60 else "HIGH"
    return f"{lo}–{hi} g CHO", "MODERATE" if hi < 60 else "HIGH"
def _weight_scaled_target(weight_context, low_gkg, high_gkg=None, per_hour=False, suffix=""):
    high_gkg = low_gkg if high_gkg is None else high_gkg
    weight_context = weight_context or {}
    kg = _nutrition_number(weight_context.get("kg"))
    usable = bool(kg and kg > 0 and not weight_context.get("needs_update"))
    unit = "g CHO/h" if per_hour else "g CHO"
    if usable:
        lo = int(round((kg * low_gkg) / 5.0) * 5)
        hi = int(round((kg * high_gkg) / 5.0) * 5)
        amount = f"~{lo} {unit}" if lo == hi else f"~{lo}–{hi} {unit}"
        return f"{amount}{suffix}"
    amount = f"~{low_gkg:.1f} g/kg" if abs(low_gkg - high_gkg) < 0.01 else f"~{low_gkg:.1f}–{high_gkg:.1f} g/kg"
    if per_hour:
        amount += "/h"
    return f"{amount} CHO{suffix}"
def _post_workout_strategy(kind, metabolic_context, next_gap_hours=None, duration_min=None):
    metabolic_context = metabolic_context or {}
    pressure = str(metabolic_context.get("glycogen_pressure") or "LOW")
    race_state = str((metabolic_context.get("race_strategy") or {}).get("state") or "OFF")
    weight = metabolic_context.get("weight") or {}
    long_ride = bool(duration_min is not None and duration_min > ATHLETE_FUELING_PROFILE["long_outdoor_min"])
    medium_ride = bool(duration_min is not None and duration_min >= ATHLETE_FUELING_PROFILE["medium_outdoor_min"])
    rapid_recovery = bool(next_gap_hours is not None and next_gap_hours < 4)
    if rapid_recovery:
        post_cho = _weight_scaled_target(
            weight,
            ATHLETE_FUELING_PROFILE["rapid_recovery_gkg_low"],
            ATHLETE_FUELING_PROFILE["rapid_recovery_gkg_high"],
            per_hour=True,
            suffix=" · first 3–4 h",
        )
        recovery_need = "VERY HIGH"
        post_label = "RAPID GLYCOGEN RESTORATION"
    elif long_ride or (kind == "RACE" and duration_min is not None and duration_min >= 90):
        gkg = ATHLETE_FUELING_PROFILE["long_ride_initial_gkg"]
        post_cho = _weight_scaled_target(weight, gkg, suffix=" initial")
        recovery_need = "HIGH"
        post_label = "LONG-RIDE INITIAL RECOVERY" if kind != "RACE" else "POST-RACE INITIAL RECOVERY"
    else:
        recovery_kind = "VO2" if kind == "RACE" and duration_min is not None and duration_min < 60 else kind
        post_cho, recovery_need = _base_post_workout_range(recovery_kind, metabolic_context, next_gap_hours)
        post_label = None
    priority_score = 0
    if long_ride or (kind == "RACE" and duration_min is not None and duration_min >= 90): priority_score += 3
    elif kind == "RACE": priority_score += 2
    elif medium_ride: priority_score += 1
    if kind in ("TEMPO", "THRESHOLD", "VO2"): priority_score += 2
    elif kind == "HEAT": priority_score += 1
    if pressure == "HIGH": priority_score += 2
    elif pressure == "MODERATE": priority_score += 1
    if rapid_recovery: priority_score += 3
    elif next_gap_hours is not None and next_gap_hours <= 10: priority_score += 2
    elif next_gap_hours is not None and next_gap_hours <= 20: priority_score += 1
    if race_state == "ACTIVE": priority_score += 2
    elif race_state == "APPROACHING": priority_score += 1
    if long_ride or (kind == "RACE" and duration_min is not None and duration_min >= 90) or rapid_recovery:
        daily_priority = "HIGH"
    elif priority_score >= 4:
        daily_priority = "HIGH"
    elif priority_score >= 2:
        daily_priority = "MODERATE"
    else:
        daily_priority = "LOW"
    if rapid_recovery:
        replenishment_guidance = "Short turnaround: distribute carbohydrate repeatedly during the first 3–4 hours. The hourly target is for rapid glycogen restoration, not a one-off meal."
    elif long_ride:
        replenishment_guidance = "The initial target is only the first recovery feeding.     elif kind == "RACE":

    elif daily_priority == "HIGH":
        replenishment_guidance = "Recent load, race context and/or short turnaround make whole-day carbohydrate availability a priority; do not judge recovery from the first snack alone."
    elif daily_priority == "MODERATE":
        replenishment_guidance = "Use the first recovery feeding as part of the next normal meals; total carbohydrate availability across the day matters more than perfectly matching exercise calories."
    else:
        replenishment_guidance = "Normal meals can cover most recovery needs. The first post-ride target is contextual, not a requirement to replace every exercise calorie immediately."

        "replenishment_guidance": replenishment_guidance,

def _v4898_note_has_athlete_request(note):
    """Detect an actionable athlete question/request even when punctuation is omitted.
    Daily Notes are natural language, not forms. A request such as
    "I don't know what wattage I need to target" must not disappear merely
    because the athlete did not type a question mark. Keep this intentionally
    conservative: uncertainty/question language must be paired with a coaching
    decision cue, or the note must contain a direct interrogative/request form.
    """
    text = str(note or "").strip()
    if not text:
        return False
    if "?" in text:
        return True
    low = re.sub(r"\s+", " ", text.lower()).strip()
    direct = (
        r"\b(?:should|could|would|can)\s+i\b",
        r"\b(?:do|did|am|was|is|are)\s+i\b.{0,90}\b(?:need|do|ride|train|target|aim|hold|recover|rest|change|adjust)\b",
        r"\b(?:what|which|how|when|where|why)\b.{0,100}\b(?:should|need|target|aim|hold|ride|train|recover|rest|do|change|adjust)\b",
        r"\b(?:tell|help|advise)\s+me\b",
    )
    if any(re.search(pattern, low, flags=re.I) for pattern in direct):
        return True
    uncertainty = bool(re.search(
        r"\b(?:i\s+(?:do\s*not|don['’]?t|dont|cannot|can['’]?t|cant)\s+know|i['’]?m\s+not\s+sure|i\s+am\s+not\s+sure|unsure|wonder(?:ing)?|not\s+clear\s+to\s+me)\b",
        low, flags=re.I,
    ))
    coaching_cue = bool(re.search(
        r"\b(?:watt(?:age|s)?|power|target|aim|pace|intensity|hard|easy|z[1-7]|zone|tte|interval|rep(?:s|etition)?|session|workout|training|ride|duration|minutes?|hours?|recovery|recover|rest)\b",
        low, flags=re.I,
    ))
    return uncertainty and coaching_cue
def _v4898_request_profile(question):
    text = re.sub(r"\s+", " ", str(question or "").lower()).strip()
    if not text:
        return {"kind":"NONE", "terms":set()}
    if re.search(r"\b(?:watt(?:age|s)?|power)\b", text) and re.search(r"\b(?:target|aim|hold|shoot|should|need|what|which)\b", text):
        kind = "POWER_TARGET"
    elif re.search(r"\b(?:how\s+long|duration|minutes?|hours?)\b", text):
        kind = "DURATION"
    elif re.search(r"\b(?:intensity|hard|easy|z[1-7]|zone)\b", text):
        kind = "INTENSITY"
    elif re.search(r"\btte\b", text):
        kind = "TTE"
    else:
        kind = "GENERAL"
    stop = {
        "the","a","an","and","or","but","if","then","than","to","of","for","in","on","at","with","from",
        "i","im","i'm","me","my","you","your","it","is","are","am","was","were","be","been","being",
        "do","does","did","dont","don't","not","know","sure","need","what","which","how","when","where","why",
        "today","tomorrow","even","though","can","could","should","would","this","that","some","any","one","more",
    }
    terms = {x for x in re.findall(r"[a-z0-9]+", text) if len(x) >= 3 and x not in stop}
    return {"kind":kind, "terms":terms}
def _v4898_request_sentence_score(sentence, question):
    sentence_low = re.sub(r"\s+", " ", str(sentence or "").lower()).strip()
    profile = _v4898_request_profile(question)
    if not sentence_low or profile["kind"] == "NONE":
        return 0
    score = 0
    terms = profile.get("terms") or set()
    sentence_terms = set(re.findall(r"[a-z0-9]+", sentence_low))
    score += min(6, 2 * len(terms & sentence_terms))
    kind = profile["kind"]
    if kind == "POWER_TARGET":
        if re.search(r"\b\d{2,4}(?:\s*[–—-]\s*\d{2,4})?\s*w\b", sentence_low, flags=re.I): score += 10
        if re.search(r"\b(?:target|hold|aim)\b", sentence_low): score += 7
        if re.search(r"\b(?:hold|aim|recommend|should)\b", sentence_low): score += 4
        elif re.search(r"\b(?:watt|power)\b", sentence_low): score += 4
        if re.search(r"\b(?:rep|reps|repeatability|tte)\b", sentence_low): score += 3
    elif kind == "DURATION":
        if re.search(r"\b\d+(?:\.\d+)?\s*(?:min|mins|minutes?|h|hr|hrs|hours?)\b", sentence_low): score += 10
    elif kind == "INTENSITY":
        if re.search(r"\b(?:easy|hard|intensity|z[1-7]|zone|threshold|vo2|recovery|aerobic)\b", sentence_low): score += 8
    elif kind == "TTE":
        if "tte" in sentence_low: score += 8
        if re.search(r"\b(?:target|hold|rep|watt|power)\b", sentence_low): score += 4
    if re.search(r"\b(?:recommend|target|aim|hold|keep|stay|use|do|skip|avoid|no|yes)\b", sentence_low):
        score += 2
    return score
def _v4898_request_answer_audit(question, recommendation):
    profile = _v4898_request_profile(question)
    kind = profile["kind"]
    if kind == "NONE":
        return {"detected":False, "kind":"NONE", "satisfied":True, "best_score":0}
    sentences = _v4844_sentence_list(recommendation)
    scores = [_v4898_request_sentence_score(x, question) for x in sentences]
    best = max(scores) if scores else 0
    rec_low = str(recommendation or "").lower()
    if kind == "POWER_TARGET":
        satisfied = bool(re.search(r"\b\d{2,4}(?:\s*[–—-]\s*\d{2,4})?\s*w\b", rec_low, flags=re.I)) and best >= 10
    elif kind == "DURATION":
        satisfied = bool(re.search(r"\b\d+(?:\.\d+)?\s*(?:min|mins|minutes?|h|hr|hrs|hours?)\b", rec_low, flags=re.I)) and best >= 8
    elif kind in {"INTENSITY","TTE"}:
        satisfied = best >= 8
    else:
        satisfied = best >= 4
    return {"detected":True, "kind":kind, "satisfied":bool(satisfied), "best_score":int(best)}

def _v4897_duration_seconds(value, unit):
    words={"one":1,"five":5,"fifteen":15,"twenty":20,"sixty":60}
    try: n=float(value)
    except Exception:
        n=words.get(str(value or "").lower())
        if n is None: return None
    u=str(unit or "").lower()
    return int(round(n*(60 if u.startswith("m") else 1)))
def _v4897_active_checkpoint_duration_mismatch(sentence, truth):
    expected=truth.get("test_seconds")
    if not expected or not (truth.get("pending_validation") or truth.get("formal_current")):
        return False
    low=str(sentence or "").lower()
    markers=list(re.finditer(r"\b(?:checkpoint|validation|benchmark|(?:planned|next|current)\s+(?:field\s+|power\s+)?test)\b",low,re.I))
    if not markers or re.search(r"\b(?:alternative|optional|option among|test options)\b",low,re.I):
        return False
    durations=[]
    for m in re.finditer(r"(?<![a-z0-9])(\d{1,3}(?:\.\d+)?|one|five|fifteen|twenty|sixty)\s*(?:[-–—]\s*)?(s|sec(?:ond)?s?|m|min(?:ute)?s?)\b",low,re.I):
        sec=_v4897_duration_seconds(m.group(1),m.group(2))
        if sec: durations.append((m.start(),m.end(),sec))
    if not durations:
        return False
    for mk in markers:
        near=min(durations,key=lambda d:min(abs(d[1]-mk.start()),abs(d[0]-mk.end())))
        dist=min(abs(near[1]-mk.start()),abs(near[0]-mk.end()))
        if dist<=55 and near[2]!=expected:
            return True
    return False
