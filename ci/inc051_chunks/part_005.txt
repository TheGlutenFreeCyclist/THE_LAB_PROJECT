def load_latest_report(include_payload=True, owner_user_id=None, include_qa_replay=False):
    owner_id = _data_owner_user_id(owner_user_id)
    needs_payload = bool(include_payload or include_qa_replay)
    # Owner requests use the separate Personal Neon bridge; retain this
    # distinction in the request-local cache key for strict backend isolation.
    personal_backend = bool(_owner_personal_data_enabled(owner_id))
    cache = _r196999_report_row_cache()
    cache_key = (str(owner_id), personal_backend)
    cached_row = cache.get(cache_key) if cache is not None else None
    if cached_row is not None and (not needs_payload or "payload_json" in cached_row):
        row = cached_row
    else:
        if needs_payload:
            projection = "singleton_id, user_id, generated_at_utc, generated_at_local, app_version, payload_json"
        else:
            # Calculate UTF-8 bytes inside Postgres/SQLite instead of sending
            # the entire multi-megabyte JSON to Render for a metadata badge.
            # Empty/NULL payloads retain the old '{}' fallback semantics.
            raw_sql = "COALESCE(NULLIF(payload_json, ''), '{}')"
            length_expr = (f"OCTET_LENGTH({raw_sql})"
                           if (personal_backend or _db_is_postgres())
                           else f"LENGTH(CAST({raw_sql} AS BLOB))")
            projection = ("singleton_id, user_id, generated_at_utc, generated_at_local, app_version, "
                          f"{length_expr} AS payload_bytes")
        rows = _data_execute(owner_id,
            f"SELECT {projection} FROM latest_report WHERE user_id=? LIMIT 1",
            (owner_id,), fetch=True,
        )
        if not rows:
            return None
        row = rows[0]
        if cache is not None:
            cache[cache_key] = row  # Store the SQL row, never the mutable parsed JSON.
    meta = {
        "generated_at_utc": row.get("generated_at_utc"),
        "generated_at_local": row.get("generated_at_local"),
        "generated_at_local_display": _format_report_timestamp(row.get("generated_at_local")),
        "app_version": row.get("app_version") or "THE LAB",
    }
    if "payload_json" in row:
        raw = row.get("payload_json") or "{}"
        meta["payload_bytes"] = len(raw.encode("utf-8"))
    else:
        meta["payload_bytes"] = int(row.get("payload_bytes") or 0)
    meta["payload_human"] = _human_bytes(meta["payload_bytes"])
    if needs_payload:
        try:
            decoded = json.loads(raw)
        except (TypeError, json.JSONDecodeError):
            decoded = None
        if isinstance(decoded, dict):
            qa_replay = decoded.pop("__qa_replay_v4831", None)
            if include_payload:
                meta["data"] = decoded
            if include_qa_replay:
                meta["qa_replay"] = qa_replay
                meta["qa_replay_available"] = bool(qa_replay)
        else:
            if include_payload:
                meta["data"] = None
            if include_qa_replay:
                meta["qa_replay"] = None
                meta["qa_replay_available"] = False
    return meta

def compute_feeling_trend(logs=None):
    logs = logs if logs is not None else load_daily_logs(240)
    if not logs:
        return {"latest": None, "avg7": None, "avg28": None, "delta": None, "label": "NO TREND YET", "points": [], "pre_avg": None, "post_avg": None}
    today = get_rome_now().date()
    recent28 = []
    recent7 = []
    pre_vals, post_vals = [], []
    points = []
    for log in logs:
        try:
            d = date.fromisoformat(str(log.get("local_date")))
        except (ValueError, TypeError):
            continue
        age = (today - d).days
        value = log.get("feeling")
        if not isinstance(value, (int, float)):
            continue
        if 0 <= age < 28:
            recent28.append(float(value))
        if 0 <= age < 7:
            recent7.append(float(value))
        phase = str(log.get("phase") or "")
        if phase.startswith("PRE"):
            pre_vals.append(float(value))
        if phase.startswith("POST"):
            post_vals.append(float(value))
        points.append({"date": d.isoformat(), "day": d.strftime("%a"), "time": log.get("local_time"), "value": int(value), "phase": phase})
    avg7 = round(statistics.mean(recent7), 1) if recent7 else None
    avg28 = round(statistics.mean(recent28), 1) if recent28 else None
    delta = round(avg7 - avg28, 1) if avg7 is not None and avg28 is not None else None
    if delta is None:
        label = "BUILDING BASELINE"
    elif delta >= 0.6:
        label = "TRENDING UP"
    elif delta <= -0.6:
        label = "TRENDING DOWN"
    else:
        label = "STABLE"
    return {
        "latest": logs[-1], "avg7": avg7, "avg28": avg28, "delta": delta, "label": label,
        "points": points[-14:],
        "pre_avg": round(statistics.mean(pre_vals[-30:]), 1) if pre_vals else None,
        "post_avg": round(statistics.mean(post_vals[-30:]), 1) if post_vals else None,
    }

def build_training_definitions(notes=None):
    notes = notes if notes is not None else load_notes()
    allowed_categories = {"TRAINING_DEFINITION", "TRAINING", "PREFERENCE", "COACH"}
    vocabulary_terms = ("tte", "time to exhaustion", "threshold", "soglia", "z4", "zone 4", "vo2", "vo₂", "over-under", "over under", "sprint")
    out = []
    for n in notes or []:
        category = str(n.get("category") or "GENERAL").strip().upper()
        body = str(n.get("text") or "").strip()
        low = body.lower()
        if category not in allowed_categories or not body:
            continue
        if category == "TRAINING_DEFINITION" or any(term in low for term in vocabulary_terms):
            out.append({"date": n.get("date"), "category": category, "text": body})
    return out[-20:]

def _v129_daily_note_availability(now=None, logs=None):
    """Extract explicit *today* availability constraints from the latest Daily Note.

    R130 scopes unavailability to the clause that actually contains the negative
    statement. Positive clauses such as "back on the trainer at 5 pm" must not be
    turned into an evening block merely because they contain a daypart/time token.
    Explicit positive availability is also retained so it can override a broad
    same-note block for that daypart.
    """
    now = now or get_rome_now()
    today_iso = now.date().isoformat()
    empty = {
        "active": False, "date": today_iso, "blocked_dayparts": [],
        "available_dayparts": [], "source": None, "note_excerpt": None,
        "parse_policy": "R130_CLAUSE_SCOPED_AVAILABILITY",
    }
    try:
        rows = list(logs if logs is not None else load_daily_logs(40))
    except Exception:
        rows = []
    today_rows = [r for r in rows if str(r.get("local_date") or "") == today_iso]
    if not today_rows:
        return empty
    today_rows.sort(key=lambda r: (str(r.get("logged_at_utc") or ""), str(r.get("local_time") or "")), reverse=True)
    note = str(today_rows[0].get("note") or "").strip()
    if not note:
        return empty
    low = re.sub(r"\s+", " ", note.lower().replace("’", "'")).strip()

    negative_re = re.compile(
        r"(?:\bcan't\s+(?:train|ride)\b|\bcannot\s+(?:train|ride)\b|\bunable\s+to\s+(?:train|ride)\b|"
        r"\bnot\s+available\b|\bunavailable\b|\bno\s+training\b|"
        r"\bnon\s+posso\s+allenarmi\b|\bnon\s+mi\s+posso\s+allenare\b|\bnon\s+riesco\s+ad\s+allenarmi\b|"
        r"\bnon\s+sono\s+disponibile\b|\bniente\s+allenamento\b|\bnon\s+posso\s+pedalare\b)",
        re.I,
    )
    positive_re = re.compile(
        r"(?:\b(?:i\s+)?can\s+(?:train|ride)\b|\bavailable\s+(?:to\s+train|to\s+ride|from|after|at)\b|"
        r"\bback\s+(?:on|to)\s+(?:the\s+)?trainer\b|\bback\s+(?:on|to)\s+(?:the\s+)?bike\b|"
        r"\bposso\s+(?:allenarmi|allenare|pedalare)\b|\bsono\s+disponibile\b|"
        r"\btorno\s+(?:sui|sul|in)\s+(?:rulli|trainer|bici)\b|\brientro\s+(?:sui|sul|in)\s+(?:rulli|trainer|bici)\b)",
        re.I,
    )

    # Punctuation and contrast conjunctions create separate semantic clauses.
    # We deliberately do not split on plain "or/e" so "can't train morning or evening"
    # remains one negative clause and blocks both named dayparts.
    clauses = [c.strip(" ,") for c in re.split(
        r"(?:[.!?;]+\s*|,\s*(?=(?:but|however|though|ma|per[oò])\b)|\s+\b(?:but|however|though|ma|per[oò])\b\s+)",
        low,
        flags=re.I,
    ) if c and c.strip(" ,")]
    if not clauses:
        clauses = [low]

    def _clock_dayparts(text):
        out = []
        # 12-hour times with am/pm.
        for m in re.finditer(r"\b(1[0-2]|0?[1-9])(?::([0-5]\d))?\s*([ap])\.?m\.?\b", text, re.I):
            hour = int(m.group(1)) % 12 + (12 if m.group(3).lower() == "p" else 0)
            out.append("MORNING" if hour < 12 else ("AFTERNOON" if hour < 17 else "EVENING"))
        # 24-hour forms, mainly useful for Italian notes (e.g. alle 17 / 17:30).
        for m in re.finditer(r"(?:\b(?:at|from|after|alle|dalle|dopo)\s+)([01]?\d|2[0-3])(?::([0-5]\d))?\b(?!\s*[ap]\.?m\.?\b)", text, re.I):
            hour = int(m.group(1))
            out.append("MORNING" if hour < 12 else ("AFTERNOON" if hour < 17 else "EVENING"))
        return out

    def _named_dayparts(text):
        out = []
        if re.search(r"\b(?:stamattina|questa\s+mattina|mattina|this\s+morning|morning|a\.?m\.?)\b", text, re.I):
            out.append("MORNING")
        if re.search(r"\b(?:questo\s+pomeriggio|pomeriggio|this\s+afternoon|afternoon)\b", text, re.I):
            out.append("AFTERNOON")
        if re.search(r"\b(?:stasera|questa\s+sera|sera|this\s+evening|evening|tonight)\b", text, re.I):
            out.append("EVENING")
        return out

    blocked = []
    available = []
    saw_negative = False
    for clause in clauses:
        neg = bool(negative_re.search(clause))
        pos = bool(positive_re.search(clause))
        if neg:
            saw_negative = True
            parts = _named_dayparts(clause) + _clock_dayparts(clause)
            if parts:
                blocked.extend(parts)
            elif re.search(r"\b(?:oggi|today|all\s+day|tutto\s+il\s+giorno|intera\s+giornata)\b", clause, re.I):
                blocked.append("ALL_DAY")
        if pos:
            # Positive availability is independent of the negative clause and can
            # explicitly restore a daypart (e.g. "Back on the trainer at 5 pm").
            available.extend(_named_dayparts(clause) + _clock_dayparts(clause))

    if not saw_negative:
        return empty
    blocked = list(dict.fromkeys(blocked))
    available = list(dict.fromkeys(available))
    if not blocked:
        return empty
    # A specific positive daypart always wins over a same-note specific block.
    blocked = [p for p in blocked if p == "ALL_DAY" or p not in set(available)]
    return {
        "active": bool(blocked),
        "date": today_iso,
        "blocked_dayparts": blocked,
        "available_dayparts": available,
        "source": "DAILY_NOTE_EXPLICIT_AVAILABILITY",
        "note_excerpt": note[:280],
        "parse_policy": "R130_CLAUSE_SCOPED_AVAILABILITY",
    }

def build_planning_context(now=None):
    now = now or get_rome_now()
    today = now.date()
    daily_availability = _v129_daily_note_availability(now=now)
    future_note_intent = _v132_future_note_intent(now=now)
    items = load_planning_items(active_only=True, limit=100)
    relevant = [i for i in items if str(i.get("end_date")) >= today.isoformat()]
    health_now = [i for i in relevant if str(i.get("kind") or "").upper() in
                  ("SICK", "INJURED", "INJURY") and _date_in_item(i, today)]
    health_override = None
    if health_now:
        health_now.sort(key=lambda i: 0 if i.get("kind") == "SICK" else 1)
        h = health_now[0]
        health_override = {
            "kind": "INJURED" if str(h.get("kind") or "").upper() in ("INJURY", "INJURED") else h.get("kind"),
            "source_kind": h.get("kind"), "title": h.get("title"), "note": h.get("note"),
            "until": h.get("end_date"), "restriction_type": h.get("restriction_type"),
        }
    blackout_dates = set()
    no_intensity_dates = set()
    race_by_date = {}
    training_by_date = {}
    horizon = today + timedelta(days=90)
    for item in relevant:
        try:
            start = date.fromisoformat(str(item.get("start_date")))
            end = date.fromisoformat(str(item.get("end_date")))
        except ValueError:
            continue
        end = min(end, horizon)
        current = max(start, today)
        while current <= end:
            ds = current.isoformat()
            if item.get("kind") == "RACE":
                race_by_date.setdefault(ds, item)
            if item.get("kind") == "TRAINING":
                training_by_date.setdefault(ds, item)
            if item.get("restriction_type") == "NO_TRAINING":
                blackout_dates.add(ds)
            elif item.get("restriction_type") == "NO_INTENSITY":
                no_intensity_dates.add(ds)
            current += timedelta(days=1)
    future_races = [i for i in relevant if i.get("kind") == "RACE" and str(i.get("start_date")) >= today.isoformat()]
    future_races.sort(key=lambda i: i.get("start_date"))
    next_race = None
    if future_races:
        r = future_races[0]
        race_date = date.fromisoformat(r["start_date"])
        race_ds = race_date.isoformat()
        conflict = "NO_TRAINING" if race_ds in blackout_dates else ("NO_INTENSITY" if race_ds in no_intensity_dates else None)
        next_race = {**r, "days_away": (race_date - today).days, "conflict": conflict}
    future_training = [i for i in relevant if i.get("kind") == "TRAINING" and str(i.get("end_date")) >= today.isoformat()]
    future_training.sort(key=lambda i: (i.get("start_date"), i.get("created_at_utc") or ""))
    next_training = future_training[0] if future_training else None
    active_today = [i for i in relevant if _date_in_item(i, today)]
    summary_parts = []
    if health_override:
        summary_parts.append(f'{health_override["kind"]} override active through {health_override["until"]}')
    if today.isoformat() in blackout_dates and not health_override:
        summary_parts.append("No-training day active")
    elif today.isoformat() in no_intensity_dates and not health_override:
        summary_parts.append("No-intensity restriction active")
    if next_race:
        summary_parts.append(f'Race in {next_race["days_away"]} day(s): {next_race.get("title") or "Race"}')
    if next_training:
        try:
            td = max(0, (date.fromisoformat(next_training.get("start_date")) - today).days)
        except Exception:
            td = None
        env = next_training.get("training_environment") or ATHLETE_FUELING_PROFILE["default_training_environment"]
        demand = _training_profile_label(next_training.get("training_demand"))
        when = f" in {td} day(s)" if td is not None else ""
        summary_parts.append(f'Planned {env.lower()} training{when}: {next_training.get("title") or "Training"} [{demand}]')
    if daily_availability.get("active"):
        parts = ", ".join(str(x).lower().replace("_", " ") for x in (daily_availability.get("blocked_dayparts") or []))
        available_parts = ", ".join(str(x).lower().replace("_", " ") for x in (daily_availability.get("available_dayparts") or []))
        msg = f"Daily Note availability: {parts} unavailable today"
        if available_parts:
            msg += f"; {available_parts} explicitly available"
        summary_parts.append(msg)
    if future_note_intent.get("active"):
        pref = ", ".join(str(x).lower() for x in (future_note_intent.get("preferred_dayparts") or [])) or "unspecified time"
        msg = f"Daily Note future intent: {future_note_intent.get('date')} quality preferred {pref}" if future_note_intent.get("quality_preferred") else f"Daily Note future availability: {future_note_intent.get('date')} preferred {pref}"
        uncertain = ", ".join(str(x).lower() for x in (future_note_intent.get("uncertain_dayparts") or []))
        if uncertain:
            msg += f"; {uncertain} availability uncertain"
        summary_parts.append(msg)
    return {
        "health_override": health_override,
        "blackout_dates": sorted(blackout_dates),
        "no_intensity_dates": sorted(no_intensity_dates),
        "race_by_date": race_by_date,
        "training_by_date": training_by_date,
        "next_race": next_race,
        "next_training": next_training,
        "active_today": active_today,
        "upcoming_items": relevant[:20],
        "postponed_workouts": [dict(i["nova_postponement"]) for i in relevant
                                if isinstance(i.get("nova_postponement"), dict)],
        "daily_availability": daily_availability,
        "future_note_intent": future_note_intent,
        "summary": " · ".join(summary_parts) if summary_parts else "Training as planned",
    }
def planning_context_text(context):
    context = context or {}
    lines = [
        "HEALTH / AVAILABILITY / EVENT CONTEXT (AUTHORITATIVE; outranks CTL/ATL/TSB and generic training readiness):",
        f'- Current summary: {context.get("summary", "Training as planned")}.',
    ]
    health = context.get("health_override")
    if health:
        lines.append(
            f'- HEALTH OVERRIDE: {health.get("kind")} through {health.get("until")}; '
            f'restriction {health.get("restriction_type")}; note: {health.get("note") or "none"}.'
        )
        lines.append("- Never describe freshness caused by illness/injury-related training reduction as tapering, peaking or improved race form.")
    if context.get("blackout_dates"):
        lines.append("- NO TRAINING dates: " + ", ".join(context["blackout_dates"]))
    if context.get("no_intensity_dates"):
        lines.append("- NO INTENSITY dates: " + ", ".join(context["no_intensity_dates"]))
    daily_availability = context.get("daily_availability") or {}
    if daily_availability.get("active"):
        lines.append("- TODAY DAILY-NOTE AVAILABILITY: unavailable during " + ", ".join(daily_availability.get("blocked_dayparts") or []) + ". This explicit same-day availability constraint outranks learned rhythm for those time windows.")
    future_intent = context.get("future_note_intent") or {}
    if future_intent.get("active"):
        lines.append("- FUTURE DAILY-NOTE INTENT: " + str(future_intent.get("date") or "future") + "; preferred dayparts " + ", ".join(future_intent.get("preferred_dayparts") or ["UNSPECIFIED"]) + "; uncertain dayparts " + ", ".join(future_intent.get("uncertain_dayparts") or ["NONE"]) + "; quality preferred=" + str(bool(future_intent.get("quality_preferred"))) + ". Explicit future intent outranks learned daypart preference, but never bypasses recovery/health/race gates.")
    race = context.get("next_race")
    if race:
        conflict = f' | CONFLICT: {race.get("conflict")}' if race.get("conflict") else ''
        race_env = str(race.get("training_environment") or "UNKNOWN").upper()
        race_duration = _race_duration_display(race.get("expected_duration_min"), race.get("expected_duration_max"))
        lines.append(f'- NEXT RACE: {race.get("title")} on {race.get("start_date")} ({race.get("days_away")} day(s) away) | environment {race_env} | {race_duration}{conflict}.')
        if race.get("note"):
            lines.append(f'- Race note: {race.get("note")}.')
        if race.get("conflict"):
            lines.append("- The active health/training restriction outranks the race. Do not prescribe racing while the conflicting restriction is active.")
    planned = [i for i in (context.get("upcoming_items") or []) if i.get("kind") == "TRAINING" and i.get("active", 1)]
    planned.sort(key=lambda i: (i.get("start_date") or "", i.get("created_at_utc") or ""))
    if planned:
        lines.append("- ATHLETE-PLANNED TRAINING COMMITMENTS:")
        for item in planned[:6]:
            span = item.get("start_date") or "date n/a"
            if item.get("end_date") and item.get("end_date") != item.get("start_date"):
                span += f' through {item.get("end_date")}'
            env = item.get("training_environment") or ATHLETE_FUELING_PROFILE["default_training_environment"]
            demand = _training_profile_label(item.get("training_demand"))
            note = item.get("note") or "none"
            dur = _race_duration_display(item.get("expected_duration_min"), item.get("expected_duration_max")) if (item.get("expected_duration_min") is not None or item.get("expected_duration_max") is not None) else "duration not entered"
            lines.append(f'  • {span}: {item.get("title") or "Planned training"} | environment {env} | ride profile {demand} | {dur} | note: {note}.')
        lines.append("- Planned-training ENVIRONMENT and RIDE PROFILE are athlete-entered context. RECOVERY is an easy ceiling; ENDURANCE/TEMPO are non-hard planned profiles; THRESHOLD/MIXED/VO2/SPRINTS are hard-intent profiles. Health/restrictions and race safety still outrank them. NOVA DECIDES means choose from context.")
    lines.append("- Priority order: health/restrictions → availability → race/event context → athlete-planned training → completed training → recovery metrics → fitness/form → workout generation.")
    return "\n".join(lines)
def _v4875_subjective_activity_lookup(activity_context):
    by_id = {}
    by_stamp = {}
    for item in (activity_context or {}).get("items") or []:
        if not isinstance(item, dict):
            continue
        log_id = str(item.get("log_id") or "").strip()
        if log_id:
            by_id[log_id] = item
        stamp = (str(item.get("date") or "")[:10], str(item.get("time") or "")[:5])
        if stamp != ("", ""):
            by_stamp[stamp] = item
    return by_id, by_stamp
def _v4875_subjective_candidate_text(item):
    if not isinstance(item, dict):
        return ""
    relation = str(item.get("relation") or "GENERAL")
    candidates = item.get("candidate_activities") or []
    rendered = []
    for c in candidates:
        if not isinstance(c, dict):
            continue
        bit = f'{c.get("time") or "time n/a"} {c.get("name") or "Activity"} · {c.get("duration_min") or 0} min'
        if c.get("load") is not None:
            bit += f' · load {c.get("load")}'
        rendered.append(bit)
    return f' | retrospective link {relation} | candidates {"; ".join(rendered) if rendered else "none"}'
def subjective_context_text(trend=None, logs=None, activity_context=None):
    trend = trend or compute_feeling_trend(logs)
    logs = logs if logs is not None else load_daily_logs(40)
    by_id, by_stamp = _v4875_subjective_activity_lookup(activity_context)
    lines = ["SUBJECTIVE DAILY LOG (structured, athlete-entered, authoritative recovery/context signal):"]
    lines.append(
        f'- 7d average {trend.get("avg7", "n/a")} | 28d average {trend.get("avg28", "n/a")} | trend {trend.get("label", "n/a")}.'
    )
    lines.append(
        "- Daily Log is retrospective subjective context. A rating/comment NEVER creates a future training slot, never proves that a second session should occur, and never overrides the history-learned Training Rhythm."
    )
    for x in logs[-12:]:
        item = by_id.get(str(x.get("id") or "").strip())
        if item is None:
            item = by_stamp.get((str(x.get("local_date") or "")[:10], str(x.get("local_time") or "")[:5]))
        lines.append(
            f'- {x.get("local_date")} {x.get("local_time")} | {x.get("feeling")}/10 | {x.get("phase")} | '
            f'completed sessions in stored context {x.get("session_count")} | load in context {x.get("today_load")} | note {x.get("note") or "none"}'
            + _v4875_subjective_candidate_text(item)
        )
    if activity_context:
        lines.append(
            "- Candidate activity linkage is retrospective only. Associate a note with a completed activity only when its wording supports that link; it never creates a future workout."
        )
    return "\n".join(lines)
def apply_health_override_to_metrics(metrics, planning_context):
    metrics = dict(metrics)
    health = (planning_context or {}).get("health_override")
    if not health:
        return metrics
    if health.get("kind") == "SICK":
        cap, label = 25, "Health Override"
    else:
        cap, label = 38, "Injury Override"
    try:
        metrics["race_readiness_score"] = min(int(metrics.get("race_readiness_score", cap)), cap)
    except (TypeError, ValueError):
        metrics["race_readiness_score"] = cap
    metrics["race_readiness_label"] = label
    metrics["race_readiness_zone"] = "red"
    metrics["race_readiness_css_color"] = "red"
    metrics["race_readiness_reason"] = (
        f'{health.get("kind")} status is active through {health.get("until")}. '
        "Health context overrides training-form signals; reduced ATL or rising TSB must not be interpreted as peaking."
    )
    return metrics
def apply_health_override_to_energy(energy, planning_context):
    energy = dict(energy)
    health = (planning_context or {}).get("health_override")
    if not health:
        return energy
    cap = 25 if health.get("kind") == "SICK" else 40
    score = min(int(energy.get("energy_score", cap)), cap)
    theta = math.pi - (score / 100.0) * math.pi
    radius = 87
    energy.update({
        "energy_score": score,
        "energy_label": "Health Hold" if health.get("kind") == "SICK" else "Injury Restricted",
        "energy_zone": "red" if health.get("kind") == "SICK" else "orange",
        "marker_x": round(100 + radius * math.cos(theta), 1),
        "marker_y": round(95 - radius * math.sin(theta), 1),
    })
    return energy

_BETA_QA_DIRECT_IDENTIFIER_RE = re.compile(r"(^|_)(?:username|email|display_name|user_id|athlete_id|external_athlete_id)(?:_|$)", re.I)
def _beta_qa_redact_direct_identifiers(node, identifiers=()):
    identifiers = tuple(str(x).strip() for x in (identifiers or ()) if str(x or "").strip())
    if isinstance(node, dict):
        out = {}
        for key, value in node.items():
            if _BETA_QA_DIRECT_IDENTIFIER_RE.search(str(key or "")):
                out[key] = "[REDACTED FOR BETA QA]"
            else:
                out[key] = _beta_qa_redact_direct_identifiers(value, identifiers)
        return out
    if isinstance(node, list):
        return [_beta_qa_redact_direct_identifiers(value, identifiers) for value in node]
    if isinstance(node, str) and identifiers:
        text = node
        for value in sorted(set(identifiers), key=len, reverse=True):
            if len(value) < 3:
                continue
            text = re.sub(re.escape(value), "[REDACTED FOR BETA QA]", text, flags=re.IGNORECASE)
        return text
    return node
def _beta_qa_report_copy(report, identifiers=()):
    safe = dict(report or {})
    safe["data"] = _beta_qa_redact_direct_identifiers((report or {}).get("data") or {}, identifiers)
    return safe

# Observation is NOT a PB, a completed prescribed test or a change to FTP/CP.
_NOVA_AUTO_EFFORT_DURATIONS = (60, 120, 180, 300, 360, 420, 480, 540, 600, 720, 900, 1200, 1800, 3600)

def _nova_r196982_activity_power_windows(payload, now=None, cycle_start=None, lookback_days=21):
    """Read exact seconds from ICU activity-power-curves; fail closed on missing provenance.

    A point is the best ROLLING average inside the activity, including a 300 s
    window in a continuous 420 s climb. No lap, interval parser or title used.
    Observations remain read-only; only authoritative execution reconciliation
    can use them as an exact-duration input for a prescribed checkpoint.
    """
    now = now or get_rome_now()
    today = now.date() if isinstance(now, datetime) else now
    try:
        cycle_day = date.fromisoformat(str(cycle_start)[:10]) if cycle_start else None
    except (ValueError, TypeError):
        cycle_day = None
    cutoff = today - timedelta(days=max(1, min(int(lookback_days), 45)))
    lower = max(cutoff, cycle_day) if cycle_day else cutoff
    if not isinstance(payload, dict):
        return {"observed_efforts": [], "featured_efforts": [], "method_note": "Activity power curves unavailable; no automatic effort claim."}
    secs = payload.get("secs")
    curves = payload.get("curves")
    if not isinstance(secs, list) or not isinstance(curves, list):
        return {"observed_efforts": [], "featured_efforts": [], "method_note": "Activity power curves unavailable; no automatic effort claim."}
    indexes = {}
    for i, value in enumerate(secs):
        try:
            sec = int(value)
            if sec in _NOVA_AUTO_EFFORT_DURATIONS and float(value) == float(sec):
                indexes[sec] = i
        except (TypeError, ValueError, OverflowError):
            continue
    observed = []
    for curve in curves:
        if not isinstance(curve, dict):
            continue
        aid = str(curve.get("id") or "").strip()
        day_text = str(curve.get("start_date_local") or "")[:10]
        try:
            day = date.fromisoformat(day_text)
        except ValueError:
            continue
        if not aid or not (lower <= day <= today):
            continue
        watts = curve.get("watts")
        if not isinstance(watts, list):
            continue
        for sec, i in indexes.items():
            if i >= len(watts):
                continue
            try:
                w = float(watts[i])
            except (TypeError, ValueError, OverflowError):
                continue
            if not (20.0 <= w <= 2500.0) or not math.isfinite(w):
                continue
            observed.append({"activity_id": aid, "date": day_text, "secs": sec,
                             "watts": int(round(w)), "source": "INTERVALS_ACTIVITY_POWER_CURVE",
                             "rolling_window": True, "manual_lap_required": False,
                             "formal_validation": False, "pb_confirmed": False})
    # One best native curve point per activity / duration, same identity as ICU.
    dedup = {}
    for row in observed:
        key = (row["activity_id"], row["date"], row["secs"])
        if key not in dedup or row["watts"] > dedup[key]["watts"]:
            dedup[key] = row
    observed = sorted(dedup.values(), key=lambda r: (r["date"], r["secs"], r["watts"]), reverse=True)[:384]
    featured = []
    for sec in _NOVA_AUTO_EFFORT_DURATIONS:
        candidates = [r for r in observed if r["secs"] == sec]
        if not candidates:
            continue
        best = max(candidates, key=lambda r: (r["watts"], r["date"]))
        featured.append(dict(best))
    return {"observed_efforts": observed, "featured_efforts": featured,
            "method_note": "Exact-duration rolling power from Intervals.icu activity curves; no LAP needed. Featured efforts are best among fetched recent rides, NOT proven YTD PBs. No automatic formal validation, FTP or CP adjustment."}

def extract_power_curve_points(payload):
    array_results = []
    point_list_results = []
    def walk(node):
        if isinstance(node, dict):
            secs = node.get("secs")
            watts = node.get("watts")
            if watts is None:
                watts = node.get("power")
            if (
                isinstance(secs, list) and isinstance(watts, list)
                and len(secs) == len(watts) and len(secs) > 0
                and isinstance(secs[0], (int, float))
            ):
                array_results.append(
                    [{"secs": s, "watts": w} for s, w in zip(secs, watts) if w]
                )
            for v in node.values():
                walk(v)
        elif isinstance(node, list):
            if node and all(isinstance(i, dict) for i in node):
                keys0 = set(node[0].keys())
                if "secs" in keys0 and ("watts" in keys0 or "power" in keys0):
                    if isinstance(node[0].get("secs"), (int, float)):
                        point_list_results.append(node)
            for item in node:
                walk(item)
    walk(payload)
    if array_results:
        return max(array_results, key=len)
    return point_list_results[0] if point_list_results else None

def best_watts_for_durations(points, target_secs_list):

        return []

    results = []
    for target in target_secs_list:
        best = None
        for p in sorted_points:
            if p.get("secs", 0) <= target:
                best = p
            else:
                break
        if best is None:
            best = sorted_points[0]
        watts = best.get("watts") or best.get("power")
        if watts:
            results.append({"label": format_duration_label(target), "watts": round(watts)})
    if results:
        valid_watts=[_num(r.get("watts")) for r in results if _num(r.get("watts")) is not None]
        max_watts = max(valid_watts) if valid_watts else None
        for r in results:
            rw=_num(r.get("watts"))
            r["pct"] = round(100 * rw / max_watts) if rw is not None and max_watts else 0
    return results

def _power_at_duration(points, target_secs):
    usable = []
    for p in points or []:
        try:
            sec = float(p.get("secs"))
            watts = float(p.get("watts") if p.get("watts") is not None else p.get("power"))
        except (TypeError, ValueError):
            continue
        if sec > 0 and watts > 0:
            usable.append((sec, watts))
    usable.sort()
    if not usable:
        return None
    for sec, watts in usable:
        if abs(sec - target_secs) < 0.5:
            return watts
    lower = None
    upper = None
    for sec, watts in usable:
        if sec < target_secs:
            lower = (sec, watts)
        elif sec > target_secs:
            upper = (sec, watts)
            break
    if not lower or not upper:
        return None
    lo_s, lo_w = lower
    hi_s, hi_w = upper
    if hi_s / lo_s > 2.5:
        return None
    frac = (math.log(target_secs) - math.log(lo_s)) / (math.log(hi_s) - math.log(lo_s))
    return lo_w + frac * (hi_w - lo_w)
def fit_critical_power_model(points, durations=(180, 300, 480, 720, 1200)):
    pairs = []
    for sec in durations:
        watts = _power_at_duration(points, sec)
        if watts is not None:
            pairs.append((float(sec), float(watts)))
    if len(pairs) < 4:
        return {"available": False, "reason": "Need at least four usable best-effort durations between 3 and 20 min."}
    xs = [1.0 / sec for sec, _ in pairs]
    ys = [watts for _, watts in pairs]
    x_mean = statistics.mean(xs)
    y_mean = statistics.mean(ys)
    denom = sum((x - x_mean) ** 2 for x in xs)
    if denom <= 0:
        return {"available": False, "reason": "Power-curve durations are not sufficiently distinct."}
    slope = sum((x - x_mean) * (y - y_mean) for x, y in zip(xs, ys)) / denom
    intercept = y_mean - slope * x_mean
    preds = [intercept + slope * x for x in xs]
    sse = sum((y - pred) ** 2 for y, pred in zip(ys, preds))
    sst = sum((y - y_mean) ** 2 for y in ys)
    r2 = 1.0 - sse / sst if sst > 0 else 1.0
    errors = [abs(y - pred) / y * 100.0 for y, pred in zip(ys, preds) if y > 0]
    mae_pct = statistics.mean(errors) if errors else None
    cp = intercept
    w_prime_j = slope
    if not (50 <= cp <= 600 and 1000 <= w_prime_j <= 80000):
        return {"available": False, "reason": "The fitted parameters fell outside a plausible cycling field-model range."}
    if r2 >= 0.985 and len(pairs) >= 5:
        confidence = "HIGH"
    elif r2 >= 0.95:
        confidence = "MODERATE"
    else:
        confidence = "LOW"
    return {
        "available": True,
        "cp": round(cp),
        "w_prime_j": round(w_prime_j),
        "w_prime_kj": round(w_prime_j / 1000.0, 1),
        "r2": round(max(-1.0, min(1.0, r2)), 3),
        "mean_error_pct": round(mae_pct, 1) if mae_pct is not None else None,
        "confidence": confidence,
        "confidence_class": confidence.lower(),
