        "weekday_scores": weekday_scores, "weekday_double_scores": weekday_double,
        "weekday_evidence": weekday_evidence, "weekday_selection_policy": "INDEPENDENT_OBSERVED_WEEKDAYS_R1964",
        "history_window_requested_days": RHYTHM_SEASON_DAYS,
        "history_first_observed_date": history_start.isoformat() if history_start else None,

Each weekday needs repeat evidence; estimated days/week is not an eligibility cap. A learned opportunity is not a mandate or recovery clearance.",
    }
def _v4895_rhythm_duration_reliable(rhythm, min_samples=3):
    r = rhythm or {}
    samples = int(r.get("training_day_duration_samples") or 0)
    confidence = str(r.get("confidence") or "LOW").upper()
    state = str(r.get("state") or "LEARNING").upper()
    return bool(samples >= int(min_samples) and confidence in ("MODERATE", "HIGH") and state != "LEARNING")

def _subjective_activity_summary(row):
    dur = row.get("duration_min")
    dur_text = f' · {round(dur)} min' if dur else ''
    load = row.get("load")
    load_text = f' · load {round(load,1)}' if load is not None else ''
    return f'{row.get("dt").strftime("%I:%M %p").lstrip("0")} {row.get("name")}{dur_text}{load_text}'
def build_subjective_activity_context(logs, activities, now=None, max_logs=12):
    now = now or get_rome_now()
    today = now.date()
    records = _rhythm_activity_records(activities, now)
    by_date = {}
    for row in records:
        by_date.setdefault(row["date"].isoformat(), []).append(row)
    rows = sorted(list(logs or []), key=lambda x: (str(x.get("local_date") or ""), str(x.get("logged_at_utc") or "")))[-max_logs:]
    items = []
    for log in rows:
        ds = str(log.get("local_date") or "")[:10]
        try:
            d = date.fromisoformat(ds)
        except ValueError:
            continue
        acts = sorted(by_date.get(ds, []), key=lambda x: x["dt"])
        relation = "GENERAL"
        candidates = []
        if d < today:
            relation = "DAY_LEVEL" if acts else "GENERAL"
            candidates = acts
        elif d == today:
            try:
                hh, mm = [int(x) for x in str(log.get("local_time") or "").split(":")[:2]]
                log_dt = datetime.combine(d, datetime.min.time()).replace(hour=hh, minute=mm)
            except Exception:
                log_dt = now.replace(tzinfo=None)
            completed = []
            for act in acts:
                duration = act.get("duration_min") or 0
                end_dt = act["dt"] + timedelta(minutes=duration)
                if end_dt <= log_dt:
                    completed.append(act)
            if len(completed) == 1:
                relation, candidates = "POSSIBLE_ACTIVITY_LINK", completed
            elif len(completed) >= 2:
                recent = [a for a in completed if 0 <= (log_dt - (a["dt"] + timedelta(minutes=a.get("duration_min") or 0))).total_seconds() <= 4*3600]
                relation = "POSSIBLE_ACTIVITY_LINK" if len(recent) == 1 else "DAY_LEVEL"
                candidates = recent if len(recent) == 1 else completed
        items.append({
            "log_id": log.get("id"), "date": ds, "time": log.get("local_time"),
            "feeling": log.get("feeling"), "note": log.get("note") or "", "relation": relation,
            "candidate_activities": [
                {"id": a.get("id"), "date": a.get("date").isoformat() if a.get("date") else ds, "name": a.get("name"), "time": a.get("dt").strftime("%H:%M"), "duration_min": round(a.get("duration_min") or 0), "load": a.get("load")}
                for a in candidates
            ],
        })
    return {"items": items, "rule": "retrospective context only; never creates a future training opportunity"}
def subjective_activity_context_text(context):
    context = context or {}
    lines = ["SUBJECTIVE ↔ COMPLETED-ACTIVITY CONTEXT (candidate linkage; not a scheduling signal):"]
    for item in context.get("items") or []:
        candidates = item.get("candidate_activities") or []
        cand_text = "; ".join(
            f'{c.get("time")} {c.get("name") or "Activity"} · {c.get("duration_min")} min' + (f' · load {c.get("load")}' if c.get("load") is not None else '')
            for c in candidates
        ) or "none"
        lines.append(
            f'- {item.get("date")} {item.get("time")} | {item.get("feeling")}/10 | {item.get("relation")} | candidates {cand_text} | note {item.get("note") or "none"}'
        )
    lines.append(
        "- AI may associate note meaning with a candidate only when the wording supports it. Never interpret a rating/comment as proof that the day has another workout remaining."
    )
    return "\n".join(lines)

def build_coach_clock(season_activities, now=None, planning_context=None, training_rhythm=None):
    now = now or get_rome_now()
    planning_context = planning_context or build_planning_context(now)
    rhythm = training_rhythm or build_training_rhythm(season_activities, now)
    today = now.date()
    today_iso = today.isoformat()
    records = [r for r in _rhythm_activity_records(season_activities, now) if _v4887_is_cycling_activity({"type": r.get("type"), "name": r.get("name")})]
    today_rows = [r for r in records if r["date"] == today]
    today_rows.sort(key=lambda x: x["dt"])
    today_load = sum(r.get("load") or 0 for r in today_rows)
    today_minutes = sum(r.get("duration_min") or 0 for r in today_rows)
    today_acts = [
        {"name": r.get("name"), "hour": r["dt"].hour, "time": r["dt"].strftime("%I:%M %p").lstrip("0"), "duration_min": round(r.get("duration_min") or 0)}
        for r in today_rows
    ]
    blackout = set(planning_context.get("blackout_dates") or [])
    easy_only = set(planning_context.get("no_intensity_dates") or [])
    race_by_date = planning_context.get("race_by_date") or {}
    training_by_date = planning_context.get("training_by_date") or {}
    health = planning_context.get("health_override")
    daily_availability = planning_context.get("daily_availability") or {}
    daily_blocked_parts = set(daily_availability.get("blocked_dayparts") or []) if daily_availability.get("active") and str(daily_availability.get("date") or "") == today_iso else set()
    daily_available_parts = set(daily_availability.get("available_dayparts") or []) if str(daily_availability.get("date") or "") == today_iso else set()
    # Calendar blackout remains a whole-day block. A broad Daily Note block may be
    # narrowed by a later explicit positive availability statement in the same note.
    blocked_today = today_iso in blackout or ("ALL_DAY" in daily_blocked_parts and not daily_available_parts)
    no_intensity_today = today_iso in easy_only
    clock_profile = rhythm.get("clock_profile") or {}
    weekday_scores = rhythm.get("weekday_scores") or {}
    weekday_double = rhythm.get("weekday_double_scores") or {}
    likely_weekdays = set(rhythm.get("likely_weekdays") or [])
    pattern_strength = rhythm.get("pattern_strength") or "LOW"
    def predicted_start(slot_date, second=False):
        mapping = clock_profile.get("second_start_min" if second else "first_start_min") or {}
        value = mapping.get(slot_date.weekday())
        if value is None:
            value = clock_profile.get("overall_second_start_min" if second else "overall_first_start_min")
        return int(round(value)) if value is not None else None
    def predicted_duration(slot_date):
        if not _v4895_rhythm_duration_reliable(rhythm):
            return None
        mapping = clock_profile.get("duration_min") or {}
        value = mapping.get(slot_date.weekday())
        if value is None:
            value = rhythm.get("median_session_duration_min") or clock_profile.get("overall_duration_min")
        return round(value, 1) if value is not None else None
    def slot_daypart(start_minute):
        if start_minute is None:
            return None
        if start_minute < 12 * 60:
            return "MORNING"
        if start_minute < 17 * 60:
            return "AFTERNOON"
        return "EVENING"
    def daily_note_blocks(slot_date, start_minute):
        if not slot_date or slot_date != today or not daily_blocked_parts:
            return False
        part = slot_daypart(start_minute)
        if part in daily_available_parts:
            return False
        if "ALL_DAY" in daily_blocked_parts:
            return True
        return part in daily_blocked_parts
    def same_day_start_is_future(start_minute):
        if start_minute is None:
            return True
        now_minute = now.hour * 60 + now.minute
        # A learned clock time is a candidate start, never an appointment in the past.
        # Five minutes of tolerance avoids flicker around Snapshot generation.
        return start_minute >= now_minute - 5
    def day_word(slot_date):
        if slot_date == today:
            return "Today"
        if slot_date == today + timedelta(days=1):
            return "Tomorrow"
        return slot_date.strftime("%A")
    def base_slot(slot_date, period, event=None, source="RHYTHM", second=False):
        ds = slot_date.isoformat() if slot_date else None
        training_environment = None
        training_demand = None
        event_note = None
        restriction = "NO_INTENSITY" if ds in easy_only else "NONE"
        start_minute = predicted_start(slot_date, second=second) if slot_date else None
        duration_hint = predicted_duration(slot_date) if slot_date else (rhythm.get("median_session_duration_min") if _v4895_rhythm_duration_reliable(rhythm) else None)
        timing_conf = rhythm.get("confidence") or "LOW"
        if period == "RACE":
            label = f'{day_word(slot_date)} · RACE · {event.get("title") or "Priority event"}'
            window = "priority event"
            restriction = "RACE"
            source = "CALENDAR"
            start_minute = None
            training_environment = str(event.get("training_environment") or "").strip().upper() or None
            if training_environment not in ("INDOOR", "OUTDOOR"):
                training_environment = None
            event_note = event.get("note") or ""
            duration_hint = _nutrition_number(event.get("expected_duration_max"))
            timing_conf = "EXPLICIT"
        elif period == "PLANNED":
            training_environment = str(event.get("training_environment") or ATHLETE_FUELING_PROFILE["default_training_environment"]).upper()
            if training_environment not in ("INDOOR", "OUTDOOR"):
                training_environment = ATHLETE_FUELING_PROFILE["default_training_environment"]
            training_demand = _training_profile_key(event.get("training_demand"))
            label = f'{day_word(slot_date)} · PLANNED · {training_environment} · {event.get("title") or "Training"}'
            window = "athlete-planned"
            event_note = event.get("note") or ""
            source = "CALENDAR"
            start_minute = None
            explicit_training_duration = _nutrition_number(event.get("expected_duration_max"))
            if explicit_training_duration is not None:
                duration_hint = explicit_training_duration
            timing_conf = "EXPLICIT"
        elif period == "SAME_DAY_SECOND":
            when = _rhythm_format_minutes(start_minute)
            label = f'{day_word(slot_date)} · likely second session' + (f' · around {when}' if when else '')
            window = "history-learned same-day opportunity"
            source = "RHYTHM"
        elif period == "RHYTHM_ALTERNATE":
            when = _rhythm_format_minutes(start_minute)
            label = f'{day_word(slot_date)} · later learned window' + (f' · around {when}' if when else '')
            window = "history-learned later same-day opportunity"
            source = "RHYTHM"
        elif period == "RHYTHM_SECOND":
            when = _rhythm_format_minutes(start_minute)
            label = f'{day_word(slot_date)} · likely second session' + (f' · around {when}' if when else '')
            window = "history-learned future second-session opportunity"
            source = "RHYTHM"
        elif period == "RHYTHM":
            when = _rhythm_format_minutes(start_minute)
            label = f'{day_word(slot_date)} · learned rhythm' + (f' · around {when}' if when else '')
            window = "history-learned opportunity"
        else:
            label = "Next training opportunity · timing still learning"
            window = "timing not inferred yet"
            source = "LEARNING"
            timing_conf = "LOW"
            start_minute = None
        return {
            "date": ds, "period": period, "window": window, "label": label,
            "restriction": restriction,
            "restriction_until": next((i.get("end_date") for i in planning_context.get("upcoming_items", []) if i.get("restriction_type") == restriction and slot_date and _date_in_item(i, slot_date)), None) if restriction == "NO_INTENSITY" else None,
            "is_race": period == "RACE", "is_planned_training": period == "PLANNED",
            "event_title": event.get("title") if event else None, "event_note": event_note,
            "training_environment": training_environment, "training_demand": training_demand,
            "expected_duration_min": _nutrition_number(event.get("expected_duration_min")) if event and period in ("PLANNED","RACE") else None,
            "expected_duration_max": _nutrition_number(event.get("expected_duration_max")) if event and period in ("PLANNED","RACE") else None,
            "planning_item_id": event.get("id") if event and period in ("PLANNED", "RACE") else None,
            "source": source, "timing_confidence": timing_conf, "start_minute": start_minute,
            "duration_hint_min": duration_hint, "duration_hint_label": (_race_duration_display(event.get("expected_duration_min"), event.get("expected_duration_max")) if event and period in ("PLANNED","RACE") and (event.get("expected_duration_min") is not None or event.get("expected_duration_max") is not None) else _rhythm_format_duration(duration_hint)),
        }
    candidates = []
    seen_dates = set()
    today_explicit = bool(race_by_date.get(today_iso) or training_by_date.get(today_iso))
    if not blocked_today:
        if race_by_date.get(today_iso) and not no_intensity_today:
            candidates.append(base_slot(today, "RACE", race_by_date[today_iso]))
            seen_dates.add(today_iso)
        elif training_by_date.get(today_iso) and not today_rows:
            candidates.append(base_slot(today, "PLANNED", training_by_date[today_iso]))
            seen_dates.add(today_iso)
    same_day_second = False
    if (
        not blocked_today and not today_explicit and len(today_rows) == 1 and rhythm.get("state") != "LEARNING"
        and float(rhythm.get("double_propensity") or 0) >= 0.45
        and float(weekday_double.get(today.weekday()) or 0) >= 0.45
    ):
        second_start = predicted_start(today, second=True)
        if (second_start is None or same_day_start_is_future(second_start)) and not daily_note_blocks(today, second_start):
            candidates.append(base_slot(today, "SAME_DAY_SECOND", source="RHYTHM", second=True))
            seen_dates.add(today_iso)
            same_day_second = True
    if not today_rows and not blocked_today and not today_explicit and rhythm.get("state") != "LEARNING":
        score = float(weekday_scores.get(today.weekday()) or 0)
        start_min = predicted_start(today)
        day_supported = today.weekday() in likely_weekdays if pattern_strength != "LOW" else score >= 0.45
        first_available = same_day_start_is_future(start_min) and not daily_note_blocks(today, start_min)
        if day_supported and first_available and today_iso not in seen_dates:
            candidates.append(base_slot(today, "RHYTHM"))
            seen_dates.add(today_iso)
        elif day_supported and today_iso not in seen_dates:
            # If the usual first window is already past or explicitly unavailable, a
            # well-established later double-day clock may still be the next plausible
            # *single* opportunity today. Do not call it a second session when the
            # first one never happened.
            second_start = predicted_start(today, second=True)
            second_supported = (
                float(rhythm.get("double_propensity") or 0) >= 0.45
                and float(weekday_double.get(today.weekday()) or 0) >= 0.45
                and second_start is not None
                and same_day_start_is_future(second_start)
                and not daily_note_blocks(today, second_start)
            )
            if second_supported:
                candidates.append(base_slot(today, "RHYTHM_ALTERNATE", source="RHYTHM", second=True))
                seen_dates.add(today_iso)
    frequency_targets = set()
    if rhythm.get("state") != "LEARNING" and pattern_strength == "LOW":
        gap = rhythm.get("median_training_gap_days")
        if gap is None:
            days_wk = float(rhythm.get("estimated_days_per_week") or 0)
            gap = max(1, round(7.0 / days_wk)) if days_wk > 0 else 2
        gap = max(1, int(round(gap)))
        last_training = max((r["date"] for r in records if r["date"] <= today), default=today - timedelta(days=gap))
        d = last_training + timedelta(days=gap)
        while d <= today:
            d += timedelta(days=gap)
        while len(frequency_targets) < 8 and d <= today + timedelta(days=42):
            frequency_targets.add(d.isoformat())
            d += timedelta(days=gap)
    d = today + timedelta(days=1)
    guard = 0
    while len(candidates) < 3 and guard < 42:
        ds = d.isoformat()
        guard += 1
        if ds in blackout:
            d += timedelta(days=1)
            continue
        race = race_by_date.get(ds)
        if race and ds not in easy_only:
            candidates.append(base_slot(d, "RACE", race))
            seen_dates.add(ds)
            d += timedelta(days=1)
            continue
        planned = training_by_date.get(ds)
        if planned:
            candidates.append(base_slot(d, "PLANNED", planned))
            seen_dates.add(ds)
            d += timedelta(days=1)
            continue
        if rhythm.get("state") != "LEARNING":
            score = float(weekday_scores.get(d.weekday()) or 0)
            if pattern_strength == "LOW":
                supported = ds in frequency_targets
            else:
                supported = d.weekday() in likely_weekdays and score >= 0.25
            if supported:
                candidates.append(base_slot(d, "RHYTHM"))
                seen_dates.add(ds)
                double_supported = (
                    len(candidates) < 3
                    and float(rhythm.get("double_propensity") or 0) >= 0.45
                    and float(weekday_double.get(d.weekday()) or 0) >= 0.45
                    and predicted_start(d, second=True) is not None
                )
                if double_supported:
                    candidates.append(base_slot(d, "RHYTHM_SECOND", source="RHYTHM", second=True))
        d += timedelta(days=1)
    if rhythm.get("state") == "LEARNING":
        has_today_explicit = bool(candidates and candidates[0].get("date") == today_iso and candidates[0].get("source") == "CALENDAR")
        if not has_today_explicit:
            candidates.insert(0, base_slot(None, "UNSCHEDULED"))
    next_slots = candidates[:3]
    if health:
        phase = "HEALTH OVERRIDE"
        headline = "Health context has priority"
        explanation = (
            f'{health.get("kind")} is active through {health.get("until")}. '
            "Learned rhythm remains visible as background, but health/restrictions decide what is actually allowed."
        )
    elif blocked_today:
        phase = "NO TRAINING"
        headline = "Today is blocked from training"
        explanation = "Calendar/health availability removes today before learned rhythm can create an opportunity."
    elif no_intensity_today:
        phase = "RESTRICTED"
        headline = "No-intensity restriction active"
        explanation = "Training may be possible, but Python limits any valid opportunity to easy aerobic/recovery work."
    elif daily_availability.get("active") and next_slots and next_slots[0].get("date") == today_iso:
        phase = "RHYTHM · AVAILABILITY"
        headline = "The next usable window respects today's availability"
        explanation = "Your explicit Daily Note availability removes the blocked daypart before learned rhythm proposes the next usable opportunity."
    elif same_day_second:
        phase = "RHYTHM · SAME-DAY"
        headline = "Another session is plausible today"
        explanation = "That possibility comes from your recent/established Intervals history — never from the Daily Log. Recovery and current context can still shorten, soften or cancel it."
    elif today_rows:
        phase = "TRAINING LOGGED"
        headline = "Today's completed work is in context"
        explanation = "THE LAB does not assume a second workout. The next opportunity is inferred from your learned rhythm or explicit Calendar commitments."
    elif rhythm.get("state") == "LEARNING":
        phase = "RHYTHM · LEARNING"
        headline = "THE LAB is learning your training rhythm"
        explanation = "There is not enough history yet to invent a training date or AM/PM slot. Timing becomes more specific as Intervals history accumulates."
    elif next_slots and next_slots[0].get("date") == today_iso:
        phase = "RHYTHM · TODAY"
        headline = "Training is plausible later today"
        explanation = "This timing comes from your observed history and remains a probability, not a commitment."
    else:
        phase = "RHYTHM · NEXT"
        headline = "Training timing follows your learned pattern"
        explanation = "Recent 21-day behavior is blended with the established 42-day rhythm and up to 180-day longer-term context; each weekday must have repeated activity evidence."
    if blocked_today:
        today_status, today_class = "BLOCKED", "missed"
    elif today_rows:
        today_status, today_class = f'{len(today_rows)} SESSION' + ("S" if len(today_rows) != 1 else ""), "done"
    elif no_intensity_today:
        today_status, today_class = "EASY ONLY", "upcoming"
    elif next_slots and next_slots[0].get("date") == today_iso:
        today_status, today_class = "LIKELY WINDOW", "open"
    else:
        # No completed workout is not a "missed" workout unless one was due.
        today_status, today_class = "NOT SCHEDULED", "upcoming"
    if today_rows:
        today_detail = f'{round(today_minutes)} min completed' + (f' · load {round(today_load,1)}' if today_load else '')
    else:
        today_detail = "No completed activity logged"

def classify_training_family(activity):
    if not _v4887_is_cycling_activity(activity):
        return "Other / Mixed"
    name = str(activity.get("name") or "").lower()
    compact = name.replace("-", " ").replace("_", " ")
    keyword_groups = [
        ("Recovery", ("recovery", "recuper", "easy spin", "scarico")),
        ("Sprint / Neuromuscular", ("sprint", "neuromus", "anaerobic", "anaerobico")),
        ("VO2max", ("vo2", "vo₂", "vo 2", "max aerobic")),
        ("Over-Under", ("over under", "overunder", "o&u", "over / under")),
        ("TTE", ("tte", "time to exhaustion", "time-to-exhaustion")),
        ("Threshold", ("threshold", "soglia", "ftp", "z4", "zone 4")),
        ("Sweet Spot", ("sweet spot", "sweetspot")),
        ("Tempo", ("tempo", "z3", "zone 3")),
        ("Endurance / Z2", ("z2", "zone 2", "endurance", "aerobic", "endurance ride")),
    ]
    for label, keys in keyword_groups:
        if any(k in compact for k in keys):
            return label
    zone_times = activity.get("icu_zone_times") or []
    if zone_times:
        try:
            secs = [float(z.get("secs") or 0) for z in zone_times]
            low, mod, high = bucket_zone_seconds(secs)
            total = low + mod + high
            if total > 0:
                high_pct = 100 * high / total
                mod_pct = 100 * mod / total
                if high_pct >= 15:
                    return "High Intensity / Mixed"
                if mod_pct >= 30:
                    return "Tempo / Threshold Mixed"
                if low / total >= 0.75:
                    return "Endurance / Z2"
        except (TypeError, ValueError):
            pass
    return "Other / Mixed"
def is_hard_training_family(family):
    f = (family or "").lower()
    return any(k in f for k in (
        "vo2", "threshold", "tte", "over-under", "sprint", "neuromuscular",
        "high intensity", "tempo / threshold"
    ))
def _v4900_activity_average_watts(activity):
    a = activity or {}
    for key in ("icu_average_watts", "average_watts"):
        value = _num(a.get(key))
        if value is not None and value > 0:
            return value
    rows = [it for it in (a.get("icu_intervals") or []) if isinstance(it, dict)]
    weighted = []
    for it in rows:
        watts = _num(it.get("average_watts"))
        secs = _num(it.get("moving_time") or it.get("elapsed_time"))
        if watts is not None and secs is not None and watts > 0 and secs > 0:
            weighted.append((watts, secs))
    if weighted:
        total = sum(sec for _, sec in weighted)
        if total > 0:
            return sum(w * sec for w, sec in weighted) / total
    return None
def _v4900_zone_low_share(activity, key):
    try:
        secs = _zone_secs((activity or {}).get(key))
        if not secs:
            return None
        low, mod, high = bucket_zone_seconds(secs)
        total = low + mod + high
        return (low / total) if total > 0 else None
    except Exception:
        return None
def _v4900_power_high_share(activity):
    try:
        secs = _zone_secs((activity or {}).get("icu_zone_times"))
        if not secs:
            return None
        low, mod, high = bucket_zone_seconds(secs)
        total = low + mod + high
        return (high / total) if total > 0 else None
    except Exception:
        return None
def _v4900_mixed_family_aerobic_contradiction(activity, family):
    """Strong evidence that a zone-mixed ride was physiologically aerobic, not a hard anchor."""
    if str(family or "") != "Tempo / Threshold Mixed":
        return False
    a = activity or {}
    blob = " ".join(str(a.get(k) or "") for k in ("name", "type", "sub_type")).lower()
    if re.search(r"\b(?:race|granfondo|gran fondo|criterium|time trial|threshold|sweet spot|over[- ]?under|tte|vo2|sprint|test|checkpoint)\b", blob, re.I):
        return False
    intensity_factor = _v40_normalized_intensity_factor(a)
    if intensity_factor is not None and intensity_factor >= 0.85:
        return False
    ftp = _num(a.get("icu_ftp") or a.get("ftp"))
    avg_watts = _v4900_activity_average_watts(a)
    avg_ratio = (avg_watts / ftp) if (avg_watts is not None and ftp is not None and ftp > 0) else None
    intervals = [it for it in (a.get("icu_intervals") or []) if isinstance(it, dict)]
    for it in intervals:
        if str(it.get("type") or "WORK").upper() != "WORK":
            continue
        secs = _num(it.get("moving_time") or it.get("elapsed_time"))
        watts = _num(it.get("average_watts"))
        if secs is not None and watts is not None and secs >= 60 and (ftp is None or ftp <= 0 or watts >= 0.88 * ftp):
            return False
    hr_low = _v4900_zone_low_share(a, "icu_hr_zone_times")
    high_share = _v4900_power_high_share(a)
    dec = _num(a.get("decoupling"))
    return bool(
        avg_ratio is not None and avg_ratio <= 0.78
        and hr_low is not None and hr_low >= 0.85
        and (high_share is None or high_share < 0.15)
        and (dec is None or abs(dec) <= 5.0)
    )
def _v4895_activity_quality_truth(activity=None, family=None):
    a = activity or {}
    fam = family or classify_training_family(a)
    if is_hard_training_family(fam):
        if _v4900_mixed_family_aerobic_contradiction(a, fam):
            return {"quality_relevant": False, "basis": "MIXED_FAMILY_AEROBIC_CONTRADICTION"}
        return {"quality_relevant": True, "basis": "TRAINING_FAMILY"}
    blob = " ".join(str(a.get(k) or "") for k in ("name", "type", "sub_type")).lower()
    if str(a.get("sub_type") or "").upper() == "RACE" or any(k in blob for k in ("race", "granfondo", "gran fondo", "criterium", "time trial")):
        return {"quality_relevant": True, "basis": "RACE"}
    explicit_test = bool(re.search(r"\b(?:test|field\s+test|checkpoint|all[- ]?out|maximal|peak[- ]power)\b", blob, re.I))
    submax_test = bool(re.search(r"\b(?:submax(?:imal)?|durability\s+benchmark|lactate|step\s+test)\b", blob, re.I))
    if explicit_test and not submax_test:
        intervals = list(a.get("icu_intervals") or [])
        if not intervals:
            return {"quality_relevant": True, "basis": "EXPLICIT_MAXIMAL_TEST_NAME"}
        ftp = _num(a.get("icu_ftp") or a.get("ftp"))
        for it in intervals:
            if str((it or {}).get("type") or "WORK").upper() != "WORK":
                continue
            secs = _num((it or {}).get("moving_time") or (it or {}).get("elapsed_time")); watts = _num((it or {}).get("average_watts"))
            if secs is not None and secs >= 10 and watts is not None and (ftp is None or watts >= 0.95 * ftp):
                return {"quality_relevant": True, "basis": "EXPLICIT_MAXIMAL_TEST"}
    intensity_factor = _v40_normalized_intensity_factor(a)
    if intensity_factor is not None and intensity_factor >= 0.90:
        return {"quality_relevant": True, "basis": "INTENSITY_FACTOR"}
    # R196.9.45: a 30-min aerobic preload can make whole-ride IF look EASY
    # even when the rider completed two actual 7-min quality repetitions.
    # Use device WORK intervals and observed watts relative to the athlete's
    # own FTP. No title inference, retrospective target or workout issuance.
    ftp = _num(a.get("icu_ftp") or a.get("ftp"))
    if ftp is not None and ftp > 0:
        observed = []
        for it in (a.get("icu_intervals") or []):
            if not isinstance(it, dict) or str(it.get("type") or "WORK").upper() != "WORK":
                continue
            secs = _num(it.get("moving_time") or it.get("elapsed_time"))
            watts = _num(it.get("average_watts"))
            if (secs is not None and watts is not None and math.isfinite(secs)
                    and math.isfinite(watts) and 300 <= secs <= 900
                    and watts >= 0.92 * ftp):
                observed.append((secs, watts))
        if len(observed) >= 2 and any(
                abs(a0[0]-a1[0]) <= max(12.0, 0.08*a0[0])
                for i,a0 in enumerate(observed) for a1 in observed[i+1:]):
            return {"quality_relevant": True,
                    "basis": "MEASURED_REPEATED_5_TO_15_MIN_WORK_ABOVE_92PCT_FTP"}
    return {"quality_relevant": False, "basis": "NOT_HARD"}
def _num(value):
    try:
        return float(value)
    except (TypeError, ValueError):

def _zone_secs(value):
    if not value:
        return []
    out = []
    for z in value:
        if isinstance(z, dict):
            out.append(int(z.get("secs") or 0))
        else:
            try:
                out.append(int(z or 0))
            except (TypeError, ValueError):
                out.append(0)
    return out

def fetch_activity_details(activity_ids):
    ids = [str(x) for x in activity_ids if x]
    if not ids:
        return {}
    runtime = _resolve_intervals_runtime()
    url = (
        f"https://intervals.icu/api/v1/athlete/{runtime['athlete_id']}/activities/"
        f"{','.join(ids)}?intervals=true"
    )
    try:
        r = requests.get(url, headers=get_intervals_headers(runtime), timeout=25)
        r.raise_for_status()
        payload = r.json()
        if isinstance(payload, list):
            found = {str(x.get("id")): x for x in payload if isinstance(x, dict) and x.get("id")}
            if found:
                return found
    except Exception:
        pass
    found = {}
    with ThreadPoolExecutor(max_workers=min(3, len(ids))) as pool:
        futures = {pool.submit(fetch_activity_detail, activity_id): activity_id for activity_id in ids}
        for future in as_completed(futures):
            activity_id = futures[future]
            try:
                detail = future.result()
            except Exception:
                detail = None
            if isinstance(detail, dict):
                found[str(activity_id)] = detail
    return found
def _v80_enrich_garmin_vo2_details(activities, details, max_single_fetches=6):
    """Fill the Garmin VO2 contract from single-activity detail only when needed.

    Intervals' batch detail endpoint is kept as the primary path. If neither the
    activity summary nor batch detail exposes VO2MaxGarmin, probe a small number
    of newest activities through the single-activity endpoint, which is known to
    expose custom activity fields. This is bounded and stops at the first FOUND.
    """
    out={str(k):v for k,v in (details or {}).items()}
    rows=[a for a in (activities or []) if isinstance(a,dict) and a.get("id")]
    def _v91_candidate_device_name(a):
        aid=str((a or {}).get("id") or "")
        detail=out.get(aid) if isinstance(out.get(aid),dict) else {}
        return str((a or {}).get("device_name") or detail.get("device_name") or "").strip()
    rows.sort(
        key=lambda a:(
            1 if "GARMIN" in _v91_candidate_device_name(a).upper() else 0,
            1 if "GARMIN" in str(a.get("source") or "").upper() else 0,
            str(a.get("start_date_local") or ""),
        ),
        reverse=True,
    )
    # If a canonical Garmin value is already present anywhere, do not add calls.
    for a in rows:
        aid=str(a.get("id"))
        for payload,scope in ((a,"ACTIVITY_SUMMARY"),(out.get(aid),"BATCH_ACTIVITY_DETAIL")):
            if isinstance(payload,dict) and _v80_garmin_vo2_contract(payload,scope).get("status")=="FOUND":
                return out,{"status":"FOUND","probe_used":False,"single_fetches":0,"activity_id":aid,"scope":scope}
    fetched=0
    last_status="ABSENT"
    for a in rows:
        if fetched>=max(0,int(max_single_fetches or 0)):break
        aid=str(a.get("id"))
        # Prefer activities that plausibly came from Garmin, but do not require
        # source metadata because older list payloads may omit it.
        single=fetch_activity_detail(aid)
        fetched+=1
        if not isinstance(single,dict):
            last_status="FETCH_FAILED"
            continue
        merged=dict(out.get(aid) or {})
        merged.update(single)
        out[aid]=merged
        contract=_v80_garmin_vo2_contract(merged,"SINGLE_ACTIVITY_DETAIL")
        last_status=contract.get("status") or "ABSENT"
        if last_status=="FOUND":
            return out,{
                "status":"FOUND","probe_used":True,"single_fetches":fetched,
                "activity_id":aid,"scope":"SINGLE_ACTIVITY_DETAIL",
                "path":contract.get("path"),"encoding":contract.get("encoding"),
            }
    return out,{"status":last_status,"probe_used":bool(fetched),"single_fetches":fetched,"activity_id":None,"scope":None}


def _v89_fit_unwrap_original(content):
    """Return raw FIT bytes from Intervals' original-file endpoint.

    Intervals documents /activity/{id}/file as the original file, commonly gzip
    compressed. requests may already decode HTTP content-encoding, so accept both.
    """
    try:
        raw=bytes(content or b"")
    except Exception:
        return b""
    if raw[:2]==b"\x1f\x8b":
        try:
            raw=zlib.decompress(raw,16+zlib.MAX_WBITS)
        except Exception:
            return b""
    return raw

def _v89_fit_garmin_vo2_contract(content, fallback_date=None):
    """Extract Garmin cycling VO2 from native FIT carriers.

    Authority order for Garmin's *display/current* cycling VO2:
      1) message 229 (max_met_data) / field 2 (vo2_max, scale 10), restricted
         to sport=cycling (FIT sport enum 2). This is the FIT SDK's explicit
         VO2-max state record and is not tied to one workout estimate.
      2) message 79 / field 0 profile metmax (scale 1024 MET), retained as a
         profile-side fallback/diagnostic.
      3) message 140 / field 7 post-activity metmax (scale 65536 MET), retained
         as the activity-estimate fallback/diagnostic.

    R98 fixes the R97 mistake of treating message 79/0 as the current Garmin
    cycling VO2 authority.  R97 QA proved 79/0 follows activity-level values
    (e.g. 80 -> 77 -> 79) even when the user's Garmin cycling VO2 remains 80.
    """
    raw=_v89_fit_unwrap_original(content)
    base={
        "status":"ABSENT","value":None,"date":str(fallback_date or "")[:10] or None,
        "source":"GARMIN_ORIGINAL_FIT","scope":"ORIGINAL_FIT_VO2",
        "path":None,"encoding":None,
        "raw_type":"uint","candidate_count":0,
        "message_79_count":0,"message_140_count":0,"message_229_count":0,
        "garmin_display_value":None,"garmin_display_raw_value":None,
        "max_met_value":None,"max_met_raw_value":None,"max_met_sport":None,
        "max_met_sub_sport":None,"max_met_update_time_raw":None,
        "profile_metmax_value":None,"profile_metmax_raw_value":None,
        "post_activity_value":None,"post_activity_raw_value":None,
        "value_policy":"CYCLING_MAX_MET_229_2_FIRST",
    }
    try:
        if len(raw)<12:return {**base,"fit_status":"EMPTY_OR_NOT_FIT"}
        offset=0
        profile_candidates=[]
        post_candidates=[]
        maxmet_candidates=[]
        message_79_count=0
        message_140_count=0
        message_229_count=0
        parsed_files=0
        while offset+12<=len(raw):
            header_size=raw[offset]
            if header_size<12 or offset+header_size>len(raw) or raw[offset+8:offset+12]!=b".FIT":
                break
            data_size=int.from_bytes(raw[offset+4:offset+8],"little",signed=False)
            start=offset+header_size;end=start+data_size
            if end>len(raw):return {**base,"fit_status":"TRUNCATED"}
            defs={};pos=start;parsed_files+=1
            while pos<end:
                rec_header=raw[pos];pos+=1
                compressed=bool(rec_header & 0x80)
                if compressed:
                    local=(rec_header>>5)&0x03
                    definition=defs.get(local)
                    if not definition:return {**base,"fit_status":"MISSING_DEFINITION"}
                    fields=definition["fields"];byteorder=definition["byteorder"];global_num=definition["global_num"]
                    record_fields={}
                    for field_num,size,_base_type in fields:
                        if field_num==253:
                            # compressed timestamp is encoded in the record header
                            continue
                        if pos+size>end:return {**base,"fit_status":"TRUNCATED_RECORD"}
                        chunk=raw[pos:pos+size];pos+=size
                        value=int.from_bytes(chunk[:min(len(chunk),8)],byteorder,signed=False) if chunk else None
                        if global_num==79 and field_num==0 and value is not None:
                            profile_candidates.append(value)
                        elif global_num==140 and field_num==7 and value is not None:
                            post_candidates.append(value)
                        elif global_num==229 and field_num in {0,2,5,6} and value is not None:
                            record_fields[field_num]=value
                    for size in definition["dev_sizes"]:
                        if pos+size>end:return {**base,"fit_status":"TRUNCATED_DEV_RECORD"}
                        pos+=size
                    if global_num==79:message_79_count+=1
                    if global_num==140:message_140_count+=1
                    if global_num==229:
                        message_229_count+=1
                        if 2 in record_fields:maxmet_candidates.append(record_fields)
                    continue
                is_definition=bool(rec_header & 0x40)
                local=rec_header & 0x0F
                if is_definition:
                    if pos+5>end:return {**base,"fit_status":"TRUNCATED_DEFINITION"}
                    pos+=1
                    arch=raw[pos];pos+=1
                    byteorder="little" if arch==0 else "big"
                    global_num=int.from_bytes(raw[pos:pos+2],byteorder,signed=False);pos+=2
                    n_fields=raw[pos];pos+=1
                    fields=[]
                    if pos+n_fields*3>end:return {**base,"fit_status":"TRUNCATED_DEFINITION_FIELDS"}
                    for _ in range(n_fields):
                        field_num=raw[pos];size=raw[pos+1];base_type=raw[pos+2];pos+=3
                        fields.append((field_num,size,base_type))
                    dev_sizes=[]
                    if rec_header & 0x20:
                        if pos>=end:return {**base,"fit_status":"TRUNCATED_DEV_DEFINITION"}
                        n_dev=raw[pos];pos+=1
                        if pos+n_dev*3>end:return {**base,"fit_status":"TRUNCATED_DEV_FIELDS"}
                        for _ in range(n_dev):
                            _field_num=raw[pos];size=raw[pos+1];_dev_idx=raw[pos+2];pos+=3
                            dev_sizes.append(size)
                    defs[local]={"global_num":global_num,"byteorder":byteorder,"fields":fields,"dev_sizes":dev_sizes}
                    continue
                definition=defs.get(local)
                if not definition:return {**base,"fit_status":"MISSING_DEFINITION"}
                global_num=definition["global_num"];byteorder=definition["byteorder"]
                record_fields={}
                for field_num,size,_base_type in definition["fields"]:
                    if pos+size>end:return {**base,"fit_status":"TRUNCATED_RECORD"}
                    chunk=raw[pos:pos+size];pos+=size
                    value=int.from_bytes(chunk[:min(len(chunk),8)],byteorder,signed=False) if chunk else None
                    if global_num==79 and field_num==0 and value is not None:
                        profile_candidates.append(value)
                    elif global_num==140 and field_num==7 and value is not None:
                        post_candidates.append(value)
                    elif global_num==229 and field_num in {0,2,5,6} and value is not None:
                        record_fields[field_num]=value
                for size in definition["dev_sizes"]:
                    if pos+size>end:return {**base,"fit_status":"TRUNCATED_DEV_RECORD"}
                    pos+=size
                if global_num==79:message_79_count+=1
                if global_num==140:message_140_count+=1
                if global_num==229:
                    message_229_count+=1
                    if 2 in record_fields:maxmet_candidates.append(record_fields)
            offset=end+2 if end+2<=len(raw) else end

        maxmet_decoded=[]
        for row in maxmet_candidates:
            raw_value=row.get(2)
            try:
                decoded=_vo2_number(float(raw_value)/10.0)
            except Exception:
                decoded=None
            if decoded is not None:
                maxmet_decoded.append({
                    "raw":raw_value,"value":round(decoded,2),
                    "update_time":row.get(0),"sport":row.get(5),"sub_sport":row.get(6),
                })
        cycling_maxmet=[r for r in maxmet_decoded if r.get("sport")==2]
        if cycling_maxmet:
            # Prefer the newest state if more than one cycling max-met record exists.
            maxmet_choice=max(cycling_maxmet,key=lambda r:(r.get("update_time") is not None,r.get("update_time") or 0))
        else:
            maxmet_choice=None

        profile_decoded=[]
        for raw_value in profile_candidates:
            try:
                decoded=_vo2_number(float(raw_value)*3.5/1024.0)
            except Exception:
                decoded=None
            if decoded is not None:
                profile_decoded.append((raw_value,round(decoded,2)))
        post_decoded=[]
        for raw_value in post_candidates:
            d=_v80_decode_garmin_vo2(raw_value)
            if d.get("valid"):post_decoded.append((raw_value,d.get("value")))

        maxmet_raw=maxmet_choice.get("raw") if maxmet_choice else None
        maxmet_value=maxmet_choice.get("value") if maxmet_choice else None
        profile_raw,profile_value=(profile_decoded[-1] if profile_decoded else (None,None))
        post_raw,post_value=(post_decoded[-1] if post_decoded else (None,None))

        if maxmet_value is not None:
            chosen_value=maxmet_value;chosen_raw=maxmet_raw
            chosen_path="$.fit.message_229.field_2"
            chosen_encoding="GARMIN_FIT_229_2_VO2MAX"
        elif profile_value is not None:
            chosen_value=profile_value;chosen_raw=profile_raw
            chosen_path="$.fit.message_79.field_0"
            chosen_encoding="GARMIN_FIT_79_0"
        else:
            chosen_value=post_value;chosen_raw=post_raw
            chosen_path="$.fit.message_140.field_7" if post_value is not None else None
            chosen_encoding="GARMIN_FIT_140_7" if post_value is not None else None

        common={
            **base,
            "message_79_count":message_79_count,"message_140_count":message_140_count,
            "message_229_count":message_229_count,
            "candidate_count":len(profile_candidates)+len(post_candidates)+len(maxmet_candidates),
            "garmin_display_value":maxmet_value if maxmet_value is not None else profile_value,
            "garmin_display_raw_value":maxmet_raw if maxmet_value is not None else profile_raw,
            "max_met_value":maxmet_value,"max_met_raw_value":maxmet_raw,
            "max_met_sport":maxmet_choice.get("sport") if maxmet_choice else None,
            "max_met_sub_sport":maxmet_choice.get("sub_sport") if maxmet_choice else None,
            "max_met_update_time_raw":maxmet_choice.get("update_time") if maxmet_choice else None,
            "profile_metmax_value":profile_value,"profile_metmax_raw_value":profile_raw,
            "post_activity_value":post_value,"post_activity_raw_value":post_raw,
            "parsed_fit_files":parsed_files,
        }
        if chosen_value is not None:
            return {
                **common,"status":"FOUND","value":chosen_value,"raw_value":chosen_raw,
                "path":chosen_path,"encoding":chosen_encoding,"fit_status":"PARSED",
            }
        if maxmet_candidates or profile_candidates or post_candidates:
            return {
                **common,"status":"INVALID","raw_value":chosen_raw,
                "path":chosen_path,"encoding":chosen_encoding,"fit_status":"PARSED",
            }
        return {**common,"fit_status":"PARSED" if parsed_files else "NOT_FIT"}
    except Exception:
        return {**base,"fit_status":"PARSE_ERROR"}

def _v89_fetch_original_fit_garmin_contract(activity, runtime=None):
    """Fetch one Intervals original activity file with bounded, non-secret diagnostics.

    R90 passes the already-resolved Intervals runtime into worker threads. This is
    critical because Flask's request/session context is not inherited by
    ThreadPoolExecutor workers; resolving credentials inside a worker can therefore
    fail before any HTTP request is made.
    """
    a=activity if isinstance(activity,dict) else {}
    aid=str(a.get("id") or "")
    day=str(a.get("start_date_local") or a.get("date") or "")[:10] or None
    source=str(a.get("source") or "") or None
    device_name=str(a.get("device_name") or "") or None
    file_type=str(a.get("file_type") or "") or None
    base={
        "activity_id":aid or None,"date":day,"source":"GARMIN_ORIGINAL_FIT","scope":"ORIGINAL_FIT_VO2",
        "activity_source":source,"device_name":device_name,"file_type":file_type,
        "http_status":None,"content_type":None,"content_length":None,
        "fetch_reason":None,"error_stage":None,
    }
    if not aid:
        return {**base,"status":"FETCH_FAILED","fetch_reason":"MISSING_ACTIVITY_ID","error_stage":"PRE_REQUEST"}
    if not runtime or not runtime.get("api_key"):
        return {**base,"status":"FETCH_FAILED","fetch_reason":"RUNTIME_UNAVAILABLE","error_stage":"PRE_REQUEST"}
    url=f"https://intervals.icu/api/v1/activity/{aid}/file"
    try:
        headers={**get_intervals_headers(runtime),"Accept":"application/octet-stream"}
        r=requests.get(url,headers=headers,timeout=20)
        http_status=int(getattr(r,"status_code",0) or 0) or None
        content_type=str((getattr(r,"headers",{}) or {}).get("Content-Type") or "")[:120] or None
        content=bytes(getattr(r,"content",b"") or b"")
        diag={**base,"http_status":http_status,"content_type":content_type,"content_length":len(content)}
        if http_status is not None and http_status>=400:
            return {**diag,"status":"FETCH_FAILED","fetch_reason":f"HTTP_{http_status}","error_stage":"HTTP_STATUS"}
        r.raise_for_status()
        c=_v89_fit_garmin_vo2_contract(content,day)
        c.update({k:v for k,v in diag.items() if k not in {"source","scope"}})
        c["activity_id"]=aid
        c["source"]="GARMIN_ORIGINAL_FIT"
        c["scope"]="ORIGINAL_FIT_VO2"
        c["fetch_reason"]="HTTP_OK"
        c["error_stage"]=None
        return c
    except requests.exceptions.RequestException as exc:
        return {**base,"status":"FETCH_FAILED","fetch_reason":type(exc).__name__.upper(),"error_stage":"REQUEST"}
    except Exception as exc:
        return {**base,"status":"FETCH_FAILED","fetch_reason":type(exc).__name__.upper(),"error_stage":"POST_REQUEST"}

def _v89_enrich_garmin_vo2_details(activities, details, max_single_fetches=1, max_fit_fetches=6, user_id=None, runtime=None):
    """Canonical Garmin enrichment: JSON custom field first, original FIT second.

    R90 preserves the R89 bounded fallback but resolves Intervals credentials once
    in the caller/request thread and passes that immutable runtime into workers.
    FIT metadata is preferred so the six-request budget is not wasted on known
    non-FIT activities. No AI provider is involved.
    """
    out,probe=_v80_enrich_garmin_vo2_details(activities,details,max_single_fetches=max_single_fetches)
    if probe.get("status")=="FOUND":
        return out,{**probe,"original_fit_used":False,"original_fit_fetches":0,"original_fit_found":0,
                   "original_fit_candidate_count":0,"original_fit_eligible_count":0,"original_fit_confirmed_fit_count":0,"original_fit_attempts":[]}
    all_rows=[a for a in (activities or []) if isinstance(a,dict) and a.get("id")]
    fit_rows=[];unknown_file_rows=[];non_fit_rows=[]
    for a in all_rows:
        ft=str(a.get("file_type") or "").strip().lower()
        if "fit" in ft:
            fit_rows.append(a)
        elif not ft:
            unknown_file_rows.append(a)
        else:
            non_fit_rows.append(a)

    def _device_name(a):
        aid=str((a or {}).get("id") or "")
        detail=out.get(aid) if isinstance(out.get(aid),dict) else {}
        return str((a or {}).get("device_name") or detail.get("device_name") or "").strip()

    def _is_garmin_device(a):
        return "GARMIN" in _device_name(a).upper()

    def _is_garmin_source(a):
        return "GARMIN" in str((a or {}).get("source") or "").upper()

    def _explicit_non_garmin(a):
        dev=_device_name(a).upper()
        src=str((a or {}).get("source") or "").upper()
        # device_name is authoritative when present. Only fall back to source
        # when device identity is absent.
        if dev:
            return "GARMIN" not in dev
        return src in {"ZWIFT","STRAVA"}

    def _candidate_key(a):
        return (
            1 if _is_garmin_device(a) else 0,
            1 if _is_garmin_source(a) else 0,
            1 if "fit" in str((a or {}).get("file_type") or "").lower() else 0,
            str((a or {}).get("start_date_local") or ""),
        )

    confirmed_garmin=[a for a in fit_rows if _is_garmin_device(a) or _is_garmin_source(a)]
    unknown_device_fit=[a for a in fit_rows if a not in confirmed_garmin and not _explicit_non_garmin(a)]
    unknown_file_garmin=[a for a in unknown_file_rows if _is_garmin_device(a) or _is_garmin_source(a)]
    unknown_file_other=[a for a in unknown_file_rows if a not in unknown_file_garmin and not _explicit_non_garmin(a)]
    skipped_explicit_non_garmin=[a for a in fit_rows+unknown_file_rows if _explicit_non_garmin(a)]

    for bucket in (confirmed_garmin,unknown_device_fit,unknown_file_garmin,unknown_file_other):
        bucket.sort(key=_candidate_key,reverse=True)

    # R91: spend the same bounded six-file budget on Garmin-device evidence first,
    # then only on genuinely unknown provenance. Do not waste it on explicit Zwift
    # originals that cannot contain Garmin native physiological message 140.
    rows=(confirmed_garmin+unknown_file_garmin+unknown_device_fit+unknown_file_other)[:max(0,int(max_fit_fetches or 0))]
    resolved_runtime=runtime
    runtime_reason=None
    if rows and resolved_runtime is None:
        try:
            resolved_runtime=_resolve_intervals_runtime(user_id)
        except Exception as exc:
            runtime_reason=type(exc).__name__.upper()
    results=[]
    if rows:
        if resolved_runtime is None:
            for a in rows:
                c={
                    "status":"FETCH_FAILED","activity_id":str(a.get("id") or ""),
                    "date":str(a.get("start_date_local") or "")[:10] or None,
                    "source":"GARMIN_ORIGINAL_FIT","scope":"ORIGINAL_FIT_VO2",
                    "activity_source":str(a.get("source") or "") or None,"device_name":str(a.get("device_name") or "") or None,"file_type":str(a.get("file_type") or "") or None,
                    "http_status":None,"content_type":None,"content_length":None,
                    "fetch_reason":runtime_reason or "RUNTIME_UNAVAILABLE","error_stage":"RUNTIME_RESOLUTION",
                }
                results.append((a,c))
        else:
            with ThreadPoolExecutor(max_workers=min(3,len(rows))) as pool:
                futures={pool.submit(_v89_fetch_original_fit_garmin_contract,a,resolved_runtime):a for a in rows}
                for future in as_completed(futures):
                    a=futures[future]
                    try:c=future.result()
                    except Exception as exc:
                        c={
                            "status":"FETCH_FAILED","activity_id":str(a.get("id") or ""),
                            "date":str(a.get("start_date_local") or "")[:10] or None,
                            "source":"GARMIN_ORIGINAL_FIT","scope":"ORIGINAL_FIT_VO2",
                            "activity_source":str(a.get("source") or "") or None,"device_name":str(a.get("device_name") or "") or None,"file_type":str(a.get("file_type") or "") or None,
                            "http_status":None,"content_type":None,"content_length":None,
                            "fetch_reason":type(exc).__name__.upper(),"error_stage":"WORKER",
                        }
                    results.append((a,c))
    found=[]
    for a,c in results:
        aid=str(a.get("id") or "")
        if c.get("status")=="FOUND":
            merged=dict(out.get(aid) or {})
            merged["garmin_vo2_contract"]=c
            merged["garmin_vo2_scope"]="ORIGINAL_FIT_VO2"
            out[aid]=merged
            # R91 projection: the activity summary itself is part of the canonical
            # VO2 source population. Project the proven contract here as well so a
            # season-window FIT hit is visible to build_vo2max_trend even when the
            # activity is older than the recent training-block detail window.
            a["garmin_vo2_contract"]=dict(c)
            a["garmin_vo2_scope"]="ORIGINAL_FIT_VO2"
            found.append(c)
    latest=max(found,key=lambda c:str(c.get("date") or "")) if found else None
    statuses={};fetch_reasons={}
    attempts=[]
    for _a,c in results:
        st=str(c.get("status") or "UNKNOWN");statuses[st]=statuses.get(st,0)+1
        fr=str(c.get("fetch_reason") or "NONE");fetch_reasons[fr]=fetch_reasons.get(fr,0)+1
        attempts.append({
            "activity_id":c.get("activity_id"),"date":c.get("date"),"activity_source":c.get("activity_source"),
            "device_name":c.get("device_name"),"file_type":c.get("file_type"),
            "status":c.get("status"),"fetch_reason":c.get("fetch_reason"),
            "error_stage":c.get("error_stage"),"http_status":c.get("http_status"),
            "content_type":c.get("content_type"),"content_length":c.get("content_length"),
            "fit_status":c.get("fit_status"),"message_79_count":c.get("message_79_count"),"message_140_count":c.get("message_140_count"),
            "garmin_display_value":c.get("garmin_display_value"),"post_activity_value":c.get("post_activity_value"),
            "value_policy":c.get("value_policy"),
        })
    attempts.sort(key=lambda x:(str(x.get("date") or ""),str(x.get("activity_id") or "")),reverse=True)
    return out,{
        "schema":"V4.8.76-R91-FIT-1",
        "status":"FOUND" if found else ("FETCH_FAILED" if results and statuses.get("FETCH_FAILED")==len(results) else "ABSENT"),
        "probe_used":True,
        "single_fetches":int(probe.get("single_fetches") or 0),
        "activity_id":latest.get("activity_id") if latest else None,
        "scope":latest.get("scope") if latest else None,
        "path":latest.get("path") if latest else None,
        "encoding":latest.get("encoding") if latest else None,
        "value":latest.get("value") if latest else None,
        "date":latest.get("date") if latest else None,
        "runtime_source":(resolved_runtime or {}).get("source") if isinstance(resolved_runtime,dict) else None,
        "runtime_resolution_status":"OK" if resolved_runtime is not None else (runtime_reason or "UNAVAILABLE"),
        "original_fit_used":bool(results),"original_fit_fetches":len(results),
        "original_fit_found":len(found),"original_fit_status_counts":statuses,
        "original_fit_fetch_reason_counts":fetch_reasons,
        "original_fit_candidate_count":len(all_rows),

        "original_fit_confirmed_fit_count":len(fit_rows),
        "original_fit_garmin_candidate_count":len(confirmed_garmin),

        "original_fit_unknown_device_candidate_count":len(unknown_device_fit)+len(unknown_file_other),
        "original_fit_skipped_explicit_non_garmin_count":len(skipped_explicit_non_garmin),
        "original_fit_unknown_file_metadata_count":len(unknown_file_rows),
        "original_fit_skipped_non_fit_count":len(non_fit_rows),
        "original_fit_attempts":attempts[:max(0,int(max_fit_fetches or 0))],
    }

def _interval_line(interval):
    secs = int(interval.get("moving_time") or interval.get("elapsed_time") or 0)
    if secs < 45:
        return None
    mins = secs / 60.0
    dur = f"{round(mins)}m" if mins >= 2 else f"{secs}s"
    watts = _num(interval.get("average_watts"))
    hr = _num(interval.get("average_heartrate"))
    zone = interval.get("zone")
    bits = [dur]
    if watts is not None:
        bits.append(f"{round(watts)} W")
    if hr is not None:
        bits.append(f"HR {round(hr)}")
    if zone is not None:
        bits.append(f"Z{zone}")
    return " · ".join(bits)
def _zone_summary(zone_secs):
    parts = []
    for idx, secs in enumerate(zone_secs or [], start=1):
        secs = int(secs or 0)
        if secs <= 0:
            continue
        if secs >= 60:
            parts.append(f"Z{idx} {round(secs / 60)}m")
        else:
            parts.append(f"Z{idx} {secs}s")
    return " · ".join(parts[:7])
def _is_heat_name(name):
    return bool(re.search(r"\bHEAT\b", str(name or ""), flags=re.IGNORECASE))
def _v79_normalize_interval_row(row, pos):
    if not isinstance(row, dict):
        return None
    sec = _num(row.get("moving_time") or row.get("elapsed_time") or row.get("duration") or row.get("duration_sec"))
    watts = _num(row.get("average_watts") or row.get("avg_watts") or row.get("power"))
    if sec is None or watts is None or sec <= 0 or watts <= 0:
        return None
    out = dict(row)
    out["_truth_pos"] = int(pos)
    out["moving_time"] = int(round(sec))
    out["average_watts"] = float(watts)
    return out

def _v125_original_fit_laps(content):
    """Read FIT lap (global 19) records from an *original* activity file.

    Independent of ICU automatically detected intervals. Accept original .FIT,
    gzip FIT and ZIP containing one FIT. Never generate lap duration or power from
    the activity name, another interval, the power curve or an athlete's screenshot.

    R128 source hierarchy for lap power:
      1) time-weighted record-level FIT power inside the lap boundaries, when record
         coverage is sufficient and the result is coherent with the native lap field;
      2) lap total_work / timer_time, when available and coherent;
      3) native integer lap avg_power.
    """
    import zipfile
    import io
    try:
        raw=bytes(content or b'')
    except (TypeError,ValueError):
        return []
    if raw[:2]==b'\x1f\x8b':
        try:
            dec=zlib.decompressobj(16+zlib.MAX_WBITS)
            raw=dec.decompress(raw,8_000_001)
            if not dec.eof or len(raw)>8_000_000:
                return []
        except Exception:
            return []
    if raw[:2] == b'PK':
        try:
            with zipfile.ZipFile(io.BytesIO(raw)) as z:
                names=[n for n in z.namelist() if n.lower().endswith('.fit') and not n.endswith('/')]
                if len(names)!=1:
                    return []
                inf=z.getinfo(names[0])
                if inf.file_size>8_000_000:
                    return []
                raw=z.read(names[0])
        except Exception:
            return []
    if len(raw)<14 or len(raw)>8_000_000:
        return []

    def _record_mean(points,start_ts,end_ts,native_watts):
        if start_ts is None or end_ts is None or end_ts<=start_ts:
            return None
        pts=sorted((float(t),float(p)) for t,p in points
                   if t is not None and p is not None and start_ts-2<=float(t)<=end_ts+2 and 0<float(p)<2500)
        if len(pts)<10:
            return None
        # De-duplicate timestamps; last sample at a timestamp wins.
        dedup={}
        for t,p in pts:
            dedup[t]=p
        pts=sorted(dedup.items())
        if len(pts)<10 or pts[0][0]>start_ts+5 or pts[-1][0]<end_ts-5:
            return None
        energy=0.0
        covered=0.0
        for i,(t,p) in enumerate(pts):
            seg_start=max(float(start_ts),t)
            next_t=pts[i+1][0] if i+1<len(pts) else float(end_ts)
            seg_end=min(float(end_ts),next_t)
            dt=max(0.0,seg_end-seg_start)
            if dt<=0:
                continue
            # FIT records can be sparse ("smart recording"). Forward-hold only
            # short gaps; long gaps do not manufacture unseen power.
            if dt>5.0:
                dt=5.0
            energy += p*dt
            covered += dt
        duration=float(end_ts)-float(start_ts)
        if duration<=0 or covered < max(10.0,0.85*duration):
            return None
        mean=energy/covered
        if not (20<=mean<=2500):
            return None
        if native_watts is not None and abs(mean-float(native_watts))>3.0:
            return None
        return mean, len(pts), round(covered,1)

    laps=[]
    records=[]
    offset=0
    try:
        while offset+12<=len(raw):
            hs=raw[offset]
            if hs<12 or offset+hs>len(raw) or raw[offset+8:offset+12]!=b'.FIT':
                break
            size=int.from_bytes(raw[offset+4:offset+8],'little')
            start=offset+hs
            end=start+size
            if end>len(raw):
                return []
            defs={}
            pos=start
            last_timestamp=None
            section_laps=[]
            section_records=[]
            while pos<end:
                h=raw[pos]
                pos+=1
                compressed=bool(h&0x80)
                compressed_offset=(h&0x1F) if compressed else None
                if compressed:
                    local=(h>>5)&3
                    d=defs.get(local)
                    if not d:
                        return []
                else:
                    local=h&15
                    if h&0x40:
                        if pos+5>end:
                            return []
                        pos+=1
                        arch=raw[pos]
                        pos+=1
                        if arch not in (0,1):
                            return []
                        bo='little' if arch==0 else 'big'
                        g=int.from_bytes(raw[pos:pos+2],bo)
                        pos+=2
                        count=raw[pos]
                        pos+=1
                        if pos+3*count>end:
                            return []
                        fields=[]
                        for _ in range(count):
                            fields.append((raw[pos],raw[pos+1],raw[pos+2]))
                            pos+=3
                        dev=[]
                        if h&0x20:
                            if pos>=end:
                                return []
                            nc=raw[pos]
                            pos+=1
                            if pos+3*nc>end:
                                return []
                            for _ in range(nc):
                                dev.append(raw[pos+1])
                                pos+=3
                        defs[local]=(g,bo,fields,dev)
                        continue
                    d=defs.get(local)
                    if not d:
                        return []
                g,bo,fields,dev=d
                row={}
                for fid,sz,typ in fields:
                    if compressed and fid==253:
                        continue
                    if pos+sz>end:
                        return []
                    b=raw[pos:pos+sz]
                    pos+=sz
                    wanted = ((g==19 and fid in {2,7,8,15,19,41,253}) or
                              (g==20 and fid in {7,253}))
                    if not wanted:
                        continue
                    if sz not in {1,2,4,8}:
                        continue
                    v=int.from_bytes(b,bo)
                    if v == (1<<(8*sz))-1:
                        continue
                    row[fid]=v
                for sz in dev:
                    if pos+sz>end:
                        return []
                    pos+=sz

                if compressed:
                    if last_timestamp is not None:
                        base=int(last_timestamp)&~0x1F
                        ts=base+int(compressed_offset)
                        if int(compressed_offset) < (int(last_timestamp)&0x1F):
                            ts+=0x20
                        row[253]=ts
                        last_timestamp=ts
                elif row.get(253) is not None:
                    last_timestamp=int(row[253])

                if g==20:
                    ts=row.get(253)
                    pwr=row.get(7)
                    if ts is not None and pwr is not None and 0<pwr<2500:
                        section_records.append((int(ts),float(pwr)))
                elif g==19:
                    raw_secs=row.get(8,row.get(7))
                    watt=row.get(19)
                    if raw_secs is None or watt is None:
                        continue
                    duration=float(raw_secs)/1000.0
                    avg_watts=float(watt)
                    avg_source='FIT_AVG_POWER'
                    total_work=row.get(41)
                    if total_work is not None and duration>0:
                        work_time_watts=float(total_work)/duration
                        if 20<=work_time_watts<=2500 and abs(work_time_watts-float(watt))<=3.0:
                            avg_watts=work_time_watts

def _v125_enrich_bounded_laps(activities, details, user_id=None):
    """One bounded original-file probe for latest unresolved repeated-work activity.

    A failed probe is explicitly surfaced in the Content Audit; never silently
    claim a damaged automatic interval was an incomplete athlete effort.
    """
    out=details if isinstance(details,dict) else {}
    ordered=sorted((a for a in activities or [] if isinstance(a,dict) and a.get('id')),
                   key=lambda a:str(a.get('start_date_local') or ''),reverse=True)
    for a in ordered[:3]:
        aid=str(a['id']);d=out.get(aid) or a
        if not _v4887_is_cycling_activity(a):continue
        native=d.get('icu_intervals') or []
        work=[r for r in native if isinstance(r,dict) and str(r.get('type') or '').upper()=='WORK']
        if len(work)<3 or _v79_activity_interval_truth(d).get('status')=='RESOLVED':continue
        secs=[_num(r.get('moving_time') or r.get('elapsed_time')) for r in work]
        valid=[v for v in secs if v is not None and v>=120]
        if len(valid)<3:continue
        med=statistics.median(valid)
        if sum(abs(v-med)<=max(15,med*.12) for v in valid)<2:continue

        probe={'status':'ATTEMPTED','source':'ORIGINAL_ACTIVITY_FILE','ai_calls':0,
               'activity_id':aid,'lap_rows_found':0,'http_status':None,
               'record_power_refined_laps':0,'lap_power_sources':[]}
        try:
            rt=_resolve_intervals_runtime(user_id)
            url=f'https://intervals.icu/api/v1/activity/{aid}/file'
            res=requests.get(url,headers={**get_intervals_headers(rt),'Accept':'application/octet-stream'},timeout=13)
            probe['http_status']=int(res.status_code)
            if res.status_code!=200:probe['status']='HTTP_UNAVAILABLE'
            elif len(res.content)>8_000_000:probe['status']='FILE_TOO_LARGE'
            else:
                laps=_v125_original_fit_laps(res.content)
                probe['lap_rows_found']=len(laps)
                probe['record_power_refined_laps']=sum(
                    1 for lap in laps if (lap or {}).get('average_watts_source')=='FIT_RECORD_POWER_TIME_WEIGHTED'
                )
                probe['lap_power_sources']=[str((lap or {}).get('average_watts_source') or '') for lap in laps[:20]]
                if laps:
                    merged['source_laps']=laps
                    merged['_v125_source_lap_provenance']='ORIGINAL_ACTIVITY_FIT_LAP'
                    probe['status']='LAPS_EXTRACTED'
                else:probe['status']='NO_READABLE_FIT_LAPS'
        except Exception as exc:
            probe['status']='PROBE_FAILED'
            probe['error_class']=type(exc).__name__
        merged['_v125_source_lap_probe']=probe
        out[aid]=merged
        break  # One source file per normal Snapshot/QA, even when unsuccessful.
    return out

def _v79_lap_candidates(detail):
    """Find device/source lap arrays without assuming one Intervals payload path."""
    found = []
    seen = set()
    allowed = {"laps", "lap", "devicelaps", "activitylaps", "fitlaps", "sourcelaps", "lapintervals"}
    def walk(node, path=(), depth=0):
        if depth > 4:
            return
        if isinstance(node, dict):
            for key, value in node.items():
                norm = re.sub(r"[^a-z0-9]", "", str(key or "").lower())
                if isinstance(value, list) and (norm in allowed or norm.endswith("laps")):
                    rows = [_v79_normalize_interval_row(x, i) for i, x in enumerate(value)]
                    rows = [x for x in rows if x]
                    if len(rows) >= 3:
                        sig = tuple((x.get("moving_time"), round(float(x.get("average_watts") or 0), 1)) for x in rows)
                        if sig not in seen:
                            seen.add(sig)
                            found.append({"path": ".".join(path + (str(key),)), "rows": rows})
                if isinstance(value, (dict, list)):
                    walk(value, path + (str(key),), depth + 1)
        elif isinstance(node, list):
            for i, value in enumerate(node[:24]):
                if isinstance(value, (dict, list)):
                    walk(value, path + (str(i),), depth + 1)
    walk(detail or {})
    found.sort(key=lambda x: len(x["rows"]), reverse=True)
    return found

def _v79_repeated_work_pattern(raw_rows, ftp=None, source="UNKNOWN", infer_work=False, allow_trimmed=False):
    """Resolve one coherent repeated-work block from ordered interval/lap rows.

    Full-duration repetitions establish the nominal work duration. ICU auto-intervals may contain
    one clipped repetition boundary; such a row is admitted only when power, alternation and
    recovery structure all support the same repeated block. This is deliberately fail-closed.
    """
    rows = []
    for i, raw in enumerate(raw_rows or []):
        row = _v79_normalize_interval_row(raw, i)
        if row:
            rows.append(row)
    if len(rows) < 3:
        return None
    ftp = _num(ftp)
    work_candidates = []
    for row in rows:
        typ = str(row.get("type") or "").upper()
        watts = float(row.get("average_watts") or 0)
        secs = int(row.get("moving_time") or 0)
        if secs < 55:
            continue
        if infer_work:
            if ftp and ftp > 0:
                is_work = watts >= 0.88 * ftp
            else:
                is_work = False
        else:
            is_work = typ == "WORK"
        if is_work:
            work_candidates.append(row)
    if len(work_candidates) < 3:
        return None
    groups = {}
    for row in work_candidates:
        bucket = max(15, int(round(int(row["moving_time"]) / 15.0) * 15))
        groups.setdefault(bucket, []).append(row)
    base_groups = [(bucket, group) for bucket, group in groups.items() if len(group) >= 3]
    if not base_groups:
        return None
    candidates = []
    for nominal, full in base_groups:
        nominal = int(nominal)
        duration_tol = max(6, int(round(0.04 * nominal)))
        full = [r for r in work_candidates if abs(int(r["moving_time"]) - nominal) <= duration_tol]
        if len(full) < 3:
            continue
        work_med = statistics.median(float(r["average_watts"]) for r in full)
        selected = list(full)
        corrected = []
        if allow_trimmed:
            clipped = []
            for row in work_candidates:
                if row in selected:
                    continue
                secs = int(row["moving_time"]); watts = float(row["average_watts"])
                if not (0.65 * nominal <= secs < nominal - duration_tol):
                    continue
                if work_med <= 0 or abs(watts - work_med) / work_med > 0.15:
                    continue
                if ftp and ftp > 0 and watts < 0.88 * ftp:
                    continue
                clipped.append(row)
            # Auto-detection boundary damage should be exceptional, not a new duration tolerance.
            max_clipped = max(1, len(full) // 4)
            clipped = sorted(clipped, key=lambda r: r["_truth_pos"])
            if len(clipped) <= max_clipped:
                selected.extend(clipped)
                corrected = clipped
        selected = sorted(selected, key=lambda r: r["_truth_pos"])
        if len(selected) < 3:
            continue
        recoveries = []
        recovery_watts = []
        coherent = True
        selected_positions = {r["_truth_pos"] for r in selected}
        for left, right in zip(selected, selected[1:]):
            between = [r for r in rows if left["_truth_pos"] < r["_truth_pos"] < right["_truth_pos"]]
            if not between:
                coherent = False; break
            if not infer_work and any(str(r.get("type") or "").upper() == "WORK" and r["_truth_pos"] not in selected_positions for r in between):
                coherent = False; break
            t = sum(int(r.get("moving_time") or 0) for r in between)
            if t <= 0:
                coherent = False; break
            rw = sum(int(r.get("moving_time") or 0) * float(r.get("average_watts") or 0) for r in between) / t
            recoveries.append(t); recovery_watts.append(rw)
        if not coherent or not recoveries:
            continue
        rec_med = statistics.median(recoveries)
        rec_spread = max(recoveries) - min(recoveries)
        rec_power_med = statistics.median(recovery_watts)
        selected_work_med = statistics.median(float(r["average_watts"]) for r in selected)
        if rec_spread > max(75, 0.30 * rec_med):
            continue
        if rec_power_med > 0:
            if ftp and ftp > 0:
                if selected_work_med < max(0.88 * ftp, rec_power_med + 20, 1.08 * rec_power_med):
                    continue
            elif selected_work_med < max(rec_power_med + 35, 1.25 * rec_power_med):
                continue
        first = selected[0]["_truth_pos"]; last = selected[-1]["_truth_pos"]
        seq = []
        correction_rows = []
        for row in rows:
            pos = row["_truth_pos"]
            if pos < first or pos > last:
                continue
            q = dict(row)
            if pos in selected_positions:
                q["type"] = "WORK"
                if row in corrected:
                    observed = int(q.get("moving_time") or 0)
                    q["_observed_moving_time"] = observed
                    q["_canonical_duration_adjusted"] = True
                    q["moving_time"] = nominal
                    correction_rows.append({"position": pos, "observed_secs": observed, "canonical_secs": nominal, "watts": round(float(q.get("average_watts") or 0), 1)})
            else:
                q["type"] = "RECOVERY"
            seq.append(q)
        candidates.append({
            "sequence": seq,
            "work_rows": [r for r in seq if str(r.get("type") or "").upper() == "WORK"],
            "interval_secs": nominal,
            "repeated_work_count": len(selected),
            "source": source,
            "corrections": correction_rows,
            "recovery_median_secs": int(round(rec_med)),
            "confidence": "HIGH" if not correction_rows else "MODERATE",
        })
    if not candidates:
        return None
    return max(candidates, key=lambda x: (x["repeated_work_count"], x["confidence"] == "HIGH", x["interval_secs"]))

def _v4884_dominant_report(values):
    vals = [str(v or "").upper() for v in values if str(v or "").strip()]
    if not vals:
        return {"value": None, "count": 0, "samples": 0, "confidence": "NONE"}
    counts = {}
    for v in vals:
        counts[v] = counts.get(v, 0) + 1
    value, count = max(counts.items(), key=lambda kv: (kv[1], kv[0]))
    samples = len(vals)
    if samples >= 4 and count / samples >= 0.75:
        confidence = "HIGH"
    elif samples >= 2 and count >= 2:
        confidence = "MODERATE"
    else:
        confidence = "LOW"
    return {"value": value, "count": count, "samples": samples, "confidence": confidence, "counts": counts}

def _daily_training_map(activities):
    days = {}
    for a in activities or []:
        ds = str(a.get("start_date_local") or "")[:10]
        if not ds:
            continue
        d = days.setdefault(ds, {"load": 0.0, "heat": False, "names": []})

def build_heat_response_pattern(season_activities, season_wellness):
    wellness = {str(w.get("id")): w for w in season_wellness or [] if w.get("id")}
    days = _daily_training_map(season_activities)
    heat_features = {}
    heat_dates = []
    for a in season_activities or []:
        if not _v4887_is_cycling_activity(a):
            continue
        name = str(a.get("name") or "")
        if not _is_heat_name(name):
            continue
        ds = str(a.get("start_date_local") or "")[:10]
        try:
            d = date.fromisoformat(ds)
        except Exception:
            continue
        heat_dates.append(d)
        rec = heat_features.setdefault(ds, {
            "duration_s": 0.0, "power_num": 0.0, "power_den": 0.0,
            "hr_num": 0.0, "hr_den": 0.0, "dec_num": 0.0, "dec_den": 0.0,
        })
        secs = max(0.0, _num(a.get("moving_time") or a.get("elapsed_time")) or 0.0)
        weight = secs if secs > 0 else 1.0
        rec["duration_s"] += secs
        watts = _num(a.get("icu_average_watts"))
        if watts is None:
            watts = _num(a.get("average_watts"))
        hr = _num(a.get("average_heartrate"))
        if watts is not None and 40 <= watts <= 600:
            rec["power_num"] += watts * weight
            rec["power_den"] += weight
        if hr is not None and 50 <= hr <= 220:
            rec["hr_num"] += hr * weight
            rec["hr_den"] += weight
        dec = _num(a.get("decoupling"))
        if dec is not None and abs(dec) <= 40:
            rec["dec_num"] += dec * weight
            rec["dec_den"] += weight
    heat_dates = sorted(set(heat_dates))
    season_start = season_end = None
    if heat_dates:
        season_end = heat_dates[-1]
        season_start = season_end
        cursor = season_end
        for d in reversed(heat_dates[:-1]):
            if (cursor - d).days > 28:
                break
            season_start = d
            cursor = d
    finalized_heat = {}
    for ds, rec in heat_features.items():
        try:
            d = date.fromisoformat(ds)
        except Exception:
            continue
        if season_start and season_end and not (season_start <= d <= season_end):
            continue
        watts = rec["power_num"] / rec["power_den"] if rec["power_den"] else None
        hr = rec["hr_num"] / rec["hr_den"] if rec["hr_den"] else None
        dec = rec["dec_num"] / rec["dec_den"] if rec["dec_den"] else None
        finalized_heat[ds] = {
            "avg_watts": watts, "avg_hr": hr,
            "power_hr_ratio": (watts / hr) if watts is not None and hr else None,
            "decoupling": dec,
            "duration_min": rec["duration_s"] / 60.0 if rec["duration_s"] else None,
        }

        row = wellness.get(ds) or {}
        if field == "restingHR" and _intervals_temp_flag(row, "tempRestingHR"):
            return None
        return _num(row.get(field))
    def _prior_median(field, exposure_day, minimum=3):
        vals = []
        for back in range(1, 8):
            v = _w_value((exposure_day - timedelta(days=back)).isoformat(), field)
            if v is not None:
                vals.append(v)
        return statistics.median(vals) if len(vals) >= minimum else None
    heat_rows = []
    hrv_samples = []
    controls = []
    if season_start and season_end:
        for ds, info in sorted(days.items()):
            try:
                d = date.fromisoformat(ds)
            except ValueError:
                continue
            if not (season_start <= d <= season_end):
                continue
            next_ds = (d + timedelta(days=1)).isoformat()
            hrv = _w_value(next_ds, "hrv")
            hrv_base = _prior_median("hrv", d)

            if not info.get("heat"):
                if hrv_delta is not None:
                    controls.append({"date": ds, "day": d, "delta_pct": hrv_delta, "load": info["load"]})

            rhr_base = _prior_median("restingHR", d)
            sleep_score = _w_value(next_ds, "sleepScore")
            sleep_score_base = _prior_median("sleepScore", d)
            sleep_secs = _w_value(next_ds, "sleepSecs")
            sleep_secs_base = _prior_median("sleepSecs", d)
            feat = finalized_heat.get(ds) or {}
            row = {
                "date": ds, "day": d, "load": info["load"],
                "hrv_delta_pct": hrv_delta,
                "rhr_delta_bpm": (rhr - rhr_base) if rhr is not None and rhr_base is not None else None,
                "sleep_score_delta": (sleep_score - sleep_score_base) if sleep_score is not None and sleep_score_base is not None else None,
                "sleep_delta_min": ((sleep_secs - sleep_secs_base) / 60.0) if sleep_secs is not None and sleep_secs_base is not None else None,
                "avg_watts": feat.get("avg_watts"), "avg_hr": feat.get("avg_hr"),
                "power_hr_ratio": feat.get("power_hr_ratio"), "decoupling": feat.get("decoupling"),
                "duration_min": feat.get("duration_min"),
            }
            heat_rows.append(row)
            if hrv_delta is not None:

    matched_controls = []
    used_control_dates = set()
    for sample in hrv_samples:
        tol = max(15.0, abs(sample["load"]) * 0.35)
        eligible = [
            c for c in controls
            if c["date"] not in used_control_dates and abs(c["load"] - sample["load"]) <= tol
        ]
        if eligible:
            chosen = min(eligible, key=lambda c: (abs(c["load"] - sample["load"]), abs((c["day"] - sample["day"]).days)))
            matched_controls.append(chosen)
            used_control_dates.add(chosen["date"])

    def _med(rows, key):
        vals = [x.get(key) for x in rows if isinstance(x.get(key), (int, float))]
        return statistics.median(vals) if vals else None
    def _sig(label, early, recent, threshold, better="higher", unit="", decimals=1, change_unit=None):
        if early is None or recent is None:
            return None, 0
        delta = recent - early
        if better == "higher":
            grade = 1 if delta >= threshold else (-1 if delta <= -threshold else 0)
        else:
            grade = 1 if delta <= -threshold else (-1 if delta >= threshold else 0)
        state = "IMPROVING" if grade > 0 else ("WORSENING" if grade < 0 else "STABLE")
        fmt = f"{{:.{decimals}f}}"
        if unit == "%":
            early_text, recent_text = fmt.format(early) + "%", fmt.format(recent) + "%"
        elif unit:
            early_text, recent_text = fmt.format(early) + f" {unit}", fmt.format(recent) + f" {unit}"
        else:
            early_text, recent_text = fmt.format(early), fmt.format(recent)
        cu = change_unit if change_unit is not None else unit
        if cu == "pp":
            change = f"{delta:+.{decimals}f} pp"
        elif cu == "%":
            change = f"{delta:+.{decimals}f}%"
        elif cu:
            change = f"{delta:+.{decimals}f} {cu}"
        else:
            change = f"{delta:+.{decimals}f}"
        return {"label": label, "early": early_text, "recent": recent_text, "change": change, "state": state, "state_class": state.lower()}, grade

            ("Next-morning sleep score", _med(early_rows, "sleep_score_delta"), _med(recent_rows, "sleep_score_delta"), 3.0, "higher", "pts", 1, "pts"),

            grade = 1 if hr_delta <= -2.0 else (-1 if hr_delta >= 2.0 else 0)

        elif er is not None and rr is not None and er > 0:

            grade = 1 if ratio_change >= 2.0 else (-1 if ratio_change <= -2.0 else 0)

                "label": "HEAT power / HR efficiency",

    else:

        improving = sum(1 for g in grades if g > 0)
        worsening = sum(1 for g in grades if g < 0)
        if score >= 2:

            adaptation_headline = f"Recent HEAT exposures show lower physiological cost on {improving} tracked signal{'s' if improving != 1 else ''} versus the early part of this HEAT season."
        elif score <= -2:

        elif improving and worsening:

            adaptation_headline = "Some tolerance signals are improving while others are not, so THE LAB does not collapse the response into a single acclimation claim."
        else:
            adaptation_state = "STABLE"
            adaptation_headline = "Early and recent HEAT responses are broadly similar across the available signals."
        if len(heat_rows) >= 12 and usable_signal_count >= 3:
            adaptation_confidence = "HIGH"
        elif len(heat_rows) >= 8 and usable_signal_count >= 2:
            adaptation_confidence = "MODERATE"
        else:
            adaptation_confidence = "LOW"
        if protocol_comparable is False and adaptation_confidence in {"HIGH", "MODERATE"}:
            adaptation_confidence = "LOW"
    adaptation_class = adaptation_state.lower().replace(" ", "-")

    if median_delta is None:
        acute_headline = "Next-morning HRV response is still assessing."
    else:
        acute_headline = f"Median next-morning HRV is {median_delta:+.1f}% versus its preceding 7-day baseline after HEAT."
    if specificity_state == "HEAT-SPECIFIC HRV PENALTY":
        interpretation = "A repeatable HEAT-specific autonomic cost is present in the current thermal season; adaptation is evaluated separately so a lower recent cost can emerge even when the acute effect remains real."
    elif specificity_state == "POSSIBLE HEAT-SPECIFIC PENALTY":
        interpretation = "A possible HEAT-specific autonomic cost is present, but exposure-to-exposure consistency is not yet strong enough for a firm call."
    elif specificity_state in {"NO HEAT-SPECIFIC HRV PENALTY", "NO CLEAR HEAT-SPECIFIC DIFFERENCE"}:
        interpretation = "HRV can fall after HEAT, but the current same-season comparison does not show a clearly larger penalty than ordinary training of similar load. Adaptation is therefore judged from the longitudinal multi-signal trend, not from HRV suppression alone."
    else:

    if not isinstance(hrv_low_ref, (int, float)):
        historical_hrv = [_w_value(ds, "hrv") for ds in hrv_dates[:-1]]
        learned_hrv = _robust_dynamic_range(historical_hrv)
        hrv_low_ref = learned_hrv[0] if learned_hrv else None
    current_relevance = bool(current_heat and latest_hrv is not None and isinstance(hrv_low_ref, (int, float)) and latest_hrv < hrv_low_ref)
    current_note = (
        "The latest below-baseline HRV was recorded the morning after a named HEAT exposure. Treat it as current thermal-load context and interpret it alongside resting HR, sleep, subjective status and performance."
        if current_relevance else None
    )
    season_exposures = len([d for d in heat_dates if season_start and season_end and season_start <= d <= season_end])
    season_definition = "latest contiguous HEAT block; a >28-day gap starts a new thermal season"
    method_note = "Season-aware observational model only: without ambient/core temperature, sweat rate or plasma-volume measurements, THE LAB can identify tolerance patterns but cannot claim a measured heat-acclimation mechanism."
    return {
        "available": bool(season_exposures),
        "confidence": specificity_confidence, "confidence_class": specificity_class,
        "samples": n, "negative_count": len(neg), "consistency_pct": round(consistency * 100),
        "median_delta_pct": round(median_delta, 1) if median_delta is not None else None,
        "control_delta_pct": round(control_delta, 1) if control_delta is not None else None,
        "control_samples": len(comparator_pool), "control_mode": comparator_mode,
        "headline": acute_headline, "acute_headline": acute_headline, "comparator": comparator,

        "adaptation_samples": len(heat_rows), "adaptation_early_samples": len(early_rows), "adaptation_recent_samples": len(recent_rows),

        "interpretation": interpretation, "comparability_note": comparability_note, "method_note": method_note,
        "current_relevance": current_relevance, "current_note": current_note,
        "confidence_note": specificity_headline,
    }

def build_aerobic_efficiency(season_activities, now=None):
    now = now or get_rome_now()
    today = now.date()
    recent_start = today - timedelta(days=13)
    prior_start = today - timedelta(days=41)
    prior_end = today - timedelta(days=14)
    candidates = []
    recent_diag = {
        "z2_detected": 0, "accepted": 0, "too_short": 0, "missing_power_hr": 0,
        "implausible_power_hr": 0, "zone_mix": 0, "variability": 0, "heat": 0,
    }
    for a in season_activities or []:
        if not _v4887_is_cycling_activity(a):
            continue
        if classify_training_family(a) != "Endurance / Z2":
            continue

        try:
            d = date.fromisoformat(raw)
        except Exception:
            continue
        is_recent = recent_start <= d <= today
        if is_recent:
            recent_diag["z2_detected"] += 1
        if _is_heat_name(a.get("name")):
            if is_recent:
                recent_diag["heat"] += 1
            continue
        secs = _num(a.get("moving_time") or a.get("elapsed_time"))
        if secs is None or secs < 2400:
            if is_recent:
                recent_diag["too_short"] += 1
            continue
        watts = _num(a.get("icu_average_watts"))
        if watts is None:
            watts = _num(a.get("average_watts"))
        hr = _num(a.get("average_heartrate"))
        if watts is None or hr is None:
            if is_recent:
                recent_diag["missing_power_hr"] += 1
            continue
        if not (80 <= watts <= 500 and 80 <= hr <= 210):
            if is_recent:
                recent_diag["implausible_power_hr"] += 1
            continue
        zone_secs = _zone_secs(a.get("icu_zone_times"))
        low_share = None
        if zone_secs:
            try:
                low, mod, high = bucket_zone_seconds(zone_secs)
                zone_total = low + mod + high
                if zone_total > 0:
                    low_share = low / zone_total
