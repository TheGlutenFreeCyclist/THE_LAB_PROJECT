def _v195_analyze(prior, current, now=None):
    """Observe contradictions; NEVER change coaching decisions or athletes' data.

    Accepts the *data* dictionaries from consecutive persisted Snapshots.
    A finding is not proof of a physiological error. Policy: precise invariants
    can be ALERT; potentially legitimate coaching changes are WATCH only.
    """
    previous = prior if isinstance(prior, dict) else {}
    incoming = current if isinstance(current, dict) else {}
    date_now = _v195_date(now.isoformat() if isinstance(now, (datetime, date)) else now)
    issues = []
    def add(code, severity, **facts):
        issues.append({'code':code, 'severity':severity, 'facts':facts})

    earlier = {_v195_id(r):r for r in _v195_rows(previous) if _v195_id(r)}
    curr_rows = _v195_rows(incoming)
    for row in curr_rows:
        fp = _v195_id(row)
        if not fp or fp not in earlier:
            continue
        old = str(earlier[fp].get('prescription_status') or '').upper()
        new = str(row.get('prescription_status') or '').upper()
        if old in R195_TERMINAL and new in {'ACTIVE', 'PLANNED', 'MISSED_UNEXECUTED'}:
            add('TERMINAL_PRESCRIPTION_REACTIVATED', 'ALERT', fingerprint=fp,
                earlier_status=old, current_status=new, workout_id=_v195_session_id(row))

    active = {}
    for row in curr_rows:
        if str(row.get('prescription_status') or '').upper() != 'ACTIVE':
            continue
        k = (str(row.get('date') or '')[:10], _v195_session_id(row))
        if not k[0] or not k[1]:
            continue
        active.setdefault(k, set()).add(_v195_id(row) or str(row.get('recorded_at_local') or 'UNKNOWN'))
    for (day, wid), fingerprints in sorted(active.items()):
        if len(fingerprints)>1:
            add('MULTIPLE_ACTIVE_PRESCRIPTION_IDENTITIES', 'WATCH',
                date=day, workout_id=wid, identities=len(fingerprints),
                note='Different period labels might describe one opportunity; review before any automatic consolidation.')

    # Consistency between athlete-visible target and canonical watt contract.
    for row in incoming.get('next_sessions') or []:
        if not _v195_hard(row):
            continue
        wid = _v195_session_id(row)
        if not wid:
            continue
        band = row.get('power_prescription') or {}
        if band.get('state') == 'PRESCRIBED':
            if (row.get('prescribed_power_low') != band.get('low_watts') or
                row.get('prescribed_power_high') != band.get('high_watts')):
                add('POWER_CONTRACT_MISMATCH', 'ALERT', date=str(row.get('date') or '')[:10], workout_id=wid)
        elif (str(row.get('stimulus_architecture_state') or '') == 'NOVA_CANONICAL_LIBRARY'
              and not wid.startswith('SPR_') and not _v19693_approved_non_watt_audit(row)):
            add('HARD_REPEAT_WITHOUT_POWER_CONTRACT', 'WATCH', date=str(row.get('date') or '')[:10], workout_id=wid)

    # Cross-snapshot changes can be legitimate; an unchanged recorded core input
    # with a changed workout is a stronger (but still reviewable) signal.
    pp, cp = _v195_primary(previous), _v195_primary(incoming)
    if pp and cp and pp != cp:
        prior_fingerprint = str((previous.get('nova_replay_fixture') or {}).get('input_fingerprint') or '')
        curr_fingerprint = str((incoming.get('nova_replay_fixture') or {}).get('input_fingerprint') or '')
        if prior_fingerprint and curr_fingerprint and prior_fingerprint == curr_fingerprint:
            add('PLAN_CHANGED_WITH_IDENTICAL_REPLAY_INPUT', 'ALERT', from_plan=pp, to_plan=cp)
        else:
            # A documented change in decision/gate/goal or completed activities can
            # justify replanning. Do not label those cases as anomalies automatically.
            prior_action = str((previous.get('nova_decision') or {}).get('action') or '')
            curr_action = str((incoming.get('nova_decision') or {}).get('action') or '')
            prior_dim = str((previous.get('nova_decision') or {}).get('dimension') or '')
            curr_dim = str((incoming.get('nova_decision') or {}).get('dimension') or '')
            prior_gate = str((previous.get('nova_prescription') or {}).get('quality_window_state') or '')
            curr_gate = str((incoming.get('nova_prescription') or {}).get('quality_window_state') or '')
            if (prior_action,prior_dim,prior_gate) == (curr_action,curr_dim,curr_gate):
                add('PLAN_CHANGED_WITHOUT_OBVIOUS_DECISION_GATE_DELTA', 'WATCH',
                    from_plan=pp, to_plan=cp,
                    note='A new athlete activity or another input change may still justify this change.')

    # Match a competition priority against any second hard prescription on the same day.
    sessions = [r for r in incoming.get('next_sessions') or [] if isinstance(r, dict)]
    race_days = {str(r.get('date') or '')[:10] for r in sessions if
        str(r.get('restriction') or '').upper()=='RACE' or r.get('is_race')}
    for row in sessions:
        if _v195_hard(row) and str(row.get('date') or '')[:10] in race_days:
            add('EXTRA_HARD_SESSION_ON_RACE_DAY', 'ALERT', date=str(row.get('date') or '')[:10], workout_id=_v195_session_id(row))

    # A dated ACTUAL report must not reissue today's already completed workout.
    fulfilled = {(str(r.get('actual_date') or '')[:10], _v195_session_id(r))
        for r in curr_rows if str(r.get('prescription_status') or '').upper()=='FULFILLED' and r.get('executed_activity_id')}
    for row in sessions:
        key = (str(row.get('date') or '')[:10], _v195_session_id(row))
        if _v195_hard(row) and key in fulfilled:
            add('EXECUTED_SAME_DAY_HARD_REISSUED', 'ALERT', date=key[0], workout_id=key[1])

    seen = set()
    distinct = []
    for item in issues:
        token = (item['code'], json.dumps(item['facts'], sort_keys=True, default=str))
        if token not in seen:
            distinct.append(item)
            seen.add(token)
    state = 'ALERT' if any(x['severity']=='ALERT' for x in distinct) else ('WATCH' if distinct else 'PASS')
    digest = hashlib.sha256(json.dumps({'previous':pp,'current':cp,'issues':distinct},
        sort_keys=True, default=str).encode('utf-8')).hexdigest()[:20]
    return {'schema':R195_SCHEMA, 'mode':'SHADOW_READ_ONLY', 'state':state,
            'issues':distinct[:24], 'issue_count':len(distinct),
            'previous_quality_plan':pp, 'current_quality_plan':cp,
            'comparison_available':bool(previous), 'fingerprint':digest,
            'policy':'Diagnostics only. Never changes workout mechanics, recovery gates, retention, or Snapshot persistence.'}

def _v195_attach_python_qa_audit(data, owner_user_id=None, now=None):
    """Compare an ephemeral Python QA candidate against the same athlete's saved Snapshot.

    Performs ONE read of existing latest_report, with explicit owner scoping
    enforced by _data_owner_user_id. No writes, no provider calls, no mutation
    of training/coaching fields. Failures stay diagnostic, never break QA.
    R115 replay remains separate and does not call this function.
    """
    if not isinstance(data, dict):
        return None
    try:
        # Explicit owner ID is essential for admin remote QA: the current
        # authenticated actor is the admin, while g scopes reads to the beta athlete.
        owner_id = _data_owner_user_id(owner_user_id)
        baseline = load_latest_report(include_payload=True, owner_user_id=owner_id)
        previous = baseline.get('data') if isinstance(baseline, dict) else None
        previous = previous if isinstance(previous, dict) else None
        diagnostic = _v195_analyze(previous, data, now=now or get_rome_now())
        diagnostic['evaluation_context'] = 'EPHEMERAL_PYTHON_QA'
        diagnostic['baseline_source'] = 'SAME_ATHLETE_LATEST_SAVED_SNAPSHOT' if previous else 'NO_SAVED_SNAPSHOT'
        diagnostic['baseline_generated_at_local'] = (baseline.get('generated_at_local')
            if previous and isinstance(baseline, dict) else None)
        diagnostic['qa_read_only'] = True
        # A first-run QA may check intra-Snapshot invariants, but cannot claim
        # a successful longitudinal comparison with a nonexistent predecessor.
        if not previous and diagnostic.get('state') == 'PASS':
            diagnostic['state'] = 'UNASSESSED'
            diagnostic['reason_code'] = 'NO_SAVED_BASELINE_FOR_COMPARISON'
        data['decision_integrity_audit'] = diagnostic
    except Exception as exc:
        data['decision_integrity_audit'] = {
            'schema': R195_SCHEMA, 'mode': 'SHADOW_READ_ONLY',
            'state': 'UNASSESSED', 'comparison_available': False,
            'issues': [{'code': 'QA_SHADOW_VALIDATOR_UNAVAILABLE', 'severity': 'WATCH',
                        'facts': {'error_class': type(exc).__name__}}],
            'issue_count': 1, 'evaluation_context': 'EPHEMERAL_PYTHON_QA',
            'baseline_source': 'UNAVAILABLE', 'qa_read_only': True,
            'policy': 'Diagnostic failure never changes QA candidate or saved Snapshot.'}
    return data['decision_integrity_audit']

def _v1952_reconcile_ledger(ledger, sessions, now=None, decision_action=None):
    """Pure copy-on-write reconciliation. Never select/modify athlete workouts.

    - Retire older exact same-day, same-time, same-mechanics reissues.
    - Annotate quality displaced by race-priority recovery, preserving its
      historical mechanics but preventing automatic missed-hard carry.
    - Do not infer race priority from merely observing a recent Zwift race.
    """
    if not isinstance(ledger,dict): return ledger
    out = copy.deepcopy(ledger)
    hist = [x for x in (out.get('prescription_history') or []) if isinstance(x,dict)]
    now = now or get_rome_now()
    stamp = now.isoformat(timespec='seconds')
    groups={}
    for r in hist:
        if str(r.get('prescription_status') or '').upper()!='ACTIVE':continue
        key=_v1952_equiv(r)
        if key is not None:groups.setdefault(key,[]).append(r)
    retired=[]
    for rows in groups.values():
        if len(rows)<2:continue
        # Unique, causally later issue required; ambiguous ties stay intact.
        issued=[]
        for r in rows:
            try:dt=datetime.fromisoformat(str(r.get('recorded_at_local') or '').replace('Z','+00:00'))
            except (ValueError,TypeError):dt=None
            if dt is None or not r.get('prescription_fingerprint'):break
            issued.append((dt,r))
        if len(issued)!=len(rows):continue
        issued.sort(key=lambda p:p[0],reverse=True)
        if len(issued)>1 and issued[0][0]==issued[1][0]:continue
        latest=issued[0][1]; winner=str(latest.get('prescription_fingerprint'))
        for _,r in issued[1:]:
            if str(r.get('prescription_fingerprint'))==winner:continue
            r.update(prescription_status='SUPERSEDED',superseded_at_local=stamp,
                     superseded_by_fingerprint=winner,superseded_by_workout_id=_v1952_wid(latest),
                     supersession_reason='R1952_EXACT_SAME_TIME_MECHANICS_LATER_ISSUANCE',
                     supersession_schema=R1952_SCHEMA)
            retired.append(str(r.get('prescription_fingerprint')))
    suspended=[]
    if str(decision_action or '').upper()=='RACE_PRIORITY':
        accepted=_v1952_accepted_hard(sessions)
        easy_days=_v1952_withdrawn_days(sessions,decision_action)
        for row in hist:
            if str(row.get('prescription_status') or '').upper() not in {'ACTIVE','MISSED_UNEXECUTED'}:continue
            day=_v1952_day(row)
            if (not day or day not in easy_days or not _v195_hard(row)
                or (day,_v1952_wid(row)) in accepted):continue
            row['automatic_carry_eligibility']='SUSPENDED_BY_RACE_PRIORITY'
            row['effective_execution_authority']='NOT_IN_CURRENT_ATHLETE_PLAN'
            row['race_displacement_schema']=R1952_SCHEMA
            suspended.append(str(row.get('prescription_fingerprint') or ''))
    # A later confirmed hard recommendation for the exact same slot is the
    # only way to release a suspended plan; a mere change to TRAIN is not enough.
    accepted=_v1952_accepted_hard(sessions)
    for row in hist:
        if str(row.get('automatic_carry_eligibility') or '')=='SUSPENDED_BY_RACE_PRIORITY':
            if (_v1952_day(row),_v1952_wid(row)) in accepted:
                row.pop('automatic_carry_eligibility',None)
                row.pop('effective_execution_authority',None)
                row.pop('race_displacement_schema',None)
            elif (_v1952_date(_v1952_day(row)) is not None and
                  _v1952_date(_v1952_day(row)) < now.date() and
                  str(row.get('prescription_status') or '').upper() in {'ACTIVE','MISSED_UNEXECUTED'}):
                row['prescription_status']='WITHDRAWN'
                row['withdrawn_at_local']=stamp
                row['withdrawal_reason']='R1952_EXPLICIT_RACE_RECOVERY_PROJECTION_SUPERSEDED_EXECUTION'
    out['prescription_history']=hist[-32:]
    a=out.get('primary_anchor')
    if isinstance(a,dict):
        ad=_v1952_day(a)
        if (str(decision_action or '').upper()=='RACE_PRIORITY'
            and ad in _v1952_withdrawn_days(sessions,decision_action)
            and str(a.get('status') or '').upper() in {'PLANNED','ACTIVE'}):
            a['effective_execution_authority']='NOT_IN_CURRENT_ATHLETE_PLAN'
            a['effective_status']='RACE_RECOVERY_SUSPENDED'
            a['race_displacement_schema']=R1952_SCHEMA
        elif (ad,_v1952_wid(a)) in accepted:
            a.pop('effective_execution_authority',None)
            a.pop('effective_status',None)
            a.pop('race_displacement_schema',None)
    prior_integrity=out.get('race_reissue_integrity') or {}
    # Stable cumulative telemetry, not a counter that disappears on the next Snapshot.
    prior_retired=[x for x in (prior_integrity.get('reissue_fingerprints_retired') or []) if isinstance(x,str)]
    all_retired=list(dict.fromkeys(prior_retired+retired))[-32:]
    pending_suspended=sum(1 for r in hist if
        str(r.get('automatic_carry_eligibility') or '')=='SUSPENDED_BY_RACE_PRIORITY'
        and str(r.get('prescription_status') or '').upper() in {'ACTIVE','MISSED_UNEXECUTED'})
    # Normal athletes with no ambiguous history or race displacement must
    # keep the exact R195.1 ledger and incur ZERO additional database writes.
    if (not all_retired and not pending_suspended and not prior_integrity
        and out.get('prescription_history',[]) == ledger.get('prescription_history',[])
        and out.get('primary_anchor') == ledger.get('primary_anchor')):
        return ledger
    out['race_reissue_integrity']={
        'schema':R1952_SCHEMA, 'exact_reissues_retired':len(all_retired),
        'reissue_fingerprints_retired':all_retired,
        'race_displaced_prescriptions':pending_suspended,
        'rule':'A race is real load. Forecasted recovery cancels automatic hard-debt carry; future recovery can be reassessed. No race is manufactured from training labels.'}
    return out

        'issues':report['issues'],'race_reissue':report.get('historical_anchor_authority')},
        sort_keys=True,default=str).encode()).hexdigest()[:20]

def _v1953_signal(watch, blocks, progression, now=None):
    """One evidence-based fatigue signal, never a medical assessment or veto."""
    out = {"schema":R1953_SCHEMA,"active":False,"state":"CLEAR",
           "reason":"NO_CONCORDANT_VERIFIED_TREND", "rpe_comparison":None,
           "autonomic_persistent":False,"recovery_variant_count":0,
           "policy":"HRV alone never changes a workout; only documented comparable subjective cost may enable a limited rest variant."}
    w=watch if isinstance(watch,dict) else {}
    if not (w.get("active") and w.get("persistent") and w.get("state")=="AUTONOMIC_TREND_WATCH"):
        return out
    out["autonomic_persistent"]=True
    comp=(progression or {}).get("comparison") or {}
    if not isinstance(comp,dict):return out
    previous=comp.get("previous_subjective_feedback") or {}
    current=(progression or {}).get("current_subjective_feedback") or {}
    # The R150 source explicitly requires valid structured feedback for both.
    if not all(isinstance(f,dict) and f.get("source")=="STRUCTURED_SESSION_FEEDBACK"
               and f.get("rpe") is not None for f in (previous,current)):
        return out
    try:
        a=float(comp.get("current_mean_watts")); b=float(comp.get("previous_mean_watts"))
        previous_rpe=int(previous["rpe"]); current_rpe=int(current["rpe"])
        comparable= b>0 and abs(a-b)/b <= 0.05 and current_rpe-previous_rpe>=2
        # A compared workout should be recent rather than a remote historical peak.
        day=str((progression or {}).get("date") or "")[:10]
        date_now=(now or get_rome_now()).date()
        age=(date_now-date.fromisoformat(day)).days
        comparable=comparable and 0<=age<=21
    except (TypeError,ValueError,KeyError,OverflowError):return out
    if not comparable:return out
    out.update({"active":True,"state":"WATCH","reason":"PERSISTENT_HRV_AND_COMPARABLE_RPE_RISE",
                "rpe_comparison":{"previous":previous_rpe,"current":current_rpe,
                                  "power_previous_w":round(b,1),"power_current_w":round(a,1)},
                "display_title":"Recovery watch"})
    return out


def _v1953_coach_prepare_variant(prescription, signal, coach_clock):
    """Nova selects a canonical rest variant BEFORE compilation, never after.

    The old post-compiler adapter changed the published workout identity and
    left the coach/compiler contract describing a different session. Variant
    choice now occurs once, on an already-authorized raw coach prescription.
    If its timing cannot be proven, keep the original dose and surface a WATCH.
    """
    if not isinstance(prescription, dict) or not (signal or {}).get('active'):
        return prescription
    if (str(prescription.get('status') or '').upper() != 'PRESCRIBED'
            or prescription.get('hard_prescription_available') is not True):
        return prescription
    try:
        index = int(prescription.get('selected_slot_index'))
        raw = (prescription.get('raw_sessions') or [])[index]
        slot = (coach_clock or {}).get('next_slots', [])[index]
        base = str(prescription.get('workout_id') or '')
        variant = _R1953_VARIANTS.get(base)
        template = next((t for t in _NOVA_WORKOUT_LIBRARY if t.get('id') == base), None)
        alternative = next((t for t in _NOVA_WORKOUT_LIBRARY if variant and t.get('id') == variant[0]), None)
        if not (variant and template and alternative and isinstance(raw,dict)
                and isinstance(slot,dict) and raw.get('_tl_nova_workout_id') == base):
            return prescription
        # Reject inconsistent library variants as a coach-side WATCH, not
        # a production AssertionError that would abort snapshot generation.
        if (template.get('work_s') != alternative.get('work_s')
                or template.get('reps') != alternative.get('reps')):
            return prescription
        core = _v19697_workout_core_seconds(alternative)
        if not core: return prescription
        original = str(raw.get('main_set') or '')
        old = f'with {variant[1]} min '
        if old not in original: return prescription
        declared = (_parse_duration_minutes(raw.get('duration')) or
            _rhythm_num(slot.get('duration_hint_min')) or
            _parse_duration_minutes(slot.get('duration_hint_label')))
        if declared is None or not math.isfinite(float(declared)) or declared <= 0:
            return prescription
        # Original expressed prior-load work must stay exactly as chosen by
        # Nova. Otherwise use a conservative 15 minute preparation minimum.
        preparation = (30 if re.search(r'30\s*min\s+low[- ]?Z2\s+aerobic\s+preload',original,re.I)
                       else 20 if re.search(r'20\s*min\s+progressive\s+aerobic\s+lead-in',original,re.I)
                       else 15)
        if float(declared)*60 + 0.01 < core[0] + preparation*60 + 5*60:
            return prescription
        out = copy.deepcopy(prescription)
        updated = dict(raw)
        updated['main_set'] = original.replace(old,f'with {variant[2]} min ',1)
        updated['_tl_nova_workout_id'] = variant[0]
        updated['_tl_nova_contract_id'] = variant[0]
        updated['_tl_nova_template_signature'] = _v109_template_signature(alternative)
        updated['_tl_nova_recovery_variant'] = R1953_SCHEMA
        out['raw_sessions'][index] = updated
        out['workout_id'] = variant[0]
        out['contract_id'] = variant[0]
        library = dict(out.get('workout_library') or {})
        library.update({'selected_id':variant[0],
            'selected_signature':updated['_tl_nova_template_signature'],
            'derived_from':base,'recovery_variant_rule':R1953_SCHEMA})
        out['workout_library'] = library
        out['recovery_variant_audit'] = {'schema':R1953_SCHEMA,'state':'COACH_SELECTED_BEFORE_COMPILER',
            'count':1,'workout_ids':[variant[0]],'original_workout_id':base,
            'source':'CONCORDANT_RECOVERY_SIGNAL'}
        return out
    except (TypeError, ValueError, KeyError, IndexError, OverflowError):
        return prescription

R1955_SCHEMA = "V4.9.79-R195.5-CHECKPOINT-TRUTH-1"

def _v1955_checkpoint_comparison(events, result_watts, duration_secs, completed_day, observed_event=None, mode="MAXIMAL"):
    """Use *earlier observed performance* only; never compare with prescribed watts.

    If no comparable earlier value is proven, the formal test still closes the
    validation question, but a numerical improvement is not asserted.
    """
    try:
        result = float(result_watts)
        secs = float(duration_secs)
    except (TypeError, ValueError):
        return {"previous_watts":None,"delta_w":None,"outcome":"BENCHMARK_COMPLETED","pb_confirmed":False,"reference_source":None}
    if not math.isfinite(result) or result <= 0 or not math.isfinite(secs) or secs <= 0:
        return {"previous_watts":None,"delta_w":None,"outcome":"BENCHMARK_COMPLETED","pb_confirmed":False,"reference_source":None}
    valid_prior = []
    try:
        completed = date.fromisoformat(str(completed_day or "")[:10])
    except (TypeError, ValueError):
        completed = None
    if completed is not None:
        for row in events or []:
            if not isinstance(row,dict) or not row.get("activity_linked"):
                continue
            try:
                d = date.fromisoformat(str(row.get("date") or "")[:10]);
                sec = float(row.get("secs") or 0); w = float(row.get("watts"))
            except (TypeError, ValueError):
                continue
            if d >= completed or abs(sec-secs)>max(2.0,min(30.0,secs*.05)) or not math.isfinite(w) or w<=0:
                continue
            # A real past effort or formal test, not a predicted model wattage.
            if str(row.get("kind") or "").upper() in {"PB","VALIDATION"}:
                valid_prior.append(w)
    native_previous = None
    ev = observed_event if isinstance(observed_event,dict) else {}
    # Only explicit, activity-linked measurements may offer a prior observed PB.
    if ev.get("activity_linked") and str(ev.get("kind") or "").upper() == "PB":
        p = _rhythm_num(ev.get("previous_watts")); delta = _rhythm_num(ev.get("delta_w"))
        if p is not None and p>0 and delta is not None and delta>0:
            native_previous = float(p)
    if native_previous is not None:
        valid_prior.append(native_previous)
    previous = max(valid_prior) if valid_prior else None
    delta = round(result-previous,1) if previous is not None else None
    gain = bool(str(mode or "").upper() == "MAXIMAL" and previous is not None and result>previous+0.05)
    # 'PB confirmed' requires an explicit native/observed prior-delta record.
    pb = bool(gain and native_previous is not None and result>native_previous+0.05)
    return {
        "previous_watts":round(previous,1) if previous is not None else None,
        "delta_w":delta,
        "outcome":"VALIDATED_GAIN" if gain else "BENCHMARK_COMPLETED",
        "pb_confirmed":pb,
        "reference_source":"PRIOR_OBSERVED_PERFORMANCE" if previous is not None else None,
    }


def _v1955_sync_checkpoint_to_stage(roadmap, now=None):
    """Keep evidence-gated stage and checkpoint copy/permission in one state.

    Recovery, health, race priority and the completed-day absorption veto win;
    this never creates a new slot, forces testing or cancels physical load.
    """
    road = roadmap if isinstance(roadmap,dict) else None
    if road is None or not road.get("available"):
        return road
    try:
        stage=int(road.get("stage_index") or 0)
    except (TypeError,ValueError):
        return road
    test = road.get("test")
    if not isinstance(test,dict):
        return road
    test = dict(test)
    old_state = str(test.get("state") or "").upper()
    quality = road.get("quality_window") or {}
    q_state = str(quality.get("state") or "").upper()
    # No change to an explicit independent safety veto or completed validation.
    if old_state in {"COMPLETED","ON HOLD","RACE PRIORITY"} or q_state in {"ON HOLD","RECOVERY HOLD","RACE PRIORITY","LEARNING"}:
        return road
    if stage>=3:
        slot = str(quality.get("preferred") or quality.get("earliest") or "").strip()
        if slot and q_state in {"WATCH","OPEN"} and quality.get("hard_session_allowed") is not False:
            test.update({"state":"WINDOW OPEN","when":slot,
                         "reason":"The evidence-gated roadmap has reached validation. The checkpoint may replace the next eligible quality window, conditional on recovery and spacing; a new PR is not required to close it."})
            src=str(test.get("reference_source") or "observed curve")
            ref=_rhythm_num(test.get("reference_watts"))
            pacing=f"Current {src} reference: {ref:.0f} W. " if ref is not None else ""
            test["instruction"]=pacing+"Use this as pacing context, not a minimum, cap, or pass mark. Complete the exact-duration test; a plateau also validates completion."
        else:
            test.update({"state":"TEST SOON","when":"When the next valid quality opportunity is exposed",
                         "reason":"The evidence-gated roadmap has reached validation, but no eligible quality opportunity is currently confirmed. Recovery and timing still decide when."})
    elif stage==2 and old_state in {"BASELINE BUILD","BUILD FIRST","TEST SOON","WINDOW OPEN"}:
        test.update({"state":"TEST SOON","when":"After the next successful race-transfer exposure and a fresh quality window",
                     "reason":"Complete the remaining race-transfer question, then validate without demanding a PR."})
    elif stage==1 and old_state in {"BASELINE BUILD","TEST SOON","WINDOW OPEN"}:
        test.update({"state":"BUILD FIRST","when":"After representative repeatability-focused quality exposures",
                     "reason":"Repeatability evidence is still being developed before validation."})
    elif stage==0 and old_state in {"BUILD FIRST","TEST SOON","WINDOW OPEN"}:
        test.update({"state":"BASELINE BUILD","when":"After representative goal-specific training has accumulated",
                     "reason":"Build sufficient comparable work before the next checkpoint."})
    else:
        return road
    test["checkpoint_sync_schema"]=R1955_SCHEMA
    road["test"]=test
    return road

    return result

# R195.6 · OBSERVED POWER UNDER PRIOR WORK (NEVER FRESH-PB EQUIVALENCE)
# No extra API calls. Do not transform sparse ICU auto-intervals into a chronology.
R1956_SCHEMA = "V4.9.80-R195.6-FATIGUED-POWER-1"

def _v1956_timed_power_segments(detail):
    """Return trusted *whole-activity* ordered power segments, or a reason.

    The FIT lap parser preserves start/end FIT timestamps and lap power. ICU
    automatic intervals often omit warm-up or only include highlights; they are
    accepted solely when explicit offsets permit checking whole-ride coverage.
    No interpolation and no assumed durations from names/PRs.
    """
    d = detail if isinstance(detail, dict) else {}
    native_laps = d.get("source_laps") or []
    proven_fit = (str(d.get("_v125_source_lap_provenance") or "") == "ORIGINAL_ACTIVITY_FIT_LAP")
    if proven_fit and len(native_laps) >= 2 and all(isinstance(r,dict) and r.get("_v125_record_origin") == "ORIGINAL_FIT_LAP" for r in native_laps):
        source, rows, start_key, end_key = "ORIGINAL_FIT_LAPS", native_laps, "fit_start_time", "fit_end_time"
    else:
        native = d.get("icu_intervals") or []
        # A timestamped, complete partition is admissible; never infer position
        # from `_truth_pos`, WORK order or start_index (sample index != seconds).
        if len(native) < 2 or not all(isinstance(r,dict) for r in native):
            return [], "NO_TRUSTED_WHOLE_RIDE_TIMELINE", None
        if all(r.get("start_secs") is not None for r in native):
            source, rows, start_key, end_key = "ICU_EXPLICIT_SECONDS", native, "start_secs", None
        elif all(r.get("start_index") is not None and r.get("end_index") is not None for r in native):
            # ICU native sample indices establish order, NOT seconds. Accept only
            # a contiguous, zero-anchored partition whose duration covers >=95%
            # of the ride, then synthesize *relative cumulative lap seconds*.
            # This is not legal for sparse automatic work-interval highlights.
            try:
                ordered=sorted(native,key=lambda r:float(r["start_index"]))
                indices=[(float(r["start_index"]),float(r["end_index"])) for r in ordered]
            except (TypeError,ValueError,OverflowError):
                return [], "INVALID_SAMPLE_INDEX", None
            if any(not (math.isfinite(a) and math.isfinite(b)) for a,b in indices):
                return [], "INVALID_SAMPLE_INDEX", None
            if indices[0][0] > 1 or indices[0][0] < 0 or any(b<=a for a,b in indices):
                return [], "INCOMPLETE_SAMPLE_INDEX_COVERAGE", None
            if any(abs(indices[i+1][0]-indices[i][1])>1 for i in range(len(indices)-1)):
                return [], "INCOMPLETE_SAMPLE_INDEX_COVERAGE", None
            source="ICU_FULL_COVERAGE_INDEXED"
            cursor=0.0; rows=[]
            for r in ordered:
                duration=_num(r.get("moving_time") or r.get("elapsed_time"))
                if duration is None or duration<=0:
                    return [], "INVALID_LAP_POWER_OR_TIME", None
                rows.append({**r,"_v1956_relative_start":cursor})
                cursor+=duration
            start_key,end_key="_v1956_relative_start",None
        else:
            return [], "NO_TRUSTED_WHOLE_RIDE_TIMELINE", None
    total = _num(d.get("moving_time") or d.get("elapsed_time"))
    if total is None or total <= 0: return [], "ACTIVITY_DURATION_UNAVAILABLE", None
    seg=[]
    for r in rows:
        start = _num(r.get(start_key)); duration = _num(r.get("moving_time") or r.get("elapsed_time")); power = _num(r.get("average_watts"))
        finish = _num(r.get(end_key)) if end_key else None
        if not all(v is not None and math.isfinite(v) for v in (start,duration,power)):
            return [], "MISSING_LAP_POWER_OR_TIME", None
        if duration <= 0 or not (0 <= power <= 2500):return [], "INVALID_LAP_POWER_OR_TIME", None
        if finish is None:finish = start+duration
        if not math.isfinite(finish) or finish <= start or abs((finish-start)-duration)>max(5.0,.06*duration):
            return [], "INCONSISTENT_LAP_BOUNDARIES", None
        seg.append({"start":float(start),"end":float(finish),"secs":float(duration),"watts":float(power),"wbal_start":_num(r.get("wbal_start"))})
    seg.sort(key=lambda r:r["start"])
    if len(seg)>200:return [], "TOO_MANY_SEGMENTS", None
    # No overlapping segments, hidden gaps or double-counting. This is a
    # conservative accounting of observed work *before* each subsequent lap.
    for a,b in zip(seg,seg[1:]):
        gap=b["start"]-a["end"]
        if gap < -3 or gap > 5:return [], "TIMELINE_GAPS_OR_OVERLAPS", None
    covered=sum(r["secs"] for r in seg)
    if covered / total < .95 or covered / total > 1.05:
        return [], "INCOMPLETE_ACTIVITY_COVERAGE", None
    if source == "ICU_EXPLICIT_SECONDS" and abs(seg[0]["start"])>5:
        return [], "ICU_TIMELINE_NOT_ZERO_ANCHORED", None
    return seg, None, source


def _v1956_fatigued_power_detail(detail, goal_key="POWER_5"):
    """Descriptive real effort power conditional on measured prior workload.

    A 365 W fatigued effort does not become a 380 W fresh PR. Prior power-zone
    time and exact W'bal trajectory are *not* inferable from lap-average watts.
    """
    seg, reason, source = _v1956_timed_power_segments(detail)
    base={"schema":R1956_SCHEMA,"available":False,"state":"DATA_NEEDED","reason":reason,
          "source":source,"best_5_min":None,"observations":[],
          "method_note":"Time-ordered original FIT laps or full-coverage explicitly timestamped native intervals only. Prior mechanical work (kJ) is observed lap power × time, not a physiological fatigue score; no fresh-power equivalence or PR claim."}
    if not seg:return base
    ftp=_num((detail or {}).get("icu_ftp") or (detail or {}).get("ftp"))
    wprime=_num((detail or {}).get("icu_w_prime"))
    previous_seconds=0.0;previous_kj=0.0; previous_hard_lap_sec=0.0
    preceding=[]; observations=[]
    for r in seg:
        secs=r["secs"];w=r["watts"]
        if 120 <= secs <= 900 and w>=100 and previous_seconds>=1800 and previous_kj>=200:
            # Exactly 5 minutes gets a separate 285-315 second bucket;
            # 3 vs 6 minutes must never be graded as comparable power.
            target="5_MIN" if 285<=secs<=315 else f"{int(round(secs/30)*30)}_SEC"
            wbal_pct = round(100*r["wbal_start"]/wprime,1) if (wprime and wprime>0 and r.get("wbal_start") is not None and 0<=r["wbal_start"]<=wprime) else None
            # Last 15 minutes: observed time-weighted lap averages. This is
            # descriptive; coarse lap boundaries are not a sample-level model.
            residual=900.0; recent_j=0.0; recent_s=0.0
            for prior in reversed(preceding):
                take=min(residual,prior["secs"])
                recent_j+=take*prior["watts"];recent_s+=take;residual-=take
                if residual<=0:break
            recent_w=round(recent_j/recent_s,1) if recent_s>=840 else None
            granular=bool(preceding and max(x["secs"] for x in preceding)<=900)
            observations.append({"duration_sec":round(secs,1),"duration_bucket":target,"watts":round(w,1),
                                 "evidence_eligible":granular,
                                 "timeline_resolution":"LAP_GRANULAR" if granular else "COARSE_LAPS_DESCRIPTIVE_ONLY",
                                 "prior_recent_15m_lap_watts":recent_w,
                                 "prior_elapsed_min":round(previous_seconds/60,1),"prior_work_kj":round(previous_kj,1),
                                 "prior_high_power_lap_min":round(previous_hard_lap_sec/60,1) if ftp and ftp>0 else None,
                                 "prior_wbal_pct":wbal_pct,
                                 "meaningful_prior_work":bool(previous_seconds>=2700 and previous_kj>=300),
                                 "workload_source":source})
        preceding.append(r)
        previous_seconds+=secs;previous_kj+=secs*w/1000
        if ftp and ftp>0 and w>=ftp:previous_hard_lap_sec+=secs
    if not observations:
        base["reason"]="NO_ELIGIBLE_LATE_QUALITY_EFFORT"
        return base
    # Retain strongest observed five-minute effort under meaningful prior work.
    eligible=[e for e in observations if e["duration_bucket"]=="5_MIN" and e["meaningful_prior_work"]]
    best=max(eligible,key=lambda e:(e["watts"],e["prior_work_kj"])) if eligible else None
    return {**base,"available":bool(best),"state":"OBSERVED" if best else "PARTIAL",
            "reason":None if best else "NO_5_MIN_EFFORT_AFTER_MEANINGFUL_WORK",
            "best_5_min":best,"observations":observations[:32],
            "method_note":base["method_note"] + " High-power lap minutes count entire laps averaging at/above FTP, not seconds above FTP; prior-15-min watts are lap-level weighted observations; no glycogen/W′bal reconstruction."}


def _v1956_compare_like_for_like(items):
    """Compare five-minute power only among independent sessions with similar lead-ins."""
    events=[]
    for e in items or []:
        if not isinstance(e,dict) or not isinstance(e.get("best_5_min"),dict) or not e["best_5_min"].get("evidence_eligible"):
            continue
        required=("duration_sec","prior_work_kj","prior_elapsed_min","watts")
        if all(_num(e["best_5_min"].get(k)) is not None and math.isfinite(_num(e["best_5_min"].get(k))) for k in required):
            events.append(e)
    # Sort chronologically; a same-day effort is not a verified earlier reference.
    events.sort(key=lambda e:(str(e.get("date") or ""),str(e.get("activity_id") or "")))
    comparisons=[]
    for cur in events:
        c=cur["best_5_min"]
        prior=[]
        for other in events:
            if not str(other.get("date") or "") < str(cur.get("date") or "") or other.get("activity_id")==cur.get("activity_id"):
                continue
            if other.get("event_kind")!=cur.get("event_kind") or other.get("source")!=cur.get("source"):
                continue
            o=other["best_5_min"]
            if abs(float(o["duration_sec"])-float(c["duration_sec"]))>15:continue
            if abs(o["prior_work_kj"]-c["prior_work_kj"])>max(80,.20*max(o["prior_work_kj"],c["prior_work_kj"])):continue
            if abs(o["prior_elapsed_min"]-c["prior_elapsed_min"])>max(10,.25*max(o["prior_elapsed_min"],c["prior_elapsed_min"])):continue
            oh=o.get("prior_high_power_lap_min");ch=c.get("prior_high_power_lap_min")
            if oh is not None and ch is not None and abs(oh-ch)>max(2,.30*max(oh,ch)):continue
            orw=o.get("prior_recent_15m_lap_watts");crw=c.get("prior_recent_15m_lap_watts")
            if orw is not None and crw is not None and abs(orw-crw)>max(30,.15*max(orw,crw)):continue
            ow=o.get("prior_wbal_pct");cw=c.get("prior_wbal_pct")
            if ow is not None and cw is not None and abs(ow-cw)>20:continue
            prior.append(other)
        if not prior:continue
        prev=max(prior,key=lambda e:e["best_5_min"]["watts"])
        delta=round(c["watts"]-prev["best_5_min"]["watts"],1)
        comparisons.append({"activity_id":cur["activity_id"],"date":cur["date"],
            "watts":c["watts"],"prior_work_kj":c["prior_work_kj"],"previous_activity_id":prev["activity_id"],
            "previous_date":prev["date"],"previous_watts":prev["best_5_min"]["watts"],
            "delta_w":delta,"outcome":"HIGHER_OBSERVED_LATE_POWER" if delta>0 else ("LOWER_OBSERVED_LATE_POWER" if delta<0 else "SIMILAR_OBSERVED_LATE_POWER"),
            "comparison":"SIMILAR_MEASURED_PRIOR_WORK","is_fresh_pb":False,
            "caveat":"Observational like-for-like comparison, not proof of physiological improvement or a fresh PR."})
    return comparisons

    blocks=_v1956_previous_training_blocks_baseline(recent_activities,limit=limit,activity_details=activity_details)
    details=activity_details if isinstance(activity_details,dict) else {}
    sources={str(a.get("id")):a for a in (recent_activities or []) if isinstance(a,dict) and a.get("id")}

        if _v1954_confirmed_competition(b):

    return blocks

def build_evidence_ledger(previous_training_blocks, overlay_blocks=None, goal_key=None, scope_start=None):
    """Conserve old canonical metrics; append independent, conservative late-power evidence."""
    led=_v1956_evidence_ledger_baseline(previous_training_blocks,overlay_blocks=overlay_blocks,goal_key=goal_key,scope_start=scope_start)
    if not isinstance(led,dict):return led
    start=str(scope_start or "")[:10]
    blocks={}
    for b in list(previous_training_blocks or [])+list(overlay_blocks or []):
        if not isinstance(b,dict):continue
        if start and str(b.get("date") or "")[:10]<start:continue
        k=_v81_evidence_block_key(b)
        blocks[k]=b
    observed=[]
    by_activity={}
    for b in blocks.values():
        fp=b.get("fatigue_power") or {}
        best=fp.get("best_5_min") if isinstance(fp,dict) else None
        if not isinstance(best,dict) or not fp.get("available") or not best.get("meaningful_prior_work") or not best.get("evidence_eligible"):continue
        event={"activity_id":b.get("activity_id"),"date":b.get("date"),"event_kind":"RACE" if _v1954_confirmed_competition(b) else "TRAINING",
               "source":fp.get("source"),"name":b.get("name"),"best_5_min":best}
        observed.append(event)
        by_activity[str(b.get("activity_id") or "")]=event
    matches=_v1956_compare_like_for_like(observed)
    match_by_id={str(x.get("activity_id")):x for x in matches}
    for row in led.get("events") or []:
        if not isinstance(row,dict):continue
        aid=str(row.get("activity_id") or "")
        if aid not in by_activity:continue
        row["fatigue_power"]={"state":"OBSERVED","best_5_min":by_activity[aid]["best_5_min"],"comparison":match_by_id.get(aid),
                             "is_fresh_pr":False}
        # Real measured load contributes to durability and, on actual races,
        # race transfer. It never enters repeated workout evidence.
        tags=list(row.get("tags") or [])
        for tag in (("MEASURED_PRIOR_WORK", "PRELOADED_QUALITY", "RACE_TRANSFER") if row.get("event_kind")=="RACE" else ("MEASURED_PRIOR_WORK", "PRELOADED_QUALITY")):
            if tag not in tags:tags.append(tag)
        row["tags"]=tags
    features=led.get("science_features")
    if isinstance(features,dict) and observed:
        features["preloaded_quality"]=True
        if any(x.get("event_kind")=="RACE" for x in observed):features["race_transfer"]=True
    led["fatigued_power"]={"schema":R1956_SCHEMA,"available":bool(observed),
        "five_min_sessions":len(observed),"five_min_race_sessions":sum(x.get("event_kind")=="RACE" for x in observed),
        "five_min_training_sessions":sum(x.get("event_kind")=="TRAINING" for x in observed),
        "observations":sorted(observed,key=lambda x:(str(x.get("date") or ""),str(x.get("activity_id") or "")),reverse=True),
        "comparable_pairs":len(matches),"comparisons":sorted(matches,key=lambda x:str(x.get("date") or ""),reverse=True),
        "source_policy":"VALIDATED_FULL_TIMELINE_ONLY",
        "method_note":"Observed 5-minute power after measured preceding kJ/duration, separate from fresh PR, work-set repeatability and test validation. Only similar-load prior independent sessions are compared; coarse prior laps (>15min), missing chronology or sparse ICU segments yield no coaching credit."}
    return led

# One evidence extraction for every configured duration; goal-scoped comparisons.
# All existing Nova planning, checkpoint, race, recovery implementations are kept.

def _v1957_goal_contracts():
    """Derive supported goals from THE LAB's authoritative roadmap definition.

    A newly registered goal gets evaluated automatically. Unsupported metadata
    fails closed rather than silently falling back to the 5-minute benchmark.
    """
    out={}
    for key, value in _V4873_ROADMAP_GOALS.items():
        t=(value or {}).get("field_test") or {}
        sec=_num(t.get("seconds"))
        mode=str(t.get("mode") or "").upper()
        if not sec or not math.isfinite(sec) or sec<10 or sec>7200 or mode not in {"MAXIMAL","SUBMAX_DURABILITY"}:
            out[str(key)]={"supported":False,"reason":"MISSING_OR_INVALID_GOAL_FIELD_TEST"}
            continue
        out[str(key)]={"supported":True,"anchor_seconds":int(sec),"mode":mode,
                       "stimulus_domain":t.get("stimulus_domain"),"checkpoint_label":t.get("label")}
    return out


def _v1957_anchor_tolerance(seconds):
    # Windows describe observed LAP averages, not exact-duration PRs.
    if seconds<=30:return 1.0
    if seconds<=90:return 2.0
    return min(90.0,max(10.0,.025*seconds))


def _v1957_prior_minimums(anchor):
    """Conservative *measurement policy*, not a universal fatigue threshold."""
    if anchor<=90:return (30.0,200.0)
    if anchor<=360:return (45.0,300.0)
    if anchor<=1800:return (60.0,400.0)
    return (60.0,400.0)


def _v1957_segment_hr(detail, segments, source):
    """Attach HR only when raw segment identity and timestamps both match.

    No HR interpolations or inheritance from the whole-activity average.
    """
    if source=="ORIGINAL_FIT_LAPS":
        rows=sorted((detail.get("source_laps") or []),key=lambda x:float(x.get("fit_start_time") or 0))
    elif source in {"ICU_EXPLICIT_SECONDS","ICU_FULL_COVERAGE_INDEXED"}:
        rows=detail.get("icu_intervals") or []
        start_field="start_secs" if source=="ICU_EXPLICIT_SECONDS" else "start_index"
        rows=sorted(rows,key=lambda x:float(x.get(start_field) or 0))
    else:return [None for _ in segments]
    if len(rows)!=len(segments):return [None for _ in segments]
    result=[]
    for seg,row in zip(segments,rows):
        duration=_num(row.get("moving_time") or row.get("elapsed_time"))
        watt=_num(row.get("average_watts"))
        hr=_num(row.get("average_heartrate") or row.get("avg_hr") or row.get("average_hr"))
        if duration is None or watt is None or abs(duration-seg["secs"])>1.5 or abs(watt-seg["watts"])>2:
            return [None for _ in segments]
        result.append(round(hr,1) if hr is not None and 45<=hr<=220 else None)
    return result


def _v1957_fatigued_power_detail(detail):
    """Goal-independent observed-effort catalog from validated whole-ride laps.

    Evidence is retained by anchor duration; no prediction of fresh watts, no
    virtual PR and no declaration that race fluctuations are workout repeats.
    """
    configs=_v1957_goal_contracts()
    anchors=sorted({int(x["anchor_seconds"]) for x in configs.values() if x.get("supported")})
    seg, reason, source=_v1956_timed_power_segments(detail)
    base={"schema":R1957_SCHEMA,"available":False,"state":"DATA_NEEDED",
          "reason":reason,"source":source,"best_by_duration":{},"observations":[],
          "goal_contracts":configs,"display_observations":[],
          "method_note":"Whole-ride indexed/elapsed chronology or FIT lap boundaries; only observed lap averages. Prior kJ is mechanical work, never a physiological fatigue score. Exact PRs and checkpoint validation are separate."}
    if not seg:return base
    hrs=_v1957_segment_hr(detail,seg,source)
    ftp=_num((detail or {}).get("icu_ftp") or (detail or {}).get("ftp"))
    wp=_num((detail or {}).get("icu_w_prime"))
    passed_s=0.0;passed_kj=0.0;passed_hard_s=0.0;history=[];found=[]
    for idx,row in enumerate(seg):
        secs=row["secs"];watts=row["watts"]
        matched=[a for a in anchors if abs(secs-a)<=_v1957_anchor_tolerance(a)]
        for anchor in matched:
            min_m,min_kj=_v1957_prior_minimums(anchor)
            if passed_s/60<min_m or passed_kj<min_kj or watts<20:
                continue
            rem=900.0;recent_watts_j=0;recent_secs=0
            for h in reversed(history):
                part=min(rem,h["secs"])
                recent_watts_j+=part*h["watts"]
                recent_secs+=part
                rem-=part
                if rem<=0:break
            recent_w=round(recent_watts_j/recent_secs,1) if recent_secs>=840 else None
            wbal = row.get("wbal_start")
            wbal_pct=round(100*wbal/wp,1) if wp and wp>0 and wbal is not None and 0<=wbal<=wp else None
            # A single giant lap prior to a 'late' effort hides its intensity.
            granularity=bool(history and max(x["secs"] for x in history)<=900)
            mode="SUBMAX_DURABILITY" if any(
                c.get("anchor_seconds")==anchor and c.get("mode")=="SUBMAX_DURABILITY"
                for c in configs.values()) else "MAXIMAL"
            hr=hrs[idx]
            eligible=bool(granularity and (mode!="SUBMAX_DURABILITY" or hr is not None))
            found.append({"anchor_seconds":anchor,"duration_sec":round(secs,1),
                 "duration_label":f"{anchor} s" if anchor<60 else (f"{anchor//60} min" if anchor%60==0 else f"{anchor} s"),
                 "duration_bucket":f"ANCHOR_{anchor}_S", "watts":round(watts,1),
                 "average_hr":hr,"analysis_mode":mode,
                 "prior_elapsed_min":round(passed_s/60,1),"prior_work_kj":round(passed_kj,1),
                 "prior_high_power_lap_min":round(passed_hard_s/60,1) if ftp and ftp>0 else None,
                 "prior_recent_15m_lap_watts":recent_w,"prior_wbal_pct":wbal_pct,
                 "timeline_resolution":"LAP_GRANULAR" if granularity else "COARSE_LAPS_DESCRIPTIVE_ONLY",
                 "evidence_eligible":eligible,"meaningful_prior_work":True,
                 "workload_source":source,"is_fresh_pr":False,
                 "interpretation":"SUBMAX_POWER_HR_CONTEXT_ONLY" if mode=="SUBMAX_DURABILITY" else "OBSERVED_POWER_AFTER_PRIOR_WORK"})
        history.append(row)
        passed_s+=secs;passed_kj+=secs*watts/1000
        if ftp and ftp>0 and watts>=ftp:passed_hard_s+=secs
    best={}
    for a in anchors:
        candidates=[x for x in found if x["anchor_seconds"]==a and x["evidence_eligible"]]
        if candidates:
            best[str(a)]=max(candidates,key=lambda x:(x["watts"],x["prior_work_kj"]))
    shown=sorted(best.values(),key=lambda x:x["anchor_seconds"],reverse=True)[:4]
    return {**base,"available":bool(best),"state":"OBSERVED" if best else ("PARTIAL" if found else "DATA_NEEDED"),
            "reason":None if best else ("INSUFFICIENT_TIMELINE_RESOLUTION_OR_HR" if found else "NO_GOAL_ANCHOR_EFFORT_AFTER_PRIOR_WORK"),
            "best_by_duration":best,"display_observations":shown,"observations":found[:100],
            "best_5_min":best.get("300"),
            "method_note":base["method_note"]+" Average 15s sprint power is not an instantaneous peak. The 60min submax benchmark additionally needs segment HR; no efficiency or decoupling improvement is inferred without comparable context."}


def _v1957_compare_like_for_like(items, mode):
    """Only matched prior independent performances for one configured anchor."""
    valid=[]
    for e in items or []:
        o=e.get("best_effort") if isinstance(e,dict) else None
        if not isinstance(o,dict) or not o.get("evidence_eligible"):continue
        if all(_num(o.get(k)) is not None and math.isfinite(_num(o[k])) for k in
               ("watts","duration_sec","prior_work_kj","prior_elapsed_min")):
            valid.append(e)
    valid.sort(key=lambda e:(str(e.get("date") or ""),str(e.get("activity_id") or "")))
    comparisons=[]
    for e in valid:
        c=e["best_effort"]; prior=[]
        for old in valid:
            o=old["best_effort"]
            if not str(old.get("date") or "")<str(e.get("date") or "") or old.get("activity_id")==e.get("activity_id"):continue
            if old.get("event_kind")!=e.get("event_kind") or old.get("source")!=e.get("source"):continue
            if o.get("anchor_seconds")!=c.get("anchor_seconds") or abs(o["duration_sec"]-c["duration_sec"])>_v1957_anchor_tolerance(c["anchor_seconds"]):continue
            if abs(o["prior_work_kj"]-c["prior_work_kj"])>max(80,.2*max(o["prior_work_kj"],c["prior_work_kj"])):continue
            if abs(o["prior_elapsed_min"]-c["prior_elapsed_min"])>max(10,.25*max(o["prior_elapsed_min"],c["prior_elapsed_min"])):continue
            oh=o.get("prior_high_power_lap_min");ch=c.get("prior_high_power_lap_min")
            if oh is not None and ch is not None and abs(oh-ch)>max(2,.3*max(oh,ch)):continue
            orw=o.get("prior_recent_15m_lap_watts");crw=c.get("prior_recent_15m_lap_watts")
            if orw is not None and crw is not None and abs(orw-crw)>max(30,.15*max(orw,crw)):continue
            ow=o.get("prior_wbal_pct");cw=c.get("prior_wbal_pct")
            if ow is not None and cw is not None and abs(ow-cw)>20:continue
            if mode=="SUBMAX_DURABILITY":
                # Without matched cardiac load this would compare different
                # physiological intensities, not submax aerobic performance.
                a=o.get("average_hr");b=c.get("average_hr")
                if a is None or b is None or abs(a-b)>min(8,.05*max(a,b)):continue
            prior.append(old)
        if not prior:continue
        prev=max(prior,key=lambda x:x["best_effort"]["watts"])
        previous=prev["best_effort"];delta=round(c["watts"]-previous["watts"],1)
        outcome=("HIGHER_OBSERVED_LATE_POWER" if delta>0 else ("LOWER_OBSERVED_LATE_POWER" if delta<0 else "SIMILAR_OBSERVED_LATE_POWER")) if mode!="SUBMAX_DURABILITY" else "SUBMAX_SIMILAR_HR_COMPARISON"
        comparisons.append({"activity_id":e.get("activity_id"),"date":e.get("date"),
             "previous_activity_id":prev.get("activity_id"),"previous_date":prev.get("date"),
             "anchor_seconds":c.get("anchor_seconds"),"watts":c["watts"],
             "previous_watts":previous["watts"],"delta_w":delta,"outcome":outcome,
             "prior_work_kj":c["prior_work_kj"],"comparison":"SIMILAR_MEASURED_PRIOR_WORK",
             "is_fresh_pb":False,"caveat":"Observed comparison, never a maximal PR, race-repeatability, or proof of physiological change."})
    return comparisons


def build_previous_training_blocks(recent_activities, limit=3, activity_details=None):
    # R195.5 baseline already includes R195.4 race gating. Avoid evaluating
    # R195.6's hard-coded five-minute module even on accounts pursuing POWER_20.
    blocks=_v1956_previous_training_blocks_baseline(recent_activities,limit=limit,activity_details=activity_details)
    details=activity_details if isinstance(activity_details,dict) else {}
    sources={str(a.get("id")):a for a in recent_activities or [] if isinstance(a,dict) and a.get("id")}
    for b in blocks or []:
        if not isinstance(b,dict):continue
        aid=str(b.get("activity_id") or "")
        d=details.get(aid) or sources.get(aid) or {}
        b["fatigue_power"]=_v1957_fatigued_power_detail(d)
        if _v1954_confirmed_competition(b):b["repeatability"]=None
    return blocks


def build_evidence_ledger(previous_training_blocks, overlay_blocks=None, goal_key=None, scope_start=None):
    """Attach only goal-relevant measured late-power observations; no fallback."""
    # Original evidence ledger excludes R195.6's fixed-5-min overlay entirely.
    led=_v1956_evidence_ledger_baseline(previous_training_blocks,overlay_blocks=overlay_blocks,
                                        goal_key=goal_key,scope_start=scope_start)
    if not isinstance(led,dict):return led
    configs=_v1957_goal_contracts()
    goal=str(goal_key or "").upper()
    cfg=configs.get(goal)
    anchor=cfg.get("anchor_seconds") if cfg and cfg.get("supported") else None
    mode=cfg.get("mode") if cfg and cfg.get("supported") else None
    start=str(scope_start or "")[:10]
    by_key={}
    for b in list(previous_training_blocks or [])+list(overlay_blocks or []):
        if not isinstance(b,dict) or (start and str(b.get("date") or "")[:10]<start):continue
        by_key[_v81_evidence_block_key(b)]=b
    observed=[]
    for b in by_key.values():
        fp=b.get("fatigue_power") or {}
        best=((fp.get("best_by_duration") or {}).get(str(anchor)) if anchor else None)
        if not isinstance(best,dict) or not fp.get("available") or not best.get("evidence_eligible"):
            continue
        observed.append({"activity_id":b.get("activity_id"),"date":b.get("date"),"name":b.get("name"),
              "event_kind":"RACE" if _v1954_confirmed_competition(b) else "TRAINING",
              "source":fp.get("source"),"goal_key":goal,"anchor_seconds":anchor,"best_effort":best})
    matches=_v1957_compare_like_for_like(observed,mode)
    by_aid={str(o.get("activity_id")):o for o in observed}
    comparisons={str(c.get("activity_id")):c for c in matches}
    for row in led.get("events") or []:
        if not isinstance(row,dict):continue
        aid=str(row.get("activity_id") or "");ev=by_aid.get(aid)
        if not ev:continue
        # Physical work is real evidence, but the historic RACE repeatability
        # contract is neither removed nor rebuilt here.
        row["fatigue_power"]={"state":"OBSERVED","goal_key":goal,"best_effort":ev["best_effort"],
                              "comparison":comparisons.get(aid),"is_fresh_pr":False}
        tags=list(row.get("tags") or [])
        tags.append("MEASURED_PRIOR_WORK") if "MEASURED_PRIOR_WORK" not in tags else None
        # Submax efficiency context is not a maximal quality effort.
        if mode=="MAXIMAL" and "PRELOADED_QUALITY" not in tags:tags.append("PRELOADED_QUALITY")
        if ev["event_kind"]=="RACE" and "RACE_TRANSFER" not in tags:tags.append("RACE_TRANSFER")
        row["tags"]=tags
    feat=led.get("science_features")
    if isinstance(feat,dict) and observed:
        if mode=="MAXIMAL":feat["preloaded_quality"]=True
        if any(e["event_kind"]=="RACE" for e in observed):feat["race_transfer"]=True
    led["fatigued_power"]={"schema":R1957_SCHEMA,"available":bool(observed),"goal_key":goal,
        "supported":bool(anchor),"status":"OBSERVED" if observed else ("DATA_NEEDED" if anchor else "UNSUPPORTED_GOAL"),
        "anchor_seconds":anchor,"mode":mode,"goal_count":len(configs),
        "observed_sessions":len(observed),"race_sessions":sum(e["event_kind"]=="RACE" for e in observed),
        "training_sessions":sum(e["event_kind"]=="TRAINING" for e in observed),
        "observations":sorted(observed,key=lambda e:(str(e.get("date") or ""),str(e.get("activity_id") or "")),reverse=True),
        "comparable_pairs":len(matches),"comparisons":sorted(matches,key=lambda c:str(c.get("date") or ""),reverse=True),
        "source_policy":"VALIDATED_FULL_TIMELINE_ONLY",
        "method_note":"Active goal and anchor come from ROADMAP_GOALS.field_test, never hard-coded 5min. Maximal observed after load differs from fresh PR; 15s is average power, not instantaneous peak. Submax requires HR; comparison is contextual only, not a formal efficiency/decoupling validation."}
    return led

# R195.8 · RECOVERY LOAD EVIDENCE TRUTH
# A hard work set with free/easy recovery is repeatability evidence, never
# proof of sustaining hard work while recovery itself carries meaningful load.
# The 70% personal FTP guard below is an intentionally conservative product
# threshold, not a universal physiological boundary or substitute for HR/zone
# assessment. Repeated independent observations remain required to close.
R1958_SCHEMA = "V4.9.82-R195.8-RECOVERY-LOAD-TRUTH-1"
R1958_MIN_MEANINGFUL_RECOVERY_FTP_RATIO = 0.70
R1958_INTRODUCTORY_RECOVERY_FTP_RATIO = 0.56


def _v1958_recovery_load_grade(event):
    """Grade *observed* between-rep recovery, not interval power or session title.

    Do not guess a recovery zone from session-global zone distribution. Ratio is
    personalised by the contemporaneous measured FTP; the product guard is
    explicitly heuristic. Unverified recovery watts/duration fail closed.
    """
    e = event if isinstance(event,dict) else {}
    ftp = _rhythm_num(e.get("ftp"))
    watts = _rhythm_num(e.get("recovery_avg_watts"))
    duration = _rhythm_num(e.get("recovery_secs"))
    if ftp is None or ftp <= 0 or watts is None or watts < 0 or duration is None or duration < 20:
        return {"grade":"UNVERIFIED","eligible":False,"successful":False,
                "reason":"MISSING_VALID_OBSERVED_RECOVERY_POWER_DURATION_OR_FTP","ratio":None}
    ratio = watts / ftp
    if ratio < R1958_INTRODUCTORY_RECOVERY_FTP_RATIO:
        grade,reason = "EASY","EASY_RECOVERY_IS_REPEATABILITY_NOT_LOADED_RECOVERY"
    elif ratio < R1958_MIN_MEANINGFUL_RECOVERY_FTP_RATIO:
        grade,reason = "INTRODUCTORY","LOW_ACTIVE_RECOVERY_DOES_NOT_ESTABLISH_LOAD_TOLERANCE"
    else:
        grade,reason = "MEANINGFUL","MEASURED_ACTIVE_RECOVERY_ABOVE_CONSERVATIVE_FTP_GUARD"
    eligible = False  # Only the ledger comparison can establish a meaningful *increase*.
    return {"grade":grade,"eligible":eligible,"successful":False,
            "reason":reason,"ratio":round(ratio,4),"threshold_ratio":R1958_MIN_MEANINGFUL_RECOVERY_FTP_RATIO,
            "work_reproduced":bool(e.get("strong")),"method":R1958_SCHEMA}


def _v1958_rescore_recovery_load(ledger):
    """Score a full baseline -> measured trial -> independent confirmation cycle.

    Thresholds are conservative *comparability policy*, not FTP-zone science:
    at least +max(10 W, 4% FTP) above a verified strong baseline, comparable
    structure, >=92% work retained and confirmation at comparable load within
    28 days. The old 70% threshold is descriptive, never an exit prerequisite.
    """
    if not isinstance(ledger,dict):return ledger
    events=[e for e in ((ledger.get('repeatability') or {}).get('events') or []) if isinstance(e,dict)]
    kinds={str(e.get('activity_id')):str(e.get('event_kind') or '').upper()
           for e in (ledger.get('events') or []) if isinstance(e,dict)}
    valid=[]; attempted=[]; successes=[]; introductory=[]
    for e in sorted(events,key=lambda x:(str(x.get('date') or '')[:10],str(x.get('activity_id') or ''))):
        grade=_v1958_recovery_load_grade(e)
        aid=str(e.get('activity_id') or '')
        expected=int(_rhythm_num(e.get('reps')) or 0)-1
        measured=int(_rhythm_num(e.get('recovery_pair_count')) or 0)
        pair_expected=_rhythm_num(e.get('recovery_pair_expected'))
        trusted=(e.get('recovery_observation_complete') is True and expected>=2
                 and measured==expected and pair_expected==expected
                 and str(e.get('activity_truth_source') or e.get('source') or '').upper()
                    in {'ICU_INTERVALS_CANONICAL','DEVICE_LAPS','ORIGINAL_ACTIVITY_FIT_LAP','ORIGINAL_FIT_LAPS'}
                 and str(e.get('activity_truth_status') or '').upper() not in {'CONFLICT','UNRESOLVED','RAW_FALLBACK'})
        if kinds.get(aid)=='RACE':
            e['recovery_load_evidence']={**grade,'grade':'EXCLUDED_RACE',
                'reason':'RACE_IS_NOT_STRUCTURED_RECOVERY_WORK'}
            continue
        if not trusted or grade['grade']=='UNVERIFIED' or not aid:
            e['recovery_load_evidence']={**grade,'grade':'UNVERIFIED',
                'reason':'INTER_REP_RECOVERY_NOT_COMPLETELY_MEASURED_OR_CANONICAL'}
            continue
        try:edate=date.fromisoformat(str(e.get('date') or '')[:10])
        except (ValueError,TypeError):
            e['recovery_load_evidence']={**grade,'grade':'UNVERIFIED','reason':'ACTIVITY_DATE_MISSING'}
            continue
        watts=_rhythm_num(e.get('recovery_avg_watts'))
        ftp=_rhythm_num(e.get('ftp'))
        work=_rhythm_num(e.get('avg_watts'))
        secs=_rhythm_num(e.get('interval_secs'))
        recsecs=_rhythm_num(e.get('recovery_secs'))
        reps=int(_rhythm_num(e.get('reps')) or 0)
        if any(v is None or v<=0 for v in (watts,ftp,work,secs,recsecs)):
            e['recovery_load_evidence']={**grade,'grade':'UNVERIFIED','reason':'INVALID_MEASURED_WORK_OR_RECOVERY'}
            continue
        baselines=[]
        for b in valid:
            try:bd=date.fromisoformat(str(b.get('date') or '')[:10])
            except (ValueError,TypeError):continue
            gap=(edate-bd).days
            if not 0<gap<=28 or not b.get('strong'):continue
            bw=_rhythm_num(b.get('avg_watts'));bf=_rhythm_num(b.get('ftp'))
            bs=_rhythm_num(b.get('interval_secs'));br=_rhythm_num(b.get('recovery_secs'))
            brec=_rhythm_num(b.get('recovery_avg_watts'))
            if (not all(v is not None and v>0 for v in (bw,bf,bs,br,brec))
                or abs(reps-int(b.get('reps') or 0))>1
                or abs(secs/bs-1)>.10 or abs(recsecs/br-1)>.25
                or abs(ftp/bf-1)>.08 or work<bw*.92):continue
            step=max(10.0,.04*ftp)
            if watts>=brec+step and (watts-brec)/ftp>=.035:
                baselines.append(b)
        baseline=min(baselines,key=lambda x:float(x['recovery_avg_watts'])) if baselines else None
        if baseline:
            grade={**grade,'eligible':True,'successful':bool(e.get('strong')),
                   'grade':'MEANINGFUL' if grade['grade']=='MEANINGFUL' else 'PROGRESSIVE',
                   'reason':'VERIFIED_RECOVERY_INCREASE_WITH_COMPARABLE_WORK',
                   'baseline_activity_id':baseline.get('activity_id'),
                   'baseline_watts':baseline.get('recovery_avg_watts'),
                   'increase_watts':round(watts-float(baseline['recovery_avg_watts']),1),
                   'work_retention_vs_baseline':round(work/float(baseline['avg_watts']),3)}
            attempted.append(e)
            if grade['successful']:successes.append(e)
        else:
            grade={**grade,'reason':'OBSERVED_BASELINE_NOT_YET_MEASURED_PROGRESS',
                   'eligible':False,'successful':False}
            introductory.append(e)
        e['recovery_load_evidence']=grade
        valid.append(e)
    # Two comparable independently completed trials with one shared older
    # baseline, not two arbitrary Z1 rides nor two separate baselines.
    confirmed=[]
    for later in successes:
        ld=date.fromisoformat(str(later['date'])[:10])
        lp=later['recovery_load_evidence']
        for earlier in successes:
            if earlier is later or earlier.get('activity_id')==later.get('activity_id'):continue
            ed=date.fromisoformat(str(earlier['date'])[:10])
            if not 0<(ld-ed).days<=28:continue
            ep=earlier['recovery_load_evidence']
            if ep.get('baseline_activity_id')!=lp.get('baseline_activity_id'):continue
            ew=_rhythm_num(earlier.get('recovery_avg_watts'))
            lw=_rhythm_num(later.get('recovery_avg_watts'))
            es=_rhythm_num(earlier.get('interval_secs'))
            ls=_rhythm_num(later.get('interval_secs'))
            er=_rhythm_num(earlier.get('recovery_secs'))
            lr=_rhythm_num(later.get('recovery_secs'))
            ework=_rhythm_num(earlier.get('avg_watts'))
            lwork=_rhythm_num(later.get('avg_watts'))
            ftp=_rhythm_num(later.get('ftp'))
            if (any(x is None or x<=0 for x in (ew,lw,es,ls,er,lr,ework,lwork,ftp))
                or abs(lw-ew)>max(10.,.05*ftp)
                or abs(ls/es-1)>.10 or abs(lr/er-1)>.25
                or lwork<ework*.92):continue
            confirmed.append((earlier,later))
    closed=bool(confirmed)
    confirmed_pair=max(confirmed,key=lambda p:(str(p[1].get('date') or ''),str(p[1].get('activity_id') or ''))) if confirmed else None
    unique_success={str(e['activity_id']):e for e in successes}
    unique_exposure={str(e['activity_id']):e for e in attempted}
    successes=list(unique_success.values());attempted=list(unique_exposure.values())
    latest_success=max((str(e.get('date') or '')[:10] for e in successes),default=None)
    latest_exposure=max((str(e.get('date') or '')[:10] for e in attempted),default=None)
    feat=ledger.get('science_features')
    if not isinstance(feat,dict):feat={};ledger['science_features']=feat
    pending=None
    if not closed and successes:
        latest=max(successes,key=lambda e:(str(e.get('date') or ''),str(e.get('activity_id') or '')))
        pending={'activity_id':latest['activity_id'],'date':latest['date'],
                 'recovery_watts':latest['recovery_avg_watts'],
                 'baseline_activity_id':latest['recovery_load_evidence'].get('baseline_activity_id'),
                 'stage':'CONFIRM_SAME_LOAD'}
    feat.update({'recovery_under_load_exposures':len(attempted),
       'recovery_under_load_successes':len(successes),'active_load_recovery':closed,
       'recovery_under_load_closed':closed,'recovery_under_load_trial':pending,
       'max_successful_recovery_ratio':round(max((float(e['recovery_avg_watts'])/float(e['ftp']) for e in successes),default=0),3) if successes else None,
       'latest_recovery_under_load_success_date':latest_success,
       'latest_recovery_under_load_exposure_date':latest_exposure,
       'recovery_under_load_introductory_sessions':len(introductory),
       'recovery_under_load_evidence_schema':R1958_SCHEMA})
    ledger['recovery_load_evidence']={'schema':R1958_SCHEMA,

def _v1959_comparable_recovery_history(evidence_ledger):
    """Only canonical, dated, completed TRAINING rep events; never race repeats.

    This is history for *dosing*, not a shortcut to RECOVERY_UNDER_LOAD awards.
    The previous session's recoveries, work and duration must all be observed.
    """
    led=evidence_ledger if isinstance(evidence_ledger,dict) else {}
    kind={str(e.get('activity_id')):str(e.get('event_kind') or '').upper()
          for e in (led.get('events') or []) if isinstance(e,dict)}
    out=[]; seen=set()
    for e in ((led.get('repeatability') or {}).get('events') or []):
        if not isinstance(e,dict):continue
        aid=str(e.get('activity_id') or '')
        if not aid or kind.get(aid)!='TRAINING' or aid in seen:continue
        source=str(e.get('activity_truth_source') or e.get('source') or '').upper()
        if source not in {'ICU_INTERVALS_CANONICAL','DEVICE_LAPS','ORIGINAL_ACTIVITY_FIT_LAP','ORIGINAL_FIT_LAPS'}:continue
        reps=int(_rhythm_num(e.get('reps')) or 0)
        if (e.get('recovery_observation_complete') is not True
            or int(_rhythm_num(e.get('recovery_pair_count')) or -1)!=reps-1
            or int(_rhythm_num(e.get('recovery_pair_expected')) or -1)!=reps-1):continue
        vals={k:_rhythm_num(e.get(k)) for k in ('reps','interval_secs','recovery_secs','avg_watts','recovery_avg_watts','ftp')}
        if any(v is None for v in vals.values()) or any(vals[k]<=0 for k in ('reps','interval_secs','recovery_secs','avg_watts','ftp')) or vals['recovery_avg_watts']<0:continue
        if vals['reps']<2 or vals['recovery_secs']<20:continue
        day=str(e.get('date') or '')[:10]
        try:date.fromisoformat(day)
        except (ValueError,TypeError):continue
        seen.add(aid)
        out.append({'activity_id':aid,'date':day,'strong':bool(e.get('strong')),
             'reps':int(round(vals['reps'])),'interval_secs':vals['interval_secs'],
             'recovery_secs':vals['recovery_secs'],'work_watts':vals['avg_watts'],
             'recovery_watts':vals['recovery_avg_watts'], 'ftp':vals['ftp'],
             'recovery_ratio':vals['recovery_avg_watts']/vals['ftp']})
    return sorted(out,key=lambda x:(x['date'],x['activity_id']),reverse=True)

def _v1959_recovery_dose_contract(goal,dimension,features):
    """Conservative observable stepped bridge; NEVER grade an athlete by a target.

    No estimate/model/FTP-scaled guessed *work* wattage: work is always derived
    from a strong completed comparable session, with a compensating reduction
    when recovery is increased. Keep structure fixed and never jump to 70% FTP.
    """
    if str(dimension or '').upper()!='RECOVERY_UNDER_LOAD':
        return {'available':False,'reason':'DIMENSION_NOT_DOSE_CONTRACTED'}
    f=features if isinstance(features,dict) else {}
    ref=f.get('latest_repeatability') or {}
    r={k:_rhythm_num(ref.get(k)) for k in ('reps','interval_secs','recovery_secs','avg_watts','recovery_avg_watts','ftp')}
    if not ref.get('strong') or any(v is None for v in r.values()) or any(r[k]<=0 for k in ('reps','interval_secs','recovery_secs','avg_watts','ftp')) or r['recovery_avg_watts']<0 or r['reps']<2 or r['recovery_secs']<20:
        return {'available':False,'schema':R1959_SCHEMA,'reason':'NO_CURRENT_STRONG_MEASURED_REPEATED_WORK','rule':'No quantified loaded recovery without a strong, recent, completely observed work/recovery pair.'}
    # For safety, a saved scientific-feature cache alone is not proof of
    # chronology or provenance. Match the most recent strong witness to ledger.
    history=list(f.get('recovery_dose_history') or [])
    if f.get('recovery_under_load_closed') is True:
        return {'available':False,'schema':R1959_SCHEMA,
                'reason':'RECOVERY_LOAD_ALREADY_ESTABLISHED',
                'rule':'The verified baseline, progression and independent confirmation closed this question; do not issue a repeat merely to raise recovery watts.'}
    ref_date=str(ref.get('date') or '')[:10]
    try:
        ref_day=date.fromisoformat(ref_date)
        today=get_rome_now().date()
        days_old=(today-ref_day).days
    except (ValueError,TypeError,AttributeError):
        return {'available':False,'schema':R1959_SCHEMA,'reason':'REFERENCE_DATE_NOT_VERIFIABLE'}
    if not -1<=days_old<=28:
        return {'available':False,'schema':R1959_SCHEMA,'reason':'STALE_OR_FUTURE_UNVERIFIED_REFERENCE',
                'rule':'No loaded-recovery watt target from a repeated-work baseline older than 28 days.'}
    structure_pool=[]
    for h in history:
        if not h.get('strong') or h['date']>ref_date:continue
        if abs(h['interval_secs']/r['interval_secs']-1)>0.10:continue
        if abs(h['recovery_secs']/r['recovery_secs']-1)>0.25:continue
        if abs(h['reps']-r['reps'])>1:continue
        if abs(h['ftp']/r['ftp']-1)>0.08:continue
        try:gap=(date.fromisoformat(ref_date)-date.fromisoformat(h['date'])).days
        except (TypeError,ValueError):continue
        if 0<=gap<=28:structure_pool.append(h)
    if not structure_pool:
        return {'available':False,'schema':R1959_SCHEMA,'reason':'REFERENCE_NOT_VERIFIED_IN_CANONICAL_LEDGER'}
    if r['avg_watts']<r['ftp']*.90 or r['avg_watts']<=r['recovery_avg_watts']*1.10:
        return {'available':False,'schema':R1959_SCHEMA,'reason':'NO_MEANINGFUL_MAIN_WORK_CAPACITY'}
    best_work=max(h['work_watts'] for h in structure_pool)
    # Preserve a recent high-quality capacity anchor. Recovery progression cannot
    # repeatedly discount work by another 7% every time the athlete succeeds.
    if r['avg_watts']<best_work*.88:
        return {'available':False,'schema':R1959_SCHEMA,'reason':'MAIN_WORK_BELOW_RECENT_CAPACITY_FLOOR',
                'rule':'Do not advance recovery load while work power falls below 88% of the best recent comparable repeated work.'}
    comparable=[h for h in structure_pool if h['work_watts']>=best_work*.88]
    # Require the canonical measured reference, not merely similar prior reps.
    matching=[h for h in comparable if h['date']==ref_date
             and abs(h['work_watts']-r['avg_watts'])<=max(3,0.015*r['avg_watts'])
             and abs(h['recovery_watts']-r['recovery_avg_watts'])<=max(3,0.025*r['ftp'])]
    if not matching:
        return {'available':False,'schema':R1959_SCHEMA,'reason':'REFERENCE_NOT_VERIFIED_IN_CANONICAL_LEDGER','rule':'Evidence must include the completed reference with matched work and recovery, not a narrative or predicted wattage.'}
    # A prior hard recovery failure at an otherwise comparable work structure
    # forbids raising load until a new successful lower-load bridge exists.
    failed=[]
    for h in history:
        if h.get('strong') or h['date']>ref_date:continue
        if abs(h['interval_secs']/r['interval_secs']-1)<=.10 and abs(h['reps']-r['reps'])<=1 and abs(h['work_watts']/r['avg_watts']-1)<=.10:
            failed.append(h)
    failed.sort(key=lambda h:(h['date'],h['activity_id']),reverse=True)
    if failed and failed[0]['date']>=ref_date:
        return {'available':False,'schema':R1959_SCHEMA,'reason':'LATEST_COMPARABLE_WORK_FAILED','rule':'Do not progress recovery after a failed comparable repetition; first re-establish quality under easy recovery.'}
    # The LAST strong comparable completed session is the recovery anchor.
    # Older higher recoveries are context, not permission to jump past a recent step.
    recent_best=max(comparable,key=lambda h:(h['date'],h['activity_id']))
    anchor_ratio=float(recent_best['recovery_ratio'])
    trial=f.get('recovery_under_load_trial') or {}
    # A real, recently completed progression needs confirmation, not another
    # automatic +5% FTP jump. No unverified/cached trial can confer closure.
    confirming=bool(trial.get('stage')=='CONFIRM_SAME_LOAD'
                    and trial.get('activity_id')==next((h['activity_id'] for h in history
                        if h.get('strong') and h['date']==ref_date),None))
    step=.05 if anchor_ratio>=.56 else .06
    candidate=anchor_ratio if confirming else min(.70,anchor_ratio+step)
    if not confirming and candidate<=anchor_ratio+.009:
        return {'available':False,'schema':R1959_SCHEMA,'reason':'NO_SAFE_PROGRESSIVE_RECOVERY_STEP'}
    def floor5(x):return int(math.floor(float(x)/5)*5)
    def ceil5(x):return int(math.ceil(float(x)/5)*5)
    rec_anchor=recent_best['recovery_watts']
    # Prescribe a narrow controlled band starting near last observed easy/loaded
    # watts. Neither end may exceed the previously proved intensity by >.06 FTP.
    rec_hi=floor5(min(r['ftp']*candidate,rec_anchor+r['ftp']*.06))
    rec_lo=floor5(max(rec_anchor-10 if confirming else rec_anchor+5,rec_hi-10))
    rec_lo=min(rec_lo,rec_hi)
    if (not confirming and rec_hi<=rec_anchor+2) or rec_hi<=rec_lo:
        return {'available':False,'schema':R1959_SCHEMA,'reason':'RECOVERY_STEP_TOO_SMALL_TO_DOSE_SAFELY'}
    # The first bounded step from a verified easy-recovery baseline is a
    # legitimate *trial*, not an earned award. Never confuse its intent with
    # observed success, and never require >55% FTP before training that skill.
    # Use a STABLE recent proven work-capacity anchor rather than compounding
    # reductions. Never increase work above the immediately observed session.
    work_hi=floor5(min(best_work*.98,r['avg_watts']-3))
    work_lo=floor5(max(best_work*.94,min(best_work*.96,work_hi-5)))
    if work_hi<=0 or work_lo<=0 or work_hi<=work_lo:
        return {'available':False,'schema':R1959_SCHEMA,'reason':'UNSAFE_OR_UNRESOLVED_WORK_BAND'}
    # Hard fail if either lever would exceed the proven main work or if
    # the recovery jump would exceed one safe increment once rounded.
    if work_hi>r['avg_watts']+1 or rec_hi>rec_anchor+r['ftp']*.06+0.01:
        return {'available':False,'schema':R1959_SCHEMA,'reason':'UNSAFE_COUPLED_LOAD_INVARIANT'}
    stage='MEASURED_RECOVERY_CONFIRMATION' if confirming else ('CONTROLLED_RECOVERY_BRIDGE' if candidate<.56 else ('ACTIVE_RECOVERY_BUILD' if candidate<.70 else 'MEANINGFUL_LOAD_TRIAL'))
    return {'available':True,'schema':R1959_SCHEMA,'stage':stage,
        'dominant_lever':'RECOVERY_LOAD_WITH_WORK_QUALITY_PROTECTION',
        'reference':{'date':ref_date,'reps':int(round(r['reps'])),'interval_secs':int(round(r['interval_secs'])),
            'recovery_secs':int(round(r['recovery_secs'])),'work_avg_watts':round(r['avg_watts'],1),
            'recovery_avg_watts':round(r['recovery_avg_watts'],1),'ftp':round(r['ftp'],1),
            'recovery_ratio':round(r['recovery_avg_watts']/r['ftp'],3),
            'matched_completed_sessions':len(comparable),
            'recent_best_repeated_work_watts':round(best_work,1),
            'minimum_meaningful_work_watts':round(best_work*.88,1),
            'last_verified_recovery_watts':round(rec_anchor,1),
            'last_verified_recovery_ratio':round(anchor_ratio,3)},
        'prescription':{'reps':int(round(r['reps'])),'interval_secs':int(round(r['interval_secs'])),
            'recovery_secs':int(round(r['recovery_secs'])),'work_watts_low':work_lo,'work_watts_high':work_hi,
            'recovery_watts_low':rec_lo,'recovery_watts_high':rec_hi,
            'recovery_ratio_low':round(rec_lo/r['ftp'],3),'recovery_ratio_high':round(rec_hi/r['ftp'],3)},
        'rule':'Preserve rep count, work/rest duration and minimum proven repeated-work quality. If the prior step succeeded, confirm its recovery load before adding another increment. The first step from easy recovery is a measured trial, not a skill award. No mandatory 70%-FTP target. If work quality fades, revert to easy recovery and do not advance.',
        'evidence_basis':'Canonical independent completed repeated work; same-structure recovery history; explicit small FTP-relative increments and measured work de-rating. Heuristic safety policy, not a clinical/physiological law.'}

    if not isinstance(out,dict) or str(out.get('status') or '').upper()!='PRESCRIBED':return out

    idx=out.get('selected_slot_index')
    try:i=int(idx);candidate=raw[i]
    except (ValueError,TypeError,IndexError):return out
    if not isinstance(candidate,dict):return out
    title=str(candidate.get('title') or '')
    orig_dim=str(candidate.get('_tl_preserved_original_dimension') or candidate.get('_tl_nova_dimension') or '').upper()
    if orig_dim!='RECOVERY_UNDER_LOAD' and 'loaded-repeatability' not in title.lower():return out

    bad=(not dc.get('available') or dc.get('schema')!=R1959_SCHEMA
         or bool(_v504_scientific_dose_violations(sig,candidate.get('main_set'),adaptive_roadmap)))
    if not bad:return out
    # R196.9.54: a stale R153 commitment is not a veto against an independent,
    # already-measured R195.9 *replacement*.  First try the ORIGINAL full
    # prescriber again against a read-only selection view that omits ONLY
    # obsolete, unresolved commitments for the exact same future slot.
    # Do not edit the actual ledger, completed evidence, primary anchor, or
    # any current safety/spacing/eligibility inputs.  If the prescriber cannot
    # make an exact canonical replacement, the original hard safety HOLD wins.

    slot=slots[i] if 0<=i<len(slots) and isinstance(slots[i],dict) else {}

# R196.0 -- strict authority handoff from the evidence-backed recovery-dose
# contract (R195.9) into R194.3's mandatory hard-work power check.
# Recovery-under-load workouts are deliberately *not* generic library templates.
# An absent library entry is NOT evidence of missing FTP or completed workouts.
# The exception is narrow: exact, immutable mechanics matching a verified dose
# contract, never a free-form power estimate or automatic approval of hard work.
R1960_SCHEMA = "V4.9.84-R196.0-COUPLED-POWER-HANDOFF-1"


def _v1960_coupled_power_band(raw, compiled, roadmap):
    if not isinstance(raw,dict) or not isinstance(compiled,dict):return None
    workout='RECOVERY_UNDER_LOAD_EVIDENCE_DOSE_V1'
    if (str(raw.get('_tl_nova_workout_id') or '')!=workout
        or str(raw.get('_tl_nova_contract_id') or '')!=workout
        or str(compiled.get('nova_workout_id') or '')!=workout
        or str(raw.get('_tl_nova_dimension') or '').upper()!='RECOVERY_UNDER_LOAD'
        or not raw.get('_tl_nova_deterministic')):return None
    sci=((roadmap or {}).get('scientific_progression') or {})
    dc=sci.get('dose_contract') or {}
    if (str(sci.get('dimension') or '').upper()!='RECOVERY_UNDER_LOAD'
        or dc.get('schema')!=R1959_SCHEMA or dc.get('available') is not True
        or str(dc.get('stage') or '')=='POST_SEVERE_INTENSITY_REENTRY'):return None
    ref=dc.get('reference') or {};p=dc.get('prescription') or {}
    def number(v,low=0,high=1000000):
        try:
            f=float(v)
            return f if math.isfinite(f) and low<=f<=high else None
        except (ValueError,TypeError,OverflowError):return None
    keys=('reps','interval_secs','recovery_secs','work_watts_low','work_watts_high','recovery_watts_low','recovery_watts_high')
    v={k:number(p.get(k)) for k in keys}
    if any(x is None for x in v.values()):return None
    if (not 2<=v['reps']<=20 or not 60<=v['interval_secs']<=1800
        or not 20<=v['recovery_secs']<=1800 or not 40<=v['recovery_watts_low']<v['recovery_watts_high']<=600
        or not 80<=v['work_watts_low']<v['work_watts_high']<=1200):return None
    best=number(ref.get('recent_best_repeated_work_watts'),80,1200)
    observed=number(ref.get('work_avg_watts'),80,1200)
    ftp=number(ref.get('ftp'),80,600)
    previous=number(ref.get('recovery_avg_watts'),0,600)
    recovery_anchor=number(ref.get('last_verified_recovery_watts'),0,600)
    count=number(ref.get('matched_completed_sessions'),1,500)
    if (None in (best,observed,ftp,previous,recovery_anchor,count)
        or v['work_watts_high']>=observed or v['work_watts_low']<best*.88-5
        or v['recovery_watts_high']>recovery_anchor+ftp*.06+5
        or v['recovery_watts_high']>=v['work_watts_low']):return None
    # The R195.9 composer publishes work and recovery as ONE indivisible dose.
    # Reject even a plausible wattage when somebody changes reps, work duration,
    # recoveries, or either band on the issued contract.
    pattern=(r'^\s*(\d+)\s*[×xX]\s*(\d+(?:\.\d+)?)\s*(min|s)'
             r'\s+free\s+at\s+(\d+)\s*[–—-]\s*(\d+)\s*W\.\s*'
             r'Recovery\s+between\s+reps:\s*(\d+(?:\.\d+)?)\s*(min|s)'
             r'\s+continuously\s+at\s+(\d+)\s*[–—-]\s*(\d+)\s*W\b')
    match=re.search(pattern,str(raw.get('main_set') or ''),re.I)
    if not match:return None
    reps,work_d,work_u,work_lo,work_hi,rec_d,rec_u,rec_lo,rec_hi=match.groups()
    work_secs=float(work_d)*(60 if work_u.lower()=='min' else 1)
    rec_secs=float(rec_d)*(60 if rec_u.lower()=='min' else 1)
    if (int(reps)!=int(v['reps']) or abs(work_secs-v['interval_secs'])>3.1
        or abs(rec_secs-v['recovery_secs'])>6.1
        or [int(work_lo),int(work_hi),int(rec_lo),int(rec_hi)]!=
        [int(v['work_watts_low']),int(v['work_watts_high']),int(v['recovery_watts_low']),int(v['recovery_watts_high'])]):return None
    # Recheck on the compiled athlete-facing text: mechanics must survive every
    # prior compiler guard, not just be present in the raw request.
    compiled_text=str(compiled.get('main_set') or '')
    # Warm-up instructions may precede the clause; they must not replace it.
    # Compare VALUES, not merely the regex shape: a rewritten 400–410 W main
    # set must fail even though it still looks syntactically well-formed.
    compiled_match=re.search(pattern[1:],compiled_text,re.I)
    if not compiled_match:return None
    cr,cwd,cwu,cwl,cwh,crd,cru,crl,crh=compiled_match.groups()
    if (int(cr)!=int(reps)
        or abs(float(cwd)*(60 if cwu.lower()=='min' else 1)-v['interval_secs'])>3.1
        or abs(float(crd)*(60 if cru.lower()=='min' else 1)-v['recovery_secs'])>6.1
        or [int(cwl),int(cwh),int(crl),int(crh)]!=
           [int(v['work_watts_low']),int(v['work_watts_high']),int(v['recovery_watts_low']),int(v['recovery_watts_high'])]):return None
    ref_date=str(ref.get('date') or '')[:10];slot_date=str(compiled.get('date') or '')[:10]
    try:
        lag=(date.fromisoformat(slot_date)-date.fromisoformat(ref_date)).days
        if not 0<=lag<=35:return None
    except (ValueError,TypeError):return None
    lo=int(v['work_watts_low']);hi=int(v['work_watts_high'])
    return {'schema':R1943_SCHEMA,'state':'PRESCRIBED','confidence':'HIGH',
        'source':'OBSERVED_COUPLED_RECOVERY_DOSE','handoff_schema':R1960_SCHEMA,
        'provenance_schema':R1959_SCHEMA,'reference_date':ref_date,
        'observed_repeated_avg_watts':round(observed,1),
        'target_watts':int(round((lo+hi)/2)),'low_watts':lo,'high_watts':hi,
        'reps':int(v['reps']),'work_seconds':int(v['interval_secs']),
        'recovery_seconds':int(v['recovery_secs']),
        'recovery_watts_low':int(v['recovery_watts_low']),
        'recovery_watts_high':int(v['recovery_watts_high']),
        'source_note':'Verified R195.9 observed coupled-dose contract; not a generic workout-library template or FTP-only guess.'}

def _v1962_repeat_guard(template, band, evidence_ledger, slot):
    """Block an immediately repeated identical main set without a new dose.

    Compare only the athlete's most recent verified repeated-work exposure.
    Different work count/duration is a different structural question. A higher
    target must exceed the 5 W prescription resolution to count as a distinct
    intensity lever. No universal power percentage or goal-specific exception.
    """
    t = template if isinstance(template,dict) else {}
    pat = str(t.get('pattern') or '').upper()
    if str(t.get('work_domain') or '').upper()=='TEMPO_Z3':
        # Do not reject Tempo because a physically different threshold block
        # has the same number of repetitions/duration. Reconciled completed
        # Tempo itself remains eligible for the ordinary duplicate guards.
        return None
    if pat not in {'STANDARD','SUSTAINED','SHORT_POWER','SPRINT'}:
        return None
    rows=[]
    for e in ((evidence_ledger or {}).get('repeatability') or {}).get('events') or []:
        if not isinstance(e,dict) or e.get('strong') is not True: continue
        if str(e.get('activity_truth_source') or e.get('source') or '').upper() not in {
             'DEVICE_LAPS','ICU_INTERVALS_CANONICAL'}:continue
        try:
            when=date.fromisoformat(str(e.get('date') or '')[:10])
            to=date.fromisoformat(str((slot or {}).get('date') or '')[:10])
            if when>to:continue
            secs=float(e.get('interval_secs')); reps=int(e.get('reps'))
        except (ValueError,TypeError):continue
        rows.append((when, str(e.get('activity_id') or ''), secs, reps,e))
    if not rows: return None
    rows.sort(key=lambda x:(x[0],x[1]),reverse=True)
    _,_,seconds,repetitions,latest=rows[0]
    if abs(seconds-float(t.get('work_s') or 0))>1 or repetitions != int(t.get('reps') or 0):
        return None
    if pat=='SPRINT':
        return {'reason':'LATEST_VERIFIED_SPRINT_STRUCTURE_ALREADY_COMPLETED',
                'reference_activity_id':latest.get('activity_id')}
    try: observed=float(latest.get('avg_watts')); target=float(band.get('target_watts'))
    except (TypeError,ValueError):return None
    if not (math.isfinite(observed) and math.isfinite(target)):return None
    if target-observed < 5:
        return {'reason':'LATEST_VERIFIED_MAIN_SET_NO_MEASURABLE_NEW_LEVER',
                'reference_activity_id':latest.get('activity_id'),
                'observed_watts':round(observed,1),'proposed_watts':target}
    return None

            work = _rhythm_num(t.get('work_s'))
            rest = _rhythm_num(t.get('recovery_s'))
            if not isinstance(reps, int) or reps < 2 or not work or rest is None:
                continue
            matches = []
            for ix, row in enumerate(rows):
                if not isinstance(row, dict) or str(row.get('type') or '').upper() != 'WORK':
                    continue
                secs, watts = _rhythm_num(row.get('duration_sec')), _rhythm_num(row.get('average_watts'))
                ftp = _rhythm_num(b.get('ftp'))
                if (secs is not None and watts is not None and watts > 0
                        and (ftp is None or watts >= ftp * .90)
                        and abs(secs-work) <= max(4.0,work*.03)):
                    matches.append(ix)
            if len(matches) < reps:
                continue
            # Work and prescribed EASY gaps must alternate. A separate short
            # warm-up surge does not create an extra interval or invalidate the
            # actual 2x7. Missing/mismatched gap is NOT a confirmed match.
            for start in range(len(matches)-reps+1):
                positions = matches[start:start+reps]
                valid = True
                for x,y in zip(positions,positions[1:]):
                    between = rows[x+1:y]
                    if (not between or any(not isinstance(z,dict) or
                            str(z.get('type') or '').upper() not in {'RECOVERY','REST'}
                            for z in between)):
                        valid=False; break
                    seconds = sum(_rhythm_num(z.get('duration_sec')) or 0 for z in between)
                    if abs(seconds-rest)>max(10.0,rest*.06):
                        valid=False; break
                if valid:
                    observed[str(t.get('id'))] = {'source':'MEASURED_NATIVE_WORK_RECOVERY',
                        'activity_id':aid,'actual_date':performed.isoformat(),
                        'measured_work_seconds':[rows[k].get('duration_sec') for k in positions],
                        'prescribed_recovery_seconds':rest,
                        'observation_only':True,'schema':R196946_SCHEMA}
                    break
    return observed

def _v1962_locked_identity(out, decision):
    if not isinstance(out, dict) or str(out.get('status') or '').upper() != 'PRESCRIBED':
        return None
    if str(out.get('source') or '') != 'PRESCRIPTION_CONTINUITY_LOCK':
        return None
    try:
        raw = out['raw_sessions'][int(out.get('selected_slot_index'))]
    except (TypeError, ValueError, IndexError, KeyError):
        return None
    if not isinstance(raw,dict): return None
    original = str(raw.get('_tl_preserved_original_dimension') or '').upper()
    current = str((decision or {}).get('dimension') or '').upper()
    band = raw.get('_tl_v1943_power_band') or {}
    # A persisted power target may have been legal yesterday but unsupported by
    # today's evidence/FTP. R153 may not resurrect it by continuity, even when
    # the scientific dimension is unchanged (the real POWER_20 QA failure).
    wid = str(out.get('workout_id') or '')
    family = next((t for t in _NOVA_WORKOUT_LIBRARY if str(t.get('id') or '') == wid), {})
    numerically_prescribed = str(family.get('pattern') or '').upper() in {'STANDARD','SUSTAINED','SHORT_POWER'}
    # Sprint, endurance and other non-numeric structures legitimately carry an
    # UNSUPPORTED_STRUCTURE audit marker; that is not a reason to invalidate
    # their plan identity or suppress a real strength/sprint training stimulus.
    stale_power = (numerically_prescribed and isinstance(band,dict) and bool(band)
                   and band.get('state') not in {'PRESCRIBED'})
    mismatched_dimension = bool(original and current and original != current)
    if mismatched_dimension or stale_power:
        return {'original_dimension':original or current, 'current_dimension':current,
                'reason':'STALE_POWER_CONTRACT' if stale_power else 'SCIENTIFIC_DIMENSION_CHANGED',
                'workout_id':out.get('workout_id'),
                'selected_slot_index':out.get('selected_slot_index')}
    return None


def build_nova_prescription(nova_decision, coach_clock, adaptive_roadmap=None, microcycle_ledger=None,
        evidence_ledger=None, power_model=None, ftp_anchor=None,
        aerobic_metabolic_range=None, training_definitions=None):
    args = (nova_decision, coach_clock, adaptive_roadmap, microcycle_ledger,
            evidence_ledger, power_model, ftp_anchor, aerobic_metabolic_range, training_definitions)
    out = _v1962_call_prescriber(*args)
    mismatch = _v1962_locked_identity(out, nova_decision)
    if not mismatch: return out
    # A new science gate may invalidate an old issued recovery protocol. Retain
    # its history (immutable), but exclude the conflicting slot from a one-time
    # pure *selection* replay. No DB write or cross-athlete state mutation.
    shadow = copy.deepcopy(microcycle_ledger or {})
    slots = list((coach_clock or {}).get('next_slots') or [])
    try: slot = slots[int(mismatch['selected_slot_index'])]
    except (IndexError, ValueError, TypeError): slot = {}
    target = (str(slot.get('date') or '')[:10], str(slot.get('period') or ''))
    kept=[];excluded=[]
    for row in shadow.get('prescription_history') or []:
        if (isinstance(row,dict) and str(row.get('nova_workout_id') or row.get('nova_contract_id') or '') == str(mismatch['workout_id'] or '')
                and _v153_plan_slot(row)==target and _v153_is_training_plan(row)):
            excluded.append(str(row.get('prescription_fingerprint') or ''))
        else:kept.append(row)
    shadow['prescription_history'] = kept
    if isinstance(shadow.get('primary_anchor'),dict) and (
         str(shadow['primary_anchor'].get('nova_workout_id') or shadow['primary_anchor'].get('nova_contract_id') or '') == str(mismatch['workout_id'] or '')):
        shadow['primary_anchor'] = None
    candidate = _v1962_call_prescriber(nova_decision, coach_clock, adaptive_roadmap, shadow,
        evidence_ledger, power_model, ftp_anchor, aerobic_metabolic_range, training_definitions)
    if not isinstance(candidate,dict) or _v1962_locked_identity(candidate, nova_decision):
        candidate = None
    if candidate is None:
        candidate = copy.deepcopy(out)
        i=mismatch.get('selected_slot_index')
        try: candidate['raw_sessions'][int(i)] = _v84_easy_raw(slot,nova_decision or {})
        except (ValueError, IndexError, TypeError, KeyError): pass
        candidate.update({'status':'EASY_ONLY','hard_prescription_available':False,
            'selected_slot_index':None,'workout_id':None,'contract_id':None,
            'fail_closed_reason':'INCOMPATIBLE_LOCK_NO_EVIDENCED_REPLACEMENT'})
        candidate.pop('single_plan_authority',None)
    candidate['prescription_continuity']={
        'schema':R1962_SCHEMA, 'active':False, 'state':'POLICY_CONFLICT_RECONCILED',
        'old_workout_id':mismatch['workout_id'], 'old_dimension':mismatch['original_dimension'],
        'current_dimension':mismatch['current_dimension'], 'excluded_history_fingerprints':excluded,
        'rule':'Historical workout identity stays in the ledger but cannot be executed against an incompatible live science decision.'}
    candidate['r1962_reconciliation'] = {'schema':R1962_SCHEMA,'state':'PRIOR_LOCK_INVALIDATED',
        'old_workout_id':mismatch['workout_id'],
        'replacement_workout_id':candidate.get('workout_id'),
        'status':candidate.get('status')}
    return candidate

def _v1965_bounded(value, low, high):
    return max(low, min(high, value))


def _v1965_number(value):
    if isinstance(value, bool):
        return None
    try:
        x = float(value)
        return x if math.isfinite(x) else None
    except (TypeError, ValueError):
        return None


def _v1965_personal_deviation(current, low, high, invert=False):
    current, low, high = map(_v1965_number, (current, low, high))
    if None in (current, low, high) or high <= low:
        return 0.0, False
    distance = (current - (low + high) / 2.0) / max(high - low, 1.0)
    return round(_v1965_bounded((-1.0 if invert else 1.0) * distance * 5.0, -6.0, 6.0), 2), True


def compute_energy_bank(form_zone, fatigue_zone, avg_sleep_hours, recent_activities=None,
                        personal_baselines=None, now=None, sleep_quality=None,
                        training_rhythm=None, wellness=None, metrics=None):
    """Heuristic energy index with continuous inputs; not a measured body battery."""
    out = _v1965_energy_previous(
        form_zone, fatigue_zone, avg_sleep_hours, recent_activities=recent_activities,
        personal_baselines=personal_baselines, now=now, sleep_quality=sleep_quality,
        training_rhythm=training_rhythm, wellness=wellness,
    )
    pb = personal_baselines or {}
    resolved = pb.get('resolved') or {}
    m = metrics or {}
    today = (wellness or [])[-1] if wellness else {}
    components = {'reference': 66.0}
    tsb = _v1965_number(m.get('tsb'))
    if tsb is not None:
        components['load_balance'] = round(_v1965_bounded(tsb * 0.68, -16, 16), 2)
    else:
        components['load_balance'] = {'green': 9, 'red': -11}.get(form_zone, 0)
    ctl = _v1965_number(m.get('ctl'))
    atl = _v1965_number(m.get('atl'))
    if ctl is not None and atl is not None and ctl > 0:
        components['fatigue_ratio'] = round(_v1965_bounded((1.0-atl/ctl)*9, -7, 7), 2)
    else:
        components['fatigue_ratio'] = {'green': 5, 'red': -6}.get(fatigue_zone, 0)
    hours = _v1965_number(avg_sleep_hours)
    habitual = _v1965_number((pb.get('sleep') or {}).get('habitual_minutes'))
    if hours is not None:
        reference_hours = habitual / 60.0 if habitual and habitual > 0 else 7.0
        components['sleep_duration'] = round(_v1965_bounded((hours-reference_hours)*5, -11, 7), 2)
    else:
        components['sleep_duration'] = 0.0
    components['sleep_quality'] = {1: 7.0, 2: 3.0, 3: -3.0, 4: -8.0}.get(sleep_quality, 0.0)
    for name, field, low, high, reverse in (
        ('hrv', 'hrv', 'hrv_low', 'hrv_high', False),
        ('resting_hr', 'restingHR', 'rhr_low', 'rhr_high', True),
    ):
        adjustment, available = _v1965_personal_deviation(today.get(field), resolved.get(low), resolved.get(high), reverse)
        components[name] = adjustment
    score_delta = _v1965_number(today.get('sleepScore'))
    score_reference = _v1965_number(resolved.get('sleep_score'))
    components['sleep_score'] = round(_v1965_bounded((score_delta-score_reference)*0.12, -4, 4), 2) if score_delta is not None and score_reference is not None else 0.0
    raw_start = sum(components.values())
    starting = round(_v1965_bounded(raw_start, 5, 100))
    score = round(_v1965_bounded(starting - float(out.get('energy_spent') or 0), 5, 100))
    label, zone = (('Charged', 'green') if score >= 72 else
                   ('Available', 'grey') if score >= 45 else
                   ('Used', 'orange') if score >= 25 else ('Drained', 'red'))
    theta = math.pi * (1 - score / 100.0)
    out.update({
        'energy_score': score, 'energy_label': label, 'energy_zone': zone,
        'energy_morning_score': starting, 'energy_starting_score': starting,
        'marker_x': round(100 + 87 * math.cos(theta), 1),
        'marker_y': round(95 - 87 * math.sin(theta), 1),
        'energy_model': 'R1965_CONTINUOUS_HEURISTIC',
        'energy_components': components,
        'energy_interpretation': 'Relative heuristic; not a measured physiological battery. Missing signals are neutral.',
    })
    return out

def _v1965_verified_quality_execution(ledger, start_date, now=None):
    """Never infer full-plan compliance from hard-only prescription history."""
    today = (now or get_rome_now()).date()
    try:
        start = date.fromisoformat(str(start_date)[:10])
    except (TypeError, ValueError):
        return {'available': False, 'reason': 'NO_DECLARED_PLAN_START', 'completed': 0, 'confirmed_missed': 0, 'unresolved': 0, 'score': None}
    start = max(start, today - timedelta(days=27))
    groups = {}
    for row in (ledger or {}).get('prescription_history') or []:
        if not isinstance(row, dict) or not (row.get('nova_workout_id') or row.get('nova_contract_id')):
            continue
        if not _v136_is_hard_summary(row) and not _v153_is_training_plan(row):
            continue
        try:
            due = date.fromisoformat(str(row.get('date') or '')[:10])
        except (TypeError, ValueError):
            continue
        if not (start <= due < today):
            continue
        status = str(row.get('prescription_status') or '').upper()
        if status in {'SUPERSEDED', 'CANCELLED', 'WITHDRAWN', 'EXPIRED_BY_NEW_HARD_EVIDENCE',
                      'DEFERRED_BY_ATHLETE'}:
            continue
        # A carried-forward instance is one commitment, not multiple missed doses.
        if row.get('carried_forward_to_fingerprint'):
            continue
        identity = (due.isoformat(), str(row.get('nova_workout_id') or row.get('nova_contract_id')))
        old = groups.get(identity)
        if old is None or (status == 'FULFILLED' and row.get('executed_activity_id')):
            groups[identity] = row
    completed = sum(str(r.get('prescription_status') or '').upper() == 'FULFILLED' and bool(r.get('executed_activity_id')) for r in groups.values())
    missed = sum(str(r.get('prescription_status') or '').upper() == 'MISSED_UNEXECUTED' for r in groups.values())
    unresolved = len(groups) - completed - missed
    # Unresolved attribution prevents a misleading numerical score.
    scored = completed + missed if unresolved == 0 else 0
    return {
        'available': bool(scored), 'score': round(100 * completed/scored) if scored else None,
        'completed': completed, 'confirmed_missed': missed, 'unresolved': unresolved,
        'observed_hard_contracts': len(groups),
        'reason': 'VERIFIED_HARD_ONLY' if scored else ('LINKS_UNRESOLVED' if unresolved else 'NO_VERIFIED_DUE_QUALITY_CONTRACTS'),
        'scope': 'VERIFIED_QUALITY_PRESCRIPTIONS_ONLY_NOT_FULL_PLAN_ADHERENCE',
    }

def _v1943_band(template, evidence_ledger=None, ftp_anchor=None, as_of_date=None, dimension=None):
    band = _r1967_band_before(template, evidence_ledger=evidence_ledger,
                              ftp_anchor=ftp_anchor, as_of_date=as_of_date,
                              dimension=dimension)
    if not isinstance(band, dict) or band.get('state') != 'PRESCRIBED':
        return band
    band = dict(band)
    band['repeatability_evidence_schema'] = R1967_SCHEMA
    source = str(band.get('source') or '')
    if source == 'COMPARABLE_VERIFIED_REPEATED_WORK':
        # A fresh best never establishes repeatability. Reconfirm a completed
        # canonical set with comparable repetitions, interval and *recovery*.
        # The earlier R194.3 selector also gates provenance, 42d age and CV.
        events = ((evidence_ledger or {}).get('repeatability') or {}).get('events') or []
        event = next((e for e in events if isinstance(e, dict)
                      and str(e.get('activity_id') or '') == str(band.get('activity_id') or '')
                      and e.get('strong') is True
                      and str(e.get('activity_truth_source') or e.get('source') or '').upper()
                      in {'ICU_INTERVALS_CANONICAL', 'DEVICE_LAPS'}
                      and abs((_v1943_finite(e.get('interval_secs'), 0) or -1)
                              - (band.get('evidence_interval_seconds') or 0)) <= 1), None)
        work = _v1943_finite(band.get('work_seconds'), 1)
        actual = _v1943_finite(band.get('evidence_interval_seconds'), 1)
        prescribed_recovery = _v1943_finite((template or {}).get('recovery_s'), 1)
        observed_recovery = _v1943_finite((event or {}).get('recovery_secs'), 1)
        # R196.9.0 — device lap boundaries commonly differ by a few seconds.
        # Recovery *duration* comparability is not a millisecond-precision test:
        # allow at most 5%, capped at 20 s (and at least 5 s) of timing jitter.
        # A materially longer/easier recovery remains noncomparable; unknown
        # recovery never silently passes. No change to reps, source, CV, age,
        # interval-duration or NOVA recovery/spacing safety gates.
        recovery_tolerance_s = (min(20.0, max(5.0, prescribed_recovery * 0.05))
                                if prescribed_recovery is not None else 0.0)
        comparable_recovery = bool(prescribed_recovery is not None
                                   and observed_recovery is not None
                                   and observed_recovery <= prescribed_recovery + recovery_tolerance_s)
        comparable_repetitions = bool(event and int(_v1943_finite(event.get('reps'), 1) or 0)
                                      >= int(band.get('reps') or 0))
        if not comparable_repetitions or not comparable_recovery or not actual or not work:
            band.update({'state':'NO_DEFENSIBLE_POWER_TARGET', 'confidence':'NONE',

                         'repeatability_reason':'MISSING_CANONICAL_REPEATS_OR_COMPARABLE_RECOVERY',
                         'reason':'MISSING_CANONICAL_REPEATS_OR_COMPARABLE_RECOVERY'})
            return band
        # Make the tolerance auditable; do not claim that a shorter verified
        # repeat proves watt-level repeatability for a longer workout.
        band['observed_recovery_seconds'] = observed_recovery
        band['planned_recovery_seconds'] = prescribed_recovery
        band['recovery_timing_tolerance_seconds'] = round(recovery_tolerance_s, 2)
        band['recovery_duration_comparable'] = comparable_recovery
        novel_context = str(dimension or '').upper() in {'RACE_TRANSFER', 'DURABILITY'}

                         'target_watts': None,
                         'power_target_mode': 'EXPLORATORY_RANGE'})
        else:
            band['repeatability_state'] = ('VERIFIED_COMPARABLE_REPEATS' if abs(actual-work) <= 1
                                            else 'SUPPORTED_BY_LONGER_REPEATS')
            band['repeatability_reason'] = 'REPEATED_WORK_AND_RECOVERY_COMPARABLE'
            band['power_target_mode'] = 'VERIFIED_REPEATED_TARGET'
            if actual > work:
                band['confidence'] = 'MODERATE'
        band['observed_single_best_is_repeatability_proof'] = False
        band['observed_repeated_work_is_direct_evidence'] = (
            band['repeatability_state'] == 'VERIFIED_COMPARABLE_REPEATS')
    elif source == 'VALID_FTP_STARTER_POLICY_NOT_OBSERVED_REPEATABILITY':
        # A supported FTP estimate can establish a starter zone; it cannot prove
        # a specific set of repetitions or license a pass/fail power grade.

    # Concordant autonomic, sleep or subjective signal -> original recovery.
    # A lone abnormal marker together with BOTH load alerts is also a
    # conservative, concrete warning pattern, not just high training load.
    if ('RECOVERY_MARKERS' in drivers or
            abnormal + int(sleep_bad) + int(subjective_bad) >= 2 or
            ((abnormal or sleep_bad or subjective_bad) and
             {'ACUTE_CHRONIC_LOAD','RECENT_LOAD_SPIKE'} <= drivers)):
        out['r196936_load_policy'] = 'CONCORDANT_RECOVERY_PATTERN_HOLD'
        return out
    out['state'] = 'BUILD' if (out.get('ctl_delta_28') or 0) >= 1.5 else 'MAINTENANCE'
    out['state_class'] = 'cyan' if out['state'] == 'BUILD' else 'green'
    out['icon'] = '⚠️'
    out['headline'] = 'Load elevated · monitor; Nova may prescribe purposeful work.'
    out['why'] = ('Elevated ATL/CTL or a weekly-load spike alone is an observation, '
                  'not proof that quality is contraindicated. Monitor the trends; '
                  'explicit restrictions, accumulated fatigue and session spacing '
                  'still apply independently.')
    out['nova_load_watch'] = {
        'schema':'NOVA-R196936-ADVISORY-1', 'state':'WATCH',
        'original_training_state':'RECOVERY',
        'drivers':sorted(drivers), 'quality_veto':False,
        'rule':'Single load ratio is advisory without corroborating signs.'}
    out['r196936_load_policy'] = 'LOAD_ONLY_WATCH_NOT_GATE'
    return out

_r19684_training_state_before = build_training_state

def build_training_state(season_activities, season_wellness, planning_context,
                         metrics, feeling_trend=None, now=None):
    raw = _r19684_training_state_before(
        season_activities, season_wellness, planning_context, metrics,
        feeling_trend=feeling_trend, now=now)
    out = _v19684_refine_load(raw, metrics=metrics, feeling_trend=feeling_trend,
                               planning_context=planning_context)
    if not isinstance(out, dict):
        return out
    p = ((metrics or {}).get('personal_baselines') or {})
    observations = []
    for key, label in (('hrv', 'HRV outside personal range'),
                       ('rhr', 'resting HR outside personal range'),
                       ('sleep', 'sleep below habitual range')):
        status = str((p.get(key) or {}).get('color') or '').lower()
        if status == 'red' or (key == 'sleep' and status == 'orange'):
            observations.append({'code': key.upper(), 'label': label,
                                 'severity': 'WATCH', 'source': 'PERSONAL_BASELINE'})
    if (feeling_trend or {}).get('label') == 'TRENDING DOWN':
        observations.append({'code': 'SUBJECTIVE_TREND', 'label': 'subjective recovery trending down',
                             'severity': 'WATCH', 'source': 'ATHLETE_FEEDBACK'})
    drivers = list(out.get('recovery_drivers') or [])
