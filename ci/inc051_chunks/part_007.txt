def percentile(sorted_values, pct):
    if not sorted_values:
        return None
    if len(sorted_values) == 1:
        return sorted_values[0]
    k = (len(sorted_values) - 1) * pct
    f, c = int(k), min(int(k) + 1, len(sorted_values) - 1)
    if f == c:
        return sorted_values[f]
    return sorted_values[f] + (sorted_values[c] - sorted_values[f]) * (k - f)

def bucket_zone_seconds(zone_seconds):
    n = len(zone_seconds)
    if n == 0:
        return 0, 0, 0
    if n == 3:
        low, mod, high = zone_seconds[0], zone_seconds[1], zone_seconds[2]
    elif n in (5, 6):
        low = sum(zone_seconds[0:2])
        mod = zone_seconds[2] if n == 5 else sum(zone_seconds[2:4])
        high = sum(zone_seconds[3:]) if n == 5 else sum(zone_seconds[4:])
    elif n == 7:
        low = sum(zone_seconds[0:2])
        mod = sum(zone_seconds[2:4])
        high = sum(zone_seconds[4:])
    else:
        third = max(1, n // 3)
        low = sum(zone_seconds[0:third])
        mod = sum(zone_seconds[third:2 * third])
        high = sum(zone_seconds[2 * third:])
    return low, mod, high

def _v4873_quality_spacing_profile(activities, now=None):
    now = now or get_rome_now()
    ends = []
    for a in activities or []:
        try:
            if _v4829_is_quality_activity(a):
                end = _v4829_activity_end(a)
                if end is not None and end <= now.replace(tzinfo=None):
                    ends.append(end)
        except Exception:
            continue
    ends = sorted(set(ends))[-16:]
    gaps = []
    for a, b in zip(ends, ends[1:]):
        hours = (b - a).total_seconds() / 3600.0

        "samples": len(gaps),

        "earliest_hours": round(earliest_h, 1),

    }
# R117: training opportunity is not the same signal as hard-session habit.
# Historical timing is a scheduling PRIOR, never proof of physiological recovery.
def _v117_hard_cadence_profile(activities, now=None):
    now = now or get_rome_now()
    cutoff = now.replace(tzinfo=None)
    rows = []
    for activity in activities or []:
        try:
            if not _v4829_is_quality_activity(activity):
                continue
            start = _v4829_parse_activity_start(activity)
            end = _v4829_activity_end(activity)
            if start is not None and end is not None and end <= cutoff:
                rows.append((end, start))
        except (TypeError, ValueError, KeyError):
            continue
    rows.sort(key=lambda item: item[0])
    # One hard training day contributes once, even if the athlete logs two activities.
    per_day = {}
    for end, start in rows:
        key = start.date()
        if key not in per_day or start < per_day[key][1]:
            per_day[key] = (end, start)
    recent = [per_day[k] for k in sorted(per_day)][-9:]
    days = [start.date() for _, start in recent]
    gaps = [(b - a).days for a, b in zip(days, days[1:]) if 1 <= (b-a).days <= 9]
    last_gaps = gaps[-6:]
    gap_mode = None
    gap_confidence = 'LOW'
    if len(last_gaps) >= 3:
        # Prefer the current repeating pattern over lifetime/weekday coincidence.
        counts = {v:last_gaps.count(v) for v in set(last_gaps)}
        best = sorted(counts, key=lambda v:(-counts[v], abs(v-last_gaps[-1]), v))[0]
        if counts[best] / len(last_gaps) >= 0.55:
            gap_mode = best
            gap_confidence = 'HIGH' if len(last_gaps) >= 5 and counts[best]/len(last_gaps) >= .75 else 'MODERATE'
    starts = [start.hour*60+start.minute for _,start in recent[-8:]]
    am = [x for x in starts if x < 12*60]
    pm = [x for x in starts if x >= 12*60]
    preferred_daypart = None
    hard_start_minute = None
    clock_confidence = 'LOW'
    if len(starts) >= 3:
        share = max(len(am),len(pm)) / len(starts)
        if share >= .70:
            preferred_daypart = 'AM' if len(am) > len(pm) else 'PM'
            hard_start_minute = int(round(statistics.median(am if preferred_daypart=='AM' else pm)))
            clock_confidence = 'HIGH' if len(starts)>=5 and share >= .80 else 'MODERATE'
    return {
        'source':'OBSERVED_CANONICAL_HARD_ACTIVITY_STARTS',
        'sample_days':len(recent), 'sample_gaps':len(last_gaps),
        'recent_day_gaps':last_gaps, 'dominant_gap_days':gap_mode,
        'gap_confidence':gap_confidence, 'hard_daypart':preferred_daypart,
        'hard_start_minute':hard_start_minute, 'time_confidence':clock_confidence,
        'last_hard_day':days[-1].isoformat() if days else None,
        'note':'Habit is a scheduling prior; activity frequency and median hard spacing cannot certify recovery.'
    }

def _v117_select_quality_window(candidates, last_end, spacing, cadence, extra_delay, second_hard=False):
    """Safety/recovery gates upstream; rank habit only among permitted candidates.

    The lower bound is a conservative PRODUCT HEURISTIC, not a clinical clearance.
    The observed median is a descriptive prior, NOT an exact hour threshold.
    """
    lower=last_end+timedelta(hours=float(spacing.get('earliest_hours') or 36)+extra_delay) if last_end else None
    preferred=last_end+timedelta(hours=float(spacing.get('preferred_hours') or 48)+extra_delay) if last_end else None
    valid=[r for r in candidates if lower is None or r['dt']>=lower]
    if not valid:
        return None,None,'NO_VALID_CLOCK_SLOT',None
    explicit=[r for r in valid if r.get('explicit')]
    if explicit:
        return valid[0]['dt'],explicit[0]['dt'],'EXPLICIT_PLANNED_HARD',explicit[0]
    first=valid[0]
    # Period and alternating-day tendencies must be learned from HARD sessions,
    # not inferred from every easy AM ride in the regular training rhythm.
    gap=cadence.get('dominant_gap_days')
    part=cadence.get('hard_daypart')
    preferred_minute=cadence.get('hard_start_minute')
    last_day=date.fromisoformat(cadence['last_hard_day']) if cadence.get('last_hard_day') else (last_end.date() if last_end else None)
    strong=gap is not None or part is not None
    if strong:
        def score(row):
            dt=row['dt']; days=(dt.date()-last_day).days if last_day else None
            minute=dt.hour*60+dt.minute
            value=0.0
            if gap is not None and days is not None:
                value+=3.0 if days==gap else max(-5.0, 1.0-2.0*abs(days-gap))
            if part:
                value+=3.0 if ('AM' if minute<720 else 'PM')==part else -3.0
            if preferred_minute is not None:
                value+=max(-2.0, 1.0-abs(minute-preferred_minute)/120.0)
            if preferred is not None and dt>=preferred: value+=.5
            # In an equivalent pattern do not needlessly defer the training goal.
            value-=(dt-first['dt']).total_seconds()/86400*0.40
            return value
        ranked=sorted(valid,key=lambda row:(-score(row),row['dt']))
        selected=ranked[0]
        # If the learned hard pattern is clear, don't force an otherwise valid
        # athlete-pattern slot to cross the exact observed median to be 'preferred'.
        reason='OBSERVED_HARD_DAY_AND_DAYPART' if gap is not None and part else ('OBSERVED_HARD_DAY' if gap is not None else 'OBSERVED_HARD_DAYPART')
        return first['dt'],selected['dt'],reason,selected
    # Sparse/irregular records: honor the conservative *planning preference*
    # without claiming that a 0.1-hour difference proves recovery.
    selected=next((r for r in valid if preferred is None or r['dt']>=preferred),first)
    return first['dt'],selected['dt'],'SPARSE_HARD_HISTORY_CONSERVATIVE_PREFERENCE',selected

def _v4873_rhythm_candidate_slots(training_rhythm, now=None, horizon_days=8):
    now = now or get_rome_now()
    rhythm = training_rhythm or {}
    if rhythm.get("state") == "LEARNING":
        return []
    scores = rhythm.get("weekday_scores") or {}
    likely = set(rhythm.get("likely_weekdays") or [])
    strength = str(rhythm.get("pattern_strength") or "LOW").upper()
    clock = rhythm.get("clock_profile") or {}
    first_map = clock.get("first_start_min") or {}
    second_map = clock.get("second_start_min") or {}
    overall_first = _rhythm_num(clock.get("overall_first_start_min"))
    overall_second = _rhythm_num(clock.get("overall_second_start_min"))
    double_scores = rhythm.get("weekday_double_scores") or {}
    double_prop = float(rhythm.get("double_propensity") or 0.0)
    out = []
    for offset in range(0, max(1, int(horizon_days)) + 1):
        d = now.date() + timedelta(days=offset)
        wd = d.weekday()
        score = float(scores.get(wd, scores.get(str(wd), 0.0)) or 0.0)
        supported = (wd in likely and score >= 0.25) if strength != "LOW" else score >= 0.45
        if not supported:
            continue
        first = _rhythm_num(first_map.get(wd, first_map.get(str(wd))))
        if first is None:
            first = overall_first
        if first is not None:
            hh, mm = divmod(max(0, min(1439, int(round(first)))), 60)
            dt = datetime.combine(d, datetime.min.time()).replace(hour=hh, minute=mm)
            if dt > now.replace(tzinfo=None):
                out.append({"dt": dt, "period": "FIRST", "score": score})
        dscore = float(double_scores.get(wd, double_scores.get(str(wd), 0.0)) or 0.0)
        second = _rhythm_num(second_map.get(wd, second_map.get(str(wd))))
        if second is None:
            second = overall_second
        if double_prop >= 0.45 and dscore >= 0.45 and second is not None:
            hh, mm = divmod(max(0, min(1439, int(round(second)))), 60)
            dt = datetime.combine(d, datetime.min.time()).replace(hour=hh, minute=mm)
            if dt > now.replace(tzinfo=None):
                out.append({"dt": dt, "period": "SECOND", "score": score})
    out.sort(key=lambda x: x["dt"])
    return out
def _v4873_slot_label(dt, now=None):
    now = now or get_rome_now()
    if dt is None:
        return "Not inferred"
    today = now.date()
    if dt.date() == today:
        day = "Today"
    elif dt.date() == today + timedelta(days=1):
        day = "Tomorrow"
    else:
        day = dt.strftime("%A")
    return f"{day} · around {dt.strftime('%H:%M')}"

def _v4896_event_id(event):
    raw = "|".join(str((event or {}).get(k) or "") for k in ("kind","date","secs","watts","activity_id","source"))

def _v4896_validation_interval_match(block, test_seconds, mode="MAXIMAL"):
    rows = list((block or {}).get("work_interval_evidence") or [])
    try:
        target = float(test_seconds)
    except (TypeError, ValueError):
        return None
    if target <= 0:
        return None
    tolerance = max(2.0, min(30.0, target * 0.05))
    matches = [
        r for r in rows
        if abs(float((r or {}).get("secs") or 0) - target) <= tolerance
        and _rhythm_num((r or {}).get("watts")) is not None
    ]
    if not matches:
        return None
    # Extra sets in the same ride do not erase an exact-duration observation.
    # This matcher supplies evidence, NOT permission to invent a checkpoint.
    best = max(matches, key=lambda r: float(r.get("watts") or 0))
    return {"secs": int(best.get("secs") or round(target)), "watts": round(float(best.get("watts")), 1)}

    if family in {"VO2max", "Sprint / Neuromuscular", "High Intensity / Mixed"}:
        return "high"
    return None
def _td_distribution_stats(activities):
    low_secs = mod_secs = high_secs = 0.0
    zone_sessions = 0
    session_bins = {"low": 0, "moderate": 0, "high": 0}
    family_counts = {}
    usable_sessions = 0
    for activity in activities or []:
        if not _v4887_is_cycling_activity(activity):
            continue
        family = classify_training_family(activity)
        family_counts[family] = family_counts.get(family, 0) + 1
        b = _td_family_bin(family)
        if b:
            session_bins[b] += 1
            usable_sessions += 1
        zt = activity.get("icu_zone_times") or []
        if zt:
            try:
                secs = [float(z.get("secs") or 0) for z in zt]
                low, mod, high = bucket_zone_seconds(secs)
                if (low + mod + high) > 0:
                    low_secs += low; mod_secs += mod; high_secs += high
                    zone_sessions += 1
            except (TypeError, ValueError):
                pass
    total = low_secs + mod_secs + high_secs
    zone_pct = None
    if total > 0:
        zone_pct = {
            "low": round(100.0 * low_secs / total, 1),
            "moderate": round(100.0 * mod_secs / total, 1),
            "high": round(100.0 * high_secs / total, 1),
        }
    sb_total = sum(session_bins.values())
    session_pct = None
    if sb_total:
        session_pct = {k: round(100.0 * v / sb_total, 1) for k, v in session_bins.items()}
    return {
        "zone_pct": zone_pct, "session_pct": session_pct,
        "zone_sessions": zone_sessions, "usable_sessions": usable_sessions,
        "family_counts": family_counts,
    }
def _td_classify_distribution(pct):
    if not pct:
        return "LEARNING"
    low = float(pct.get("low") or 0); mod = float(pct.get("moderate") or 0); high = float(pct.get("high") or 0)
    if low >= 65 and high >= 5 and high >= mod + 2:
        return "POLARIZED"
    if low >= 58 and mod >= high + 2 and mod < 25:
        return "PYRAMIDAL"
    if mod >= 25 and mod >= high + 8:
        return "THRESHOLD"
    if low >= 68 and mod >= high + 2:
        return "PYRAMIDAL"
    return "MIXED"
def _td_alignment_score(pct, target):
    if not pct or target not in _TD_DISTRIBUTION_PROFILES:
        return None
    centre = _TD_DISTRIBUTION_PROFILES[target]
    moved_points = sum(abs(float(pct.get(k) or 0) - centre[k]) for k in ("low", "moderate", "high")) / 2.0
    score = max(0.0, 100.0 - 1.55 * moved_points)
    low = float(pct.get("low") or 0); mod = float(pct.get("moderate") or 0); high = float(pct.get("high") or 0)
    if target == "POLARIZED" and not (low > high > mod):
        score -= 10
    elif target == "PYRAMIDAL" and not (low > mod > high):
        score -= 10
    elif target == "THRESHOLD" and mod < 22:
        score -= 10
    return round(max(0.0, min(100.0, score)))
def _td_goal_score(activities, goal_key):
    cfg = _TD_GOALS.get(goal_key)
    if not cfg:
        return None, {"primary_count": 0, "hard_count": 0, "usable_sessions": 0}
    weighted = []
    primary_count = hard_count = 0
    for activity in activities or []:
        if not _v4887_is_cycling_activity(activity):
            continue
        family = classify_training_family(activity)
        if family == "Other / Mixed":
            continue
        if family in cfg["primary"]:
            weight = 1.0; primary_count += 1
        elif family in cfg["support"]:
            weight = 0.72
        else:
            weight = 0.32
        if _v4895_activity_quality_truth(activity, family).get("quality_relevant"):
            hard_count += 1
        weighted.append(weight)
    if not weighted:
        return None, {"primary_count": 0, "hard_count": 0, "usable_sessions": 0}
    base = 100.0 * statistics.mean(weighted)
    minimum = int(cfg.get("anchor_min_14d") or 1)
    if primary_count == 0:
        base *= 0.58
    elif primary_count < minimum:
        base *= 0.82
    if hard_count > 5:
        base -= min(15.0, (hard_count - 5) * 4.0)
    return round(max(0.0, min(100.0, base))), {
        "primary_count": primary_count, "hard_count": hard_count, "usable_sessions": len(weighted),
    }
def _td_score_label(score):

        return "LEARNING", "learning"
    if score >= 85:
        return "STRONG ALIGNMENT", "strong"

def _td_distribution_correction(target, pct):
    if target == "AUTO" or not pct or target not in _TD_DISTRIBUTION_PROFILES:
        return None
    centre = _TD_DISTRIBUTION_PROFILES[target]
    deltas = {k: float(pct.get(k) or 0) - centre[k] for k in centre}
    excess = max(deltas, key=deltas.get)
    deficit = min(deltas, key=deltas.get)
    if deltas[excess] < 6 and deltas[deficit] > -6:
        return "The current intensity balance is already close to the selected distribution; preserve the pattern rather than chasing exact percentages."
    if excess == "moderate":
        return "Moderate/threshold time is the main drift. Reduce unplanned medium-hard accumulation before adding more quality."
    if excess == "high":
        return "High-intensity time is the main drift. Protect quality by reducing hard-session density and restoring low-intensity space."
    if deficit == "low":
        return "Low-intensity volume is under-represented. Rebuild the aerobic floor before increasing the density of quality work."
    if target == "POLARIZED" and deficit == "high":
        return "Low intensity is dominant, but the polarized high-intensity pole is under-represented. Add purposeful high-intensity work only when recovery permits."
    if target in {"PYRAMIDAL", "THRESHOLD"} and deficit == "moderate":
        return "The selected model calls for more controlled moderate/threshold work; add it deliberately rather than turning easy rides into accidental tempo."
    return "The current distribution is drifting from the selected model. Adjust the largest excess first instead of forcing every zone toward an exact percentage."

def _wellness_value_rows(wellness, field, temp_field=None):
    rows = []
    for w in sorted(wellness or [], key=lambda x: x.get("id", "")):
        value = w.get(field)
        if isinstance(value, bool) or not isinstance(value, (int, float)):
            continue
        value_f = float(value)
        if not math.isfinite(value_f):
            continue
        ds = str(w.get("id") or "")[:10]
        try:
            day = date.fromisoformat(ds)
        except ValueError:
            continue
        rows.append({
            "date": day, "value": value,
            "temporary": bool(temp_field and _intervals_temp_flag(w, temp_field)),
            "row": w,
        })
    return rows
def _latest_wellness_observation(wellness, field, temp_field=None):
    rows = _wellness_value_rows(wellness, field, temp_field)
    if not rows:
        return {"value": None, "date": None, "temporary_only": False, "carried_forward": False, "latest_row_date": None}
    genuine = [r for r in rows if not r["temporary"]]
    latest_any = rows[-1]
    if not genuine:
        return {
            "value": latest_any["value"], "date": None, "temporary_only": True,
            "carried_forward": True, "latest_row_date": latest_any["date"],
        }
    latest = genuine[-1]
    carried = bool(latest_any["date"] > latest["date"] and latest_any["temporary"])
    return {
        "value": latest["value"], "date": latest["date"], "temporary_only": False,
        "carried_forward": carried, "latest_row_date": latest_any["date"],
    }
def _latest_weight_provenance(wellness):
    rows = _wellness_value_rows(wellness, "weight", "tempWeight")
    if not rows:
        return {
            "value": None, "date": None, "latest_row_date": None,
            "unchanged_since": None, "same_value_rows": 0,
            "latest_row_temporary": False, "carried_forward": False,
        }
    latest = rows[-1]
    latest_display = round(float(latest["value"]), 1)
    block = [latest]
    for row in reversed(rows[:-1]):
        if round(float(row["value"]), 1) != latest_display:
            break
        block.append(row)
    block.reverse()
    unchanged_since = block[0]["date"]
    latest_row_date = latest["date"]
    latest_row_temporary = bool(latest.get("temporary"))
    return {
        "value": latest["value"],
        "date": unchanged_since,
        "unchanged_since": unchanged_since,
        "latest_row_date": latest_row_date,
        "same_value_rows": len(block),
        "latest_row_temporary": latest_row_temporary,
        "carried_forward": bool(latest_row_temporary or len(block) > 1),

def build_adaptive_sleep_baseline(wellness, lookback_days=SLEEP_BASELINE_DAYS):
    rows = []
    for w in sorted(wellness or [], key=lambda x: str(x.get("id") or "")):
        raw_day = str(w.get("id") or "")[:10]
        secs = w.get("sleepSecs")
        if not isinstance(secs, (int, float)) or not raw_day:
            continue
        try:
            d = date.fromisoformat(raw_day)
        except Exception:
            continue
        minutes = float(secs) / 60.0
        if not 180 <= minutes <= 720:
            continue
        rows.append({"date": d, "minutes": minutes, "kind": "WEEKDAY" if d.weekday() < 5 else "WEEKEND"})
    manual_sleep_reference = GARMIN_BASELINES.get("sleep_minutes")
    reference_label = (_fmt_hours(float(manual_sleep_reference) / 60.0) + " manual reference") if isinstance(manual_sleep_reference, (int, float)) else "No manual sleep reference"
    if not rows:
        return {
            "available": False, "days": lookback_days, "kind": "UNKNOWN",
            "current_minutes": None, "habitual_minutes": None, "current_delta_min": None,
            "weekday_minutes": None, "weekend_minutes": None, "overall_minutes": None,
            "weekday_label": "n/a", "weekend_label": "n/a", "overall_label": "n/a",
            "weekday_iqr": None, "weekend_iqr": None, "status": "NO DATA", "color": "grey", "pct": 0,
            "reference_minutes": manual_sleep_reference, "reference_label": reference_label,
            "reference_delta_min": None, "sample_count": 0, "weekday_samples": 0, "weekend_samples": 0,
            "trend_label": "NO TREND", "trend_delta_min": None,
        }
    current = rows[-1]
    cutoff = current["date"] - timedelta(days=lookback_days)
    baseline_rows = [r for r in rows[:-1] if r["date"] >= cutoff]
    if not baseline_rows:
        baseline_rows = rows[:-1]
    weekday = [r["minutes"] for r in baseline_rows if r["kind"] == "WEEKDAY"]
    weekend = [r["minutes"] for r in baseline_rows if r["kind"] == "WEEKEND"]
    overall = [r["minutes"] for r in baseline_rows]
    overall_median = _median_minutes(overall)
    weekday_median = _median_minutes(weekday) if len(weekday) >= 5 else overall_median
    weekend_median = _median_minutes(weekend) if len(weekend) >= 2 else overall_median
    habitual = weekday_median if current["kind"] == "WEEKDAY" else weekend_median
    delta = current["minutes"] - habitual if habitual is not None else None
    if delta is None:
        status, color, pct = "INSUFFICIENT DATA", "grey", 0
    else:
        if delta >= -20:
            status, color = "WITHIN HABITUAL", "green"
        elif delta >= -45:
            status, color = "SLIGHTLY SHORT", "grey"
        elif delta >= -75:
            status, color = "SHORT VS HABITUAL", "orange"
        else:
            status, color = "WELL BELOW HABITUAL", "red"
        pct = max(10, min(100, round(current["minutes"] / habitual * 100))) if habitual else 0
    prior_values = [r["minutes"] for r in rows[:-1]][-28:]
    recent14 = prior_values[-14:]
    prior14 = prior_values[-28:-14]
    trend_delta = None
    trend_label = "NO TREND"
    if len(recent14) >= 5 and len(prior14) >= 5:
        trend_delta = round(statistics.mean(recent14) - statistics.mean(prior14))
        if trend_delta >= 15:
            trend_label = "LONGER"
        elif trend_delta <= -15:
            trend_label = "SHORTER"
        else:
            trend_label = "STABLE"
    ref_delta = round(current["minutes"] - manual_sleep_reference) if isinstance(manual_sleep_reference, (int, float)) else None
    return {
        "available": habitual is not None,
        "days": lookback_days,
        "kind": current["kind"],
        "current_minutes": round(current["minutes"]),
        "habitual_minutes": round(habitual) if habitual is not None else None,
        "current_delta_min": round(delta) if delta is not None else None,
        "weekday_minutes": round(weekday_median) if weekday_median is not None else None,
        "weekend_minutes": round(weekend_median) if weekend_median is not None else None,
        "overall_minutes": round(overall_median) if overall_median is not None else None,
        "weekday_label": _fmt_hours(weekday_median / 60.0) if weekday_median is not None else "n/a",
        "weekend_label": _fmt_hours(weekend_median / 60.0) if weekend_median is not None else "n/a",
        "overall_label": _fmt_hours(overall_median / 60.0) if overall_median is not None else "n/a",
        "weekday_iqr": _sleep_iqr_label(weekday),
        "weekend_iqr": _sleep_iqr_label(weekend),
        "habitual_label": _fmt_hours(habitual / 60.0) if habitual is not None else "n/a",
        "status": status, "color": color, "pct": pct,
        "reference_minutes": manual_sleep_reference, "reference_label": reference_label,
        "reference_delta_min": ref_delta,
        "sample_count": len(baseline_rows), "weekday_samples": len(weekday), "weekend_samples": len(weekend),
        "trend_label": trend_label, "trend_delta_min": trend_delta,
    }
def _range_signal(value, low, high, kind="neutral"):
    if not isinstance(value, (int, float)):
        return {"color": "grey", "status": "No current data", "pct": 0}
    if not isinstance(low, (int, float)) or not isinstance(high, (int, float)) or high < low:
        return {"color": "grey", "status": "Learning baseline", "pct": 0}
    width = max(high - low, 1)
    if low <= value <= high:
        return {"color": "green", "status": "Within baseline", "pct": 100}
    if value < low:
        distance = low - value
        pct = max(10, round(100 - (distance / width) * 75))
        if kind == "hrv":
            color = "red" if distance > width * 0.25 else "grey"
            status = "Below baseline"
        elif kind == "heart_rate":
            color = "grey" if distance <= 3 else "red"
            status = "Below baseline"
        else:
            color, status = "grey", "Below baseline"
        return {"color": color, "status": status, "pct": pct}
    distance = value - high
    pct = max(10, round(100 - (distance / width) * 75))
    if kind == "hrv":
        color = "green" if distance <= width * 0.25 else "grey"
        status = "Above baseline"
    elif kind == "heart_rate":
        color = "red" if distance > 2 else "grey"
        status = "Above baseline"
    else:
        color, status = "grey", "Above baseline"
    return {"color": color, "status": status, "pct": pct}
def _target_signal(value, target, metric):
    if not isinstance(value, (int, float)):
        return {"color": "grey", "status": "No current data", "pct": 0}
    if not isinstance(target, (int, float)):
        return {"color": "grey", "status": "Learning baseline", "pct": 0}
    if metric == "sleep_minutes":
        diff = value - target
        pct = max(10, min(100, round(value / target * 100))) if target else 0
        if diff >= -30:
            return {"color": "green", "status": "On target", "pct": pct}
        if diff >= -75:
            return {"color": "grey", "status": "Below target", "pct": pct}
        return {"color": "red", "status": "Short sleep", "pct": pct}
    if metric == "sleep_score":
        pct = max(10, min(100, round(value)))
        if value >= target - 3:
            return {"color": "green", "status": "Near baseline", "pct": pct}
        if value >= target - 10:
            return {"color": "grey", "status": "Below baseline", "pct": pct}
        return {"color": "red", "status": "Low vs baseline", "pct": pct}
    if metric == "spo2":
        pct = max(10, min(100, round(value)))
        if value >= target - 1:
            return {"color": "green", "status": "Near baseline", "pct": pct}
        if value >= target - 2:
            return {"color": "grey", "status": "Below baseline", "pct": pct}
        return {"color": "red", "status": "Low vs baseline", "pct": pct}
    return {"color": "grey", "status": "Reference", "pct": 0}
def _robust_dynamic_range(values, min_samples=7):
    vals = sorted(float(v) for v in values if isinstance(v, (int, float)) and math.isfinite(float(v)))
    if len(vals) < min_samples:
        return None
    q1, med, q3 = statistics.quantiles(vals, n=4, method="inclusive")
    if q3 <= q1:
        spread = max(abs(med) * 0.02, 1.0)
        q1, q3 = med - spread, med + spread
    return float(q1), float(q3)
def _robust_dynamic_target(values, min_samples=7):
    vals = [float(v) for v in values if isinstance(v, (int, float)) and math.isfinite(float(v))]
    return float(statistics.median(vals)) if len(vals) >= min_samples else None
def _reference_range(manual_low, manual_high, history_values):
    if isinstance(manual_low, (int, float)) and isinstance(manual_high, (int, float)) and manual_high >= manual_low:
        return float(manual_low), float(manual_high), "MANUAL REFERENCE"
    learned = _robust_dynamic_range(history_values)
    if learned:
        return learned[0], learned[1], "DYNAMIC BASELINE"
    return None, None, "LEARNING"
def _reference_target(manual_target, history_values):
    if isinstance(manual_target, (int, float)):
        return float(manual_target), "MANUAL REFERENCE"
    learned = _robust_dynamic_target(history_values)
    return (learned, "DYNAMIC BASELINE") if learned is not None else (None, "LEARNING")
def _range_label(source, low, high, unit=""):
    if low is None or high is None:
        return "Learning · data needed"
    precision = 1 if any(abs(v - round(v)) > 1e-6 for v in (low, high)) else 0
    return f"{source.title()} · {low:.{precision}f}–{high:.{precision}f}{unit}"
def _target_label(source, target, unit=""):
    if target is None:
        return "Learning · data needed"
    precision = 1 if abs(target - round(target)) > 1e-6 else 0
    return f"{source.title()} · {target:.{precision}f}{unit}"

        "rhr": {**rhr_signal, "garmin": _range_label(rhr_source, rhr_low, rhr_high, " bpm"), "recent": _fmt_number(statistics.median(rhr_recent), 0, " bpm") if rhr_recent else "n/a", "reference_source": rhr_source},

def compute_health_rings(metrics):
    pb = metrics.get("personal_baselines") or {}
    rings = {}
    for key in ("rhr", "hrv", "sleep"):
        item = pb.get(key)
        if item:
            rings[key] = {"pct": item.get("pct", 0), "color": item.get("color", "grey"), "status": item.get("status", "Reference")}
    return rings

def compute_recent_trend(wellness, n=5, form_thresholds=None, form_median=None):
    daily = []
    for w in sorted(wellness, key=lambda x: x.get("id", "")):
        ctl, atl = w.get("ctl"), w.get("atl")
        if ctl is None or atl is None:
            continue
        tsb = round(ctl - atl, 1)
        try:
            weekday = date.fromisoformat(w["id"]).strftime("%a")
        except (ValueError, KeyError):
            weekday = w.get("id", "")[-2:]
        delta = round(tsb - form_median, 1) if form_median is not None else None
        daily.append({
            "date": w.get("id", ""), "weekday": weekday, "tsb": tsb, "delta": delta,
            "zone": classify_form(tsb, form_thresholds),
        })
    return daily[-n:]
def _energy_training_day_context(today_load, today_minutes, training_rhythm=None):
    rhythm = training_rhythm or {}
    typical_load = _rhythm_num(rhythm.get("typical_training_day_load"))
    q25_load = _rhythm_num(rhythm.get("training_day_load_q25"))
    q75_load = _rhythm_num(rhythm.get("training_day_load_q75"))
    typical_minutes = _rhythm_num(rhythm.get("typical_training_day_minutes"))
    q25_minutes = _rhythm_num(rhythm.get("training_day_minutes_q25"))
    q75_minutes = _rhythm_num(rhythm.get("training_day_minutes_q75"))
    load_samples = int(rhythm.get("training_day_load_samples") or 0)
    duration_samples = int(rhythm.get("training_day_duration_samples") or 0)
    learned = (
        str(rhythm.get("state") or "LEARNING") != "LEARNING"
        and (load_samples >= 3 or duration_samples >= 3)
        and (typical_load is not None or typical_minutes is not None)
    )
    if today_load <= 0 and today_minutes <= 0:
        return {
            "label": "NO TRAINING YET", "class": "grey", "learned": learned,
            "extra_cost": 0.0, "load_ratio": None, "duration_ratio": None,
            "typical_load": typical_load, "typical_minutes": typical_minutes,
        }
    if not learned:
        return {
            "label": "BASELINE LEARNING", "class": "grey", "learned": False,
            "extra_cost": 0.0,
            "load_ratio": round(today_load / typical_load, 2) if typical_load and today_load > 0 else None,
            "duration_ratio": round(today_minutes / typical_minutes, 2) if typical_minutes and today_minutes > 0 else None,
            "typical_load": typical_load, "typical_minutes": typical_minutes,
        }
    load_ratio = today_load / typical_load if typical_load and typical_load > 0 else None
    duration_ratio = today_minutes / typical_minutes if typical_minutes and typical_minutes > 0 else None
    ratios = [r for r in (load_ratio, duration_ratio) if r is not None]
    max_ratio = max(ratios) if ratios else 1.0
    light_load = q25_load is not None and today_load <= q25_load
    light_duration = q25_minutes is not None and today_minutes <= q25_minutes
    above_load = q75_load is not None and today_load > q75_load
    above_duration = q75_minutes is not None and today_minutes > q75_minutes
    if max_ratio >= 1.55 or (above_load and above_duration and max_ratio >= 1.30):
        label, css = "HIGH VS YOUR NORM", "orange"
    elif above_load or above_duration or max_ratio >= 1.20:
        label, css = "ABOVE YOUR NORM", "orange"
    elif (light_load or q25_load is None) and (light_duration or q25_minutes is None) and max_ratio <= 0.75:
        label, css = "LIGHT VS YOUR NORM", "green"
    else:
        label, css = "TYPICAL VS YOUR NORM", "grey"
    load_anchor = q75_load if q75_load and q75_load > 0 else (typical_load * 1.15 if typical_load else None)
    minute_anchor = q75_minutes if q75_minutes and q75_minutes > 0 else (typical_minutes * 1.15 if typical_minutes else None)
    load_excess = max(0.0, today_load - load_anchor) if load_anchor is not None else 0.0
    minute_excess = max(0.0, today_minutes - minute_anchor) if minute_anchor is not None else 0.0
    extra_cost = min(16.0, load_excess * 0.08 + minute_excess * 0.015)
    return {
        "label": label, "class": css, "learned": True, "extra_cost": extra_cost,
        "load_ratio": round(load_ratio, 2) if load_ratio is not None else None,
        "duration_ratio": round(duration_ratio, 2) if duration_ratio is not None else None,
        "typical_load": round(typical_load, 1) if typical_load is not None else None,
        "typical_minutes": round(typical_minutes, 1) if typical_minutes is not None else None,
    }
def _energy_ambulatory_activity(activity):
    activity = activity or {}
    kind = " ".join(str(activity.get(k) or "") for k in ("type", "sub_type")).strip().lower()
    if kind:
        return any(key in kind for key in ("run", "walk", "hike", "hiking"))
    name = str(activity.get("name") or "").strip().lower()
    return any(key in name for key in ("run", "walk", "hike", "hiking"))
def _energy_steps_context(wellness, now=None, today_activities=None):
    now = now or get_rome_now()
    today = now.date()
    history_start = today - timedelta(days=42)
    by_day = {}
    for row in wellness or []:
        raw_day = str((row or {}).get("id") or "")[:10]
        try:
            day = date.fromisoformat(raw_day)
        except ValueError:
            continue
        raw_steps = (row or {}).get("steps")
        if isinstance(raw_steps, bool):
            continue
        try:
            steps = int(round(float(raw_steps)))
        except (TypeError, ValueError):
            continue
        if steps < 0:
            continue
        by_day[day] = steps
    today_steps = by_day.get(today)
    if today_steps is None:
        return {
            "available": False, "steps": None, "steps_label": None, "label": None,
            "class": "grey", "learned": False, "samples": 0, "typical_steps": None,
            "q75_steps": None, "ratio": None, "cost": 0.0, "overlap_guard": False,
        }
    history = sorted(
        steps for day, steps in by_day.items()
        if history_start <= day < today and isinstance(steps, int) and steps > 0
    )
    learned = len(history) >= 7
    typical = statistics.median(history) if history else None
    q75 = percentile(history, 0.75) if learned else None
    ratio = (today_steps / typical) if typical and typical > 0 else None
    base_cost = min(4.5, max(0.0, float(today_steps)) / 4000.0)
    extra_cost = 0.0
    if learned and typical and typical > 0:
        anchor = q75 if q75 and q75 > 0 else typical * 1.20
        excess = max(0.0, float(today_steps) - float(anchor))
        extra_cost = min(3.0, excess / 2500.0)
    if learned and ratio is not None:
        if ratio >= 1.55 or (q75 is not None and today_steps >= q75 * 1.30):
            label, css = "HIGH MOVEMENT VS YOUR NORM", "orange"
        elif ratio >= 1.20 or (q75 is not None and today_steps > q75):
            label, css = "ABOVE MOVEMENT VS YOUR NORM", "orange"
        else:
            label, css = "WITHIN YOUR USUAL MOVEMENT SO FAR", "grey"
    else:
        label, css = "MOVEMENT BASELINE LEARNING", "grey"
    overlap_guard = any(_energy_ambulatory_activity(a) for a in (today_activities or []))
    cost = min(7.5, base_cost + extra_cost)
    if overlap_guard:
        cost *= 0.55
    return {
        "available": True,
        "steps": today_steps,
        "steps_label": f"{today_steps:,}",
        "label": label,
        "class": css,
        "learned": learned,
        "samples": len(history),
        "typical_steps": round(typical) if typical is not None else None,

def compute_energy_bank(form_zone, fatigue_zone, avg_sleep_hours, recent_activities=None, personal_baselines=None, now=None, sleep_quality=None, training_rhythm=None, wellness=None):
    now = now or get_rome_now()
    recent_activities = recent_activities or []
    wellness = wellness or []
    pb = personal_baselines or {}
    starting = 62
    starting += {"green": 12, "grey": 0, "red": -14}.get(form_zone, 0)
    starting += {"green": 9, "grey": 0, "red": -11}.get(fatigue_zone, 0)
    if isinstance(avg_sleep_hours, (int, float)):
        sleep_target_minutes = (pb.get("sleep") or {}).get("habitual_minutes")
        if isinstance(sleep_target_minutes, (int, float)) and sleep_target_minutes > 0:
            sleep_delta = avg_sleep_hours * 60.0 - sleep_target_minutes
            if sleep_delta >= 20:
                starting += 3
            elif sleep_delta < -75:
                starting -= 6
            elif sleep_delta < -45:
                starting -= 3
            elif sleep_delta < -20:
                starting -= 1
        else:
            if avg_sleep_hours >= 7.5:
                starting += 4
            elif avg_sleep_hours < 6.5:
                starting -= 6
            elif avg_sleep_hours < 7.0:
                starting -= 2
    if sleep_quality == 1:
        starting += 10
    elif sleep_quality == 2:
        starting += 7
    elif sleep_quality == 3:
        starting -= 3
    elif sleep_quality == 4:
        starting -= 10
    hrv = pb.get("hrv", {})
    rhr = pb.get("rhr", {})
    sleep = pb.get("sleep", {})
    if hrv.get("color") == "red": starting -= 10
    elif hrv.get("color") == "grey": starting -= 4
    if rhr.get("color") == "red": starting -= 7
    elif rhr.get("color") == "grey" and rhr.get("status") == "Above baseline": starting -= 3
    if sleep.get("color") == "red":
        starting -= 2 if sleep_quality in (1, 2) else 7
    elif sleep.get("color") == "grey":
        starting -= 0 if sleep_quality in (1, 2) else 3
    starting = max(25, min(96, round(starting)))
    today = now.date().isoformat()
    today_activities = [a for a in recent_activities if str(a.get("start_date_local", ""))[:10] == today]
    today_load = 0.0
    today_minutes = 0.0
    hard_sessions = 0
    for a in today_activities:
        load = a.get("icu_training_load")
        if isinstance(load, (int, float)):
            today_load += max(0, float(load))
        secs = a.get("moving_time") or a.get("elapsed_time") or 0
        if isinstance(secs, (int, float)):
            today_minutes += max(0, float(secs)) / 60.0
        try:
            family = classify_training_family(a)
            if _v4895_activity_quality_truth(a, family).get("quality_relevant"):
                hard_sessions += 1
        except Exception:
            pass
    training_context = _energy_training_day_context(today_load, today_minutes, training_rhythm)
    steps_context = _energy_steps_context(wellness, now=now, today_activities=today_activities)
    load_cost = min(42.0, today_load * 0.26)
    duration_cost = min(10.0, max(0.0, today_minutes - 45.0) * 0.035)
    hard_cost = 5.0 if hard_sessions else 0.0
    rhythm_excess_cost = float(training_context.get("extra_cost") or 0.0)
    steps_cost = float(steps_context.get("cost") or 0.0)
    spent = min(62.0, load_cost + duration_cost + hard_cost + rhythm_excess_cost + steps_cost)
    score = max(5, min(100, round(starting - spent)))
    if score >= 72:
        label, zone = "Charged", "green"
    elif score >= 45:
        label, zone = "Available", "grey"
    elif score >= 25:
        label, zone = "Used", "orange"
    else:
        label, zone = "Drained", "red"
    theta = math.pi - (score / 100.0) * math.pi
    marker_radius = 87
    marker_x = round(100 + marker_radius * math.cos(theta), 1)
    marker_y = round(95 - marker_radius * math.sin(theta), 1)
    return {
        "energy_score": score, "energy_label": label, "energy_zone": zone,
        "marker_x": marker_x, "marker_y": marker_y,
        "energy_morning_score": starting,  # legacy key retained for stored-report/template compatibility
        "energy_starting_score": starting,
        "energy_today_load": round(today_load),
        "energy_today_minutes": round(today_minutes),
        "energy_today_sessions": len(today_activities),
        "energy_today_steps": steps_context.get("steps"),
        "energy_today_steps_label": steps_context.get("steps_label"),
        "energy_steps_cost": round(steps_cost, 1),
        "energy_steps_context": steps_context.get("label"),
        "energy_steps_context_class": steps_context.get("class", "grey"),
        "energy_steps_baseline_learned": bool(steps_context.get("learned")),
        "energy_steps_samples": steps_context.get("samples", 0),
        "energy_typical_steps": steps_context.get("typical_steps"),
        "energy_steps_q75": steps_context.get("q75_steps"),
        "energy_steps_ratio": steps_context.get("ratio"),
        "energy_steps_overlap_guard": bool(steps_context.get("overlap_guard")),
        "energy_spent": round(spent),
        "energy_sleep_quality": sleep_quality,
        "energy_training_context": training_context.get("label"),
        "energy_training_context_class": training_context.get("class", "grey"),
        "energy_training_baseline_learned": bool(training_context.get("learned")),
        "energy_typical_day_load": training_context.get("typical_load"),
        "energy_typical_day_minutes": training_context.get("typical_minutes"),
        "energy_today_load_ratio": training_context.get("load_ratio"),
        "energy_today_duration_ratio": training_context.get("duration_ratio"),
        "energy_rhythm_excess_cost": round(rhythm_excess_cost, 1),
        "energy_cost_method": "real load + duration + intensity + daily steps; learned training-day/movement excess; ambulatory overlap guard; no session-count penalty",
    }
def sleep_quality_details(value):
    try:
        q = int(float(value))
    except (TypeError, ValueError):
        return "Q–", "No quality data", "qna"
    labels = {1: "Excellent", 2: "Good", 3: "Average", 4: "Poor"}
    q = max(1, min(4, q))
    return f"Q{q}", labels[q], f"q{q}"
def compute_race_readiness(form_zone, fatigue_zone, sleep_quality, personal_baselines):
    score = 50
    signals = []
    score += {"green": 18, "grey": 0, "red": -18}.get(form_zone, 0)
    score += {"green": 15, "grey": 0, "red": -15}.get(fatigue_zone, 0)
    if sleep_quality == 1:
        score += 12
    elif sleep_quality == 2:
        score += 9
    elif sleep_quality == 3:
        score -= 3; signals.append((2, "sleep quality average"))
    elif sleep_quality == 4:
        score -= 10; signals.append((3, "sleep quality poor"))
    pb = personal_baselines or {}
    hrv, rhr, sleep, spo2 = (pb.get(k, {}) for k in ("hrv", "rhr", "sleep", "spo2"))
    if hrv.get("status") == "Within baseline": score += 6
    elif hrv.get("status") == "Above baseline": score += 5
    elif hrv.get("status") == "Below baseline":
        score += -9 if hrv.get("color") == "red" else -4
        signals.append((3 if hrv.get("color") == "red" else 2, "HRV below personal range"))
    if rhr.get("status") == "Within baseline": score += 4
    elif rhr.get("status") == "Above baseline":
        score += -8 if rhr.get("color") == "red" else -3
        signals.append((3 if rhr.get("color") == "red" else 2, "resting HR elevated"))
    if sleep.get("color") == "green":
        score += 4
    elif sleep.get("color") == "grey":
        # Missing sleepQuality is unknown, NOT a poor-quality observation.
        if sleep_quality in (3, 4):
            score -= 3; signals.append((1, "sleep slightly short"))
    elif sleep.get("color") == "red":
        if sleep_quality in (1, 2):
            score -= 3; signals.append((1, "sleep short but quality good"))
        elif sleep_quality in (3, 4):
            score -= 8; signals.append((3, "sleep well below habitual opportunity"))
        else:
            # Duration is observed short, quality is not observed. Keep the
            # legitimate duration warning without assuming bad sleep quality.
            score -= 3; signals.append((1, "sleep short; quality not reported"))
    if spo2.get("status") != "No current data":
        if spo2.get("color") == "green": score += 2
        elif spo2.get("color") == "red":
            score -= 5; signals.append((2, "SpO2 below usual level"))
    score = max(0, min(100, round(score)))
    signals.sort(key=lambda x: x[0], reverse=True)
    if signals:
        reason = " · ".join(text for _, text in signals[:3])
    elif form_zone == "green" and fatigue_zone == "green":
        reason = "Load balance and recovery signals are supportive."
    else:
        reason = "Recovery signals are broadly stable."
    if score >= 70: return score, "Ready to Train", "green", reason
    if score <= 40: return score, "Recovery Bias", "red", reason
    return score, "Train Smart", "grey", reason
def compute_metrics(wellness, form_thresholds=None, fatigue_thresholds=None, baseline_wellness=None):

def _v4876_prompt_scalar(value, suffix=""):
    if value is None:
        return "-"
    if isinstance(value, str):
        stripped = value.strip()
        if not stripped or stripped.lower() in {"n/a", "none", "null"}:
            return "-"
        return stripped + suffix
    if isinstance(value, float):
        if not math.isfinite(value):
            return "-"
        rendered = f"{value:.2f}".rstrip("0").rstrip(".")
        return rendered + suffix
    return str(value) + suffix

def _rhythm_num(value):
    try:
        value = float(value)
    except (TypeError, ValueError):
        return None
    return value if math.isfinite(value) else None
def _rhythm_quantile(values, q):
    vals = sorted(float(v) for v in values if _rhythm_num(v) is not None)
    if not vals:
        return None
    if len(vals) == 1:
        return vals[0]
    pos = (len(vals) - 1) * max(0.0, min(1.0, float(q)))
    lo = int(math.floor(pos))
    hi = int(math.ceil(pos))
    if lo == hi:
        return vals[lo]
    frac = pos - lo
    return vals[lo] + (vals[hi] - vals[lo]) * frac
def _rhythm_weighted_median(pairs):
    clean = []
    for value, weight in pairs:
        v, w = _rhythm_num(value), _rhythm_num(weight)
        if v is not None and w is not None and w > 0:
            clean.append((v, w))
    if not clean:
        return None
    clean.sort(key=lambda x: x[0])
    half = sum(w for _, w in clean) / 2.0
    acc = 0.0
    for value, weight in clean:
        acc += weight
        if acc >= half:
            return value
    return clean[-1][0]
def _rhythm_format_minutes(minutes):
    if minutes is None:

    minutes = int(round(minutes)) % 1440
    hour, minute = divmod(minutes, 60)

def _rhythm_format_duration(minutes):
    if minutes is None:
        return None
    minutes = max(1, int(round(minutes / 5.0) * 5))
    if minutes < 90:
        return f"~{minutes} min"
    h, m = divmod(minutes, 60)
    return f"~{h}h{m:02d}" if m else f"~{h} h"

def _rhythm_span_stats(records, start_date, end_date):
    if not start_date or not end_date or end_date < start_date:
        return {
            "days_observed": 0, "training_days": 0, "sessions": 0, "hours": 0.0,
            "days_per_week": 0.0, "sessions_per_week": 0.0, "hours_per_week": 0.0,
            "sessions_per_training_day": 0.0, "double_days": 0, "double_day_pct": 0.0,
            "median_duration_min": None, "duration_q25_min": None, "duration_q75_min": None,
            "median_day_load": None, "day_load_q25": None, "day_load_q75": None, "load_day_samples": 0,
            "median_day_minutes": None, "day_minutes_q25": None, "day_minutes_q75": None, "duration_day_samples": 0,
            "weekday_probability": {i: 0.0 for i in range(7)},
            "weekday_double_probability": {i: 0.0 for i in range(7)},
        }
    days_observed = (end_date - start_date).days + 1
    selected = [r for r in records if start_date <= r["date"] <= end_date]
    by_day = {}
    for r in selected:
        by_day.setdefault(r["date"], []).append(r)
    training_days = len(by_day)
    sessions = len(selected)
    hours = sum((r.get("duration_min") or 0.0) for r in selected) / 60.0
    weeks = max(days_observed / 7.0, 1 / 7.0)
    double_days = sum(1 for rows in by_day.values() if len(rows) >= 2)
    weekday_occurrences = {i: 0 for i in range(7)}
    cursor = start_date
    while cursor <= end_date:
        weekday_occurrences[cursor.weekday()] += 1
        cursor += timedelta(days=1)
    weekday_training = {i: 0 for i in range(7)}
    weekday_double = {i: 0 for i in range(7)}
    for d, rows in by_day.items():
        wd = d.weekday()
        weekday_training[wd] += 1
        if len(rows) >= 2:
            weekday_double[wd] += 1
    durations = [r["duration_min"] for r in selected if r.get("duration_min")]
    day_loads = []
    day_minutes = []
    for rows in by_day.values():
        raw_loads = [_rhythm_num(r.get("load")) for r in rows]
        raw_minutes = [_rhythm_num(r.get("duration_min")) for r in rows]
        if raw_loads and all(v is not None for v in raw_loads):
            day_loads.append(sum(max(0.0, float(v)) for v in raw_loads))
        if raw_minutes and all(v is not None for v in raw_minutes):
            day_minutes.append(sum(max(0.0, float(v)) for v in raw_minutes))
    return {
        "start_date": start_date.isoformat(), "end_date": end_date.isoformat(),
        "days_observed": days_observed, "training_days": training_days, "sessions": sessions,
        "hours": round(hours, 1),
        "days_per_week": round(training_days / weeks, 2),
        "sessions_per_week": round(sessions / weeks, 2),
        "hours_per_week": round(hours / weeks, 2),
        "sessions_per_training_day": round(sessions / training_days, 2) if training_days else 0.0,
        "double_days": double_days,
        "double_day_pct": round(double_days / training_days, 3) if training_days else 0.0,
        "median_duration_min": round(statistics.median(durations), 1) if durations else None,
        "duration_q25_min": round(_rhythm_quantile(durations, 0.25), 1) if durations else None,
        "duration_q75_min": round(_rhythm_quantile(durations, 0.75), 1) if durations else None,
        "median_day_load": round(statistics.median(day_loads), 1) if day_loads else None,
        "day_load_q25": round(_rhythm_quantile(day_loads, 0.25), 1) if day_loads else None,
        "day_load_q75": round(_rhythm_quantile(day_loads, 0.75), 1) if day_loads else None,
        "load_day_samples": len(day_loads),
        "median_day_minutes": round(statistics.median(day_minutes), 1) if day_minutes else None,
        "day_minutes_q25": round(_rhythm_quantile(day_minutes, 0.25), 1) if day_minutes else None,
        "day_minutes_q75": round(_rhythm_quantile(day_minutes, 0.75), 1) if day_minutes else None,
        "duration_day_samples": len(day_minutes),
        "weekday_probability": {
            i: round(weekday_training[i] / weekday_occurrences[i], 3) if weekday_occurrences[i] else 0.0
            for i in range(7)
        },
        "weekday_double_probability": {
            i: round(weekday_double[i] / weekday_training[i], 3) if weekday_training[i] else 0.0
            for i in range(7)
        },
    }

        first = rows[0]
        first_by_weekday[d.weekday()].append((first["start_minute"], weight))
        first_all.append((first["start_minute"], weight))
        if len(rows) >= 2:
            second_by_weekday[d.weekday()].append((rows[1]["start_minute"], weight))
            second_all.append((rows[1]["start_minute"], weight))
        for row in rows:
            if row.get("duration_min"):
                duration_by_weekday[d.weekday()].append((row["duration_min"], weight))
                duration_all.append((row["duration_min"], weight))
    return {
        "first_start_min": {i: _rhythm_weighted_median(first_by_weekday[i]) for i in range(7)},
        "second_start_min": {i: _rhythm_weighted_median(second_by_weekday[i]) for i in range(7)},
        "duration_min": {i: _rhythm_weighted_median(duration_by_weekday[i]) for i in range(7)},
        "overall_first_start_min": _rhythm_weighted_median(first_all),
        "overall_second_start_min": _rhythm_weighted_median(second_all),
        "overall_duration_min": _rhythm_weighted_median(duration_all),
    }
def _rhythm_training_gap(records, today):
    dates = sorted({r["date"] for r in records if r["date"] < today})
    if len(dates) < 2:
        return None
    dates = dates[-20:]
    gaps = [(b - a).days for a, b in zip(dates, dates[1:]) if (b - a).days > 0]
    return round(statistics.median(gaps), 1) if gaps else None

def _v4883_strength_structure_signals(text):
    s = str(text or "")
    schemes = []
    rep_values = []
    for m in re.finditer(r"(?<!\d)(\d{1,2})\s*[×xX]\s*(\d{1,2})(?:\s*[–—-]\s*(\d{1,2}))?(?!\d)", s):
        sets = int(m.group(1))
        lo = int(m.group(2))
        hi = int(m.group(3) or lo)
        if 1 <= sets <= 10 and 1 <= lo <= 30 and 1 <= hi <= 30 and lo <= hi:
            schemes.append({"sets": sets, "reps_low": lo, "reps_high": hi})
            rep_values.append((lo + hi) / 2.0)
    failure = bool(re.search(r"\b(?:to failure|until failure|training to failure|cedimento|a cedimento|failure reps?)\b", s, flags=re.I))
    if re.search(r"\b(?:no failure|not to failure|without failure|senza cedimento)\b", s, flags=re.I):
        failure = False
    rir = []
    for m in re.finditer(r"\b([0-5])\s*(?:rir|reps? in reserve)\b", s, flags=re.I):
        rir.append(int(m.group(1)))
    load_mentions = [float(x.replace(',', '.')) for x in re.findall(r"(?<!\d)(\d{1,3}(?:[.,]\d+)?)\s*kg\b", s, flags=re.I)]
    return {"schemes": schemes, "rep_values": rep_values, "failure": failure, "rir": rir, "load_mentions_kg": load_mentions}
def _v4883_strength_structure_assessment(goal_key, rep_values, structured_sessions, failure_sessions):
    if structured_sessions <= 0 or not rep_values:
        return {"status": "UNKNOWN", "median_reps": None, "confidence": "NONE"}
    med = float(statistics.median(rep_values))
    goal = str(goal_key or "ENDURANCE_BASE").upper()
    ideal_max = 6 if goal in {"SPRINTER", "POWER_1"} else (8 if goal in {"POWER_5", "POWER_20", "TIME_TRIAL"} else 10)
    failure_ratio = float(failure_sessions or 0) / max(1.0, float(structured_sessions))
    if med <= ideal_max + 1 and failure_ratio <= 0.25:
        status = "ALIGNED"
    elif med <= 12 and failure_ratio <= 0.50:
        status = "MIXED"
    else:
        status = "HIGH_FATIGUE_BIAS"
    confidence = "HIGH" if structured_sessions >= 4 else ("MODERATE" if structured_sessions >= 2 else "LOW")
    return {"status": status, "median_reps": round(med, 1), "confidence": confidence, "failure_session_ratio": round(failure_ratio, 2)}
