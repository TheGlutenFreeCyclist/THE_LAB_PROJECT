def _v196919_authorize_request(snapshot, agreement=None):
    """Pure gate for explicit consent AND independent scientific clearance.

    No workout is made by this function. It can only authorize a subsequent
    call to the existing R27 authority. Missing fields always mean NOT cleared.
    """
    snap = snapshot if isinstance(snapshot, dict) else {}
    intent = _v196918_calibration_request(snap)
    base = {'schema': R196919_SCHEMA, 'request_id': _v196919_request_token(snap),
            'gap_state': intent.get('state'), 'canonical_contract': None,
            'state': 'UNASSESSABLE', 'can_ask_consent': False, 'can_request_r27': False,
            'can_execute': False, 'issued': False, 'watts': None}
    if intent.get('state') == 'NO_POWER_GAP':
        return dict(base, state='NOT_NEEDED')
    proposed = intent.get('test') if isinstance(intent.get('test'), dict) else {}
    if (intent.get('gap_kind') == 'COUPLED_REPEATABILITY' or
            proposed.get('protocol') == 'COUPLED_REPEATABILITY_ASSESSMENT_PROPOSAL'):
        return dict(base, state='PROTOCOL_DEFINITION_REQUIRED',
                    reason='R27 single effort cannot prove coupled work/recovery')
    if proposed.get('protocol') != 'EXISTING_R27_PRIMARY_VALIDATION':
        return dict(base, state='EVIDENCE_REVIEW_REQUIRED')
    road = snap.get('adaptive_roadmap') if isinstance(snap.get('adaptive_roadmap'), dict) else {}
    original = _v4893_validation_contract(road)
    if (original.get('available') is not True or
            str(original.get('mode') or '') not in {'MAXIMAL', 'SUBMAX_DURABILITY'} or
            float(original.get('seconds') or 0) != proposed.get('duration_seconds') or
            str(original.get('goal_key') or '') != intent.get('goal_key')):
        return dict(base, state='R27_CONTRACT_MISMATCH', can_ask_consent=False)
    # Attest the actual goal-owned R27 roadmap family, rather than trusting
    # a free-form seconds/mode pair to masquerade as its scientific test.
    field = ((_V4873_ROADMAP_GOALS.get(intent.get('goal_key')) or {})
             .get('field_test') or {})
    anchor = float(field.get('seconds') or 0)
    configured_mode = str(field.get('mode') or '')
    raw_test = road.get('test') if isinstance(road.get('test'), dict) else {}
    role = str(raw_test.get('checkpoint_role') or 'PRIMARY').upper()
    try:
        configured_anchor = float(raw_test.get('primary_anchor_seconds') or 0)
    except (TypeError, ValueError, OverflowError):
        configured_anchor = 0
    if (anchor <= 0 or original.get('mode') != configured_mode or
            (role == 'PRIMARY' and float(original['seconds']) != anchor) or
            (role == 'EXTENSION' and configured_anchor != anchor) or
            role not in {'PRIMARY', 'EXTENSION'}):
        return dict(base, state='GOAL_SPECIFIC_R27_PROTOCOL_NOT_ATTESTED')
    base.update(canonical_contract='VALIDATION_CONTRACT_V1', can_ask_consent=True)
    response = agreement if isinstance(agreement, dict) else {}
    action = str(response.get('decision') or '').upper()
    if action in {'DECLINE', 'WITHDRAW'}:
        return dict(base, state='ATHLETE_DECLINED' if action == 'DECLINE' else 'ATHLETE_WITHDREW')
    if (action != 'ACCEPT' or response.get('request_id') != base['request_id'] or
            response.get('scope') != 'ONE_CANONICAL_VALIDATION'):
        return dict(base, state='AWAITING_EXPLICIT_ATHLETE_AGREEMENT')
    decision = snap.get('nova_decision') if isinstance(snap.get('nova_decision'), dict) else {}
    q = road.get('quality_window') if isinstance(road.get('quality_window'), dict) else {}
    context = snap.get('planning_context') if isinstance(snap.get('planning_context'), dict) else {}
    if (decision.get('hard_session_allowed') is not True or
            str(decision.get('action') or '').upper() not in {'TRAIN', 'VALIDATE'} or
            str(q.get('state') or '').upper() in _R196918_NO_HARD or
            context.get('health_override') or
            str((road.get('test') or {}).get('state') or '').upper() not in _R196919_ACTIVE_TEST_STATES):
        return dict(base, state='WAIT_FOR_SAFETY_OR_SCIENTIFIC_STAGE')
    if not _v4893_validation_due(road, snap.get('microcycle_ledger')):
        return dict(base, state='WAIT_FOR_CANONICAL_R27_WINDOW')
    return dict(base, state='CONSENTED_R27_CAN_REVIEW', can_request_r27=True)


def _v196919_try_canonical_r27(snapshot, agreement=None):
    """Invoke *original* R27 alignment; do not forge its workout or ledger.

    Output is a candidate raw-session composition, NOT a completed plan.
    A new Snapshot still must compile, validate, and reconcile the actual plan.
    """
    snap = snapshot if isinstance(snapshot, dict) else {}
    gate = _v196919_authorize_request(snap, agreement)
    if not gate['can_request_r27']:
        return dict(gate, raw_sessions=None)
    pres = snap.get('nova_prescription') if isinstance(snap.get('nova_prescription'), dict) else {}
    clock = snap.get('coach_clock') if isinstance(snap.get('coach_clock'), dict) else {}
    rows = pres.get('raw_sessions') if isinstance(pres.get('raw_sessions'), list) else []
    slots = clock.get('next_slots') if isinstance(clock.get('next_slots'), list) else []
    # No known slot or time envelope => no issuance, even after consent.
    if not rows or not slots:
        return dict(gate, state='SLOT_UNAVAILABLE', can_request_r27=False, raw_sessions=None)
    road = snap.get('adaptive_roadmap') if isinstance(snap.get('adaptive_roadmap'), dict) else {}
    rewritten = _v4893_align_due_validation_to_window(
        rows, clock, adaptive_roadmap=road, microcycle_ledger=snap.get('microcycle_ledger'))
    matches = [i for i, r in enumerate(rewritten or []) if isinstance(r, dict)
               and r.get('_tl_deterministic_validation') is True]
    if len(matches) != 1:
        return dict(gate, state='R27_NO_ELIGIBLE_SLOT', can_request_r27=False, raw_sessions=None)
    idx = matches[0]
    slot = slots[idx] if idx < len(slots) and isinstance(slots[idx], dict) else {}
    d = str(slot.get('date') or '')[:10]
    context = snap.get('planning_context') if isinstance(snap.get('planning_context'), dict) else {}
    if (not d or d in set(context.get('blackout_dates') or []) or
            d in set(context.get('no_intensity_dates') or []) or
            slot.get('availability_confirmed') is not True or
            slot.get('is_race') or slot.get('restriction') in {'NO_TRAINING', 'NO_INTENSITY'}):
        return dict(gate, state='SLOT_NOT_EXPLICITLY_CONFIRMED', can_request_r27=False, raw_sessions=None)
    # An already issued or active plan owns its slot. Consent cannot
    # supersede continuity or quietly create a second quality exposure.
    led = snap.get('microcycle_ledger') if isinstance(snap.get('microcycle_ledger'), dict) else {}
    for historical in led.get('prescription_history') or []:
        if not isinstance(historical, dict):
            continue
        if (str(historical.get('prescription_status') or '').upper()
                in {'ACTIVE', 'PLANNED', 'ISSUED', 'MISSED_UNEXECUTED'} and
                str(historical.get('planned_date') or historical.get('date') or '')[:10] == d):
            return dict(gate, state='EXISTING_LIVE_PLAN_OWNS_SLOT',
                        can_request_r27=False, raw_sessions=None)
    contract = rewritten[idx].get('_tl_validation_contract') or {}
    actual = _v4893_validation_contract(road)
    if (not isinstance(contract, dict) or not contract.get('available') or
            contract.get('seconds') != actual.get('seconds') or
            contract.get('goal_key') != actual.get('goal_key')):
        return dict(gate, state='R27_OUTPUT_FAILED_VERIFICATION', can_request_r27=False, raw_sessions=None)
    return dict(gate, state='R27_CANDIDATE_REQUIRES_NORMAL_COMPILATION',
                candidate_slot_index=idx, raw_sessions=rewritten,
                can_execute=False, issued=False)


def _v196919_reconcile_observed_evidence(snapshot, execution=None, previous_blocks=None,
                                          performance_evidence=None, ledger=None):
    """Only canonical R149 may promote observed power. No input PB becomes a dose.

    Distinguish exact activity-link proof from an independently validated power
    band; the latter requires an entirely new upstream scientific evaluation.
    """
    snap = snapshot if isinstance(snapshot, dict) else {}
    intent = _v196918_calibration_request(snap)
    base = {'schema': R196919_SCHEMA, 'state': 'EVIDENCE_UNVERIFIED',
            'can_recalculate': False, 'prescription_changed': False,
            'new_workout': None, 'new_watt_band': None, 'verified_event': None}
    ex = execution if isinstance(execution, dict) else {}
    # A just-completed R27 test may already have closed the earlier gap in
    # the current Snapshot. Evidence must follow the immutable fulfilled plan,
    # not require an obsolete 'missing watts' state to remain visible.
    if intent.get('gap_kind') == 'COUPLED_REPEATABILITY':
        return dict(base, state='CANONICAL_COUPLED_OR_OTHER_VALIDATOR_REQUIRED')
    # A plan and the real event must have already been linked and FULFILLED
    # through the existing ledger reconciliation, not through a form claim.
    lid = str(ex.get('activity_id') or '').strip()
    day = str(ex.get('actual_date') or '')[:10]
    fingerprint = str(ex.get('prescription_fingerprint') or '').strip()
    if not lid or not fingerprint or not re.fullmatch(r'\d{4}-\d{2}-\d{2}', day):
        return dict(base, state='ACTIVITY_LINK_REQUIRED')
    led = ledger if isinstance(ledger, dict) else {}
    fulfilled = [x for x in led.get('prescription_history') or [] if isinstance(x, dict)
                 and x.get('prescription_status') == 'FULFILLED'
                 and str(x.get('prescription_fingerprint') or '') == fingerprint
                 and str(x.get('executed_activity_id') or '') == lid
                 and str(x.get('actual_date') or '')[:10] == day]
    if len(fulfilled) != 1:
        return dict(base, state='FULFILLED_LEDGER_MATCH_REQUIRED')
    plan = fulfilled[0]
    if (str(plan.get('nova_workout_id') or plan.get('nova_contract_id') or '').upper()
            not in {'VALIDATION_CONTRACT_V1'} and
            str(plan.get('execution_match_basis') or '') != 'VALIDATION_SINGLE_EFFORT'):
        return dict(base, state='NOT_AN_R27_VALIDATION')
    road = snap.get('adaptive_roadmap') if isinstance(snap.get('adaptive_roadmap'), dict) else {}
    contract = _v4893_validation_contract(road)
    if (not contract.get('available') or contract.get('mode') != 'MAXIMAL'):
        return dict(base, state='CANONICAL_NON_MAXIMAL_REVIEW_REQUIRED')
    observation = _v149_exact_checkpoint_observation(
        performance_evidence or {}, previous_blocks or [], lid, day, contract['seconds'])
    if not isinstance(observation, dict) or observation.get('watts') is None:
        return dict(base, state='RAW_POWER_EVIDENCE_MISSING_OR_INCOMPARABLE')
    # Reuse the actual pre-existing reconciler to promote the linked checkpoint.
    # It can decline independently. Never mutate the caller's evidence.
    fresh = _v149_promote_reconciled_validation(
        performance_evidence or {}, {'primary_goal':intent.get('goal_key')},
        previous_blocks or [], led)
    new_events = [x for x in (fresh or {}).get('events') or [] if isinstance(x, dict)
                  and x.get('formal_validation') and x.get('activity_linked')
                  and str(x.get('activity_id') or '') == lid
                  and str(x.get('date') or '')[:10] == day]
    if not new_events:
        return dict(base, state='R149_PROMOTION_NOT_PROVEN')
    return dict(base, state='FRESH_VALIDATION_READY_FOR_FULL_RECOMPUTE',
                can_recalculate=True, verified_event=dict(new_events[-1]),
                candidate_performance_evidence=fresh,
                next_step='RUN_FULL_CANONICAL_SNAPSHOT_WITH_FRESH_ACTIVITIES_AND_EVIDENCE')

def _v196919_latest_agreement(user_id, request_id):
    _db_execute('''CREATE TABLE IF NOT EXISTS nova_evidence_agreements (
        id TEXT PRIMARY KEY, user_id TEXT NOT NULL, request_id TEXT NOT NULL,
        decision TEXT NOT NULL, scope TEXT NOT NULL, created_at_utc TEXT NOT NULL)''')
    rows = _db_execute('''SELECT decision,request_id,scope,created_at_utc
        FROM nova_evidence_agreements WHERE user_id=? AND request_id=?
        ORDER BY created_at_utc DESC, id DESC LIMIT 1''',
        (str(user_id), str(request_id)), fetch=True)
    if not rows:
        return None
    result = rows[0]
    # Consent applies to this request only and expires; fail closed on bad clocks.
    try:
        stamp = datetime.fromisoformat(str(result['created_at_utc']).replace('Z', '+00:00'))
        now = datetime.now(ZoneInfo('UTC'))
        if stamp.tzinfo is None or not (0 <= (now - stamp).total_seconds() <= 7 * 24 * 3600):
            return None
    except (ValueError, TypeError, OverflowError):
        return None
    return dict(result)

def _v196920_prepare_consented_r27(snapshot, user_id=None, read_only=False, agreement=None):
    """Returns (candidate_precompile_or_None, review). No I/O in read-only QA.

    A user approval cannot replace a prescription that is already available,
    override a locked slot, or synthesize a workout template / watt target.
    """
    snap = snapshot if isinstance(snapshot, dict) else {}
    pres = snap.get('nova_prescription') if isinstance(snap.get('nova_prescription'), dict) else {}
    base = {'schema':R196920_SCHEMA, 'state':'NO_ELIGIBLE_REQUEST',
            'issued':False, 'persisted':False, 'can_execute':False,
            'request_id':None, 'canonical_plan_id':None, 'watt_band':None}
    if pres.get('hard_prescription_available') is True and str(pres.get('status') or '').upper() == 'PRESCRIBED':
        return None, dict(base, state='EXISTING_QUALITY_PRESERVED')
    gate = _v196919_authorize_request(snap)
    if not gate.get('can_ask_consent'):
        return None, dict(base, state=gate.get('state') or 'NO_ELIGIBLE_REQUEST')
    token = gate['request_id']
    if agreement is None and not read_only and user_id is not None:
        agreement = _v196919_latest_agreement(user_id, token)
    if read_only and agreement is None:
        return None, dict(base, state='READ_ONLY_NO_CONSENT_EVALUATION', request_id=token)
    bridge = _v196919_try_canonical_r27(snap, agreement)
    if bridge.get('state') != 'R27_CANDIDATE_REQUIRES_NORMAL_COMPILATION':
        return None, dict(base, state=bridge.get('state'), request_id=token)
    index = bridge.get('candidate_slot_index')
    raw = bridge.get('raw_sessions') or []
    if (type(index) is not int or index < 0 or index >= len(raw) or
            not isinstance(raw[index], dict) or
            raw[index].get('_tl_deterministic_validation') is not True):
        return None, dict(base, state='R27_RAW_INVALID', request_id=token)
    real_contract = _v4893_validation_contract(snap.get('adaptive_roadmap'))
    if not real_contract.get('available'):
        return None, dict(base, state='R27_CONTRACT_UNAVAILABLE', request_id=token)
    # Never add a new test over an already compiled/active quality; R27 replaces
    # exactly the raw slot it canonically selected, as in the preexisting engine.
    candidate = copy.deepcopy(pres)
    candidate.update({'status':'PRESCRIBED', 'source':'VALIDATION_CONTRACT',
        'workout_id':_R196920_VALIDATION_ID, 'contract_id':_R196920_VALIDATION_ID,
        'selected_slot_index':index, 'raw_sessions':copy.deepcopy(raw),

        'fail_closed_reason':None, 'evidence_cycle_source':'R27_AFTER_SCOPED_CONSENT',
        'evidence_cycle_request_id':token})
    return candidate, dict(base, state='R27_NORMAL_COMPILATION_PENDING',
        request_id=token, selected_slot_index=index,
        goal_key=real_contract.get('goal_key'),
        seconds=real_contract.get('seconds'), mode=real_contract.get('mode'))


def _v196920_validation_row(session, goal, seconds):
    s = session if isinstance(session, dict) else {}
    comp = s.get('session_composition') if isinstance(s.get('session_composition'),dict) else {}
    dur = s.get('session_duration_feasibility') if isinstance(s.get('session_duration_feasibility'),dict) else {}
    contract = s.get('validation_contract') if isinstance(s.get('validation_contract'),dict) else {}
    # Validate the REAL normal compiler output, not a raw _tl marker alone.
    if (str(s.get('nova_workout_id') or s.get('nova_contract_id') or '') != _R196920_VALIDATION_ID or
            str(s.get('stimulus_role') or '').upper() != 'VALIDATE' or
            str(comp.get('state') or '').upper() != 'COMPLETE' or
            str(dur.get('state') or '').upper() != 'PASS' or
            s.get('is_race') or str(s.get('restriction') or 'NONE').upper() != 'NONE' or
            str(s.get('date') or '')[:10] == ''):
        return False
    try:
        day = date.fromisoformat(str(s.get('date'))[:10])
        _ = day
        if goal and str(s.get('goal_key') or contract.get('goal_key') or goal).upper() != str(goal).upper():
            return False
        primary = float(comp.get('primary_duration_min')) * 60
        if abs(primary - float(seconds)) > max(2, .01 * float(seconds)):
            return False
    except (ValueError, TypeError, OverflowError):
        return False
    return True


def _v196920_attest_compiled_r27(prescription, review, clock=None):
    p = prescription if isinstance(prescription,dict) else {}
    rv = review if isinstance(review,dict) else {}
    base = dict(rv, issued=False, persisted=False, can_execute=False,
        state='FINAL_COMPILATION_REJECTED')
    idx = rv.get('selected_slot_index')
    rows = p.get('sessions') if isinstance(p.get('sessions'),list) else []
    if type(idx) is not int or idx < 0 or idx >= len(rows):
        return dict(base, state='R27_COMPILED_SLOT_MISSING')
    if (str(p.get('status') or '').upper()!='PRESCRIBED' or
            str(p.get('workout_id') or '') != _R196920_VALIDATION_ID or
            not p.get('canonical_plan_id') or
            not _v196920_validation_row(rows[idx], rv.get('goal_key'), rv.get('seconds'))):
        return base
    expected = (clock or {}).get('next_slots') or []
    if (idx >= len(expected) or not isinstance(expected[idx],dict) or
            expected[idx].get('availability_confirmed') is not True or
            str(expected[idx].get('date') or '')[:10] != str(rows[idx].get('date') or '')[:10]):
        return dict(base, state='R27_COMPILED_AVAILABILITY_MISMATCH')
    other_hard = [i for i,r in enumerate(rows) if i!=idx and isinstance(r,dict) and
        (r.get('hard_spacing_relevant') or str(r.get('intensity_class') or '').lower() in {'tempo','threshold','vo2'})]
    if other_hard:
        return dict(base, state='R27_MULTIPLE_HARD_EXPOSURES')
    if rv.get('mode')=='MAXIMAL' and (p.get('execution_status')!='READY' or
            p.get('hard_prescription_available') is not True):
        return dict(base, state='R27_MAXIMAL_NOT_READY')
    return dict(base, issued=True, state='R27_CANONICAL_PLAN_COMPILED',
        canonical_plan_id=p['canonical_plan_id'],
        planned_date=str(rows[idx].get('date') or '')[:10],
        prescription_fingerprint=_v136_prescription_fingerprint(_v4840_ledger_session_summary(rows[idx])))

def _v196920_evidence_cycle_status(performance_evidence, ledger, direction):

def _v196946_supersede_redundant_issued_future(ledger, sessions, user_id=None, now=None):
    # This is ledger reconciliation, not a future-training veto. A plan issued
    # AFTER the athlete finished an independently measured matching dose is
    # superseded (NEVER marked FULFILLED). Respect the original real activity.
    out = ledger if isinstance(ledger,dict) else {}
    pairs=[]
    for session in sessions or []:
        if not isinstance(session,dict) or session.get('hard_spacing_relevant'):
            continue
        credit=session.get('observed_quality_credit') or {}
        if credit.get('schema')!=R196946_SCHEMA:
            continue
        identifiers=_v196946_recent_observed_workout_ids(session,
            out.get('primary_goal'),session.get('nova_prescription_dimension'))
        matched={wid:ref for wid,ref in identifiers.items()
            if ref.get('activity_id')==credit.get('activity_id')
            and ref.get('actual_date')==credit.get('actual_date')}
        if not matched:continue
        pairs.append((str(session.get('date') or '')[:10],matched))
    if not pairs:return out
    rows=[dict(x) if isinstance(x,dict) else x for x in out.get('prescription_history') or []]
    superseded=[]
    for row in rows:
        if not isinstance(row,dict) or str(row.get('prescription_status') or '').upper()!='ACTIVE':
            continue
        ident=str(row.get('nova_workout_id') or row.get('nova_contract_id') or '')
        for planned,observed in pairs:
            if str(row.get('date') or '')[:10]!=planned or ident not in observed:continue
            ref=observed[ident]
            block=next((b for b in (_V146_BLOCK_CONTEXT.get() or ())
                if isinstance(b,dict) and str(b.get('activity_id') or '')==ref['activity_id']),None)
            if not isinstance(block,dict):continue
            issued=_v153_parse_local(row.get('recorded_at_local'))
            try:
                start=datetime.strptime(str(block.get('date') or '')[:10]+' '+str(block.get('time') or ''),
                                        '%Y-%m-%d %I:%M %p')
                finished=start+timedelta(minutes=float(block['duration_min']))
                if issued is None or finished>issued.replace(tzinfo=None):continue
            except (TypeError,ValueError,KeyError,AttributeError,OverflowError):
                continue
            row.update(prescription_status='SUPERSEDED',
                supersession_reason='RECENT_OBSERVED_HARD_DOSE_ALREADY_PERFORMED',
                superseded_at_local=(now or get_rome_now()).isoformat(timespec='seconds'),
                observed_activity_reference={'date':ref['actual_date'],
                    'activity_id':ref['activity_id'], 'source':ref['source'],
                    'retroactive_fulfillment':False})
            superseded.append(row)
            break
    if not superseded:return out
    out['prescription_history']=rows
    anchor=out.get('primary_anchor')
    if isinstance(anchor,dict) and str(anchor.get('status') or '').upper()=='PLANNED':
        for row in superseded:
            if (str(anchor.get('date') or '')[:10]==str(row.get('date') or '')[:10]
                    and str(anchor.get('main_set') or '')==str(row.get('main_set') or '')):
                anchor['status']='SUPERSEDED'
                anchor['supersession_reason']='RECENT_OBSERVED_HARD_DOSE_ALREADY_PERFORMED'
                anchor['prescription_note']='Matching quality was already performed. Do not repeat this hard main set.'
                out['primary_anchor']=anchor
                out['continuity_state']='OBSERVED QUALITY ABSORBED'
                break
    if out.get('ledger_id'):
        _db_execute('UPDATE microcycle_ledgers SET ledger_json=?, updated_at_utc=? WHERE id=? AND user_id=?',
            (json.dumps(out,ensure_ascii=False,separators=(',',':'),default=_report_json_default),
             _now_utc_text(),out['ledger_id'],_data_owner_user_id(user_id)))
    return out


def _nova_reconcile_duplicate_active_commitments(ledger, sessions, now=None):
    """One active identity per date + actual canonical main-set workout ID.

    Preserve all historical records and any different workouts. A committed
    session in the current final plan wins; otherwise newest issuance wins.
    Multiple period labels cannot make one identical workout two commitments.
    """
    out = ledger if isinstance(ledger,dict) else {}
    history = [dict(r) if isinstance(r,dict) else r
        for r in out.get('prescription_history') or []]
    groups = {}
    for i,entry in enumerate(history):
        if not isinstance(entry,dict) or str(entry.get('prescription_status') or '').upper()!='ACTIVE':
            continue
        day = str(entry.get('date') or '')[:10]
        wid = str(entry.get('nova_workout_id') or entry.get('nova_contract_id') or '')
        if day and wid:groups.setdefault((day,wid),[]).append(i)
    current = {}
    for sess in sessions or []:
        if not isinstance(sess,dict) or sess.get('hard_spacing_relevant') is not True:
            continue
        day = str(sess.get('date') or '')[:10]
        wid = str(sess.get('nova_workout_id') or '')
        if day and wid:
            current[(day,wid)] = _v136_prescription_fingerprint(
                _v4840_ledger_session_summary(sess))
    changed=[]
    for key, indices in groups.items():
        if len(indices) < 2:continue
        preferred = current.get(key)
        selected = max(indices, key=lambda i:(
            1 if preferred and str(history[i].get('prescription_fingerprint') or '')==preferred else 0,
            str(history[i].get('recorded_at_local') or ''), i))
        for i in indices:
            if i == selected:continue
            history[i]['prescription_status']='SUPERSEDED'
            history[i]['supersession_reason']='DUPLICATE_ACTIVE_SAME_DAY_CANONICAL_WORKOUT'
            history[i]['superseded_by_fingerprint']=history[selected].get('prescription_fingerprint')
            history[i]['superseded_at_local']=(now or get_rome_now()).isoformat(timespec='seconds')
            changed.append(i)
    if not changed:return out, False
    out=dict(out,prescription_history=history)
    out['active_identity_reconciliation']={
        'schema':'NOVA-R196948-ACTIVE-IDENTITY-1',
        'duplicate_commitments_superseded':len(changed),
        'prescription_history_preserved':True}
    return out, True


def update_microcycle_ledger_from_snapshot(ledger, sessions, user_id=None, now=None):

def _v196920_cancel_issued_validation(user_id, snapshot, request_id):
    """CAS-cancel exactly one UNEXECUTED ledger plan; never erase history.

    Caller already established the owning authenticated athlete and CSRF.

        state=str(plan.get('prescription_status') or '').upper()
        if state=='WITHDRAWN':
            conn.rollback();return {'state':'ALREADY_WITHDRAWN'}
        if state not in {'ACTIVE','PLANNED','ISSUED'} or plan.get('executed_activity_id'):
            conn.rollback();return {'state':'PLAN_ALREADY_EXECUTED_OR_NOT_ACTIVE'}
        plan['prescription_status']='WITHDRAWN'

# R196.9.24 — EVIDENCE → COUPLED-ASSESSMENT → CANONICAL PRESCRIPTION
# A scoped coupled assessment is not an ordinary hard repeat and is never
# silently enabled from green readiness. Work band is exploratory, not a pass
# mark or physiological ceiling. Recovery progression stays bounded.
R196924_SCHEMA='V4.9.125-R196.9.24-COUPLED-EVIDENCE-BRIDGE-1'
_R196924_ASSESSMENT='INTRODUCTORY_COUPLED_EVIDENCE_ASSESSMENT'
_R196924_WORKOUT='RECOVERY_UNDER_LOAD_EVIDENCE_DOSE_V1'
_R196924_SCOPE='ONE_COUPLED_ASSESSMENT_AND_EXACT_SLOT'
_r196924_prior_dose=_v1959_recovery_dose_contract


def _v196924_reference(features):
    """Select an actual *comparable* completed strong pair, not latest raw 30/15.

    The chronological most recent repeatability remains visible in the ledger;
    short floats and noncanonical history are not promoted to 3×5' evidence.
    """
    f=features if isinstance(features,dict) else {}
    good=[]
    for h in f.get('recovery_dose_history') or []:
        if not isinstance(h,dict) or not h.get('strong'):continue
        try:
            age=(get_rome_now().date()-date.fromisoformat(str(h['date'])[:10])).days
            if not 0<=age<=28:continue
            if not (2<=int(h['reps'])<=20 and 20<=float(h['recovery_secs'])<=1800
                    and 60<=float(h['interval_secs'])<=1800 and float(h['ftp'])>0
                    and 80<=float(h['work_watts'])<=1200
                    and 40<=float(h['recovery_watts'])<600
                    and 80<=float(h['ftp'])<=600
                    and float(h['work_watts'])>float(h['recovery_watts'])*1.1
                    and 0<=float(h['recovery_watts'])/float(h['ftp'])<.70):continue
        except (KeyError,TypeError,ValueError,OverflowError,ZeroDivisionError):continue
        good.append(h)
    return sorted(good,key=lambda h:(h['date'],h['activity_id']),reverse=True)[0] if good else None


def _v1959_recovery_dose_contract(goal, dimension, features):
    base=_r196924_prior_dose(goal,dimension,features)
    if str(dimension or '').upper()!='RECOVERY_UNDER_LOAD' or base.get('available'):
        return base
    h=_v196924_reference(features)
    if h is None:return base
    f=dict(features or {})
    f['latest_repeatability']={
        'date':h['date'],'strong':True,'reps':h['reps'],
        'interval_secs':h['interval_secs'],'recovery_secs':h['recovery_secs'],
        'avg_watts':h['work_watts'],'recovery_avg_watts':h['recovery_watts'],
        'ftp':h['ftp'],'source': 'CANONICAL_MATCHED_WORK_RECOVERY_PAIR',
    }
    coupled=_r196924_prior_dose(goal,dimension,f)
    if coupled.get('available'):
        return dict(coupled,reference_selection={'schema':R196924_SCHEMA,
            'activity_id':h['activity_id'],'date':h['date'],
            'reason':'LATEST_COMPARABLE_FULL_WORK_RECOVERY_NOT_LATEST_UNRELATED_INTERVAL'})
    # Do not promote an already-rejected easy-recovery repeat into an
    # ordinary hard quality. The existing R196.1 scientific selector can
    # choose a different *already evaluated* stimulus. No added gates/paths.
    return coupled

def _v196924_request(decision,road,clock,prescription=None):
    """Stable exact-slot, exact-evidence token; never a global athlete switch."""
    dec=decision if isinstance(decision,dict) else {}
    sci=((road or {}).get('scientific_progression') or {})
    dose=sci.get('dose_contract') or {}
    slots=(clock or {}).get('next_slots') or []
    match=(prescription or {}).get('quality_slot_match') or {}
    idx=match.get('index')
    # The quality-window authority owns selection. A stale future-quality
    # match from a historical plan is never permission to use a different day.
    iso=str(((road or {}).get('quality_window') or {}).get('preferred_iso') or '')
    try:
        canonical_minute=int(iso[11:13])*60+int(iso[14:16])
        canonical_day=date.fromisoformat(iso[:10]).isoformat()
    except (ValueError,TypeError,OverflowError):
        return None
    eligible_indices=[i for i,s in enumerate(slots) if isinstance(s,dict) and
       str(s.get('date') or '')[:10]==canonical_day and
       str(s.get('start_minute') or '')==str(canonical_minute)]
    if len(eligible_indices)!=1:return None
    idx=eligible_indices[0]  # same exact slot for prompt, agreement and compiler
    if (dose.get('stage')!=_R196924_ASSESSMENT or
            str(sci.get('dimension') or '').upper()!='RECOVERY_UNDER_LOAD' or
            dec.get('hard_session_allowed') is not True or
            str(dec.get('action') or '').upper()!='TRAIN' or
            type(idx) is not int or idx<0 or idx>=len(slots)):
        return None
    slot=slots[idx]
    if not isinstance(slot,dict):return None
    day=str(slot.get('date') or '')[:10]
    try:
        if date.fromisoformat(day)<get_rome_now().date():return None
        minute=int(slot.get('start_minute'))
        if not 0<=minute<1440:return None
    except (ValueError,TypeError):return None
    if slot.get('is_race') or str(slot.get('restriction') or 'NONE').upper()!='NONE':return None
    window=(road or {}).get('quality_window') or {}
    if str(window.get('state') or '').upper() not in {'WATCH','OPEN'}:return None
    proof=dose.get('reference') or {};p=dose.get('prescription') or {}
    signature={'schema':R196924_SCHEMA,'goal':dec.get('goal_key'),
        'dimension':'RECOVERY_UNDER_LOAD','stage':dose.get('stage'),
        'reference':{k:proof.get(k) for k in ('date','activity_id','reps','interval_secs','recovery_secs','work_avg_watts','recovery_avg_watts','ftp')},
        'prescription':{k:p.get(k) for k in ('reps','interval_secs','recovery_secs','work_watts_low','work_watts_high','recovery_watts_low','recovery_watts_high')},
        'slot':{'date':day,'start_minute':minute},
        'window':window.get('preferred_iso'),
        'last_hard_end':window.get('last_hard_end'),
    }
    token=hashlib.sha256(json.dumps(signature,sort_keys=True,ensure_ascii=False,
                              separators=(',',':'),default=str).encode()).hexdigest()
    return {'schema':R196924_SCHEMA,'request_id':token,'index':idx,'date':day,
            'start_minute':minute,'reps':p.get('reps'),
            'interval_seconds':p.get('interval_secs'),
            'work_low':p.get('work_watts_low'),'work_high':p.get('work_watts_high'),
            'recovery_low':p.get('recovery_watts_low'),
            'recovery_high':p.get('recovery_watts_high'),
            'reference_date':proof.get('date'),'reference_activity_id':proof.get('activity_id'),
            'availability_confirmed':bool(slot.get('availability_confirmed'))}

def build_nova_prescription(nova_decision,coach_clock,adaptive_roadmap=None,
        microcycle_ledger=None,evidence_ledger=None,power_model=None,
        ftp_anchor=None,aerobic_metabolic_range=None,training_definitions=None):
    p=_r196924_before_prescription(nova_decision,coach_clock,
        adaptive_roadmap=adaptive_roadmap,microcycle_ledger=microcycle_ledger,
        evidence_ledger=evidence_ledger,power_model=power_model,
        ftp_anchor=ftp_anchor,aerobic_metabolic_range=aerobic_metabolic_range,
        training_definitions=training_definitions)
    if not isinstance(p,dict):return p
    sci=((adaptive_roadmap or {}).get('scientific_progression') or {})
    dose=sci.get('dose_contract') or {}
    if dose.get('stage')!=_R196924_ASSESSMENT:return p
    request_info=_v196924_request(nova_decision,adaptive_roadmap,coach_clock,p)
    result=copy.deepcopy(p)
    result['coupled_evidence_bridge']={'schema':R196924_SCHEMA,'state':'EVIDENCE_PROTOCOL_DEFINED' if request_info else 'AWAITING_SAFE_SLOT',
        'request':request_info,'requires_consent':True,'assessed':False,
        'reference_id':(dose.get('reference') or {}).get('activity_id'),
        'assessment_only':True,'not_a_performance_grade':True}
    # No request, agreement, confirmed availability, or secure user request:
    # never create an executable assessment just because science wants data.
    agreement=_v196924_agreement(request_info['request_id']) if request_info else None
    authorized=bool(request_info and agreement and agreement.get('decision')=='ACCEPT' and
        agreement.get('scope')==_R196924_SCOPE and
        p.get('status')=='PRESCRIBED' and p.get('workout_id')==_R196924_WORKOUT)
    if authorized:
        idx=request_info['index']; slots=(coach_clock or {}).get('next_slots') or []
        # Explicit acceptance is consent to this exact date/time too. A changed
        # window changes the signed token and invalidates the earlier acceptance.
        slot=slots[idx] if idx<len(slots) else None
        if not isinstance(slot,dict) or str(slot.get('restriction') or 'NONE').upper()!='NONE' or slot.get('is_race'):
            authorized=False
        else:
            slot['availability_confirmed']=True
            slot['availability_source']='ATHLETE_EXPLICIT_COUPLED_SLOT_CONFIRMATION'
            # R196.9.15 reads the ROADMAP quality window, not the session.
            # Consent must not become a false second availability authority.
            window=(adaptive_roadmap or {}).get('quality_window') or {}
            if (not isinstance(window,dict) or
                    str(window.get('preferred_iso') or '')[:16] !=
                    f"{request_info['date']}T{request_info['start_minute']//60:02d}:{request_info['start_minute']%60:02d}"):
                authorized=False
            else:
                window['availability_confirmed']=True
                window['availability_source']='ATHLETE_EXPLICIT_COUPLED_SLOT_CONFIRMATION'
    if authorized:
        result['coupled_evidence_bridge']['state']='CONSENTED_EXACT_SLOT_FOR_CANONICAL_COMPILATION'
        result['coupled_evidence_bridge']['agreed_request_id']=request_info['request_id']
        result['coupled_evidence_bridge']['assessment_only']=True
        return result
    raw=[_v84_easy_raw(s,nova_decision or {}) for s in (coach_clock or {}).get('next_slots') or []]
    result.update({'status':'EASY_ONLY','source':'R196924_COUPLED_EVIDENCE_CONSENT_GATE',
        'selected_slot_index':None,'workout_id':None,'contract_id':None,
        'hard_prescription_available':False,'raw_sessions':raw,
        'fail_closed_reason':'COUPLED_ASSESSMENT_REQUIRES_EXACT_SLOT_CONSENT',
        'future_quality_candidate':None})
    return result

def _v196930_coupled_compile_checks(compiled,clock,request_info):
    out=compiled if isinstance(compiled,dict) else {}
    req=request_info if isinstance(request_info,dict) else {}
    slots=(clock or {}).get('next_slots') or []
    rows=out.get('sessions') or []
    idx=req.get('index')
    errors=[]
    if out.get('status')!='PRESCRIBED':errors.append('UPSTREAM_NOT_PRESCRIBED')
    if out.get('workout_id')!=_R196924_WORKOUT:errors.append('WORKOUT_ID_MISMATCH')
    if type(idx) is not int or not 0<=idx<len(rows) or not 0<=idx<len(slots):
        errors.append('SIGNED_SLOT_NOT_IN_COMPILED_PLAN')
        return {'blocking_checks':errors,'provisional':False,'future_slot':False,
            'safe_to_issue':False,'candidate_status':str(out.get('status') or 'UNKNOWN'),
            'schema':'R196930-COMPILE-DIAGNOSTIC-1'}
    row=rows[idx] if isinstance(rows[idx],dict) else {}
    slot=slots[idx] if isinstance(slots[idx],dict) else {}
    if str(row.get('nova_workout_id') or '')!=_R196924_WORKOUT:errors.append('SESSION_WORKOUT_ID_MISMATCH')
    if str(row.get('date') or '')[:10]!=req.get('date'):
        errors.append('SIGNED_DATE_MISMATCH')
    if str(row.get('intensity_class') or '').lower() not in {'vo2','threshold','tempo'}:
        errors.append('SESSION_NOT_CANONICAL_HARD')
    if (row.get('power_prescription') or {}).get('source')!='OBSERVED_COUPLED_RECOVERY_DOSE':
        errors.append('NATIVE_COUPLED_POWER_EVIDENCE_MISSING')
    if (row.get('session_duration_feasibility') or {}).get('state')=='FAIL':
        errors.append('SESSION_DURATION_UNFEASIBLE')
    if slot.get('availability_confirmed') is not True:
        errors.append('EXACT_SLOT_AVAILABILITY_NOT_CONFIRMED')
    if slot.get('is_race') or str(slot.get('restriction') or 'NONE').upper()!='NONE':
        errors.append('SLOT_SAFETY_RESTRICTION')
    if str(row.get('restriction') or 'NONE').upper()!='NONE':
        errors.append('COMPILED_SAFETY_RESTRICTION')

def _v196925_observation_blueprint(goal):
    """Read-only mechanics FROM the versioned workout catalog, not ad-hoc test."""
    pool=[]
    for row in _NOVA_WORKOUT_LIBRARY:
        if str(row.get('goal') or '')!=str(goal or '').upper():continue
        if str(row.get('pattern') or '') not in {'STANDARD','SUSTAINED'}:continue
        try:
            reps=int(row['reps']);work=int(row['work_s']);recovery=int(row['recovery_s'])
            if not (2<=reps<=5 and 120<=work<=900 and recovery>=work*.4 and recovery<=1800):continue
        except (KeyError,TypeError,ValueError,OverflowError):continue
        pool.append((reps*work,work,reps,str(row['id']),row))
    if not pool:return None
    _,_,_,_,selected=sorted(pool,key=lambda x:(x[0],x[1],x[2],x[3]))[0]
    return {'library_version':NOVA_WORKOUT_LIBRARY_VERSION,'template_id':selected['id'],
            'reps':int(selected['reps']),'work_seconds':int(selected['work_s']),
            'recovery_seconds':int(selected['recovery_s']),
            'effort':'SELF_PACED_SUBMAXIMAL_OBSERVATION',
            'work_watts':None,'recovery_watts':None,
            'work_is_uncapped':True,'score_or_fail_target':None,
            'required_evidence':['linked_owner_activity_id','raw_time_aligned_power',
               'exact_repetition_boundaries','measured_recovery_power','observed_work_power',
               'full_activity_date','source_provenance'],
            'notes':'This is an evidence-acquisition blueprint, not an issued workout. '
                    'Until the canonical prescription is explicitly cleared, do not replace an easy session '
                    'with hard intervals. Athlete-recorded power is evidence; no invented target or pass mark.'}


def _v196925_request(snapshot):
    """Single deterministic request for any athlete with no comparable pair."""
    d=snapshot if isinstance(snapshot,dict) else {}
    dec=d.get('nova_decision') if isinstance(d.get('nova_decision'),dict) else {}
    road=d.get('adaptive_roadmap') if isinstance(d.get('adaptive_roadmap'),dict) else {}
    pres=d.get('nova_prescription') if isinstance(d.get('nova_prescription'),dict) else {}
    sci=road.get('scientific_progression') if isinstance(road.get('scientific_progression'),dict) else {}
    dose=sci.get('dose_contract') if isinstance(sci.get('dose_contract'),dict) else {}
    test=road.get('test') if isinstance(road.get('test'),dict) else {}
    goal=str(dec.get('goal_key') or road.get('goal_key') or '').upper()
    dim=str(dec.get('dimension') or sci.get('dimension') or '').upper()
    state=str(dose.get('stage') or '').upper()
    result={'schema':R196925_SCHEMA,'goal_key':goal,'dimension':dim,
            'state':'NOT_REQUIRED','consent_required':False,'issued':False,
            'executable':False,'numeric_watts_prescribed':False}
    if goal not in R1962_GOALS or dim!='RECOVERY_UNDER_LOAD':return result
    if state==_R196924_ASSESSMENT or (pres.get('coupled_evidence_bridge') or {}).get('request'):
        return dict(result,state='MEASURED_COUPLED_REFERENCE_ALREADY_AVAILABLE')
    if dose.get('available') is True:return dict(result,state='RECOVERY_DOSE_AVAILABLE')
    # Already completed formal primary capacity validation is not a new R27.
    if str(test.get('state') or '').upper() not in {'COMPLETE','COMPLETED'}:
        return dict(result,state='PRIMARY_REFERENCE_REQUIRED',
               action='USE_EXISTING_GOAL_OWNED_R27_IF_ELIGIBLE',
               next_step='REQUEST_GOAL_OWNED_PRIMARY_TEST_WITH_EXISTING_R27_CONSENT')
    template=_v196925_observation_blueprint(goal)
    if not template:
        signature={'schema':R196925_SCHEMA,'goal':goal,'dimension':dim,
                   'mode':'LINK_EXISTING_ACTIVITY_ONLY'}
        token=hashlib.sha256(json.dumps(signature,sort_keys=True,
              separators=(',',':')).encode()).hexdigest()
        return dict(result,state='ACTIVITY_EVIDENCE_REQUESTED',request_id=token,
                    action='LINK_COMPLETED_CANONICAL_WORK_RECOVERY_ACTIVITY',
                    next_step='IMPORT_ACTUAL_REPEATED_POWER_ACTIVITY')
    window=road.get('quality_window') if isinstance(road.get('quality_window'),dict) else {}
    # The candidate is diagnostic and NEVER an executable workout by itself.
    preferred_iso=str(window.get('preferred_iso') or '')
    try:
        when=datetime.fromisoformat(preferred_iso)
        if when.date()<get_rome_now().date():preferred_iso=''
    except (TypeError,ValueError,OverflowError):preferred_iso=''
    slots=((d.get('coach_clock') or {}).get('next_slots') or [])
    candidates=[]
    if preferred_iso:
        for slot in slots:
            if not isinstance(slot,dict):continue
            try:
                minute=int(slot.get('start_minute'))
                exact=(str(slot.get('date') or '')[:10]==preferred_iso[:10]
                       and minute==int(preferred_iso[11:13])*60+int(preferred_iso[14:16]))
            except (ValueError,TypeError,OverflowError):exact=False
            if exact and not slot.get('is_race') and str(slot.get('restriction') or 'NONE').upper()=='NONE':
                candidates.append(slot)
    context=d.get('planning_context') or {}
    ready=(dec.get('hard_session_allowed') is True and
           str(dec.get('action') or '').upper()=='TRAIN' and
           str(window.get('state') or '').upper() in {'OPEN','WATCH'} and
           len(candidates)==1 and not context.get('health_override') and
           not (str(pres.get('status') or '').upper()=='PRESCRIBED'
                and pres.get('hard_prescription_available') is True))
    sig={'schema':R196925_SCHEMA,'goal':goal,'dimension':dim,
         'template_id':template['template_id'],'library_version':template['library_version'],
         'test_identity':{k:test.get(k) for k in ('state','seconds','mode')},
         'gap':str(dose.get('reason') or pres.get('fail_closed_reason') or ''),
         'quality_window':preferred_iso if ready else None}
    token=hashlib.sha256(json.dumps(sig,sort_keys=True,ensure_ascii=False,
                  separators=(',',':'),default=str).encode('utf-8')).hexdigest()
    return dict(result,state='REQUEST_INITIAL_WORK_RECOVERY_BASELINE' if ready else 'WAIT_FOR_SAFETY',
        action='LINK_VERIFIED_ACTIVITY_OR_REQUEST_INITIAL_MEASUREMENT',
        next_step='OBTAIN_FIRST_MEASURED_REPEATABILITY_SESSION',
        blueprint=template,request_id=token,window_candidate=preferred_iso if ready else None,
        consent_required=True,evidence_required=True,
        gap_reason=sig['gap'],
        note='The blueprint is not a scheduled workout. A new safely cleared canonical quality must be issued before replacing an easy prescription.')


def _v196925_assess_activity(snapshot,activity_id):
    """Owner's current fetched activity only; never accept a claimed watt number."""
    d=snapshot if isinstance(snapshot,dict) else {}
    activity_id=str(activity_id or '').strip()
    if not re.fullmatch(r'[A-Za-z0-9_-]{3,80}',activity_id):
        return {'state':'INVALID_ACTIVITY_ID','verified':False}
    blocks=d.get('previous_training_blocks') or []
    matching=[b for b in blocks if isinstance(b,dict) and
              str(b.get('activity_id') or '')==activity_id]
    if len(matching)!=1:return {'state':'AWAITING_OWNER_ACTIVITY_IMPORT' if not matching
                                else 'DUPLICATE_ACTIVITY_IDENTITIES','verified':False}
    block=matching[0]
    truth=block.get('activity_truth') or {}
    repetition=block.get('repeatability') or {}
    if not isinstance(truth,dict) or not isinstance(repetition,dict):
        return {'state':'STRUCTURED_INTERVALS_MISSING','verified':False}
    source=str(truth.get('source') or block.get('interval_structure_source') or '').upper()
    if (str(truth.get('status') or '').upper()!='RESOLVED' or
            str(truth.get('confidence') or '').upper()!='HIGH' or
            source not in _R196925_TRUSTED):
        return {'state':'NEEDS_AUTHORITATIVE_RAW_POWER','verified':False,
                'source':source or None}
    def number(key):
        try:
            value=float(repetition.get(key));return value if math.isfinite(value) else None
        except (TypeError,ValueError,OverflowError):return None
    reps=number('reps'); seconds=number('interval_secs'); rest=number('recovery_secs')
    watts=number('avg_watts');restwatts=number('recovery_avg_watts')
    if None in (reps,seconds,rest,watts,restwatts):
        return {'state':'FULL_WORK_RECOVERY_STREAM_REQUIRED','verified':False}
    if (not 2<=reps<=30 or not 30<=seconds<=1800 or not 20<=rest<=1800 or
        not 50<=watts<=1500 or not 0<=restwatts<watts):
        return {'state':'WORK_RECOVERY_STRUCTURE_NOT_COMPARABLE','verified':False}
    try:
        day=date.fromisoformat(str(block['date'])[:10]);age=(get_rome_now().date()-day).days
    except (KeyError,TypeError,ValueError):return {'state':'ACTIVITY_DATE_INVALID','verified':False}
    if not 0<=age<=28:return {'state':'EVIDENCE_OUTSIDE_CURRENT_28D_WINDOW','verified':False}

            'activity_id':activity_id,'date':day.isoformat(),'reps':int(reps),
            'interval_seconds':int(seconds),'recovery_seconds':int(rest),
            'observed_work_watts':round(watts,1),'observed_recovery_watts':round(restwatts,1),
            'observed_provenance':source,'strong':bool(repetition.get('strong')),
            'next_step':'RUN_FULL_NOVA_SNAPSHOT_TO_REEVALUATE_QUESTION_STATE',
            'mastery_awarded':False,'workout_issued':False}

def _v196926_canonical_baseline_template(goal,slot=None):
    """Choose a real versioned library identity. Never infer an unattested watt target.

    Choose a mechanically appropriate form for the goal, and fit it to the
    observed slot envelope. The use of a library template is auditable and
    does not claim that the athlete has demonstrated its watts.
    """
    goal=str(goal or '').upper(); candidates=[]
    for x in _NOVA_WORKOUT_LIBRARY:
        if not isinstance(x,dict) or x.get('goal')!=goal or x.get('adaptive_only'):continue
        pattern=str(x.get('pattern') or '')
        if pattern not in {'STANDARD','SUSTAINED','SHORT_POWER','SPRINT'}:continue
        try:
            reps=int(x.get('reps')); work=int(x.get('work_s')); rec=int(x.get('recovery_s'))
        except (ValueError,TypeError):continue
        if reps<2 or reps>8 or work<6 or work>900 or rec<20 or rec>900:continue
        if goal in {'POWER_5','VO2MAX','POWER_20','TIME_TRIAL'} and not (work>=120 and reps<=5):continue
        if goal=='POWER_1' and not (20<=work<=120 and reps<=6):continue
        if goal=='SPRINTER' and not (6<=work<=20 and 3<=reps<=8):continue
        session_min=20+(reps*work+(reps-1)*rec)/60+8
        try:
            max_min=float((slot or {}).get('duration_hint_min') or 60)
        except (TypeError,ValueError):max_min=60
        if session_min>max_min+1:continue
        # De-risk starting observations: avoid excessive intervals and volume.
        objective=(abs(work-300) if goal=='POWER_5' else
                   abs(work-240) if goal=='VO2MAX' else
                   abs(work-480) if goal in {'POWER_20','TIME_TRIAL'} else
                   abs(work-45) if goal=='POWER_1' else abs(work-8))
        candidates.append((objective,session_min,reps,str(x['id']),x))
    if not candidates:return None
    _,minutes,reps,_,template=min(candidates,key=lambda r:(r[0],r[1],r[2],r[3]))
    return {'schema':R196926_SCHEMA,'template_id':template['id'],
        'library_version':NOVA_WORKOUT_LIBRARY_VERSION,
        'template_signature':_v109_template_signature(template),
        'reps':reps,'work_seconds':int(template['work_s']),
        'recovery_seconds':int(template['recovery_s']),
        'duration_minutes':int(math.ceil(minutes)),
        'work_watts_low':None,'work_watts_high':None,
        'recovery_watts_low':None,'recovery_watts_high':None,
        'effort':'SUBMAXIMAL_SELF_PACED_REPEATED_WORK',
        'required_evidence':list(_R196926_REQUIRED),
        'stop_rule':'Stop if pain, illness, unsafe symptoms, loss of control, or inability to recover.',
        'no_watt_ceiling':True,'pass_mark':None}


def _v196926_initial_request(decision,road,clock,prescription=None):
    """Build one deterministic, single-slot request from CURRENT authority."""
    d=decision if isinstance(decision,dict) else {}
    r=road if isinstance(road,dict) else {}
    c=clock if isinstance(clock,dict) else {}
    p=prescription if isinstance(prescription,dict) else {}
    sci=r.get('scientific_progression') or {};dose=sci.get('dose_contract') or {}
    dim=str(d.get('dimension') or '').upper()
    science_dim=str(sci.get('dimension') or '').upper()
    goal=str(d.get('goal_key') or r.get('goal_key') or '').upper()
    base={'schema':R196926_SCHEMA,'goal_key':goal,'dimension':dim,
          'issued':False,'executable':False,'purpose':'FIRST_MEASURED_REPEATED_WORK_RECOVERY'}
    # R196.9.52: an INITIAL work/recovery test belongs exclusively to the
    # RECOVERY_UNDER_LOAD science question.  An upstream alternative may set
    # dose.reason=NON_RECOVERY_DIMENSION_WORKOUT_LIBRARY while an older
    # Decision still carries RECOVERY_UNDER_LOAD; that MUST NOT open a test.
    if (goal not in R1962_GOALS or goal=='ENDURANCE_BASE' or
            dim!='RECOVERY_UNDER_LOAD' or science_dim!=dim):
        return dict(base,state='OTHER_SCIENCE_OWNER')
    # A missing dose does not necessarily mean missing *measurements*: it may
    # mean the question has already closed, the comparison failed, or a safe
    # step was refused. Never create an initial test for any of those cases.
    if dose.get('available') is True:
        return dict(base,state='EXISTING_CONTRACT_AVAILABLE')
    if (dose.get('schema')!=R1959_SCHEMA or
            str(dose.get('reason') or '') not in {
                'NO_CURRENT_STRONG_MEASURED_REPEATED_WORK',
                'REFERENCE_NOT_VERIFIED_IN_CANONICAL_LEDGER',
                'STALE_OR_FUTURE_UNVERIFIED_REFERENCE'}):
        return dict(base,state='OTHER_SCIENCE_OWNER')
    rec=((r.get('question_state') or {}).get('dimensions') or {}).get(dim) or {}
    if (rec.get('exit_satisfied') is True or rec.get('closed') is True):
        return dict(base,state='OTHER_SCIENCE_OWNER')
    if (p.get('coupled_evidence_bridge') or {}).get('request'):
        return dict(base,state='COMPARABLE_EVIDENCE_PATH_AVAILABLE')
    test=r.get('test') or {}
    if str(test.get('state') or '').upper() not in {'COMPLETED','COMPLETE'}:
        return dict(base,state='PRIMARY_REFERENCE_TEST_REQUIRED',
                    next_action='ISSUE_EXISTING_R27_PRIMARY_TEST_WITH_GOAL_CONSENT')
    window=r.get('quality_window') or {}
    iso=str(window.get('preferred_iso') or '')
    try:
        dt=datetime.fromisoformat(iso)
        if dt.date()<get_rome_now().date():raise ValueError('past window')
        minute=dt.hour*60+dt.minute; day=dt.date().isoformat()
    except (ValueError,TypeError,OverflowError):
        return dict(base,state='WAIT_FOR_SAFE_SLOT',next_action='REASSESS_NEXT_QUALITY_WINDOW')
    slots=c.get('next_slots') or []
    matches=[(i,s) for i,s in enumerate(slots) if isinstance(s,dict) and
        str(s.get('date') or '')[:10]==day and s.get('start_minute')==minute]
    if len(matches)!=1:
        return dict(base,state='WAIT_FOR_SAFE_SLOT',next_action='RECONCILE_EXACT_CLOCK_WINDOW')
    idx,slot=matches[0]
    if (d.get('hard_session_allowed') is not True or
        str(d.get('action') or '').upper()!='TRAIN' or
        str(window.get('state') or '').upper() not in {'WATCH','OPEN'} or
        str(slot.get('restriction') or 'NONE').upper()!='NONE' or slot.get('is_race') or
        (p.get('hard_prescription_available') is True and p.get('status')=='PRESCRIBED')):
        return dict(base,state='WAIT_FOR_SAFETY',next_action='REASSESS_SAFE_WINDOW')
    template=_v196926_canonical_baseline_template(goal,slot)
    if not template:
        return dict(base,state='REQUIRES_SMALLER_CANONICAL_SLOT',
                    next_action='FIND_SAFE_ENVELOPE_OR_EXISTING_ACTIVITY')
    sig={'schema':R196926_SCHEMA,'goal':goal,'dimension':dim,
         'template_id':template['template_id'],'signature':template['template_signature'],
         'library_version':template['library_version'],'date':day,'minute':minute,
         'quality_window':iso,'last_hard_end':window.get('last_hard_end'),
         'primary_state':test.get('state'),'primary_when':test.get('when'),
         'gap':dose.get('reason')}
    token=hashlib.sha256(json.dumps(sig,sort_keys=True,ensure_ascii=False,
                          default=str,separators=(',',':')).encode('utf-8')).hexdigest()
    return dict(base,state='CONSENT_REQUIRED',request_id=token,
        index=idx,date=day,start_minute=minute,window_iso=iso,
        template=template,consent_scope=_R196926_SCOPE,
        next_action='REQUEST_FIRST_CANONICAL_OBSERVATION',
        safety_eligible=True,availability_confirmed=False,
        observed_watts_only=True,not_a_performance_grade=True)

def _v196926_final_attestation(sessions,prescription):

    The input compiled canonical session is retained for exact comparison.

    rows=[dict(x) if isinstance(x,dict) else x for x in (sessions or [])]
    p=prescription if isinstance(prescription,dict) else {}
    bridge=p.get('initial_evidence_bridge') or {}
    if bridge.get('state')!='CANONICAL_OBSERVATION_COMPILED':return rows
    req=bridge.get('request') or {};t=req.get('template') or {};idx=req.get('index')
    originals=p.get('sessions') or []
    if type(idx) is not int or idx<0 or idx>=len(rows) or idx>=len(originals):
        raise ValueError('R196926_FINAL_EVIDENCE_SESSION_MISSING')
    canonical=originals[idx]; final=rows[idx]
    core=('date','start_minute','title','intensity_class','main_set',
          'nova_workout_id','nova_contract_id','nova_canonical_plan_id')
    if (not isinstance(canonical,dict) or not isinstance(final,dict) or
        any(final.get(key)!=canonical.get(key) for key in core) or
        canonical.get('nova_evidence_protocol')!=_R196926_PROTOCOL or
        canonical.get('nova_evidence_request_id')!=req.get('request_id') or
        final.get('nova_workout_id')!=t.get('template_id') or
        final.get('nova_canonical_plan_id')!=bridge.get('canonical_plan_id') or
        final.get('date')!=req.get('date') or
        final.get('start_minute')!=req.get('start_minute') or
        final.get('prescribed_power_low') is not None or
        final.get('prescribed_power_high') is not None or
        final.get('provisional_quality') or not final.get('hard_spacing_relevant') or
        sum(str(x.get('intensity_class') or '').lower() in {'vo2','tempo','threshold'}
            for x in rows if isinstance(x,dict))!=1):
        raise ValueError('R196926_FINAL_EVIDENCE_PROTOCOL_DIVERGED')
    # Projection may remove optional provenance fields. Restore those *only*
    # after validating actual identical mechanics and authority.
    for key in ('nova_evidence_protocol','nova_evidence_request_id',
        'assessment_only','work_band_exploratory','watt_band_is_a_ceiling',
        'no_performance_pass_mark','evidence_required'):
        if key in final and final[key]!=canonical[key]:
            raise ValueError('R196926_FINAL_EVIDENCE_METADATA_CONFLICT')
        final[key]=copy.deepcopy(canonical[key])
    rows[idx]=final
    return rows

    """Persistent explainable next action for any science dimension, not a fake test.

    b=((d.get('nova_prescription') or {}).get('initial_evidence_bridge') or {})
    req=b.get('request') or {}
    if req.get('request_id')!=request_id:return {'state':'MISMATCHED_REQUEST'}
    led=d.get('microcycle_ledger') or {}
    if not led.get('ledger_id'):return {'state':'NO_ISSUED_PLAN'}
    conn=_db_connect()
    try:
        cur=conn.cursor()
        if not _db_is_postgres():cur.execute('BEGIN IMMEDIATE')
        sql='SELECT ledger_json FROM microcycle_ledgers WHERE id=? AND user_id=?'
        if _db_is_postgres():sql+=' FOR UPDATE'
        cur.execute(_db_sql(sql),(str(led['ledger_id']),str(user_id)))
        fetched=cur.fetchone()
        if not fetched:conn.rollback();return {'state':'NO_ISSUED_PLAN'}
        old=fetched[0];payload=json.loads(old)
        matches=[r for r in payload.get('prescription_history') or [] if isinstance(r,dict) and
                 r.get('nova_evidence_protocol')==_R196926_PROTOCOL and
                 hmac.compare_digest(str(r.get('nova_evidence_request_id') or ''),request_id)]
        if not matches:conn.rollback();return {'state':'NO_ISSUED_PLAN'}
        if len(matches)!=1:conn.rollback();return {'state':'DUPLICATE_PLAN_IDENTITY'}
        row=matches[0];status=str(row.get('prescription_status') or '').upper()
        if status=='WITHDRAWN':conn.rollback();return {'state':'ALREADY_WITHDRAWN'}
        if (status not in {'ACTIVE','PLANNED','ISSUED'} or row.get('executed_activity_id')):
            conn.rollback();return {'state':'ALREADY_EXECUTED_OR_NOT_ACTIVE'}
        row.update({'prescription_status':'WITHDRAWN','withdrawn_request_id':request_id,
          'withdrawn_reason':'ATHLETE_WITHDREW_INITIAL_EVIDENCE_CONSENT',
          'withdrawn_at_local':get_rome_now().isoformat(timespec='seconds')})
        anchor=payload.get('primary_anchor') or {}

            anchor.update({'status':'WITHDRAWN','prescription_lock':False})
        payload['primary_anchor']=anchor
        new=json.dumps(payload,ensure_ascii=False,separators=(',',':'),default=_report_json_default)
        cur.execute(_db_sql('UPDATE microcycle_ledgers SET ledger_json=?,updated_at_utc=? WHERE id=? AND user_id=? AND ledger_json=?'),
          (new,_now_utc_text(),str(led['ledger_id']),str(user_id),old))
        if cur.rowcount!=1:conn.rollback();return {'state':'CONCURRENT_LEDGER_UPDATE'}
        conn.commit();return {'state':'WITHDRAWN'}
    except (ValueError,TypeError,KeyError,json.JSONDecodeError):
        conn.rollback();return {'state':'MALFORMED_LEDGER'}
    finally:conn.close()

def _v196927_attest_final_contract(sessions, compiled):
    """Fail closed if a signed evidence workout changes in semantic projection.

    The caller passes a *deep copy* taken BEFORE any coach/semantic transforms.
    Missing optional provenance is restored by legacy adapters after this check;
    an altered mechanical field, schedule, contract or prescribed power is NEVER
    restored by this function. Works for any workout goal in the catalog.
    """
    p = compiled if isinstance(compiled, dict) else {}
    rows = sessions if isinstance(sessions, list) else []
    originals = p.get('sessions') or []
    bridges = (
        ('initial_evidence_bridge', 'CANONICAL_OBSERVATION_COMPILED',
         _R196926_PROTOCOL, 'R196927_INITIAL'),
        ('coupled_evidence_bridge', 'CANONICAL_ASSESSMENT_COMPILED',
         'COUPLED_CANONICAL_MEASUREMENT_V1', 'R196927_COUPLED'),
    )
    active = [(key, p[key], protocol, code) for key, state, protocol, code in bridges
              if isinstance(p.get(key), dict) and p[key].get('state') == state]
    if not active:
        return [dict(x) if isinstance(x,dict) else x for x in rows]
    if len(active) != 1 or len(originals) != len(rows):
        raise ValueError('R196927_CANONICAL_PLAN_CARDINALITY_CHANGED')
    key, bridge, protocol, code = active[0]
    request = bridge.get('request') or {}
    index = request.get('index')
    if (type(index) is not int or index < 0 or index >= len(rows) or
        not isinstance(originals[index], dict) or
        not isinstance(rows[index], dict)):
        raise ValueError(code + '_SESSION_MISSING')
    original = originals[index]
    final = rows[index]
    signed_token = bridge.get('agreed_request_id')
    if (not bridge.get('issued') or not signed_token or
        signed_token != request.get('request_id') or
        original.get('nova_evidence_protocol') != protocol or
        original.get('nova_evidence_request_id') != signed_token or

def _v196927_attest_issued_ledger(ledger, sessions, compiled):
    """A signed hard evidence test must have one matching ledger record.

    The existing ledger writer remains the sole persistence owner. Never write
    independent history or fabricate a completed activity as an alternative.
    """
    p=compiled if isinstance(compiled,dict) else {}
    bridges=(('initial_evidence_bridge',_R196926_PROTOCOL),
             ('coupled_evidence_bridge','COUPLED_CANONICAL_MEASUREMENT_V1'))
    active=[(p[key],protocol) for key,protocol in bridges if
            isinstance(p.get(key),dict) and p[key].get('issued') and
            p[key].get('state') in {'CANONICAL_OBSERVATION_COMPILED','CANONICAL_ASSESSMENT_COMPILED'}]
    if not active:return {'state':'NOT_APPLICABLE'}
    if len(active)!=1:raise ValueError('R196927_LEDGER_MULTIPLE_EVIDENCE_CONTRACTS')
    bridge,protocol=active[0]
    request=bridge.get('request') or {}
    matches=[x for x in sessions or [] if isinstance(x,dict) and
             x.get('nova_evidence_protocol')==protocol and
             x.get('nova_evidence_request_id')==bridge.get('agreed_request_id')]
    if len(matches)!=1:raise ValueError('R196927_LEDGER_SESSION_NOT_UNIQUE')
    ses=matches[0]
    history=(ledger or {}).get('prescription_history') or []
    recorded=[x for x in history if isinstance(x,dict) and
              x.get('nova_evidence_protocol')==protocol and
              x.get('nova_evidence_request_id')==bridge.get('agreed_request_id')]
    if len(recorded)!=1:raise ValueError('R196927_LEDGER_EVIDENCE_NOT_PERSISTED')
    stored=recorded[0]
    if (str(stored.get('prescription_status') or '').upper() in
            {'WITHDRAWN','SUPERSEDED','FULFILLED','CANCELLED','CANCELED'} or
            stored.get('executed_activity_id')):
        raise ValueError('R196927_LEDGER_NOT_AN_ACTIVE_MEASUREMENT')
    for field in ('date','main_set','nova_workout_id','nova_contract_id',
                  'nova_canonical_plan_id'):
        if stored.get(field)!=ses.get(field):
            raise ValueError('R196927_LEDGER_IDENTITY_DIVERGED_' + field.upper())
    if ((request.get('date') and stored.get('date')!=request['date']) or
        stored.get('nova_canonical_plan_id')!=bridge.get('canonical_plan_id')):
        raise ValueError('R196927_LEDGER_SIGNED_WINDOW_DIVERGED')
    return {'state':'ATTESTED','protocol':protocol,
            'request_id':bridge.get('agreed_request_id')}

def _v196926_build_evidence_backlog(snapshot):
    d=snapshot if isinstance(snapshot,dict) else {}
    decision=d.get('nova_decision') or {}
    roadmap=d.get('adaptive_roadmap') or {}

    scientific=roadmap.get('scientific_progression') or {}
    goal=str(decision.get('goal_key') or roadmap.get('goal_key') or '').upper()

    qs=d.get('question_state') or roadmap.get('question_state') or {}
    qs=qs if isinstance(qs,dict) else {}
    dimensions=qs.get('dimensions') or {}
    selected=set(str(x).upper() for x in qs.get('open_dimensions') or [])
    initial=prescription.get('initial_evidence_bridge') or {}
    coupled=prescription.get('coupled_evidence_bridge') or {}
    primary=roadmap.get('test') or {}
    if initial:
        state=initial.get('state') or 'NEEDS_EVIDENCE'
        req=initial.get('request') or {}
        next_action=('EXECUTE_AND_LINK_MEASURED_OBSERVATION' if initial.get('issued')
                     else req.get('next_action') or 'REQUEST_OR_IMPORT_EVIDENCE')
        active_protocol=_R196926_PROTOCOL
    elif coupled:
        state=coupled.get('state') or 'NEEDS_EVIDENCE'
        next_action=('EXECUTE_MEASURED_COUPLED_PROTOCOL' if coupled.get('issued')
                     else 'REQUEST_CONSENT_TO_MEASURE_COUPLED_RECOVERY')
        active_protocol='COUPLED_CANONICAL_MEASUREMENT_V1'
    elif str(primary.get('state') or '').upper() not in {'COMPLETE','COMPLETED'}:
        state='PRIMARY_TEST_REQUIRED'
        next_action='RUN_EXISTING_R27_GOAL_TEST'
        active_protocol='R27_GOAL_PRIMARY_TEST' if primary else None
    else:
        state='EVIDENCE_MONITORING'
        next_action='EVALUATE_NEW_ACTIVITIES_AND_PROGRESS_OPEN_QUESTION'
        active_protocol=None
    issues=[]
    for key,record in (dimensions.items() if isinstance(dimensions,dict) else []):
        if not isinstance(record,dict):continue
        name=str(key).upper(); status=str(record.get('state') or 'UNKNOWN').upper()
        exiting=record.get('exit_satisfied')
        closed=(exiting is True or (exiting is None and record.get('closed') is True))
        if closed:continue
        relevant=(name==dimension)
        protocol=(active_protocol if relevant else
                  ('R27_GOAL_PRIMARY_TEST' if name=='FRESH_CAPACITY' and primary else None))
        authorized=bool(relevant and (
            initial.get('issued') or coupled.get('issued')))
        is_priority=name in selected or relevant
        issues.append({'dimension':name,'question':name,'state':status,
            'exit_satisfied':False,'evidence_count':record.get('evidence_count'),
            'last_evidence_date':record.get('last_evidence_date'),
            'missing_reason':record.get('reason') or 'DIRECT_CANONICAL_EVIDENCE_NOT_YET_SUFFICIENT',
            'evidence_source':record.get('source'),
            'priority_scope':'ACTIVE_ROADMAP' if is_priority else 'GOAL_SCIENCE_TRACKING',
            'acquisition_protocol':protocol,
            'protocol_state':('ISSUED_AWAITING_MEASURED_ACTIVITY' if authorized else
                'EXISTING_PROTOCOL_REQUIRES_CURRENT_SAFETY_AND_CONSENT' if protocol else
                'OWNER_VALIDATION_REQUIRED'),
            'next_action':next_action if relevant else
                'SEARCH_CANONICAL_NATIVE_ACTIVITIES_THEN_REQUEST_GOAL_SPECIFIC_VALIDATED_PROTOCOL',
            'science_owner_must_validate':not bool(protocol),
            'auto_promotion_from_consent':False})
    issues.sort(key=lambda i:(0 if i['dimension']==dimension else
                             1 if i['priority_scope']=='ACTIVE_ROADMAP' else 2,
                             i['dimension']))
    previous=d.get('evidence_cycle_review') or prescription.get('evidence_cycle_review') or {}
    return {'schema':'V4.9.128-R196.9.27-GENERAL_EVIDENCE_CONTINUITY-1','goal_key':goal,'current_dimension':dimension,
        'state':state,'next_action':next_action,'issues':issues,
        'open_issue_count':len(issues),
        'active_roadmap_issue_count':sum(x['priority_scope']=='ACTIVE_ROADMAP' for x in issues),
        'question_state_source':'CANONICAL_QUESTION_STATE_DIMENSIONS' if isinstance(dimensions,dict) and dimensions else
            'QUESTION_STATE_UNAVAILABLE',
        'legacy_validation_review':{'owner':'R196920_R27_VALIDATION_ONLY',
            'state':previous.get('state'),'issued':bool(previous.get('issued'))},
        'acquisition_owner':'R196924_R196926_CANONICAL_EVIDENCE',
        'never_infer_watts':True,'execution_requires_current_safety_and_consent':True,
        'no_automatic_success_from_consent':True,
        'activity_reconciliation':'R149_NATIVE_ORIGINAL_ACTIVITY'}
