def _v4897_filter_ai_text(text, truth, field_name=""):
    kept=[]; removed=[]
    for sentence in _v4844_sentence_list(text):

def _v110_polish_nova_recommendation(base_analysis, nova_prescription=None, snapshot_question=None, ai_runtime_override=None, usage_kind="SNAPSHOT"):
    """One tiny AI call: language-only commentary around Nova's final canonical advice."""
    base_analysis=base_analysis if isinstance(base_analysis,dict) else {}
    nova_prescription=nova_prescription if isinstance(nova_prescription,dict) else {}
    canonical=_v106_recommendation_projection(base_analysis.get("next_sessions"))
    plan_id=str(nova_prescription.get("canonical_plan_id") or "none")
    workout_id=str(nova_prescription.get("workout_id") or "none")
    dimension=str(nova_prescription.get("dimension") or "none")
    question=str(snapshot_question or "NONE").strip()[:700]
    prompt=(
        "You are THE LAB's language editor, not a coach. Nova/Python has already made the final training decision. "
        "Write ONLY a short explanatory commentary that can be appended after the canonical recommendation below. "
        "Do not prescribe, change, repeat or add workout mechanics, intensity, duration, reps, watts, timing, scheduling or alternatives. "
        "Do not introduce any number. Do not tell the athlete to add, remove, move, skip, shorten, extend, increase, decrease or replace anything. "
        "Explain the rationale in natural concise language. If the athlete question is relevant and can be answered from the canonical recommendation, address it without adding new training instructions. "
        "If no useful language improvement is needed, return an empty string. Return ONLY valid JSON with exactly one key: commentary.\n\n"
        f"PLAN ID: {plan_id}\nWORKOUT ID: {workout_id}\nDIMENSION: {dimension}\n"
        f"CANONICAL RECOMMENDATION:\n{canonical}\n\nATHLETE QUESTION:\n{question}"
    )
    ai_runtime=ai_runtime_override or _resolve_ai_runtime()
    result=_call_ai_provider(
        ai_runtime["provider"],ai_runtime["api_key"],ai_runtime["model"],prompt,max_tokens=320,
        usage_kind=str(usage_kind or "SNAPSHOT")+"_RECOMMENDATION_ONLY", user_id=ai_runtime.get("user_id"),
        credential_source=ai_runtime.get("source"), prompt_profile={
            "canonical_recommendation_chars":len(canonical),"snapshot_question_chars":len(question),
            "static_rules_chars":max(0,len(prompt)-len(canonical)-len(question)),"prompt_budget_chars":6000,
        }, return_event_id=True,
    )
    raw,event_id=result if isinstance(result,tuple) else (result,None)
    raw=str(raw or "").strip()
    if raw.startswith("```"):
        raw=raw.strip("`")
        if raw.startswith("json"):raw=raw[4:]
    try:
        payload=json.loads(raw)
    except Exception:
        _set_ai_usage_status(event_id,"parse_error","JSON_DECODE_ERROR")
        return "",event_id
    commentary=_v110_recommendation_commentary_guard(payload.get("commentary"))
    return commentary,event_id

def _normalize_chat_history(raw_history, max_messages=10):
    clean = []
    if not isinstance(raw_history, list):
        return clean
    for item in raw_history[-max_messages:]:
        if not isinstance(item, dict):
            continue
        role = str(item.get("role") or "").strip().lower()
        text = str(item.get("text") or "").strip()
        if role not in ("user", "assistant") or not text:
            continue
        clean.append({"role": role, "text": text[:2500]})
    return clean
# R119 STEP 2 · Read-only explanation layer. The coach engine must NOT call any of

def _v119_nova_intent(question):
    """R120 language intent gate; preserve R119 function contract for existing routes.

    Prioritise an explicit progression obstacle over incidental "back" language.
    Never route injury / prescription-changing questions to templated explanations.
    """
    text=str(question or "").strip().lower()
    for accent,plain in (("à","a"),("è","e"),("é","e"),("ì","i"),("ò","o"),("ù","u")):
        text=text.replace(accent,plain)
    text=re.sub(r"\s+", " ", text)
    if not text or len(text)>1200:
        return None
    if re.search(r"\b(fever|injur|illness|pain|sick|malat|infortun|dolor|febbr|modify|swap|spost|cambia|modific|should i|posso fare)\w*",text):
        return None
    stage_words=bool(re.search(r"\b(?:roadmap|repeatabil\w*|transfer|trasfer\w*|fas[ei]|stage|step|progress\w*|progres\w*|avanz\w*)\b",text))
    # Explicit 'why back to stage' asks for stage history, not future requirements.
    rollback=bool(re.search(r"\b(?:back\s+(?:in|to)|(?:went|go|going)\s+back|move\w*\s+back|returned\s+to|regress\w*|tornat\w*|indietro)\b",text))
    # Match the exact live user question 'What is holding me back from progressing
    # in my roadmap?' as well as other phrasing. Bare 'back' is NOT rollback.
    obstacle=bool(re.search(
        r"(?:\bwhat\b.*\b(?:missing|need\w*|require\w*|prevent\w*|block\w*|holding|stopping|keeping)\b|"
        r"\bwhat\b.*\b(?:must|should)\b.*\bdo\b|"
        r"\b(?:holding\s+(?:me\s+)?back|prevent\w*|block\w*|stop\w*|keep\w*\s+me)\b|"
        r"\bhow\b.*\b(?:advance|progress|reach|move\s+(?:to|forward))\b|"
        r"\bwhat\b.*\b(?:advance|progress|reach|transfer|move\s+forward)\b|"
        r"\b(?:cosa|che\s+cosa)\b.*\b(?:manc\w*|serv\w*|imped\w*|bloc\w*|occor\w*|devo)\b|"
        r"\b(?:come|perche|per)\b.*\b(?:avanz\w*|pass\w*|arriv\w*|progred\w*)\b|"
        r"\b(?:manc\w*|imped\w*|bloc\w*)\b.*\b(?:avanz\w*|pass\w*|progress\w*|progres\w*)\b)",text))
    if stage_words and obstacle:
        # A question with two explicit independent demands remains for AI Coach.
        if rollback and re.search(r"\b(?:and|e)\b.*\b(?:what|cosa|che cosa)\b",text):
            return None
        return "NEXT_EVIDENCE"
    if stage_words and (rollback or re.search(r"\b(?:why|perche|come mai|explain|spiega|stato|where|dove|which|quale)\b",text) or len(text)<56):
        return "ROADMAP"
    if re.search(r"\b(?:why|perche|come mai|explain|spiega)\b",text) and re.search(r"\b(?:session|workout|training|allenament\w*|sedut\w*|3\s*[×x]\s*6|3x6)\b",text):
        return "NEXT_SESSION"
    return None

def _v119_list(value):
    if isinstance(value,list):
        return value
    if isinstance(value,dict):
        numeric=[(int(re.sub(r"[^0-9]", "",str(k))),v) for k,v in value.items() if re.fullmatch(r"\[?\d+\]?",str(k))]
        return [v for _,v in sorted(numeric)] if len(numeric)==len(value) else []
    return []

_CONTENT_AUDIT_SENSITIVE_KEY_RE = re.compile(
    r"(^|_)(?:api_?key|password|secret|token|authorization|cookie|credential|private_?key|license_?code|reset_?code)(?:_|$)",
    re.I,
)
_CONTENT_AUDIT_SIGNAL_KEY_RE = re.compile(
    r"(^|_)(?:status|state|confidence|assessment|class|zone|signal|trend|label|source|mode|quality|readiness|restriction|environment|demand|available|eligible|eligibility|retest|warning)(?:_|$)",
    re.I,
)
def _content_audit_sensitive_key(key):
    return bool(_CONTENT_AUDIT_SENSITIVE_KEY_RE.search(str(key or "")))
def _content_audit_scalar(value):
    if value is None:
        return "null"
    if isinstance(value, bool):
        return "true" if value else "false"
    if isinstance(value, (int, float)) and not isinstance(value, bool):
        if isinstance(value, float) and not math.isfinite(value):
            return json.dumps(str(value), ensure_ascii=False)
        return str(value)
    return json.dumps(str(value), ensure_ascii=False)
def _content_audit_render_node(node, indent=0):
    pad = "  " * indent
    lines = []
    if isinstance(node, dict):
        if not node:
            return [pad + "- {}"]
        for key, value in node.items():
            key_text = str(key)
            prefix = f"{pad}- `{key_text}`"
            if _content_audit_sensitive_key(key_text):
                lines.append(prefix + ": [REDACTED]")
            elif isinstance(value, (dict, list)):
                lines.append(prefix + ":")
                lines.extend(_content_audit_render_node(value, indent + 1))
            else:
                lines.append(prefix + ": " + _content_audit_scalar(value))
        return lines
    if isinstance(node, list):
        if not node:
            return [pad + "- []"]
        for idx, value in enumerate(node):
            prefix = f"{pad}- `[{idx}]`"
            if isinstance(value, (dict, list)):
                lines.append(prefix + ":")
                lines.extend(_content_audit_render_node(value, indent + 1))
            else:
                lines.append(prefix + ": " + _content_audit_scalar(value))
        return lines
    return [pad + "- " + _content_audit_scalar(node)]
def _content_audit_signal_index(node, path="data"):
    out = []
    if isinstance(node, dict):
        for key, value in node.items():
            key_text = str(key)
            child_path = f"{path}.{key_text}" if path else key_text
            if _content_audit_sensitive_key(key_text):
                continue
            if isinstance(value, (dict, list)):
                out.extend(_content_audit_signal_index(value, child_path))
            elif _CONTENT_AUDIT_SIGNAL_KEY_RE.search(key_text):
                out.append((child_path, value))
    elif isinstance(node, list):
        for idx, value in enumerate(node):
            out.extend(_content_audit_signal_index(value, f"{path}[{idx}]"))
    return out
def _content_audit_compact_payload(data):
    try:
        out = json.loads(json.dumps(data, ensure_ascii=False, default=str))
    except Exception:
        return data
    ap = out.get("advanced_physiology") if isinstance(out, dict) else None
    if isinstance(ap, dict) and isinstance(ap.get("cards"), list):
        def card_key(card):
            return card.get("key") if isinstance(card, dict) else None
        groups = []
        for g in ap.get("groups") or []:
            if not isinstance(g, dict):
                continue
            groups.append({k:v for k,v in g.items() if k != "cards"} | {"card_keys":[card_key(c) for c in (g.get("cards") or []) if card_key(c)]})
        sections = []
        for s in ap.get("display_sections") or []:
            if not isinstance(s, dict):
                continue
            sections.append({k:v for k,v in s.items() if k != "cards"} | {"card_keys":[card_key(c) for c in (s.get("cards") or []) if card_key(c)]})
        ap["groups"] = groups
        ap["display_sections"] = sections
        ap["gap_card_keys"] = [card_key(c) for c in (ap.get("gap_cards") or []) if card_key(c)]
        ap.pop("gap_cards", None)
        ap["audit_structure_note"] = "Full metric bodies are stored once in advanced_physiology.cards; groups/display_sections/gap_card_keys reference those canonical cards by key."
    return out

# Pending requests/declines are tracked independently from the QA consent field.
def _r170_consent_queue(user_id, action=None):
    _db_execute("""CREATE TABLE IF NOT EXISTS beta_qa_consent_requests (
        user_id TEXT PRIMARY KEY, requested_at_utc TEXT, responded_at_utc TEXT,
        last_response TEXT)""")
    if action == "REQUEST":
        now = _now_utc_text()
        _db_execute("""INSERT INTO beta_qa_consent_requests(user_id,requested_at_utc,responded_at_utc,last_response)
            VALUES(?,?,NULL,NULL) ON CONFLICT(user_id) DO UPDATE SET
            requested_at_utc=excluded.requested_at_utc, responded_at_utc=NULL, last_response=NULL""", (str(user_id), now))
    elif action in {"DECLINED", "ACCEPTED"}:
        _db_execute("""INSERT INTO beta_qa_consent_requests(user_id,requested_at_utc,responded_at_utc,last_response)
            VALUES(?,?,?,?) ON CONFLICT(user_id) DO UPDATE SET
            responded_at_utc=excluded.responded_at_utc,last_response=excluded.last_response""",
            (str(user_id), _now_utc_text(), _now_utc_text(), action))
    rows = _db_execute("SELECT * FROM beta_qa_consent_requests WHERE user_id=?", (str(user_id),), fetch=True)
    return dict(rows[0]) if rows else {}

# R177: stable code = what failed; correlation_id = this particular occurrence.
# Never put raw exception messages, athlete identifiers, or upstream response bodies here.
def _r177_snapshot_incident(stage, category, error_class=None, actor_id=None, diagnostic=None):
    safe_stage = re.sub(r"[^A-Z0-9_]+", "_", str(stage or "UNKNOWN").upper())[:48]
    family = str(category or "UNEXPECTED").upper()
    if family not in {"FETCH", "DATA_SHAPE", "DECISION", "AUTHORITY", "RENDER", "SAVE", "UNEXPECTED"}:
        family = "UNEXPECTED"
    code = "TL-SNP-" + family + "-" + ("FAILED" if family != "UNEXPECTED" else "PIPELINE")
    incident_id = "SNP-" + uuid.uuid4().hex[:16].upper()
    retryable = family in {"FETCH", "UNEXPECTED"}
    # R196.9.1: only explicitly supplied, machine-generated diagnostic codes.
    # Never log raw exception messages, upstream bodies or athlete data.
    safe_diagnostic = re.sub(r"[^A-Z0-9_+\-]+", "_", str(diagnostic or "").upper())[:160]
    if safe_diagnostic and not re.fullmatch(
        r"(?:TL-CMAG-[A-Z0-9_+]+|AI_(?:TIMEOUT|CONNECTION|EMPTY_RESPONSE|TRUNCATED|UNAVAILABLE))",
        safe_diagnostic,
    ):
        safe_diagnostic = ""
    detail = {"code":code, "stage":safe_stage,"severity":"ERROR", "release":APP_VERSION,
              "correlation_id":incident_id,"recoverable":True,"retryable":retryable,
              "error_class":re.sub(r"[^A-Za-z0-9_]+", "", str(error_class or ""))[:64]}
    if safe_diagnostic:
        detail["diagnostic"] = safe_diagnostic
    try:
        app.logger.error("SNAPSHOT_INCIDENT code=%s stage=%s correlation_id=%s release=%s class=%s diagnostic=%s",
                         code,safe_stage,incident_id,APP_VERSION,detail["error_class"],safe_diagnostic or "NONE")
    except Exception:
        pass
    try:
        audit_event("SNAPSHOT_INCIDENT",actor_user_id=actor_id,target_user_id=actor_id,details=detail)
    except Exception:
        pass
    instruction = "Try again later; if this repeats, send support the incident ID." if retryable else "Contact support with the incident ID; do not repeat the same action until it is checked."
    message = ("THE LAB could not complete the Snapshot. The new report was not confirmed as saved. Please check Last Full Report before relying on it. "
               + instruction + " Diagnostic " + code + " · stage " + safe_stage
               + ((" · detail " + safe_diagnostic) if safe_diagnostic else "") + " · incident " + incident_id + ".")
    return {"message":message,**detail}
