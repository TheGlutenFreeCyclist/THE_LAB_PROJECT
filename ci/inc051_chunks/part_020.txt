def _r196966_recover_morning_slot(clock,rhythm):
    """A single explicit PM event must not erase a HIGH-confidence AM habit.

    Only create a clearly unconfirmed RHYTHM opportunity. Do not fabricate
    athlete availability or overwrite the calendar's PM booking. If the
    rhythm is weak, an explicit restriction exists, or AM already exists,
    preserve the original Clock unchanged.
    """
    out = copy.deepcopy(clock) if isinstance(clock,dict) else clock
    if not isinstance(out,dict) or not isinstance(rhythm,dict):
        return out
    if (str(rhythm.get('confidence') or '').upper() != 'HIGH' or
        str(rhythm.get('state') or '').upper() not in {'STABLE','ADAPTING'}):
        return out
    cp = rhythm.get('clock_profile') or {}
    first = cp.get('first_start_min') or {}
    dur = cp.get('duration_min') or {}
    scores = rhythm.get('weekday_double_scores') or {}
    recent_double = _rhythm_num((rhythm.get('recent') or {}).get('double_day_pct'))
    if recent_double is None or recent_double < .75:
        return out
    slots = list(out.get('next_slots') or [])
    groups = {}
    for s in slots:
        if isinstance(s,dict) and s.get('date'):
            groups.setdefault(str(s['date'])[:10],[]).append(s)
    rebuilt = []
    for day, items in groups.items():
        if (not any(x.get('is_planned_training') is True and
                    str(x.get('source') or '').upper() == 'CALENDAR' for x in items)
            or any(str(x.get('restriction') or 'NONE').upper() != 'NONE'
                   or x.get('is_race') is True for x in items)):
            continue
        try:
            weekday = str(date.fromisoformat(day).weekday())
        except ValueError:
            continue
        score = _rhythm_num(scores.get(weekday))
        evidence = (rhythm.get('weekday_evidence') or {}).get(weekday) or {}
        if score is None or score < .75 or evidence.get('supported') is not True:
            continue
        v = _rhythm_num(first.get(weekday))
        length = _rhythm_num(dur.get(weekday))
        # The expected AM opportunity must be independently supported; no
        # generic 6AM/60m fantasy for athletes without weekday evidence.
        if v is None or length is None or not 0 <= v < 720 or length <= 0:
            continue
        if any((_rhythm_num(x.get('start_minute')) is not None and
                abs(float(x['start_minute']) - v) < 120) for x in items):
            continue
        if any(_rhythm_num(x.get('start_minute')) is not None and
               float(x['start_minute']) <= v + max(90.0,length+30.0)
               for x in items):
            continue
        # The calendar may be a fully exclusive availability limit. Do not
        # infer AM if explicit daily availability declares a single session.
        availability = out.get('daily_availability') or {}
        if isinstance(availability,dict):
            constraints = availability.get(day) or {}
            if isinstance(constraints,dict) and (constraints.get('single_session_only') is True
                 or constraints.get('morning_available') is False):
                continue
        minute=int(round(v))
        rebuilt.append(dict(date=day,period='RHYTHM',
            window='history-learned opportunity',
            label=f'{date.fromisoformat(day).strftime("%A")} · learned morning rhythm · around {minute//60}:{minute%60:02d}',
            restriction='NONE',is_race=False,is_planned_training=False,
            event_title=None,event_note=None,training_environment=None,
            training_demand=None,expected_duration_min=None,
            expected_duration_max=None,planning_item_id=None,source='RHYTHM',
            timing_confidence='HIGH',start_minute=minute,
            duration_hint_min=round(length,1),
            duration_hint_label=_rhythm_format_duration(length),
            duration_estimate_source='LEARNED_WEEKDAY_SINGLE_SESSION',
            duration_estimate_not_availability=True,
            availability_confirmed=False,r196966_recovered_morning=True))
    if rebuilt:
        out['next_slots']=sorted(slots+rebuilt,
            key=lambda s:(str(s.get('date') or ''),
                          _rhythm_num(s.get('start_minute')) if _rhythm_num(s.get('start_minute')) is not None else 1440))
        out['r196966_recovered_morning_slots']=[x['date'] for x in rebuilt]
        out['next_slot_label']=out['next_slots'][0].get('label')
    return out

def _nova_r196975_cross_goal_display_offer(view):
    """Read-only display filter: reject demonstrably foreign-goal prior offers.

    The current decision/roadmap remains authoritative; known catalogue
    identities and explicit goal tags are only consistency evidence. Missing
    provenance is NOT converted to a new execution authorization. Never
    mutate stored offers, scheduling, clearance or historical audit.
    """
    d = view if isinstance(view, dict) else {}
    offer = d.get('next_quality_offer') or {}
    if not isinstance(offer, dict):
        return None
    workout = offer.get('workout') or {}
    if not isinstance(workout, dict):
        return None
    decision = d.get('nova_decision') or {}
    roadmap = d.get('adaptive_roadmap') or {}
    current_goal = str((decision if isinstance(decision, dict) else {}).get('goal_key')
                       or (roadmap if isinstance(roadmap, dict) else {}).get('goal_key') or '').upper()
    if not current_goal:
        return offer  # unknown goal does not justify rewriting a historic offer
    explicit_goals = [str(g).upper() for g in (
        offer.get('goal_key'), workout.get('goal_key'), workout.get('goal')) if g]
    if any(g != current_goal for g in explicit_goals):
        return None
    identity = str(offer.get('workout_id') or workout.get('nova_workout_id') or '')
    catalog_goals = {str(t.get('goal') or '').upper() for t in _NOVA_WORKOUT_LIBRARY
                     if t.get('id') == identity and t.get('goal')}
    if catalog_goals and current_goal not in catalog_goals:
        return None
    return offer


def _nova_r196975_quality_direction(decision, road, offer=None):
    """Read-only next-QUALITY communication; never use it as training clearance.

    Priority is an already-selected canonical workout, then an existing formal
    validation step, then a known goal/dimension-specific catalogue identity,
    then the actual roadmap stage. No invented watts, dosing, date or booking.
    Even HOLD/VALIDATE keep a direction; their decision remains unchanged.
    """
    d = decision if isinstance(decision, dict) else {}
    r = road if isinstance(road, dict) else {}
    goal = str(d.get('goal_key') or r.get('goal_key') or '').upper()
    action = str(d.get('action') or '').upper()
    science = r.get('scientific_progression') or {}
    if not isinstance(science, dict):
        science = {}
    dimension = str(science.get('dimension') or d.get('dimension') or '').upper()
    current_offer = _nova_r196975_cross_goal_display_offer({
        'nova_decision':d,'adaptive_roadmap':r,'next_quality_offer':offer}) or {}

                           and x.get('label') and str(x.get('state') or '').upper() not in {'BUILT', 'COMPLETED'}), None)

            source = 'ROADMAP_FORMAL_VALIDATION_STEP'

    if not title:
        title = 'Next quality · ' + _R196975_GOAL_LABELS.get(goal,'specific development')
        source = 'GOAL_DIRECTION_NO_WORKOUT_CLAIM'
    return {
        'schema':R196975_DIRECTION_SCHEMA,
        'title':title,'goal_key':goal,'decision_action_observed':action,

        'execution_authorized_by_preview':False,
        'adds_workout_to_ledger':False,'adds_calendar_event':False,
    }

def _nova_r196979_quality_day_advisor(data):
    """Read-only assessment of independent physiological systems, not permission.

    Conservative heuristic, NOT a clinically validated score. Explicit baseline
    state outranks naked values; two measurements of autonomic balance count as
    one system. No decision, workout, or prescription field is ever written.
    """
    d = data if isinstance(data, dict) else {}
    decision = d.get('nova_decision') or {}
    pres = d.get('nova_prescription') or {}
    offer_wrapper = d.get('next_quality_offer') or {}
    offer = (offer_wrapper.get('workout') or {}) if isinstance(offer_wrapper, dict) else {}
    if not isinstance(offer, dict):
        offer = {}
    direction = d.get('next_quality_direction') or {}
    clock = d.get('coach_clock') or {}
    rhythm = clock.get('rhythm') or d.get('training_rhythm') or {}
    recent = rhythm.get('recent') or {}
    health = d.get('health_observatory') or {}
    signals = health.get('signals') or {}
    system_list = (d.get('physiological_state') or {}).get('systems') or []
    by_label = {str(x.get('label') or '').upper(): x for x in system_list if isinstance(x, dict)}

    def number(x):
        try:
            n = float(x)
            return n if n == n and abs(n) != float('inf') else None
        except (ValueError, TypeError):
            return None

    count_raw = d.get('energy_today_sessions')
    if count_raw is None:
        count_raw = clock.get('today_activity_count')
    try:
        count = max(0, int(count_raw or 0))
    except (ValueError, TypeError):
        count = 0
    pct = number(recent.get('double_day_pct'))
    double_habit = pct is not None and pct >= .65 and (number(recent.get('training_days')) or 0) >= 10
    ratio = number(d.get('energy_today_load_ratio'))
    duration_ratio = number(d.get('energy_today_duration_ratio'))
    # A count alone (including two micro sessions) never proves completion.
    workload_context = [x for x in (ratio, duration_ratio) if x is not None]
    day_complete = bool(double_habit and count >= 2 and workload_context
                        and max(workload_context) >= .65)
    heat_today = 'HEAT' in str(clock.get('today_activities_text') or '').upper()

    # Signal states are already assessed against this athlete's baseline by
    # the upstream observatory. No uncalibrated universal numeric thresholds.
    adverse_states = {'LOW', 'BELOW_USUAL', 'BELOW_BASELINE', 'ADVERSE',
                      'DEVIATION', 'OUTSIDE_RANGE', 'ABNORMAL', 'WORSE'}
    favorable_states = {'IMPROVED', 'FAVORABLE', 'BETTER_THAN_USUAL'}
    usual_states = {'USUAL', 'BASELINE', 'NORMAL', 'WITHIN_BASELINE', 'STABLE'}

    def signal_state(key):
        x = signals.get(key) or {}
        if not isinstance(x, dict) or x.get('available') is not True or x.get('stale') is True:
            return 'missing'
        state = str(x.get('state') or '').upper()
        if state in usual_states:
            return 'baseline'
        if state in adverse_states or (key == 'restingHR' and state in {'HIGH', 'ABOVE_USUAL'}):
            return 'adverse'
        # Improvement requires explicit trustworthy upstream evidence, not
        # "HRV higher" or a guessed comparison of raw numeric values.
        if state in favorable_states and x.get('favorable_confirmed') is True:
            return 'favorable'
        return 'unknown'

    def system_state(name):
        item = by_label.get(name) or {}
        if not item:
            return 'missing'
        status = str(item.get('status') or '').upper()
        color = str(item.get('status_class') or '').lower()
        if color == 'red':
            return 'adverse'
        if color == 'purple' and item.get('favorable_confirmed') is True and \
                str(item.get('confidence') or '').upper() in ('HIGH', 'VERIFIED'):
            return 'favorable'
        if color in ('green', 'cyan') or status in ('BALANCED', 'CONTROLLED', 'BASELINE'):
            return 'baseline'
        # Orange WATCH is a caution flag, not by itself a measured adverse
        # physiological deviation. Avoid recoding mild sleep debt as RED.
        if color == 'orange' or status == 'WATCH':
            return 'watch'
        return 'missing'

    hrv = signal_state('hrv')
    rhr = signal_state('restingHR')
    sleep = signal_state('sleepSecs')
    autonomic = system_state('AUTONOMIC')
    recovery = system_state('RECOVERY')
    load = system_state('TRAINING LOAD')

    def consolidate(system, measurements):
        if system == 'adverse' or 'adverse' in measurements:
            return 'adverse'
        known = [v for v in measurements if v in ('baseline', 'favorable')]
        if system == 'favorable':
            return 'favorable' if not measurements or known else 'baseline'
        if system == 'baseline' or known:
            # A red system has already been handled; orange WATCH with actual
            # usual measurements is context, not an extra adverse vote.
            return 'favorable' if system == 'missing' and len(known) >= 2 and all(
                v == 'favorable' for v in known) else 'baseline'
        return 'missing' if system == 'missing' else 'watch'

    autonomic = consolidate(autonomic, [hrv, rhr])
    recovery = consolidate(recovery, [sleep])
    # A quantified genuinely excessive same-day load is relevant even if
    # the old broad system label still says CONTROLLED.
    if count > 0 and ratio is not None and ratio > 1.35:
        load = 'adverse'
    elif load == 'watch' and ratio is not None and ratio <= 1.0:
        load = 'baseline'
    elif load == 'missing' and ratio is not None and 0 <= ratio <= 1.0:
        load = 'baseline'

    heat = d.get('heat_response') or {}
    # Only a documented acute physiological response; historical HEAT
    # association/HEAT activity names never serve as a veto.
    if (isinstance(heat, dict) and heat.get('measured_adverse_response') is True
            and str(heat.get('confidence') or '').upper() in ('HIGH', 'VERIFIED')):
        load = 'adverse'

    # Optional vital signs are a fourth independent domain ONLY when the
    # observatory explicitly supplies a fresh, reliable athlete-specific state.
    # Do not turn old SpO2/temperature or missing confidence into a veto.
    vitals = []
    for key in ('spO2', 'respirationRate', 'skinTemp', 'temperature'):
        item = signals.get(key) or {}
        if not isinstance(item, dict) or item.get('available') is not True:
            continue
        age = number(item.get('age_days'))
        fresh = item.get('fresh') is True or (age is not None and 0 <= age <= 3)
        trusted = str(item.get('confidence') or '').upper() in ('HIGH', 'VERIFIED')
        if not (fresh and trusted):
            continue
        state = str(item.get('state') or '').upper()
        if state in adverse_states or (key != 'spO2' and state in ('HIGH', 'ABOVE_USUAL')):
            vitals.append('adverse')
        elif state in usual_states:
            vitals.append('baseline')
    vitals_state = ('adverse' if 'adverse' in vitals else
                    'baseline' if 'baseline' in vitals else 'missing')

    safety = (health.get('acute_red_flag') is True or
              health.get('safety_restriction') is True or
              health.get('verified_critical_deviation') is True)
    groups = {'autonomic': autonomic, 'recovery': recovery, 'load': load}
    if vitals_state != 'missing':
        groups['vitals'] = vitals_state
    adverse = sum(v == 'adverse' for v in groups.values())
    stable = sum(v in ('baseline', 'favorable') for v in groups.values())
    observed = sum(v != 'missing' for v in groups.values())
    favorable = sum(v == 'favorable' for v in groups.values())

    if safety or adverse >= 2:
        tier = 4
    elif adverse or stable < 2 or observed < 2:
        tier = 3
    elif favorable and adverse == 0 and all(v in ('baseline', 'favorable') for v in groups.values()):
        tier = 1
    else:
        tier = 2
    tiers = {
        1: ('PRIMED', 'Primed to do quality/racing', 'purple', '⚡',
            'Recovery systems are on baseline, with a confirmed positive shift.'),
        2: ('READY', 'Ready for quality', 'green', '✓',
            'Your main recovery markers are holding their personal range.'),
        3: ('MIXED', 'Mixed signals, quality may be allowed', 'amber', '↔',
            'A few markers need attention before considering intensity.'),
        4: ('NOT_SUGGESTED', 'Quality is not suggested', 'red', '!',
            'Independent warning signs call for a recovery-first approach.'),
    }
    code, title, tone, icon, message = tiers[tier]
    if tier == 4 and safety:
        message = 'An observed safety alert takes priority over otherwise normal markers.'
    if tier == 3 and not adverse and observed < 2:
        message = 'Only a limited set of fresh recovery signals is available.'
    elif tier == 3 and not adverse:
        message = 'Some recovery signals are still worth monitoring.'

    # Separate current authority from physiology; do not infer authorization
    # from the color of this card or an unscheduled roadmap direction.
    today = str(clock.get('iso') or '')[:10]
    offer_date = str(offer.get('date') or offer.get('planned_date') or '')[:10]
    is_today = bool(offer) and (not today or not offer_date or today == offer_date)
    canonical_today = (str(decision.get('action') or '').upper() == 'TRAIN'
                       and str(pres.get('status') or '').upper() == 'PRESCRIBED'
                       and decision.get('hard_session_allowed') is not False
                       and decision.get('quality_window_eligible') is not False
                       and is_today and offer.get('provisional_quality') is not True)
    name = str(direction.get('athlete_title') or direction.get('title') or '').strip()
    # Direction title is source data displayed escaped in the Jinja template.
    next_quality = ('Next quality: ' + name) if name else 'Next quality: Awaiting roadmap direction'
    permission = ('A quality session is issued for today — follow the official workout.'
                  if canonical_today else 'No quality session prescribed for today.')
    if day_complete:
        context = 'Training day complete · Follow your planned recovery.'
    elif double_habit and count == 1:
        context = 'First of your usual two sessions · The day is still in progress.'
    elif count == 0:
        context = 'No sessions logged today.'
    else:
        context = f'{count} session' + ('' if count == 1 else 's') + ' logged today.'
    # These are context-only variables, never votes in the model.
    def chip(label, icon, state, value):

                   'unknown':'Unclassified', 'missing':'Unavailable'}[state]
        return {'label':label,'icon':icon,'value':value or display,'tone':color,
                'state':state}
    checks = [chip('Autonomic', '♥', autonomic, ''),
              chip('Sleep & recovery', '☾', recovery, ''),
              chip('Training load', '↗', load, ''),
              chip('HRV', '∿', hrv, ''),
              chip('Resting HR', '♥', rhr, ''),
              chip('Sleep', '☾', sleep, '')]
    if vitals_state != 'missing':
        checks.append(chip('Vital signs', '✧', vitals_state, ''))
    if safety:
        checks.insert(0,chip('Health alert','!', 'adverse', 'Observed safety flag'))
    return {'schema':R196979_QUALITY_ADVISOR_SCHEMA,
            'code':code,'tier':tier,'tone':tone,'icon':icon,
            'title':title,'message':message,'context':context,
            'next_quality':next_quality,'permission':permission,
            'checks':checks,'systems':groups,'evidence_coverage':observed,
            'canonical_quality_today':canonical_today,
            'sessions_today':count,'double_habit':double_habit,
            'day_complete':day_complete,'heat_today':heat_today,
            'session_count_is_never_a_veto':True,'read_only':True,
            'changes_decision':False,'changes_prescription':False}

    except (TypeError, ValueError):
        return None
    observed = {(str(o.get('activity_id') or ''), str(o.get('date') or '')[:10]): o
        for o in ((d.get('performance_evidence') or {}).get('observed_efforts') or [])
        if isinstance(o, dict) and o.get('source') == 'INTERVALS_ACTIVITY_POWER_CURVE'
        and o.get('rolling_window') is True and o.get('secs') == seconds
        and isinstance(o.get('watts'), (int, float)) and 20 <= o['watts'] <= 2500
        and math.isfinite(float(o['watts']))}
    candidates = []
    for b in d.get('previous_training_blocks') or []:
        if not isinstance(b, dict) or b.get('quality_relevant') is not True:
            continue
        if str(b.get('quality_basis') or '') != 'EXPLICIT_MAXIMAL_TEST':
            continue
        aid, day = str(b.get('activity_id') or ''), str(b.get('date') or '')[:10]
        try:
            if not cutoff <= date.fromisoformat(day) <= date.fromisoformat(today):
                continue
        except ValueError:
            continue
        name = str(b.get('name') or '').lower()
        # Athlete explicitly designates both test/maximal purpose and duration.
        if not (('test' in name or 'checkpoint' in name) and
                ('maximal' in name or 'all-out' in name or 'all out' in name) and
                re.search(r'(?:\b5[ -]?(?:min(?:ute)?s?|m)\b|\bfive[ -]min)', name)):
            continue
        point = observed.get((aid, day))
        if not point:
            continue
        candidates.append((day, aid, b, point))
    if not candidates:
        return None
    day, aid, block, point = max(candidates, key=lambda row: (row[0], row[1]))
    return {'schema': R196983_ATTEMPT_SCHEMA, 'state': 'MAXIMAL_ATTEMPT_RECORDED',
            'goal_key': goal, 'activity_id': aid, 'date': day,
            'activity_name': str(block.get('name') or ''),
            'seconds': seconds, 'watts': round(float(point['watts'])),

            'effort_recorded': True, 'pb_confirmed': False,

            'not_an_issued_workout': True,

def _nova_r196983_hard_today(data):
    """Hard-work proof, not number of sessions, determines same-day recovery."""
    d = data if isinstance(data, dict) else {}
    today = str((d.get('coach_clock') or {}).get('iso') or '')[:10]
    if not re.fullmatch(r'\d{4}-\d{2}-\d{2}', today):
        return None
    hard_rows = [r for r in ((d.get('week_memory') or {}).get('activities') or [])
                 if isinstance(r, dict) and str(r.get('date') or '')[:10] == today
                 and r.get('hard') is True]
    blocks = [b for b in d.get('previous_training_blocks') or []
              if isinstance(b, dict) and str(b.get('date') or '')[:10] == today
              and b.get('quality_relevant') is True]
    for r in hard_rows:
        # Match on date + activity name, since weekly memory has no activity_id.
        name = str(r.get('name') or '').strip().lower()
        for b in blocks:
            if name and name == str(b.get('name') or '').strip().lower():
                return {'state':'HARD_QUALITY_COMPLETED_TODAY',
                        'activity_id':b.get('activity_id'), 'date':today,
                        'reason':b.get('quality_basis') or 'CANONICAL_HARD_ACTIVITY'}
    return None

def _nova_r196983_review_current(data, attempt):
    """Owner-stored pending attempt remains visible until true reconciliation.

        return None

def _v4896_verified_unplanned_checkpoint(goal_key, direction, plan, test_cfg,

    """An athlete-run maximal 5-minute test is a real checkpoint without a PB.

    Return ONE ordinary FORMAL_VALIDATION event, or None. Only the existing

    def _day(value):
        if not isinstance(value,str):return None
        try:return date.fromisoformat(value[:10])
        except ValueError:return None

    def _timestamp(value):
        if not isinstance(value,str):return None
        try:return datetime.fromisoformat(value.replace('Z','+00:00'))
        except ValueError:return None

    def _q(v):
        if type(v) is int and v in (1,2,3,4):return v
        if isinstance(v,str) and v.upper() in ('Q1','Q2','Q3','Q4'):return int(v[1])
        return None

    def _owned(observation,owner):
        # Reject explicitly cross-owner records, accept dedicated single-owner fixtures.
        return observation.get('athlete_id')==owner

    def _clean_q(rows,owner,asof):
        """Keep one unique Q per day; contradictory duplicates invalidate that day.
        Provider mappings must be verified for a valid comparison.
        """
        byday={}; bad_mapping=False; conflicting=set(); invalid=0; future=0
        for x in rows or []:
            if not isinstance(x,dict) or not _owned(x,owner):continue
            dt=_day(x.get('date') or x.get('day'))
            if dt is None:invalid+=1;continue
            if dt>asof:future+=1;continue
            q=_q(x.get('q')); source=x.get('provider','Intervals.icu')
            if q is None:invalid+=1;continue
            if x.get('mapping_verified') is not True:bad_mapping=True;continue
            v=(q,source,x.get('mapping_version',''))
            if dt in byday and byday[dt]!=v:conflicting.add(dt)
            else:byday[dt]=v
        for day in conflicting:byday.pop(day,None)
        return byday,{'invalid_q_count':invalid,'future_q_excluded':future,
                      'conflicting_dates':sorted(str(d) for d in conflicting),
                      'unverified_mapping':bad_mapping}

    def _qperiod(qmap,start,end):
        found={d:qmap[d][0] for d in qmap if start<=d<=end}
        return found

    def _q_distribution(qs):
        c=Counter(qs.values()); n=len(qs)
        return {'n':n,'Q1':c[1],'Q2':c[2],'Q3':c[3],'Q4':c[4],
                'poor_tail_fraction':round((c[3]+c[4])/n,3) if n else None}

    def _qtrend(qmap,asof,plan_start,issues):
        cfg=CONFIG
        recent_start=asof-timedelta(days=cfg['recent_days']-1)
        recent=_qperiod(qmap,recent_start,asof)
        candidate_pre_end=plan_start-timedelta(days=1) if plan_start else None
        pre=None
        # Anchor historical pre-plan window when present; must have coverage.
        if candidate_pre_end and candidate_pre_end < recent_start:
            pre=_qperiod(qmap,candidate_pre_end-timedelta(days=cfg['reference_days']-1),candidate_pre_end)
        if pre and len(pre)>=cfg['q_reference_min']:
            reference=pre;reference_kind='PRE_NOVA_ANCHOR';reference_end=candidate_pre_end
        else:
            # Search up to a year back for the EARLIEST sufficiently covered anchor,
            # rather than constantly moving a degraded baseline forward.
            reference_end=recent_start-timedelta(days=1)
            reference=_qperiod(qmap,reference_end-timedelta(days=cfg['reference_days']-1),reference_end)
            reference_kind='HISTORICAL_ANCHOR' if len(reference)>=cfg['q_reference_min'] else 'INSUFFICIENT'
            start_search=max(min(qmap, default=recent_start),asof-timedelta(days=364))
            scan_end=recent_start-timedelta(days=cfg['reference_days'])
            cursor=start_search
            while cursor<=scan_end:
                candidate=_qperiod(qmap,cursor,cursor+timedelta(days=27))
                if len(candidate)>=cfg['q_reference_min']:
                    reference=candidate;reference_end=cursor+timedelta(days=27)
                    reference_kind='EARLIEST_QUALIFIED_ANCHOR'
                    break
                cursor+=timedelta(days=1)
        rq=_q_distribution(reference);cq=_q_distribution(recent)
        wk1={d:q for d,q in recent.items() if d<recent_start+timedelta(days=7)}
        wk2={d:q for d,q in recent.items() if d>=recent_start+timedelta(days=7)}
        compatible=len({qmap[d][1:] for d in set(reference)|set(recent)})<=1
        enough=len(reference)>=cfg['q_reference_min'] and len(recent)>=cfg['q_recent_min'] and compatible and not issues['unverified_mapping']
        persistence=(len(wk1)>=cfg['week_min'] and len(wk2)>=cfg['week_min'] and
                     sum(q>=3 for q in wk1.values())>=cfg['q_persistent_week_min'] and
                     sum(q>=3 for q in wk2.values())>=cfg['q_persistent_week_min'])
        poor_shift=(cq['poor_tail_fraction']-rq['poor_tail_fraction']) if enough else None
        # Persistent small erosion: detect loss of excellent nights + repeated Q3,
        # even when the full poor-tail threshold has not been reached.
        softer_persistence=(len(wk1)>=cfg['week_min'] and len(wk2)>=cfg['week_min'] and
                           sum(q>=3 for q in wk1.values())>=2 and
                           sum(q>=3 for q in wk2.values())>=2)
        subtle=bool(enough and softer_persistence and poor_shift>=.12 and
                    (rq['Q1']/rq['n']-cq['Q1']/cq['n'])>=.20)
        decline=bool(enough and ((persistence and poor_shift>=cfg['q_poor_shift'] and
                     cq['Q1']/cq['n']<rq['Q1']/rq['n']) or subtle))
        # Persistent poor quality should not be redefined as a healthy baseline.
        burden=bool(len(recent)>=cfg['q_recent_min'] and persistence and
                    cq['poor_tail_fraction']>=cfg['q_burden_poor_fraction'] and compatible and not issues['unverified_mapping'])
        return {'comparable':enough,'reference_kind':reference_kind,
                'reference_window':{'start':str(reference_end-timedelta(days=27)),'end':str(reference_end)},
                'recent_window':{'start':str(recent_start),'end':str(asof)},
                'reference':rq,'recent':cq,'recent_week_samples':[len(wk1),len(wk2)],
                'persistent_poor_both_weeks':persistence,'poor_tail_change':round(poor_shift,3) if poor_shift is not None else None,
                'declining':decline,'subtle_erosion_detected':subtle,'persistent_burden':burden,
                'reference_pre_nova':reference_kind=='PRE_NOVA_ANCHOR',
                'source_compatible':compatible,'mapping_verified':not issues['unverified_mapping']}

    def _longitudinal_profile(qmap,asof,plan_start):
        """Full history is descriptive; short-horizon decisions remain auditable.
        Do not infer a pre-plan Q history from an aggregate training history.
        """
        blocks=[]
        for i in range(13):
            end=asof-timedelta(days=i*28)
            start=end-timedelta(days=27)
            qs=_qperiod(qmap,start,end)
            if not qs and (not qmap or end<min(qmap)):break
            blocks.append({'start':str(start),'end':str(end),'coverage':len(qs),
                           'distribution':_q_distribution(qs),'reliable':len(qs)>=CONFIG['q_reference_min']})
        pre={d:qmap[d][0] for d in qmap if plan_start and d<plan_start}
        post={d:qmap[d][0] for d in qmap if plan_start and d>=plan_start}
        dates=sorted(qmap)
        transitions=Counter()
        for left,right in zip(dates,dates[1:]):
            if (right-left).days!=1:continue
            from_band='Q1_Q2' if qmap[left][0]<=2 else 'Q3_Q4'
            to_band='Q1_Q2' if qmap[right][0]<=2 else 'Q3_Q4'
            transitions[f'{from_band}__TO__{to_band}']+=1
        return {'observed_days_total':len(qmap),'observed_first_date':str(min(qmap)) if qmap else None,
                'observed_last_date':str(max(qmap)) if qmap else None,
                'pre_nova':_q_distribution(pre),'post_nova':_q_distribution(post),
                'pre_post_is_causal_comparison':False,
                'last_364_days_28day_blocks_newest_first':blocks,
                'consecutive_day_transition_counts':dict(transitions),
                'missing_nights_are_not_imputed':True}


    def _clean_activities(rows,owner,asof):
        seen={}
        for a in rows or []:
            if not isinstance(a,dict) or not _owned(a,owner):continue
            d=_day(a.get('date') or a.get('start_date_local'))
            key=a.get('id') or a.get('activity_id')
            if not d or d>asof or not key or a.get('verified_completed') is not True:continue
            # Never duplicate load if identical activity ID appears more than once.
            if key not in seen:seen[key]=a
        return seen

    def _completed_session_scores(row):
        """One validated, paired RPE/Execution/After record for one completed activity.

        Daily Log fields are optional for storage, but are NOT eligible for
        NOVA Health subjective comparisons until all three are present 1–10.
        Boolean, floating, missing and out-of-range scores cannot pass.
        """
        if not isinstance(row,dict):return None
        scores=(row.get('rpe'),row.get('execution',row.get('execution_score')),
                row.get('after',row.get('post_session_state')))
        return scores if all(type(v) is int and 1<=v<=10 for v in scores) else None

    def _feedback_trend(activities,rows,owner,asof):
        # Compare post-session recovery on actual, completed, similarly demanding sessions.
        # Use fixed coarse workload bins to avoid comparing a maximal test with easy Z2.
        bins={}; feedback_by_id={}; conflicts=set()
        # Conflicting repeated uploads cannot be chosen by list ordering.
        for f in rows or []:
            if not isinstance(f,dict) or not _owned(f,owner):continue
            id_=f.get('activity_id')
            if not id_ or id_ not in activities:continue
            logged_at=f.get('logged_at')
            if logged_at and _day(logged_at) and _day(logged_at)>asof:continue
            scores=_completed_session_scores(f)
            if scores is None:continue
            # A changed complete tuple for the same session is ambiguous, even
            # when its After value happens to match. Never pick by fetch order.
            if id_ in feedback_by_id and feedback_by_id[id_]!=scores:conflicts.add(id_)
            feedback_by_id[id_]=scores
        for id_,scores in sorted(feedback_by_id.items()):
            if id_ in conflicts:continue
            score=scores[2]
            a=activities[id_]
            dt=_day(a.get('date') or a.get('start_date_local'))
            if not dt or dt>asof:continue
            sport=(a.get('sport') or a.get('type') or 'UNKNOWN').upper()
            intensity=str(a.get('intensity_class') or a.get('family') or 'UNKNOWN').upper()
            if intensity=='UNKNOWN':continue
            duration=a.get('duration_min')
            if duration is None:
                seconds=a.get('moving_time')
                duration=float(seconds)/60 if type(seconds) in (int,float) and seconds>0 else None
            load=a.get('load',a.get('icu_training_load'))
            if duration is None or load is None:continue
            try:duration=float(duration);load=float(load)
            except (TypeError,ValueError):continue
            if duration<=0 or load<0:continue
            # Within a class, compare relative to its own observed baseline bins.
            key=(sport,intensity)
            bins.setdefault(key,[]).append((dt,score,duration,load,id_))
        candidates=[]
        for key,values in bins.items():
            recent_start=asof-timedelta(days=13)
            ref_start=recent_start-timedelta(days=42)
            recent=[v for v in values if recent_start<=v[0]<=asof]
            reference=[v for v in values if ref_start<=v[0]<recent_start]
            if len(reference)<CONFIG['after_min_per_period'] or len(recent)<CONFIG['after_min_per_period']:continue
            dur_ref=median(v[2] for v in reference);load_ref=median(v[3] for v in reference)
            def within(v):
                d=v[2]/dur_ref; l=v[3]/load_ref if load_ref>0 else 1 if v[3]==0 else 999
                return CONFIG['matched_duration_ratio_low']<=d<=CONFIG['matched_duration_ratio_high'] and CONFIG['matched_load_ratio_low']<=l<=CONFIG['matched_load_ratio_high']
            reference=[v for v in reference if within(v)]
            recent=[v for v in recent if within(v)]
            if len(reference)<CONFIG['after_min_per_period'] or len(recent)<CONFIG['after_min_per_period']:continue
            delta=median(v[1] for v in recent)-median(v[1] for v in reference)
            candidates.append({'family':list(key),'reference_n':len(reference),'recent_n':len(recent),
                               'after_median_change':round(delta,2),'declining':delta<=-CONFIG['after_decline']})
        if not candidates:return {'comparable':False,'declining':False,'matched_groups':0,'groups':[]}
        # Any valid declining cohort is observed; stable cohorts do not erase it.
        return {'comparable':True,'declining':any(x['declining'] for x in candidates),
                'matched_groups':len(candidates),'groups':candidates}

    def _feeling_trend(rows,owner,asof):
        # Daily Log is not post-session After; compare only same-session-count phases.
        recent_start=asof-timedelta(days=13); ref_start=recent_start-timedelta(days=28)
        perday={};conflicts=set()
        for p in rows or []:
            if not isinstance(p,dict) or not _owned(p,owner):continue
            day=_day(p.get('date'));value=p.get('value',p.get('feeling'))
            if not day or day>asof or type(value) not in (int,float) or not 1<=value<=10:continue
            phase=(p.get('session_count') or (2 if '2 SESSIONS' in p.get('phase','') else 1 if '1 SESSION' in p.get('phase','') else None))
            if phase not in (1,2):continue
            # Conflicting duplicates without a canonical log version are excluded.
            key=(day,phase)
            if key in perday and perday[key]!=value:conflicts.add(key)
            perday[key]=value
        for key in conflicts:perday.pop(key,None)
        results=[]
        for phase in (1,2):
            ref=[v for (day,p),v in perday.items() if p==phase and ref_start<=day<recent_start]
            rec=[v for (day,p),v in perday.items() if p==phase and recent_start<=day<=asof]
            if len(ref)<CONFIG['feeling_min_per_period'] or len(rec)<CONFIG['feeling_min_per_period']:continue
            diff=median(rec)-median(ref)
            results.append({'phase_sessions':phase,'reference_n':len(ref),'recent_n':len(rec),
                            'median_change':round(diff,2),'declining':diff<=-CONFIG['feeling_decline']})
        return {'comparable':bool(results),'declining':any(x['declining'] for x in results), 'groups':results}

    def _safe_context(snapshot):
        return deepcopy(snapshot.get('training_context') or {})



    def _q1_compass(qtrend):
        """Aspirational indicator, never an independent alert or automatic workout veto.
        Q is ordinal: percentages are descriptive, not a fabricated numeric sleep score.
        """
        a=qtrend['reference']; b=qtrend['recent']
        if not qtrend['comparable'] or not a['n'] or not b['n']:
            return {'available':False,'principle':'Q1 IS AN ASPIRATION, NOT A MANDATE'}
        before=a['Q1']/a['n']; now=b['Q1']/b['n']
        return {'available':True,'reference_q1_fraction':round(before,3),
                'recent_q1_fraction':round(now,3),
                'lost_excellent_nights':before-now>=.20,
                'q2_is_not_failure':True,
                'principle':'Q1 is the preferred quality direction; stable Q2 is not automatically inadequate'}


    def _verified_sleep_opportunity(data,owner,asof):
        """Sleep duration is contextual, not a second independent recovery vote.
        Sleep need is explicitly athlete-declared, never inferred as an 8h target.
        """
        prefs=data.get('sleep_opportunity_context') or {}
        need=prefs.get('athlete_declared_sleep_need_min')
        if type(need) not in (int,float) or not 300<=need<=720:need=None
        flexible=prefs.get('feasible_extra_minutes_per_night')
        if type(flexible) not in (int,float) or not 0<=flexible<=180:flexible=None
        constraint=prefs.get('main_constraint') if prefs.get('main_constraint') in ('WORK','FAMILY','SHIFT','OTHER','NONE') else 'UNSPECIFIED'
        values={};conflict=set()
        for x in data.get('sleep_duration',[]) or []:
            if not isinstance(x,dict) or not _owned(x,owner) or x.get('verified') is not True:continue
            d=_day(x.get('date'));mins=x.get('minutes')
            if d is None or not asof-timedelta(days=27)<=d<=asof or type(mins) not in (int,float) or not 120<=mins<=900:continue
            if d in values and values[d]!=mins:conflict.add(d)
            else:values[d]=mins
        for d in conflict:values.pop(d,None)
        recent=[v for d,v in values.items() if d>=asof-timedelta(days=13)]
        typical=median(recent) if len(recent)>=10 else None
        gap=max(0,need-typical) if need is not None and typical is not None else None
        limited=gap is not None and gap>=45
        constrained=constraint in ('WORK','FAMILY','SHIFT') and flexible is not None and flexible<30
        return {'verified_recent_n':len(recent),'personal_need_declared_min':need,
                'observed_recent_median_min':round(typical) if typical is not None else None,
                'personal_opportunity_gap_min':round(gap) if gap is not None else None,
                'limited_opportunity':limited,'constraint':constraint,'feasible_extra_minutes_per_night':flexible,
                'low_flexibility':constrained,'is_independent_recovery_vote':False,
                'recommendation_minutes_validated':False}


    def _load_trajectory(snapshot):
        """Load direction as context, not a penalty for high training volume."""
        ctx=snapshot.get('training_context') or {}
        recent=ctx.get('load_7d'); prior=ctx.get('load_prev_week')
        if type(recent) not in (int,float) or type(prior) not in (int,float) or recent<0 or prior<=0:
            return {'available':False,'does_not_set_health_card_alone':True}
        ratio=recent/prior
        return {'available':True,'completed_load_7d':recent,'previous_week_load':prior,
                'relative_change_pct':round(100*(ratio-1),1),
                'direction':'INCREASED' if ratio>1.15 else 'REDUCED' if ratio<.85 else 'SIMILAR',
                'does_not_set_health_card_alone':True,'causality_not_established':True}


    def _exposure_lag(qmap,activities,asof):
        """Descriptive WHOLE-DAY completed exposure vs the FOLLOWING night's Q.

        Group every verified activity by local day, not by Snapshot or feedback
        row. HIGH/legacy hard is intense; an easy-only control day must have
        exclusively EASY/legacy Z1-Z2 sessions. MODERATE or UNKNOWN days are
        deliberately not mislabeled as easy. No direct card/veto or causal claim.
        """
        high_classes={'HIGH','HARD','QUALITY','VO2MAX','THRESHOLD','TEMPO'}
        easy_classes={'EASY','Z1','Z2','RECOVERY'}
        by_day={}
        for a in activities.values():
            d=_day(a.get('date') or a.get('start_date_local'))
            if not d or d>=asof:continue # No following-night observation yet.
            intensity=str(a.get('intensity_class') or a.get('family') or 'UNKNOWN').upper()
            item=by_day.setdefault(d,{'sessions':0,'hard':False,'easy':0,'uncertain':False,
                                       'duration_min':0.0,'load':0.0,'load_n':0})
            item['sessions']+=1
            duration=a.get('duration_min')
            load=a.get('load',a.get('icu_training_load'))
            try:
                if duration is not None and float(duration)>0:item['duration_min']+=float(duration)
            except (TypeError,ValueError,OverflowError):pass
            try:
                if load is not None and 0<=float(load)<float('inf'):
                    item['load']+=float(load);item['load_n']+=1
            except (TypeError,ValueError,OverflowError):pass
            if intensity in high_classes:item['hard']=True
            elif intensity in easy_classes:item['easy']+=1
            else:item['uncertain']=True # MODERATE and UNKNOWN aren't easy controls.
        hard=[];easy=[];ambiguous=0
        for d,day in by_day.items():
            following=d+timedelta(days=1)
            if following not in qmap:continue
            if day['hard']:hard.append(qmap[following][0]>=3)
            elif not day['uncertain'] and day['easy']==day['sessions']:
                easy.append(qmap[following][0]>=3)
            else:ambiguous+=1
        available=(len(hard)>=5 and len(easy)>=5 and
                   len({v[1:] for v in qmap.values()})<=1)
        return {'available':available,'intense_day_next_night_n':len(hard),
                'easy_only_day_next_night_n':len(easy),
                'unclassified_next_night_n':ambiguous,
                'whole_days_with_completed_training_n':len(by_day),
                'double_session_days_n':sum(x['sessions']>=2 for x in by_day.values()),
                'intense_day_q3_q4_fraction':round(sum(hard)/len(hard),3) if available else None,
                'easy_day_q3_q4_fraction':round(sum(easy)/len(easy),3) if available else None,
                'is_causal':False,'sets_card_alone':False,'double_sessions_deduplicated':True,
                'whole_day_load_is_context_only':True}


    def _confirmed_followup(data,owner,asof,qtrend,after,feeling,activities,qmap,session_feedback):
        """Positive follow-up is an association, not an inferred rest day.
        No activity is NOT evidence a rest day was taken.
        """
        actions=[]; contradictions=[]
        active_dates={_day(a.get('date') or a.get('start_date_local')) for a in activities.values()}
        for x in data.get('recovery_actions',[]) or []:
            if not isinstance(x,dict) or not _owned(x,owner) or x.get('athlete_confirmed') is not True:continue
            d=_day(x.get('date'))
            if d is None or not asof-timedelta(days=13)<=d<=asof:continue
            if x.get('kind') not in ('REST_DAY_TAKEN','SLEEP_OPPORTUNITY_ADDED','VOLUME_REDUCED'):continue
            if d>asof-timedelta(days=4):
                # Actions reported too recently have no 4-day observation period.
                continue
            if x['kind']=='REST_DAY_TAKEN' and d in active_dates:
                contradictions.append({'date':str(d),'kind':x['kind'],'reason':'VERIFIED_COMPLETED_ACTIVITY_ON_CLAIMED_REST_DAY'})
                continue
            actions.append({'date':str(d),'kind':x['kind'],'athlete_confirmed':True})
        q_up=qtrend['comparable'] and qtrend['poor_tail_change'] is not None and qtrend['poor_tail_change']<=-.25
        feedback_up=any(g.get('after_median_change',0)>=2 for g in after.get('groups',[])) or any(g.get('median_change',0)>=1.5 for g in feeling.get('groups',[]))
        valid_actions=[]
        for a in actions:
            dt=_day(a['date'])
            subsequent_q=sum(dt<day<=asof for day in qmap)
            subsequent_feedback=sum(1 for x in session_feedback or [] if isinstance(x,dict) and _owned(x,owner) and
               x.get('activity_id') in activities and _day(activities[x['activity_id']].get('date') or activities[x['activity_id']].get('start_date_local')) and
               dt<_day(activities[x['activity_id']].get('date') or activities[x['activity_id']].get('start_date_local'))<=asof and
               _completed_session_scores(x) is not None)
            if subsequent_q>=3 and subsequent_feedback>=2:valid_actions.append(a)
        rebound=bool(valid_actions and q_up and feedback_up)
        return {'confirmed_actions':actions,'contradictions':contradictions,'improving_q':bool(q_up),'improving_feedback':bool(feedback_up),
                'observed_rebound':rebound,'sufficient_followup_actions':valid_actions,'causal_effect_proven':False,'no_activity_is_not_rest':True}




    def _effort_cost_trend(activities,rows,owner,asof):
        """RPE and Execution agree on deterioration for similar completed sessions.
        RPE alone, including a high score in a purposeful quality, is never penalized.
        """
        feedback={};conflicts=set()
        for x in rows or []:
            if not isinstance(x,dict) or not _owned(x,owner):continue
            id_=x.get('activity_id')
            if id_ not in activities:continue
            scores=_completed_session_scores(x)
            if scores is None:continue
            if id_ in feedback and feedback[id_]!=scores:conflicts.add(id_)
            feedback[id_]=scores
        rows_by_group={}
        for id_,(rpe,exe,after) in feedback.items():
            if id_ in conflicts:continue
            a=activities[id_];d=_day(a.get('date') or a.get('start_date_local'))
            dur=a.get('duration_min');load=a.get('load',a.get('icu_training_load'))
            intensity=str(a.get('intensity_class') or a.get('family') or 'UNKNOWN').upper()
            if not d or dur is None or load is None or intensity=='UNKNOWN':continue
            try:dur=float(dur);load=float(load)
            except (TypeError,ValueError):continue
            if dur<=0 or load<0:continue
            group=(str(a.get('sport') or a.get('type') or 'UNKNOWN').upper(),intensity)
            rows_by_group.setdefault(group,[]).append((d,dur,load,rpe,exe))
        recent_start=asof-timedelta(days=13);ref_start=recent_start-timedelta(days=42)
        groups=[]
        for group,allv in rows_by_group.items():
            ref=[x for x in allv if ref_start<=x[0]<recent_start]
            rec=[x for x in allv if recent_start<=x[0]<=asof]
            if len(ref)<3 or len(rec)<3:continue
            ref_dur=median(x[1] for x in ref);ref_load=median(x[2] for x in ref)
            def similar(x):
                ld=x[2]/ref_load if ref_load>0 else 1 if x[2]==0 else 999
                return .65<=x[1]/ref_dur<=1.5 and .65<=ld<=1.5
            ref=[x for x in ref if similar(x)];rec=[x for x in rec if similar(x)]
            if len(ref)<3 or len(rec)<3:continue
            rpe_delta=median(x[3] for x in rec)-median(x[3] for x in ref)
            exe_delta=median(x[4] for x in rec)-median(x[4] for x in ref)
            groups.append({'family':list(group),'reference_n':len(ref),'recent_n':len(rec),
                           'rpe_median_change':rpe_delta,'execution_median_change':exe_delta,
                           'cost_increasing':rpe_delta>=2 and exe_delta<=-2})
        return {'comparable':bool(groups),'cost_increasing':any(g['cost_increasing'] for g in groups),
                'groups':groups,'high_rpe_alone_is_not_bad':True}

    def _feedback_presence(activities, rows, owner, asof, observed_qualities):
        """Presence/freshness ledger, not a verdict or proxy for adherence.

        Only complete, activity-linked trios are observations. Quality sessions
        are observed from verified completed activity IDs; do not infer a future
        quality, or label every hard session as prescribed quality.
        """
        by_id={}; conflicting=set()
        for x in rows or []:
            if not isinstance(x,dict) or not _owned(x,owner):continue
            ident=x.get('activity_id')
            if ident not in activities:continue
            scores=_completed_session_scores(x)
            if scores is None:continue
            if ident in by_id and by_id[ident]!=scores:conflicting.add(ident)
            by_id[ident]=scores
        for ident in conflicting:by_id.pop(ident,None)
        dated=[]
        for ident,scores in by_id.items():
            d=_day(activities[ident].get('date') or activities[ident].get('start_date_local'))
            if d is not None and d<=asof:dated.append((d,str(ident),scores))
        dated.sort()
        recent_start=asof-timedelta(days=13)
        latest=dated[-1] if dated else None
        valid_quality={}
        for q in observed_qualities or []:
            if not isinstance(q,dict) or q.get('verified_completed') is not True or q.get('quality_relevant') is not True:continue
            ident=q.get('activity_id')
            a=activities.get(ident)
            if a is None:continue
            d=_day(a.get('date') or a.get('start_date_local'))
            if d and d<=asof and str(q.get('date'))==str(d):valid_quality[ident]=d
        recent_quality=sorted((d,str(ident)) for ident,d in valid_quality.items() if d>=recent_start)
        pending=[(d,ident) for d,ident in recent_quality if ident not in by_id]
        latest_quality=recent_quality[-1] if recent_quality else None
        # Do not assert feedback is absent for qualities outside observed marker coverage.
        status=('NO_COMPLETED_QUALITY_IDENTIFIED_IN_SOURCE' if not recent_quality else
                'LATEST_COMPLETED_QUALITY_FEEDBACK_RECORDED' if latest_quality[1] in by_id else
                'LATEST_COMPLETED_QUALITY_FEEDBACK_MISSING')
        return {'complete_activity_feedback_n':len(dated),
                'complete_feedback_recent_14d_n':sum(d>=recent_start for d,_,_ in dated),
                'last_complete_feedback_activity_date':str(latest[0]) if latest else None,
                'last_complete_feedback_age_days':(asof-latest[0]).days if latest else None,
                'latest_complete_feedback_scores':dict(zip(('rpe','execution','after'),latest[2])) if latest else None,
                'observed_completed_quality_recent_14d_n':len(recent_quality),
                'observed_quality_feedback_complete_recent_14d_n':sum(ident in by_id for _,ident in recent_quality),
                'observed_quality_feedback_missing_recent_14d_n':len(pending),
                'last_observed_completed_quality_date':str(latest_quality[0]) if latest_quality else None,
                'status':status, 'quality_source':'COMPLETED_QUALITY_RELEVANT_ACTIVITY_IDS_ONLY',
                'not_a_recovery_penalty':True,'no_scheduled_quality_assumed':True}

    def _evidence_labels(card_id,qtrend,after,feeling,sleep,load,compass,feedback_presence):
        signals=[]
        if qtrend['comparable']:
            signals.append('SLEEP Q · VERIFIED HISTORY')
        if after['comparable']:signals.append('AFTER · MATCHED SESSIONS')
        elif feedback_presence['complete_feedback_recent_14d_n']:
            signals.append('SESSION FEEDBACK · RECORDED')
        if feeling['comparable']:signals.append('FEELING · SAME-DAY PHASE')
        if sleep['limited_opportunity']:signals.append('SLEEP OPPORTUNITY · PERSONAL GAP')
        if load['available']:signals.append('COMPLETED LOAD · CONTEXT')
        if compass.get('lost_excellent_nights'):signals.append('Q1 FREQUENCY · DECLINING')
        return signals

    def interpret(snapshot:dict[str,Any])->dict[str,Any]:
        """Deterministic pure interpretation. No score or hidden training side effect."""
        data=deepcopy(snapshot)
        owner=data.get('athlete_id')
        asof=_day(data.get('as_of'));plan=_day(data.get('plan_started_on'))
        if not isinstance(owner,str) or not owner or not asof:raise ValueError('athlete_id and ISO as_of required')
        qmap,issues=_clean_q(data.get('sleep_q'),owner,asof)
        qtrend=_qtrend(qmap,asof,plan,issues)
        history=_longitudinal_profile(qmap,asof,plan)
        activities=_clean_activities(data.get('activities'),owner,asof)
        after=_feedback_trend(activities,data.get('session_feedback'),owner,asof)
        feeling=_feeling_trend(data.get('daily_feeling'),owner,asof)
        effort_cost=_effort_cost_trend(activities,data.get('session_feedback'),owner,asof)
        feedback_presence=_feedback_presence(activities,data.get('session_feedback'),owner,asof,data.get('observed_completed_quality'))
        events=[e for e in data.get('health_events',[]) if isinstance(e,dict) and _owned(e,owner) and _day(e.get('date')) and asof-timedelta(days=13)<=_day(e['date'])<=asof]
        restriction=[e['restriction'] for e in events if e.get('restriction') in ('NO_INTENSITY','NO_TRAINING')]
        has_sickness=any(e.get('kind')=='SICKNESS' for e in events)
        symptomatic=any(e.get('medical_followup_requested') is True for e in events)
        feedback_declining=after['declining'] or feeling['declining']
        compass=_q1_compass(qtrend)
        sleep_opportunity=_verified_sleep_opportunity(data,owner,asof)
        workload=_load_trajectory(data)
        exposure_lag=_exposure_lag(qmap,activities,asof)
        followup=_confirmed_followup(data,owner,asof,qtrend,after,feeling,activities,qmap,data.get('session_feedback'))
        moderate_sleep_headroom=bool(qtrend['comparable'] and
              qtrend['recent']['poor_tail_fraction'] is not None and
              qtrend['recent']['poor_tail_fraction']>=.50)
        if symptomatic:card_id='HEALTH_FOLLOWUP'
        elif qtrend['declining'] and feedback_declining:card_id='RECOVERY_MISMATCH'
        elif qtrend['declining'] or feedback_declining or effort_cost['cost_increasing']:card_id='RECOVERY_DRIFT'
        elif qtrend['persistent_burden']:card_id='SLEEP_PATTERN_CONCERN'
        elif followup['observed_rebound']:card_id='RECOVERY_REBOUND'
        elif qtrend['comparable'] or after['comparable'] or feeling['comparable']:
            # A complete longitudinal Q comparison (or reliable comparable
            # feedback) is enough for a *qualified* judgment. No mandatory
            # match-feedback gate. Strong load-absorption language requires
            # independent, comparable post-session evidence + Q stability.
            sufficiently_supported=bool(qtrend['comparable'] and after['comparable'] and workload['available'])
            card_id=('LOAD_WELL_ABSORBED' if sufficiently_supported and not
                     (moderate_sleep_headroom or sleep_opportunity['limited_opportunity'])
                     else 'RECOVERY_STABLE_WITH_HEADROOM')
        else:card_id='EVIDENCE_BUILDING'
        # Consequential missingness is disclosed; no 'well absorbed' from HRV alone.
        observations=[]
        if qtrend['comparable']:
            observations.append({'domain':'sleep_q','reference':qtrend['reference'], 'recent':qtrend['recent'],
               'date_window':qtrend['recent_window'],'reference_kind':qtrend['reference_kind'],
               'poor_tail_change':qtrend['poor_tail_change'],'persistent':qtrend['persistent_poor_both_weeks']})
        else:observations.append({'domain':'sleep_q','unavailable':'NO_COMPARABLE_LONGITUDINAL_Q',
               'recent_samples':qtrend['recent']['n'],'reference_samples':qtrend['reference']['n'],
               'mapping_verified':qtrend['mapping_verified'],'source_compatible':qtrend['source_compatible']})
        if after['comparable']:observations.append({'domain':'post_session_after','matched_groups':after['groups']})
        if feeling['comparable']:observations.append({'domain':'daily_feeling','same_phase_groups':feeling['groups']})
        if effort_cost['comparable']:observations.append({'domain':'rpe_execution_cost','matched_groups':effort_cost['groups']})
        observations.append({'domain':'complete_session_feedback_presence','coverage':deepcopy(feedback_presence),
                             'interpretation':'SUPPORTING_EVIDENCE_NOT_A_GATE'})
        wellness=data.get('current_wellness') or {}
        if wellness:observations.append({'domain':'current_wellness','current':deepcopy(wellness),
            'longitudinal_inference':False})
        if has_sickness:observations.append({'domain':'health_events','sickness_reported':True,
          'restrictions_are_explicit_only':restriction})
        # Missing baseline is a strong limit even when training context is abundant.
        insufficient=[]
        if not qtrend['comparable']:insufficient.append('Longitudinal Q history, same verified mapping, >=20 reference and >=10 recent nights')
        # Feedback is required from athletes after quality work, but it is an
        # optional *evidence domain*: sparse quality does not mean poor health.
        # Missing feedback is not classified as insufficient general recovery.
        if not after['comparable']:insufficient.append('Comparable quality-session feedback not yet sufficient for an independent absorption trend')
        card=deepcopy(CARDS[card_id]);card['id']=card_id
        q_ref=qtrend['reference']; q_now=qtrend['recent']
        if card_id=='EVIDENCE_BUILDING':
            parts=[f"Training context is available, but this record contains {q_now['n']}/14 recent Q-rated nights and {q_ref['n']}/28 reference nights"]
            if feedback_presence['complete_activity_feedback_n']:
                parts.append(f"{feedback_presence['complete_activity_feedback_n']} complete activity-linked feedback records are available, but a comparable longitudinal trend is not established")
            else:parts.append('no complete activity-linked feedback is available yet')
            card['explanation']='; '.join(parts)+'. Current HRV and RHR cannot replace missing longitudinal evidence.'
        elif card_id=='RECOVERY_STABLE_WITH_HEADROOM':
            why=[]
            if qtrend['comparable']:
                why.append(f"verified Q history compares {q_now['Q3']+q_now['Q4']}/{q_now['n']} recent Q3–Q4 nights with {q_ref['Q3']+q_ref['Q4']}/{q_ref['n']} in the historical reference; no qualifying persistent Q decline was detected")
            else:
                why.append('comparable session-level subjective feedback is available, but verified longitudinal Q comparison is not')
            if moderate_sleep_headroom:
                why.append('the recent Q3–Q4 share still leaves potential recovery headroom')
            if sleep_opportunity['limited_opportunity']:
                why.append(f"observed sleep is {sleep_opportunity['personal_opportunity_gap_min']} minutes below your self-declared sleep need")
            if after['comparable']:
                why.append('matched complete post-session feedback supports this interpretation without showing a qualifying decline')
            elif feedback_presence['complete_activity_feedback_n']:
                why.append(f"{feedback_presence['complete_activity_feedback_n']} complete session feedback record(s) contribute context, but matched periods are not yet sufficient to confirm sustainable load absorption")
            else:
                why.append('there is no complete post-session feedback yet to independently check load absorption')
            card['description']='Available longitudinal evidence does not show a qualifying sustained recovery decline; the degree of load absorption remains qualified by the evidence.'
            card['explanation']='; '.join(why)+'. This is a provisional observation, not training clearance.'
        elif card_id=='RECOVERY_REBOUND':
            card['explanation']=(f"Q3–Q4 decreased from {q_ref['Q3']+q_ref['Q4']}/{q_ref['n']} to "
              f"{q_now['Q3']+q_now['Q4']}/{q_now['n']} nights, with improvement in comparable subjective feedback. "
              f"An athlete-confirmed recovery adjustment occurred during the recent window; timing alone cannot establish causation.")
        elif card_id=='LOAD_WELL_ABSORBED':
            card['explanation']=(f"Across the recent 14-day window, Q3–Q4 occurred "
               f"{q_now['Q3']+q_now['Q4']}/{q_now['n']} nights compared with "
               f"{q_ref['Q3']+q_ref['Q4']}/{q_ref['n']} in the personal reference. "
               f"Available comparable recovery feedback shows no sustained deterioration. "
               f"This is an observation, not permission for any specific workout.")
        elif card_id=='RECOVERY_MISMATCH':
            domains=[]
            if after['declining']:domains.append('comparable post-session After declined')
            if feeling['declining']:domains.append('same-phase Daily Feeling declined')
            card['explanation']=(f"The share of Q3–Q4 nights rose from "
               f"{q_ref['Q3']+q_ref['Q4']}/{q_ref['n']} to {q_now['Q3']+q_now['Q4']}/{q_now['n']}, "
               f"with recurrence in both recent weeks; {' and '.join(domains)}. "
               f"Normal daily HRV does not erase the observed changes; their cause remains unproven.")
        elif card_id=='RECOVERY_DRIFT':
            reasons=[]
            if qtrend['declining']:
                reasons.append(f"Q3–Q4 increased from {q_ref['Q3']+q_ref['Q4']}/{q_ref['n']} to "
                               f"{q_now['Q3']+q_now['Q4']}/{q_now['n']} nights across the compared periods")
            if after['declining']:reasons.append('comparable post-session After deteriorated')
            if feeling['declining']:reasons.append('same-phase daily Feeling deteriorated')
            if effort_cost['cost_increasing']:reasons.append('RPE increased while execution declined for matched sessions')
            card['explanation']='; '.join(reasons)+'. The remaining domains do not yet independently confirm a recovery mismatch.'
        elif card_id=='SLEEP_PATTERN_CONCERN':
            card['explanation']=(f"Q3–Q4 nights remain frequent ({q_now['Q3']+q_now['Q4']}/{q_now['n']}) "
                f"in both recent weeks. A persistently unfavorable pattern is not assumed to be "
                f"healthy merely because it resembles the reference period.")
        else:
            card['explanation']='Recent reported symptoms request follow-up; a medical condition is not inferred from training data.'
        if sleep_opportunity['low_flexibility'] and card_id in ('RECOVERY_DRIFT','RECOVERY_MISMATCH','SLEEP_PATTERN_CONCERN'):
            card['action']='Sleep opportunity is constrained by your current schedule. Consider reducing training density or reviewing recovery flexibility before setting an unrealistic sleep target.'
        elif sleep_opportunity['limited_opportunity'] and card_id in ('LOAD_WELL_ABSORBED','RECOVERY_STABLE_WITH_HEADROOM'):
            card['action']='Where feasible, look for more sleep opportunity; stable training response does not make a short sleep window ideal.'
        if compass.get('available') and compass['recent_q1_fraction']==0 and not qtrend['declining']:
            card['sleep_compass_note']='Q1 remains the aspiration; Q2 is a good night, not a failure. No Q1 quota is imposed.'
        if feedback_presence['status']=='LATEST_COMPLETED_QUALITY_FEEDBACK_MISSING' and card_id!='HEALTH_FOLLOWUP':
            card['action'] += ' Record RPE, Execution and After for the latest completed quality session to refine the next evaluation.'
        card['feedback_evidence_note']=(
            f"Complete session feedback: {feedback_presence['complete_activity_feedback_n']} recorded; "
            f"{feedback_presence['complete_feedback_recent_14d_n']} in the last 14 days. "
            + ('The latest identifiable completed quality has no complete feedback yet.' if feedback_presence['status']=='LATEST_COMPLETED_QUALITY_FEEDBACK_MISSING' else
               'The latest identifiable completed quality has complete feedback.' if feedback_presence['status']=='LATEST_COMPLETED_QUALITY_FEEDBACK_RECORDED' else
               'Recent quality-session coverage cannot be established from the available activity markers.')
        )
        card['signal_labels']=_evidence_labels(card_id,qtrend,after,feeling,sleep_opportunity,workload,compass,feedback_presence)
        return {'schema':VERSION,'analysis_as_of':str(asof),'card':card,
           'provisional':True,'clinical_validation':False,'dose_validated':False,
           'independent_evidence_policy':{'sleep_quality_provider_score_and_duration':'ONE_COMPOSITE_DOMAIN',
              'rpe_plus_execution':'ONE_PAIRED_SESSION_COST_DOMAIN',
              'after_and_daily_feeling':'CORRELATED_SUBJECTIVE_DOMAINS_NOT_ADDED_AS_INDEPENDENT_POINTS',
              'workload_and_sleep_duration':'CONTEXT_NOT_AUTOMATIC_RISK_VOTES',
              'hrv_and_rhr':'SUPPORTING_CONTEXT_NEVER_A_SINGLE_DAY_VETO',
              'numeric_weighted_health_score':False},
           'analysis':{'sleep_q':qtrend,'longitudinal_q':history,'post_session_after':after,'daily_feeling':feeling,'effort_cost':effort_cost,
                       'data_quality':issues,'insufficient':insufficient,'feedback_presence':feedback_presence,
                       'completed_activity_count_in_fixture':len(activities),
                       'training_context':_safe_context(data),
                       'q1_compass':compass,'sleep_opportunity':sleep_opportunity,
                       'completed_load_trajectory':workload,'next_night_exposure_association':exposure_lag,
                       'athlete_confirmed_followup':followup,
                       'pre_nova_training_context_available':bool(data.get('training_context',{}).get('pre_nova_history_days',0)),
                       'pre_nova_sleep_comparison_available':qtrend['reference_pre_nova'],
                       'sickness_reported':has_sickness,'explicit_restrictions':restriction},
           'evidence':observations,'recommendation':{'kind':card_id,
                 'text':card['action'],'days':None,'hours':None,'dose_evidence':'NOT_CALIBRATED',
                 'athlete_discretion':True},
           'quality_readiness_display_only':deepcopy(data.get('quality_readiness_display_only')),
           'quality_readiness_modified':False,
           'is_medical_clearance':False,'is_training_prescription':False,
           'training_writes':[],'calendar_writes':[],'ledger_writes':[]}

    def dump(snapshot):
        return json.dumps(interpret(snapshot),sort_keys=True,ensure_ascii=False,indent=2)
    return interpret

def _nova_health_input_v06(data, historical_wellness, completed_activities, owner_logs):
    """Owner-local projection; no generated Q, After or rest evidence."""
    d=data if isinstance(data,dict) else {}
    clock=d.get('coach_clock') or {}
    as_of=str(clock.get('iso') or get_rome_now().date().isoformat())[:10]
    direction=d.get('training_direction') or {}
    rhythm=d.get('training_rhythm') or {}
    state=d.get('training_state') or {}
    metrics=d.get('metrics') or {}
    owner='CURRENT_AUTHENTICATED_ATHLETE' # local marker, no actual user ID persisted
    wellness=[]
    for w in historical_wellness or []:
        if not isinstance(w,dict):continue
        day=str(w.get('id') or '')[:10]
        value=w.get('sleepQuality')
        # Intervals.icu's four-category native ordinal field. Missing/unexpected
        # values never become Q3/Q4 and unverified data never become healthy.
        if type(value) is int and value in (1,2,3,4) and len(day)==10 and day<=as_of:
            wellness.append({'athlete_id':owner,'date':day,'q':value,
                'provider':'Intervals.icu','mapping_version':'NATIVE_SLEEPQUALITY_Q1_Q4',
                'mapping_verified':True})
    latest_by_id={}
    for a in completed_activities or []:
        if not isinstance(a,dict):continue
        act_id=str(a.get('id') or '')
        day=str(a.get('start_date_local') or '')[:10]
        if not act_id or len(day)!=10 or day>as_of:continue
        # We require an actual completed source entry. No projected workouts.
        seconds=a.get('moving_time')
        try:mins=float(seconds)/60
        except (ValueError,TypeError):continue
        if mins<=0:continue
        load=a.get('icu_training_load')
        # Observed intensity bin for feedback matching only; never a quality gate.
        intensity='UNKNOWN'
        try:
            intensity_value=float(a.get('icu_intensity'))
            if 0<intensity_value<=250:
                intensity='HIGH' if intensity_value>=90 else 'MODERATE' if intensity_value>=75 else 'EASY'
        except (TypeError,ValueError):pass
        # Match exact native type and measured dose. Do not infer a prescribed quality.
        latest_by_id[act_id]={'athlete_id':owner,'id':act_id,'date':day,
            'sport':str(a.get('type') or 'UNKNOWN').upper(),
            'intensity_class':intensity,'duration_min':round(mins,2),
            'load':load,'verified_completed':True}
    # Quality designation must originate from a completed, recognized activity,
    # never from NOVA's upcoming quality, coach slots or planned calendar.
    qualities=[]
    for block in d.get('previous_training_blocks') or []:
        if not isinstance(block,dict) or block.get('quality_relevant') is not True:continue
        ident=str(block.get('activity_id') or '')
        act=latest_by_id.get(ident)
        if not act:continue
        day=str(block.get('date') or '')[:10]
        if day!=act['date']:continue
        qualities.append({'activity_id':ident,'date':day,
                         'quality_relevant':True,'verified_completed':True})
    logs=owner_logs if isinstance(owner_logs,list) else []
    feedback=[];feelings=[];events=[]
    incomplete_logs=0;completed_logs=0
    for row in logs:
        if not isinstance(row,dict):continue
        day=str(row.get('local_date') or '')[:10]
        if len(day)!=10 or day>as_of:continue
        # An RPE-only, After-only, Execution-only or Feeling-only Daily Log
        # remains in the DB, but is NOT subjective NOVA Health evidence.
        # All 3 scores must be valid and refer to an exact completed activity.
        act_id=str(row.get('execution_activity_id') or '')
        scores=(row.get('session_rpe'),row.get('session_execution_score'),row.get('post_session_state'))
        complete=all(type(x) is int and 1<=x<=10 for x in scores)
        linked=bool(act_id and act_id in latest_by_id)
        if complete and linked:
            completed_logs+=1
            feedback.append({'athlete_id':owner,'activity_id':act_id,
                'rpe':scores[0],'execution':scores[1],'after':scores[2],
                'complete_session_feedback':True})
            value=row.get('feeling')
            if type(value) is int and 1<=value<=10:
                phase=str(row.get('phase') or '')
                count=row.get('session_count')
                if type(count) is not int or count not in (1,2):
                    count=2 if '2 SESSIONS' in phase else 1 if '1 SESSION' in phase else None
                if count in (1,2):
                    feelings.append({'athlete_id':owner,'date':day,'value':value,
                        'phase':phase,'session_count':count,'complete_session_feedback':True})
        else:
            incomplete_logs+=1
        # Explicit health events are independent safety/context observations.
        # A partial feedback log MUST NOT erase a real reported illness.
        status=str(row.get('status') or '').upper()
        restrict=str(row.get('restriction_type') or '').upper()
        if status in ('SICK','SICKNESS','INJURED') or restrict in ('NO_TRAINING','NO_INTENSITY'):
            events.append({'athlete_id':owner,'date':day,
                'kind':'SICKNESS' if status in ('SICK','SICKNESS') else ('INJURY' if status=='INJURED' else 'RESTRICTION'),
                'restriction':restrict if restrict in ('NO_TRAINING','NO_INTENSITY') else None,
                'medical_followup_requested':False})

def _nova_r196999_phase_rows(workout):
    '''Preserve verified canonical duration and phases; no generated dose.

    `session_composition` is a canonical upstream object. If the block total
    differs from its declared total or its state is not COMPLETE, fail closed.
    Work/recovery repeats are shown only when the existing canonical library
    or power-prescription proves them; otherwise show aggregate minutes.
    '''
    w = workout if isinstance(workout, dict) else {}
    comp = w.get('session_composition') or {}
    if not isinstance(comp, dict) or comp.get('state') != 'COMPLETE':
        return []
    blocks = comp.get('blocks') or []
    if not isinstance(blocks, list) or not blocks:
        return []
    total = _nova_r196999_finite(comp.get('total_seconds'))
    durations = [_nova_r196999_finite(b.get('duration_seconds')) if isinstance(b, dict) else None for b in blocks]
    if (total is None or total <= 0 or any(x is None or x < 0 for x in durations)
        or abs(sum(durations) - total) > 1):
        return []
    def mins(seconds):
        if seconds is None:
            return None
        return f'{seconds / 60:g} min' if seconds % 60 else f'{int(seconds / 60)} min'
    label = {'PREPARATION': 'Warm-up', 'CANONICAL_INTERVALS': 'Main set',
             'AEROBIC_TOPPER': 'Aerobic completion', 'COOLDOWN': 'Cool-down'}
    power = w.get('power_prescription') or {}
    feasibility = w.get('session_duration_feasibility') or {}
    p_ok = (isinstance(power, dict) and str(power.get('state') or '') == 'PRESCRIBED'
            and isinstance(feasibility, dict) and feasibility.get('state') == 'PASS')
    rows = []
    for block, sec in zip(blocks, durations):
        typ = str(block.get('type') or '').upper()
        duration = mins(sec)
        intensity = str(block.get('intensity') or '').upper()
        if typ == 'PREPARATION':
            detail = 'Progressive Z1 to low Z2' if intensity == 'Z1_TO_LOW_Z2' else intensity.replace('_',' ').title()
        elif typ == 'COOLDOWN':
            detail = 'Very easy Z1' if intensity == 'Z1' else intensity.replace('_',' ').title()
        elif typ == 'AEROBIC_TOPPER':
            detail = 'Conversational Z1 to low Z2 · no additional hard reps' if intensity == 'Z1_TO_LOW_Z2' else intensity.replace('_',' ').title()
        elif typ == 'CANONICAL_INTERVALS':
            work_sec = _nova_r196999_finite(block.get('quality_work_seconds'))
            rec_sec = _nova_r196999_finite(block.get('recovery_seconds'))
            detail = (f'{mins(work_sec)} of quality work' if work_sec is not None else 'Canonical intervals')
            if rec_sec is not None and rec_sec > 0:
                detail += f' + {mins(rec_sec)} easy recovery'
            if p_ok:
                reps = _nova_r196999_finite(power.get('reps'))
                rep_sec = _nova_r196999_finite(power.get('work_seconds'))
                if (reps is not None and int(reps)==reps and 1 <= reps <= 100
                    and rep_sec is not None and rep_sec > 0 and work_sec is not None
                    and abs(reps * rep_sec - work_sec) <= 1):
                    detail = f'{int(reps)} × {mins(rep_sec)} work'
                    if rec_sec is not None and rec_sec > 0:
                        detail += f' + {mins(rec_sec)} easy recovery'
            if p_ok:
                low = _nova_r196999_finite(power.get('low_watts'))
                high = _nova_r196999_finite(power.get('high_watts'))
                if low is not None and high is not None and 0 < low <= high:
                    detail += f' · {low:g}–{high:g} W'
                    if power.get('exploratory') is True:
                        detail += ' (exploratory, not pass/fail)'
        else:
            detail = intensity.replace('_', ' ').title() if intensity else 'Canonical phase'
        rows.append({'label': label.get(typ, typ.replace('_',' ').title() or 'Phase'),
                     'duration': duration, 'detail': detail})
    return rows

    n = int(seconds)

def _nova_r1969102_verified_personal_watts(workout, canonical_id):
