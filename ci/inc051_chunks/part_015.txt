# R167 · ATHLETE-MATURITY-AWARE FUTURE QUALITY + AUTONOMIC TREND WATCH
# General multi-athlete correction. A current recovery hold may block today's hard
# work without turning every later Clock opportunity into definitive Z2. For
# sufficiently established athletes, THE LAB can project one future canonical
# quality candidate, but it remains provisional until a later recovery check.
# Separately, markedly suppressed rMSSD/HRV must surface as an autonomic watch;
# persistence can suggest systemic stress/illness context but is never diagnostic.
R168_SCHEMA = "V4.9.44-R168-1"


def _v167_num(value):
    try:
        x = float(value)
        return x if math.isfinite(x) else None
    except (TypeError, ValueError):
        return None


def _v167_clock_tz(coach_clock):
    raw = str((coach_clock or {}).get("iso") or "")
    try:
        dt = datetime.fromisoformat(raw.replace("Z", "+00:00"))
        return dt.tzinfo or _runtime_timezone()
    except Exception:
        return _runtime_timezone()


def _v167_slot_dt(slot, coach_clock):
    if not isinstance(slot, dict):
        return None
    try:
        d = date.fromisoformat(str(slot.get("date") or "")[:10])
    except Exception:
        return None
    minute = _v167_num(slot.get("start_minute"))
    if minute is None:
        # Day-level Calendar/Rhythm opportunities remain comparable at midday for
        # projection ranking only; this never grants execution authority.
        minute = 12 * 60
    return datetime.combine(d, datetime.min.time(), tzinfo=_v167_clock_tz(coach_clock)) + timedelta(minutes=int(round(minute)))


def _v167_training_maturity(coach_clock, adaptive_roadmap=None):
    rhythm = ((coach_clock or {}).get("rhythm") or {}) if isinstance((coach_clock or {}).get("rhythm"), dict) else {}
    recent = rhythm.get("recent") if isinstance(rhythm.get("recent"), dict) else {}
    established = rhythm.get("established") if isinstance(rhythm.get("established"), dict) else {}
    q = ((adaptive_roadmap or {}).get("quality_window") or {}) if isinstance((adaptive_roadmap or {}).get("quality_window"), dict) else {}
    score = 0
    reasons = []
    days = _v167_num(recent.get("days_per_week"))
    hours = _v167_num(recent.get("hours_per_week"))
    recent_days = int(recent.get("training_days") or 0)
    established_days = int(established.get("training_days") or 0)
    spacing_samples = int(q.get("spacing_samples") or 0)
    if days is not None and days >= 4.0:
        score += 1; reasons.append("RECENT_FREQUENCY_ESTABLISHED")
    if hours is not None and hours >= 5.0:
        score += 1; reasons.append("RECENT_VOLUME_ESTABLISHED")
    if recent_days >= 8 or established_days >= 18:
        score += 1; reasons.append("MULTI_WEEK_TRAINING_HISTORY")
    if spacing_samples >= 6:
        score += 1; reasons.append("HARD_SPACING_PERSONAL_HISTORY")
    if str(rhythm.get("confidence") or "").upper() in {"MODERATE", "HIGH"}:
        score += 1; reasons.append("RHYTHM_CONFIDENCE")
    level = "HIGH" if score >= 4 else ("MODERATE" if score >= 2 else "LOW")
    return {
        "schema": R168_SCHEMA,
        "level": level,
        "score": score,
        "recent_days_per_week": days,
        "recent_hours_per_week": hours,
        "recent_training_days": recent_days,
        "established_training_days": established_days,
        "hard_spacing_samples": spacing_samples,
        "reasons": reasons,
        "rule": "Training maturity is learned from repeated frequency/volume/rhythm and hard-spacing history, not from one absolute watt value or athlete identity.",
    }


def _v167_quality_rate_for_slot(coach_clock, slot):
    try:
        d = date.fromisoformat(str((slot or {}).get("date") or "")[:10])
    except Exception:
        return 0.0
    rhythm = ((coach_clock or {}).get("rhythm") or {}) if isinstance((coach_clock or {}).get("rhythm"), dict) else {}
    st = rhythm.get("session_type_rhythm") if isinstance(rhythm.get("session_type_rhythm"), dict) else {}
    cycling = ((st.get("modalities") or {}).get("CYCLING") or {}) if isinstance(st.get("modalities"), dict) else {}
    rates = cycling.get("learned_quality_rate_by_weekday") if isinstance(cycling.get("learned_quality_rate_by_weekday"), dict) else {}
    raw = rates.get(d.weekday(), rates.get(str(d.weekday())))
    return float(_v167_num(raw) or 0.0)


def _v167_last_hard_dt(adaptive_roadmap, coach_clock):
    q = ((adaptive_roadmap or {}).get("quality_window") or {}) if isinstance((adaptive_roadmap or {}).get("quality_window"), dict) else {}
    raw = str(q.get("last_hard_end") or "")
    if not raw:
        return None
    try:
        dt = datetime.fromisoformat(raw.replace("Z", "+00:00"))
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=_v167_clock_tz(coach_clock))
        return dt.astimezone(_v167_clock_tz(coach_clock))
    except Exception:
        return None


def _v167_future_quality_slot(coach_clock, adaptive_roadmap, maturity):
    if (maturity or {}).get("level") == "LOW":
        return None
    q = ((adaptive_roadmap or {}).get("quality_window") or {}) if isinstance((adaptive_roadmap or {}).get("quality_window"), dict) else {}
    if str(q.get("state") or "").upper() != "RECOVERY HOLD":
        return None
    slots = list((coach_clock or {}).get("next_slots") or [])
    if not slots:
        return None
    try:
        now = datetime.fromisoformat(str((coach_clock or {}).get("iso") or "").replace("Z", "+00:00"))
    except Exception:
        now = get_rome_now()
    if now.tzinfo is None:
        now = now.replace(tzinfo=_v167_clock_tz(coach_clock))
    today = now.date()
    last_hard = _v167_last_hard_dt(adaptive_roadmap, coach_clock)
    earliest = float(_v167_num(q.get("learned_earliest_hours")) or 36.0)
    preferred = float(_v167_num(q.get("learned_preferred_hours")) or earliest)
    delay = float(_v167_num(q.get("recovery_delay_hours")) or 0.0)
    required = (earliest + delay) if (maturity or {}).get("level") == "HIGH" else (preferred + delay)
    candidates = []
    for idx, slot in enumerate(slots):
        if not isinstance(slot, dict):
            continue
        try:
            d = date.fromisoformat(str(slot.get("date") or "")[:10])
        except Exception:
            continue
        if d <= today:
            continue
        restriction = str(slot.get("restriction") or "NONE").upper()
        if slot.get("is_race") or restriction in {"NO_TRAINING", "NO_INTENSITY", "HEALTH"}:
            continue
        dt = _v167_slot_dt(slot, coach_clock)
        if dt is None:
            continue
        hours_since = ((dt - last_hard).total_seconds() / 3600.0) if last_hard else None
        if hours_since is not None and hours_since < required:
            continue
        rate = _v167_quality_rate_for_slot(coach_clock, slot)
        explicit_bonus = 12.0 if str(slot.get("timing_confidence") or "").upper() == "EXPLICIT" else 0.0
        # Prefer the athlete's learned quality rhythm when it is meaningfully
        # established, while still penalising unnecessary delay.
        delay_days = max(0, (d - today).days - 1)
        score = rate * 100.0 + explicit_bonus - 6.0 * delay_days
        candidates.append((score, -idx, idx, slot, hours_since, rate, required))
    if not candidates:
        return None
    candidates.sort(reverse=True, key=lambda x: (x[0], x[1]))
    _, _, idx, slot, hours_since, rate, required = candidates[0]
    return {
        "index": idx,
        "slot": dict(slot),
        "hours_since_last_hard": round(hours_since, 1) if hours_since is not None else None,
        "learned_quality_rate": round(rate, 3),
        "required_spacing_hours": round(required, 1),
    }


def _v167_autonomic_watch(season_wellness, metrics, now=None):
    now = now or get_rome_now(); today = now.date(); metrics = metrics or {}
    series = _v41_wellness_series(season_wellness or [], "hrv", 5, 350)
    if not series:
        return {"schema": R168_SCHEMA, "active": False, "state": "NO_DATA"}
    recent = [(x[0], float(x[1])) for x in series if today - timedelta(days=13) <= x[0] <= today]
    if not recent:
        return {"schema": R168_SCHEMA, "active": False, "state": "NO_RECENT_DATA"}
    latest_day, latest = recent[-1]
    resolved = ((metrics.get("personal_baselines") or {}).get("resolved") or {}) if isinstance((metrics.get("personal_baselines") or {}).get("resolved"), dict) else {}
    low = _v167_num(resolved.get("hrv_low")); high = _v167_num(resolved.get("hrv_high"))
    if low is None:
        prior_vals = [float(x[1]) for x in series if today - timedelta(days=42) <= x[0] < latest_day]
        if len(prior_vals) >= 7:
            s = sorted(prior_vals)
            low = statistics.quantiles(s, n=4, method="inclusive")[0] if len(s) >= 4 else min(s)
            high = statistics.quantiles(s, n=4, method="inclusive")[2] if len(s) >= 4 else max(s)
    if low is None or low <= 0:
        return {"schema": R168_SCHEMA, "active": False, "state": "BASELINE_LEARNING", "latest_ms": round(latest, 1)}
    ratio = latest / low
    below_pct = (1.0 - ratio) * 100.0
    hrv7 = [(x[0], float(x[1])) for x in series if today - timedelta(days=6) <= x[0] <= today]
    prior = [(x[0], float(x[1])) for x in series if today - timedelta(days=34) <= x[0] <= today - timedelta(days=7)]
    gm7 = math.exp(statistics.mean(math.log(v) for _, v in hrv7)) if len(hrv7) >= 3 else None
    prior_gm = math.exp(statistics.mean(math.log(v) for _, v in prior)) if len(prior) >= 3 else None
    weekly_delta = ((gm7 / prior_gm - 1.0) * 100.0) if gm7 and prior_gm else None
    last4 = recent[-4:]
    low_count = sum(1 for _, v in last4 if v < low)
    severe_acute = ratio <= 0.80
    repeated_low = len(last4) >= 3 and low_count >= 3
    trend_low = bool(gm7 is not None and gm7 < low and weekly_delta is not None and weekly_delta <= -8.0)
    active = bool(latest < low and (severe_acute or repeated_low or trend_low))
    if not active:
        return {
            "schema": R168_SCHEMA, "active": False, "state": "WITHIN_TREND_TOLERANCE",
            "latest_ms": round(latest, 1), "baseline_low_ms": round(low, 1),
        }
    # A depressed weekly mean can still be pulled down by one extreme morning.
    # Escalate to persistent/trend watch only when at least two recent mornings
    # are below the personal floor, or when 3/4 are low.
    persistent = bool(repeated_low or (trend_low and low_count >= 2))
    state = "AUTONOMIC_TREND_WATCH" if persistent else "ACUTE_AUTONOMIC_WATCH"
    illness_context = "MONITOR_IF_PERSISTENT_OR_SYMPTOMATIC" if persistent else "NOT_INFERRED_FROM_ONE_READING"
    return {
        "schema": R168_SCHEMA,
        "active": True,
        "state": state,
        "severity": "HIGH" if severe_acute else "MODERATE",
        "latest_date": latest_day.isoformat(),
        "latest_ms": round(latest, 1),
        "baseline_low_ms": round(low, 1),
        "baseline_high_ms": round(high, 1) if high is not None else None,
        "below_personal_floor_pct": round(max(0.0, below_pct), 1),
        "seven_day_geomean_ms": round(gm7, 1) if gm7 is not None else None,
        "seven_day_vs_prior_pct": round(weekly_delta, 1) if weekly_delta is not None else None,
        "low_readings_last4": low_count,
        "persistent": persistent,
        "illness_context": illness_context,
        "message": (
            "HRV/rMSSD is persistently suppressed relative to the athlete's own baseline. Accumulated load, recovery debt, heat, stress, measurement conditions or illness can all contribute; THE LAB should monitor symptoms and the next comparable mornings rather than diagnose the cause."
            if persistent else
            "Today's HRV/rMSSD is markedly below the athlete's own baseline. One morning is not diagnostic; repeat comparable measurements and cross-check load, resting HR, sleep and symptoms."
        ),
        "rule": "Personal longitudinal baseline first. A single extreme reading creates an acute watch; repeated suppression or a materially depressed 7-day level escalates to trend watch. Illness is a possible context only when the pattern persists or symptoms/other signals align, never a diagnosis from HRV alone.",
    }

def _v167_attach_future_quality_candidate(out, nova_decision, coach_clock, adaptive_roadmap, evidence_ledger, power_model, ftp_anchor, aerobic_metabolic_range, training_definitions):
    if not isinstance(out, dict):
        return out
    dec = nova_decision or {}; action = str(dec.get("action") or "").upper()
    if action != "HOLD" or str((((adaptive_roadmap or {}).get("quality_window") or {}).get("state") or "")).upper() != "RECOVERY HOLD":
        return out
    maturity = _v167_training_maturity(coach_clock, adaptive_roadmap)
    slot_info = _v167_future_quality_slot(coach_clock, adaptive_roadmap, maturity)
    out["athlete_training_maturity"] = maturity
    if not slot_info:
        return out
    idx = int(slot_info["index"]); slots = list((coach_clock or {}).get("next_slots") or [])
    if idx < 0 or idx >= len(slots):
        return out
    goal = str(dec.get("goal_key") or (adaptive_roadmap or {}).get("goal_key") or "ENDURANCE_BASE").upper()
    dim = str(dec.get("dimension") or "NONE").upper()
    template, library_audit = _v109_select_workout_template(goal, dim, evidence_ledger=evidence_ledger, training_definitions=training_definitions, slot=slots[idx])
    quality = _v109_render_library_workout(template, goal, dim, power_model=power_model, ftp_anchor=ftp_anchor, slot=slots[idx], aerobic_metabolic_range=aerobic_metabolic_range) if template else None
    if not isinstance(quality, dict) or not str(quality.get("intensity") or "").upper().startswith("HARD"):
        return out

# This wraps the native selector instead of bypassing any upstream health, recovery,

def _v172_quality_earlier_slot(first, selected, candidates, cadence, extra_delay):
    """Permit earlier quality only for comparably suitable slots after recovery normalizes.

    The caller has already passed all hard gates and the native spacing check.
    Rhythm is a preference, not an additional recovery hold. When role evidence is
    absent or conflicting, retain the native result rather than invent authority.
    """
    if not isinstance(first, dict) or not isinstance(selected, dict) or first is selected:
        return False
    try:
        # In the existing quality-window contract, zero added delay means the
        # readiness-based recovery penalty has cleared (green readiness).
        if float(extra_delay) != 0.0:
            return False
        early_dt, later_dt = first["dt"], selected["dt"]
        if not isinstance(early_dt, datetime) or not isinstance(later_dt, datetime):
            return False
        if early_dt >= later_dt or (later_dt - early_dt) > timedelta(hours=48):
            return False
        # Explicit scheduling is higher authority than learned weekday rhythm.
        if first.get("explicit") or selected.get("explicit"):
            return False
        # Only use genuinely learned per-day quality rates, never a default 0.5.
        profile = cadence or {}
        rates = profile.get("quality_rate_by_weekday") or {}
        early_raw = rates.get(early_dt.weekday(), rates.get(str(early_dt.weekday())))
        later_raw = rates.get(later_dt.weekday(), rates.get(str(later_dt.weekday())))
        if early_raw is None or later_raw is None:
            return False
        early_rate, later_rate = float(early_raw), float(later_raw)
        if not all(math.isfinite(x) and 0 <= x <= 1 for x in (early_rate, later_rate)):
            return False
        # Require a positive observed quality tendency at the earlier slot and
        # absence of a meaningful quality-preference gap.
        return early_rate >= 0.5 and later_rate - early_rate <= 0.25
    except (TypeError, ValueError, KeyError, OverflowError):
        return False


def _v117_select_quality_window(candidates, last_end, spacing, cadence, extra_delay, second_hard=False):
    result = _v117_select_quality_window_r171_final(
        candidates, last_end, spacing, cadence, extra_delay, second_hard=second_hard
    )
    if not isinstance(result, tuple) or len(result) != 4:
        return result
    earliest_dt, selected_dt, reason, selected_row = result
    if earliest_dt is None or selected_dt is None or earliest_dt >= selected_dt:
        return result
    # Mirror the native selector's spacing gate exactly: never use a candidate
    # which was excluded by its lower bound.
    try:
        lower = (last_end + timedelta(hours=float((spacing or {}).get("earliest_hours") or 36) + extra_delay)) if last_end else None
        valid = [r for r in candidates if isinstance(r, dict) and isinstance(r.get("dt"), datetime) and (lower is None or r["dt"] >= lower)]
        first = next((r for r in valid if r["dt"] == earliest_dt), None)
    except (TypeError, ValueError, OverflowError):
        return result
    if _v172_quality_earlier_slot(first, selected_row, candidates, cadence, extra_delay):
        return earliest_dt, earliest_dt, "R172_RECOVERED_EARLIER_COMPARABLE_QUALITY_SLOT", first
    return result

# R174 · DAILY NOTE EVENT-TIME AUTHORITY
# A historical note's relative dates are resolved against when it was logged,
# never against the current Snapshot time. Expired targets are kept as audit
# provenance but may not become tomorrow's availability/intensity restriction.

def _v174_note_event_anchor(row, now):
    """Resolve the native Daily Log's local_date/local_time to a parsing clock.

    The log's local_date is the highest-authority event date. Fall back to its
    timestamp only when the local date is unavailable; current time only when
    neither exists. Do not infer any athlete-specific location from the log.
    """
    event_day = None
    try:
        event_day = date.fromisoformat(str(row.get("local_date") or ""))
    except (TypeError, ValueError):
        pass
    if event_day is None:
        try:
            utc = datetime.fromisoformat(str(row.get("logged_at_utc") or "").replace("Z", "+00:00"))
            if utc.tzinfo is not None and now.tzinfo is not None:
                event_day = utc.astimezone(now.tzinfo).date()
            else:
                event_day = utc.date()
        except (TypeError, ValueError):
            pass
    if event_day is None or event_day > now.date() + timedelta(days=1):
        return now
    hour, minute = now.hour, now.minute
    match = re.fullmatch(r"\s*([01]?\d|2[0-3]):([0-5]\d)\s*", str(row.get("local_time") or ""))
    if match:
        hour, minute = int(match.group(1)), int(match.group(2))
    return now.replace(year=event_day.year, month=event_day.month, day=event_day.day,
                       hour=hour, minute=minute, second=0, microsecond=0)

def _v175_duration_minutes(row):

    if not isinstance(row, dict):
        return None
    for key in ("duration", "expected_duration_min", "duration_hint_min"):
        value = row.get(key)
        if isinstance(value, (int, float)) and math.isfinite(value) and value > 0:
            return float(value)
        if isinstance(value, str):

            if m:

            if m:
                return float(m.group(1)) * 60
    return None


def _v175_optional_volume_advice(sessions, coach_clock, execution_model, nova_decision):

def _v176_session_minutes(session):
    """Native compact labels include '~2h30'; never silently drop the 30 minutes."""
    if not isinstance(session, dict):
        return None
    for key in ("duration", "duration_hint_min", "expected_duration_min"):
        raw = session.get(key)
        if isinstance(raw, (int, float)) and not isinstance(raw, bool):
            if math.isfinite(raw) and 0 < raw < 24 * 60:
                return float(raw)
        if not isinstance(raw, str):
            continue
        hours = re.search(r"(?<!\d)(\d+(?:\.\d+)?)\s*h(?:ours?)?\s*(?:(\d{1,2})\s*(?:m|min)?)?", raw, re.I)
        if hours:
            value = float(hours.group(1)) * 60 + (float(hours.group(2)) if hours.group(2) else 0)
            if 0 < value < 24 * 60:
                return value
        mins = re.search(r"(?<!\d)(\d+(?:\.\d+)?)\s*(?:min|minutes)\b", raw, re.I)
        if mins:
            value = float(mins.group(1))
            if 0 < value < 24 * 60:
                return value
    return None


def _v176_completed_volume_profile(activities, snapshot_day):

    The upstream input is reconciled physical cycling history (Intervals native

        return profile

    if end is None:
        return profile

    prior_start = end - timedelta(days=41)
    recent_total = prior_total = 0.0
    latest_by_day = {}
    weekly_hours = [0.0] * 6  # fixed complete calendar-aligned 7-day bins ending snapshot_day
    weekly_training_days = [set() for _ in range(6)]
    ids_seen = set()
    oldest = None
    for row in activities:
        if not isinstance(row, dict) or not _v4887_is_cycling_activity(row):
            continue

        if _v141_is_race_activity(row) or _v40_is_hard_candidate(row):
            continue
        intensity = _v40_normalized_intensity_factor(row)
        if intensity is not None and intensity > 0.82:
            continue
        latest_by_day[d] = max(latest_by_day.get(d, 0.0), minutes)
    last_easy = max(latest_by_day) if latest_by_day else None
    ranked = sorted(latest_by_day.items(), key=lambda kv: kv[1], reverse=True)
    # TWO distinct completed easy-ride days earn a duration step. Single spikes
    # do not advance the anchor, and only the last six weeks count.
    repeated = ranked[1][1] if len(ranked) >= 2 else None
    repeated_day = max((d for d, m in latest_by_day.items() if repeated is not None and m >= 0.95 * repeated), default=None)
    previous_hours = prior_total / 60.0 / 4.0
    recent_hours = recent_total / 60.0 / 2.0
    ratio = (recent_hours / previous_hours) if previous_hours >= 1.0 else None
    immediate_prior_weekly = (weekly_hours[2] + weekly_hours[3]) / 2.0
    short_ratio = (recent_hours / immediate_prior_weekly
                   if immediate_prior_weekly >= 1.0 else None)
    # Provider data cannot certify that an empty week was really an off-week.
    # Require cycling observed in every comparator week before using an
    # assertive trend description. An absence of evidence is not low volume.
    # Compare both time scales and an actual hours/week change, not a 15% blip.
    active_weeks = [len(week) for week in weekly_training_days]
    coverage_supported = (min(active_weeks) >= 1 and oldest is not None
                          and oldest <= end - timedelta(days=35))
    observed_increase = recent_hours - previous_hours
    short_increase = recent_hours - immediate_prior_weekly
    growth_brake = bool(coverage_supported and ratio is not None and short_ratio is not None
                        and ratio >= 1.30 and short_ratio >= 1.30
                        and observed_increase >= 2.0 and short_increase >= 2.0)
    profile.update({"available": bool(latest_by_day), "easy_ride_count": len(latest_by_day),
                    "recent_easy_days": len(latest_by_day),
                    "repeated_duration_min": round(repeated, 1) if repeated else None,
                    "last_repeated_ride_date": repeated_day.isoformat() if repeated_day else None,
                    "last_easy_ride_date": last_easy.isoformat() if last_easy else None,
                    "week_recent_hours": round(recent_hours, 2),
                    "week_prior_hours": round(previous_hours, 2),
                    "growth_ratio": round(ratio, 3) if ratio is not None else None,
                    "two_week_comparator_ratio": round(short_ratio, 3) if short_ratio is not None else None,
                    "immediately_prior_week_hours": round(immediate_prior_weekly, 2),
                    "weekly_training_days": active_weeks,
                    "coverage_state": ('ACTIVITY_SPREAD_SUPPORTED_NOT_COMPLETE_PROOF'
                                       if coverage_supported else 'COVERAGE_UNKNOWN'),
                    "growth_evidence": ('CORROBORATED_RISE_BOTH_WINDOWS' if growth_brake else
                                        'UNSUPPORTED_OR_MODEST_CHANGE'),
                    "growth_brake": growth_brake,
                    "history_days_covered": (end - oldest).days + 1 if oldest else 0})
    return profile


def _v176_progressive_volume_advice(sessions, coach_clock, execution_model,
                                      nova_decision, cycling_history_activities=None,
                                      snapshot_day=None):
    """Progress only on repeated completed aerobic rides, with fresh-load brakes.

    A learned window does not establish athlete availability. Suggestions are
    independent of roadmap goals and never join the hard/session carry ledger.
    """
    # Preserve R175's native guardrails and human-facing continuity contract.
    out = _v175_optional_volume_advice(sessions, coach_clock, execution_model, nova_decision)
    out = dict(out)
    out["schema"] = R176_SCHEMA
    out.update({"non_binding": True, "creates_session": False,
                "counts_as_missed_workout": False, "alters_hard_prescription": False,
                "progression_source": "COMPLETED_EASY_RIDES_NOT_PRESCRIPTION_HISTORY",
                "stage": "NO_WINDOW", "stage_label": "NO OPTIONAL WINDOW"})
    if not out.get("available"):
        out.update({"stage_label": "LEARNING / NO CONFIRMED WINDOW",
                    "purpose": "Aerobic volume matters alongside any cycling goal; the learned rhythm is a feasibility clue, not a limit on potential development.",
                    "headline": "No suitable future easy-duration opportunity is currently identified. Do not add one to an existing hard session or assume free time.",
                    "guidance": "When time and recovery allow, consider a separate easy longer ride; further guidance needs reliable completed activities and a suitable opportunity.",
                    "continuity": "This is educational and entirely optional: no missed-volume debt, adherence penalty or hard-workout displacement."})
        return out
    if isinstance(snapshot_day, datetime):
        snapshot_day = snapshot_day.date()
    if not isinstance(snapshot_day, date):
        # No silently invented 'today' in deterministic replay.
        out.update({"stage": "TIME_UNKNOWN", "stage_label": "RECHECK DATE",
                    "suggested_minutes": None})
        return out
    profile = _v176_completed_volume_profile(cycling_history_activities, snapshot_day)
    out["completed_volume_evidence"] = profile
    base = _v176_session_minutes((sessions or [])[out["session_index"]])
    if base is None:
        out.update({"stage": "UNUSABLE_DURATION", "stage_label": "DURATION UNKNOWN",
                    "suggested_minutes": None})
        return out
    out["baseline_minutes"] = round(base)
    hold = bool(out.get("recovery_hold_now"))
    hard_near = bool(out.get("next_day_quality_candidate"))
    earned = profile.get("repeated_duration_min")
    if earned is None:
        # When history is truncated or sparse, the R175 conservative cue may
        # remain; it does not claim that duration has been earned/repeated.
        out.update({"stage": "EXPLORATION", "stage_label": "START / REPEAT",
                    "earned_minutes": None})
        out["purpose"] = ("Aerobic volume can be developed alongside every cycling power goal. "
                          "This is an initial optional exploration, NOT an achieved level or an obligation.")
    else:
        out["earned_minutes"] = round(earned)
        recent = profile.get("last_repeated_ride_date")
        stale = not recent or (snapshot_day - date.fromisoformat(recent)).days > 21
        # Earned level is repeated. A longer isolated ride never unlocks a new
        # step. After a long break we restart conservatively, without punishment.
        level = max(base, earned) if not stale else base
        increment = min(20.0, max(10.0 if level >= 60 else 5.0, level * 0.12), level * 0.20)
        if hard_near:
            increment = min(increment, 10.0)
        step = int(increment // 5) * 5
        if stale:
            out.update({"stage": "REBUILD", "stage_label": "RESTART GENTLY",
                        "suggested_minutes": round(base) + (step if step >= 5 else 0)})
        elif profile.get("growth_brake"):
            out.update({"stage": "CONSOLIDATE", "stage_label": "ABSORB RECENT VOLUME",
                        "suggested_minutes": None})
        else:
            out.update({"stage": "PROGRESS", "stage_label": "REPEATED EXPOSURE",
                        "suggested_minutes": round(level) + (step if step >= 5 else 0)})
        out["purpose"] = ("Gradual aerobic-volume development is useful even when the primary "
                          "goal is power. This step is based on repeated COMPLETED easy rides, "
                          "not on the proposed plan, and it is never mandatory.")
    target = out.get("suggested_minutes")
    if target is not None and target <= base:
        target = None
        out["suggested_minutes"] = None
    if out["stage"] == "CONSOLIDATE":
        out["headline"] = ("Completed ride data suggest a substantial rise in weekly cycling hours "
                           "across both comparison windows. Consider consolidating easy volume "
                           "before an optional longer ride; this does not change Nova's hard plan.")
    elif target is not None:
        out["headline"] = (f"On the suggested easy day, you may REPLACE the existing ~{round(base)}-min "
                           f"easy ride with one easy Z1–Z2 ride gradually approaching ~{round(target)} min "
                           "ONLY if you genuinely have a long enough free window and recovery supports it. "
                           "The extra time is NOT scheduled or presumed available. Do not add a second ride, "
                           "change any hard session or count a skipped option as missed training.")
    else:
        out["headline"] = ("Keep easy riding consistent; no numeric duration increase is "
                           "justified by the current observations. Reassess after more completed rides.")
    if hold:
        out["stage_before_recovery_guard"] = out["stage"]
        out["stage"] = "RECOVERY_GATED"
        out["stage_label"] = "HOLD · RECHECK BEFORE EXTRA VOLUME"
        out["guidance"] = ("Today's recovery HOLD remains in force: this is NOT authorization "
                           "to add volume now. For the named future day, only consider the extra "
                           "time if genuinely available and fresh recovery indicators and symptoms "
                           "clearly support easy training. With poor/unknown morning signals, "
                           "use the original easy duration or shorten/rest. A new Snapshot is "
                           "the reliable way to review the full plan; this optional suggestion "
                           "never independently reopens hard workouts.")
    else:
        out["guidance"] = ("Only if there is genuinely enough time, you feel well and Z1–Z2 "
                           "remains comfortably conversational. Shorten/stop for abnormal fatigue, "
                           "symptoms or unexpectedly high HR. More hours are not automatically "
                           "more benefit. No physiological optimal-volume ceiling is inferred.")
    if hard_near:
        out["guidance"] += (" A hard candidate follows within a day; protect freshness for "
                            "that workout and skip the optional extension if needed.")
    out["continuity"] = ("No missed-volume debt, no compliance penalty, no carry-forward and "
                         "no hard-session rescheduling if this is skipped. Completed rides update "
                         "the next assessment; unavailable time is NEVER inferred as disinterest.")
    out["dose_policy"] = ("Repeat successful easy durations on distinct days before growing again; "
                          "hold progression after a sharp rise in completed weekly hours, or if "
                          "recovery worsens. There is no universal time cap or guaranteed payoff.")
    return out

def _v181_supplemental_effort(block, plan):
    """Classify ONLY provider-detected short, elevated-power work after a valid main set.

    Observed power relative to the canonical work prescription is a screening
    heuristic; it never proves a maximal sprint, readiness, or an extra prescribed
    stimulus. No inference from titles, athlete IDs, or missing power data.
    """
    if not isinstance(block, dict) or not isinstance(plan, dict):
        return None
    ev = _v155_main_set_evidence(block, plan)
    if not ev or len(ev.get("main") or []) != int(ev.get("reps") or 0):
        return None
    rows = list(ev.get("rows") or [])
    main = list(ev.get("main") or [])
    if not main:
        return None
    last_main_index = max(int(x["index"]) for x in main)
    watts_target = _num(ev.get("power_high"))
    if watts_target is None or not math.isfinite(watts_target) or watts_target <= 0:
        return None  # no individual reference; never invent a high-power threshold
    threshold = 1.20 * watts_target  # detection heuristic, not an optimal dose
    extra = []
    for r in rows:
        if int(r.get("index", -1)) <= last_main_index:
            continue  # warmup openers are not post-session accelerations
        secs = _num(r.get("secs"))
        watts = _num(r.get("watts"))
        if secs is None or watts is None or not math.isfinite(secs) or not math.isfinite(watts) or not (5 <= secs <= 45):
            continue
        if watts < threshold:
            continue
        extra.append({"seconds": round(secs, 1), "average_watts": round(watts, 1)})
    if not extra:
        return None
    load = _num(block.get("load"))
    if load is not None and not math.isfinite(load):
        load = None
    zone_summary = str(block.get("power_zone_summary") or "")
    return {
        "schema": R181_SCHEMA,
        "available": True,
        "state": "OBSERVED_EXTRA_SHORT_HIGH_POWER",
        "source": "INTERVALS_NATIVE_WORK_INTERVAL_EVIDENCE",
        "count": len(extra),
        "total_seconds": round(sum(x["seconds"] for x in extra), 1),
        "max_interval_average_watts": max(x["average_watts"] for x in extra),

def _v182_equivalence_key(plan):
    if not isinstance(plan, dict):
        return None
    text = " ".join(str(plan.get("main_set") or "").split())
    if not text:
        return None
    # Comparisons use immutable prescribed mechanics, NOT an athlete name or
    # arbitrary power/interval tolerances that could merge different workouts.
    def norm_number(value):
        try:
            return round(float(value), 5) if value is not None else None
        except (TypeError, ValueError, OverflowError):
            return str(value or "")
    return (
        str(plan.get("date") or "")[:10], text,
        norm_number(plan.get("prescribed_power_low")),
        norm_number(plan.get("prescribed_power_high")),
        norm_number(plan.get("interval_minutes")),
    )


def _v182_issued_dt(plan):
    try:
        stamp = str(plan.get("recorded_at_local") or "").strip()
        if not stamp:
            return None
        dt = datetime.fromisoformat(stamp.replace("Z", "+00:00"))
        return dt if dt.tzinfo is not None else dt.replace(tzinfo=ROME_TZ)
    except (TypeError, ValueError, OverflowError):
        return None


def _v182_canonicalize_active_reissues(ledger, blocks):
    """Return a cloned ledger with only provable superseded versions retired.

    This is a reconciliation input normalizer, NOT a new recommendation engine.
    A missing issue timestamp does not outrank a causally dated exact equivalent.
    If two DIFFERENT workout IDs/contents plausibly match, leave them ambiguous.
    """
    out = _v4831_json_clone(ledger)
    if not isinstance(out, dict) or not isinstance(blocks, (list, tuple)):
        return out
    actives = _v146_active_prescriptions(out)
    retired = {}  # fingerprint -> newer fingerprint
    for block in blocks:
        if not isinstance(block, dict) or not block.get("quality_relevant"):
            continue
        groups = {}
        for p in actives:
            if str(p.get("date") or "")[:10] != str(block.get("date") or "")[:10]:
                continue
            key = _v182_equivalence_key(p)
            if key is None or _v146_match_plan_block(p, block, allow_fulfilled=False) is None:
                continue
            groups.setdefault(key, []).append(p)
        for rows in groups.values():
            if len(rows) < 2:
                continue
            # No merging across genuinely different canonical workouts.
            ids = {str(p.get("nova_workout_id") or p.get("nova_contract_id") or "").strip()
                   for p in rows if p.get("nova_workout_id") or p.get("nova_contract_id")}
            if len(ids) > 1:
                continue
            issued = [(dt, p) for p in rows if (dt := _v182_issued_dt(p)) is not None]
            if not issued:
                continue
            issued.sort(key=lambda item: item[0], reverse=True)
            # If two distinct prescriptions share the most recent timestamp,
            # their issuance order is not provable. Fail closed.
            latest = issued[0][0]
            if sum(dt == latest and _v136_prescription_fingerprint(p) !=
                   _v136_prescription_fingerprint(issued[0][1]) for dt, p in issued) > 0:
                continue
            winner = issued[0][1]
            wfp = str(winner.get("prescription_fingerprint") or _v136_prescription_fingerprint(winner))
            for p in rows:
                pfp = str(p.get("prescription_fingerprint") or _v136_prescription_fingerprint(p))
                if pfp != wfp:
                    retired[pfp] = wfp
    if not retired:
        return out
    for row in out.get("prescription_history") or []:
        if not isinstance(row, dict):
            continue
        fp = str(row.get("prescription_fingerprint") or _v136_prescription_fingerprint(row))
        if fp in retired and str(row.get("prescription_status") or "ACTIVE").upper() in {"ACTIVE", "PLANNED", ""}:
            row["prescription_status"] = "SUPERSEDED"
            row["superseded_by_fingerprint"] = retired[fp]
            row["superseded_reason"] = "R182_IDENTICAL_MECHANICS_LATER_ISSUANCE"
    for field in ("primary_anchor", "next_cycle_anchor"):
        row = out.get(field)
        if not isinstance(row, dict):
            continue
        fp = str(row.get("prescription_fingerprint") or _v136_prescription_fingerprint(row))
        if fp in retired:
            row["prescription_status"] = "SUPERSEDED"
            row["superseded_by_fingerprint"] = retired[fp]
    out["equivalent_reissue_resolution"] = {
        "schema": R182_SCHEMA, "retired_versions": len(retired),
        "method": "IDENTICAL_DATE_MECHANICS_AND_UNIQUE_LATEST_ISSUANCE",
        "different_workouts_still_fail_closed": True,
    }
    return out


def _v146_reconcile_prescription_execution(ledger, blocks, now=None):

def _v181_supplemental_effort(block, plan):
    """R181 canonical evidence first; R182 native W′bal boundaries if canonical omits the extra.

    R182 requires four canonical reps in the original device-lap evidence plus
    complete ordered native intervals before interpreting a post-main-set burst.
    No training load, adherence score or recovered work is manufactured here.

def _v183_adherence_axes(block, plan, prior):

        return None

                  and math.isfinite(high) and 0 < low <= high)

        watts.append(float(x) if x is not None and math.isfinite(x) and x > 0 else None)

    elif above == total:

        state, label = "BELOW_TARGET", "BELOW TARGET"
    else:

             "prescribed_power_high_w": float(high) if band_valid else None,
             "observed_power_watts": [round(x, 1) if x is not None else None for x in watts],

def _v185_pto_item(item):
    """Only an explicitly labelled OTHER/PTO Calendar range is a vacation context.

    Restriction type is never modified. E.g. OTHER 'photo shoot' or 'NO PTO'
    must not silently change training scheduling.
    """
    title = str((item or {}).get('title') or '').strip().upper()
    return bool(str((item or {}).get('kind') or '').upper() == 'OTHER'
                and (title == 'PTO' or title.startswith('PTO ') or title.startswith('PTO·')
                     or title.startswith('PTO:') or title.startswith('PTO-'))
                and (item or {}).get('active', 1))


def _v185_pto_dates(items, start=None, end=None):
    out = set()
    for item in items or []:
        if not _v185_pto_item(item):
            continue
        try:
            first = date.fromisoformat(str(item.get('start_date') or '')[:10])
            last = date.fromisoformat(str(item.get('end_date') or item.get('start_date') or '')[:10])
            if last < first:
                first, last = last, first
            lo = max(first, start) if start else first
            hi = min(last, end) if end else last
            if hi < lo:
                continue
            # Calendar windows are finite; cap absurdly long entries defensively.
            for i in range(min((hi - lo).days + 1, 366)):
                out.add((lo + timedelta(days=i)).isoformat())
        except (ValueError, TypeError, OverflowError):
            continue
    return out


def _v185_load_pto_dates(start, end):
    try:
        items = load_planning_items(active_only=True, limit=240)
        return _v185_pto_dates(items, start=start, end=end)
    except Exception:
        # No PTO exception is better than inventing one when Calendar is unavailable.
        return set()

def _v117_hard_cadence_profile(activities, now=None):
    """Preserve real last-hard/spacing while isolating holiday timing outliers.

    Existing training loads, activities, last-hard end, spacing, and recovery are
    untouched; only observed scheduling priors are conditioned on non-PTO days.
    """
    base = _v117_hard_cadence_profile_r184_final(activities, now=now)
    if not isinstance(base, dict) or not activities:
        return base
    try:
        now = now or get_rome_now()
        excluded = _v185_load_pto_dates(now.date() - timedelta(days=100), now.date())
        if not excluded:
            return base
        ordinary = [a for a in activities if str((a or {}).get('start_date_local') or '')[:10] not in excluded]
        if len(ordinary) == len(activities):
            return base
        learned = _v117_hard_cadence_profile_r184_final(ordinary, now=now)
        if not isinstance(learned, dict):
            return base
        # Never overwrite last_hard_day, gaps or spacing with a filtered history.
        prior_keys = ('hard_daypart','hard_start_minute','time_confidence',
                      'hard_week_profile','hard_week_confidence','preferred_hard_weekdays',
                      'preferred_quality_weekdays','aerobic_biased_weekdays',
                      'quality_rate_by_weekday','weekend_session_role','weekly_session_role',
                      'weekday_quality_day_rate','weekend_quality_day_rate',
                      'session_role_shift_detected')
        out = dict(base)
        for key in prior_keys:
            if key in learned:
                out[key] = learned[key]
        out['pto_timing_excluded_days'] = len(excluded)
        out['pto_timing_model'] = 'NON_PTO_OBSERVED_SCHEDULING_ONLY'
        return out
    except Exception:
        return base

        habitual = str((cadence or {}).get('hard_daypart') or '').upper()
        confidence = str((cadence or {}).get('time_confidence') or 'LOW').upper()
        if habitual in {'AM','PM'} and confidence in {'MODERATE','HIGH'}:
            if early_part != habitual:
                return False
        if early_part != late_part:
            return False
        minute = (cadence or {}).get('hard_start_minute')
        if minute is not None and confidence in {'MODERATE', 'HIGH'}:
            minute = float(minute)
            distance_early = abs((early.hour * 60 + early.minute) - minute)
            distance_late = abs((late.hour * 60 + late.minute) - minute)
            # A markedly less habitual time is not a comparable opportunity.
            if distance_early > distance_late + 90:
                return False
        return True
    except (ValueError, TypeError, KeyError, OverflowError):
        return False

def _v185_slot_time_from_label(label):
    """Historical persisted slot is a localized display string, not a time field.

    If time cannot be proven, decline same-day re-timing rather than guessing.
    """
    m = re.search(r'\b(?:around|at)\s+(\d{1,2}):(\d{2})\s*(AM|PM)\b',
                  str(label or ''), re.I)
    if not m:
        return None
    h, minute = int(m.group(1)), int(m.group(2))
    if h < 1 or h > 12 or minute > 59:
        return None
    return (h % 12) * 60 + (720 if m.group(3).upper() == 'PM' else 0) + minute


def _v185_carry_same_day(out, decision, clock, ledger, now=None):
    """Do not drop a previously issued AM hard commitment after AM is missed.

    Requires a new, eligible PM hard slot from Nova. This does NOT authorize new
    intensity by itself and cannot reassign races, validations or Calendar events.
    """
    if not isinstance(out, dict) or out.get('status') != 'PRESCRIBED' \
            or not out.get('hard_prescription_available') \
            or str((decision or {}).get('action') or '').upper() != 'TRAIN':
        return out
    now = now or get_rome_now()
    idx, slot = _v166_target_slot(out, clock)
    if idx is None or not isinstance(slot, dict) or slot.get('is_race') or slot.get('is_planned_training') \
            or str(slot.get('source') or '').upper() == 'CALENDAR' \
            or str(slot.get('restriction') or 'NONE').upper() != 'NONE':
        return out
    if str(slot.get('date') or '') != now.date().isoformat():
        return out
    try:
        selected_min = int(round(float(slot.get('start_minute'))))
        if selected_min <= now.hour*60+now.minute:
            return out
    except (TypeError, ValueError):
        return out
    rows = []
    for r in (ledger or {}).get('prescription_history') or []:
        if not isinstance(r, dict) or str(r.get('date') or '')[:10] != now.date().isoformat():
            continue
        if str(r.get('prescription_status') or '').upper() in {'SUPERSEDED','FULFILLED','CANCELLED','WITHDRAWN','EXPIRED_BY_NEW_HARD_EVIDENCE'}:
            continue
        if r.get('superseded_by_fingerprint') or r.get('executed_activity_id') or not _v153_is_training_plan(r):
            continue
        prior_min = _v185_slot_time_from_label(r.get('slot'))
        if prior_min is None or selected_min <= prior_min + 60 or now.hour*60+now.minute < prior_min + 90:
            continue
        issued = _v153_parse_local(r.get('recorded_at_local'))
        if issued is None or issued > now:
            continue
        if _v153_material_hard_after(r, ledger or {}):
            continue
        rows.append((issued, r, prior_min))
    if not rows:
        return out
    rows.sort(key=lambda x:x[0], reverse=True)
    _, prior, prior_min = rows[0]
    preserved = _v153_preserved_raw(prior, out.get('dimension'))
    raw = list(out.get('raw_sessions') or [])
    if idx < 0 or idx >= len(raw) or not isinstance(raw[idx], dict):
        return out
    wid = str(prior.get('nova_workout_id') or prior.get('nova_contract_id') or '')
    if not wid:
        return out
    raw[idx] = preserved
    out['raw_sessions'] = raw
    out['workout_id'] = wid
    out['contract_id'] = prior.get('nova_contract_id') or wid
    out['source'] = 'SAME_DAY_UNEXECUTED_HARD_CONTINUITY'
    out['prescription_continuity'] = {
        'schema': R185_SCHEMA, 'active': True, 'state': 'SAME_DAY_UNEXECUTED_MOVED_LATER',
        'workout_id': wid, 'target_date': now.date().isoformat(),
        'original_start_minute': prior_min, 'new_start_minute': selected_min,
        'prior_recorded_at_local': prior.get('recorded_at_local'),
        'new_hard_workouts_added': 0, 'availability_confirmed': bool(slot.get('availability_confirmed')),
        'rule': 'Preserve a missed morning hard commitment at the next independently eligible same-day quality slot; ' 
                'do not infer the ride was completed or impose a second hard session.'
    }
    out['prescription_continuity_schema'] = R185_SCHEMA
    return out

def _v187_snapshot_journal(prior, incoming, now):
    previous = (prior or {}).get("data") if isinstance(prior, dict) else None
    previous = previous if isinstance(previous, dict) else None
    old_history = previous.get("decision_change_history") if previous else None
    history = [copy.deepcopy(v) for v in (old_history or []) if isinstance(v, dict)][-11:]
    if previous is None:
        return history, None
    before = _v187_plan_view(previous)
    after = _v187_plan_view(incoming)
    bg, ag = before["gate"], after["gate"]
    p1, p2 = before.get("plan"), after.get("plan")
    keys = ("date", "period", "title", "workout_id", "main_set")
    try:
        shift = abs(float((p1 or {}).get("start_minute")) - float((p2 or {}).get("start_minute")))
    except (TypeError, ValueError):
        shift = 0.0
    if ((tuple((p1 or {}).get(k) for k in keys) == tuple((p2 or {}).get(k) for k in keys))
        and shift < 30.0
        and (bg.get("quality_window_state"), bg.get("nova_action"), bg.get("quality_allowed")) ==
            (ag.get("quality_window_state"), ag.get("nova_action"), ag.get("quality_allowed"))):
        return history, None
    changed = _v187_changed_metrics(bg, ag)
    explanation = _v187_decision_explanation(before, after, changed)
    stamp = (now or get_rome_now()).isoformat(timespec="seconds")
    item = {"schema": R187_SCHEMA, "at_local": stamp,
            "previous_snapshot_local": prior.get("generated_at_local"),
            "before": before, "after": after,
            "evidence_changes": changed,
            "explanation": explanation,
            "reason_scope": "EVIDENCE_DELTA_AND_AUTHORITY_GATE_NOT_HISTORICAL_CAUSAL_PROOF"}
    # Stable id avoids duplicate entries on equivalent consecutive commits.
    item["change_id"] = hashlib.sha256(json.dumps(
        [prior.get("generated_at_utc"), before["summary"], after["summary"],
         bg.get("quality_window_state"), ag.get("quality_window_state")],
        sort_keys=True, default=str).encode()).hexdigest()[:20]
    if not history or history[-1].get("change_id") != item["change_id"]:
        history.append(item)
    return history[-12:], item

# A historic PB may remain a valid lifetime/YTD reference without being an

def _v189_recent_progress_reference(snapshot):
    """Use the athlete's persisted Snapshot clock, not server/visitor wall clock."""
    if not isinstance(snapshot, dict):
        return None
    clock = snapshot.get('coach_clock') or {}
    candidates = [
        (clock.get('iso') if isinstance(clock, dict) else None),
        snapshot.get('snapshot_generated_at_local'),
        snapshot.get('generated_at_local'),
    ]
    for value in candidates:
        raw = str(value or '').strip()
        try:
            if len(raw) >= 10 and re.fullmatch(r'\d{4}-\d{2}-\d{2}', raw[:10]):
                return date.fromisoformat(raw[:10])
        except (ValueError, TypeError):
            continue
    return None


def _v163_recent_progress_badges(snapshot, limit=3):
    """No stale, pre-plan or unrecognized progress in the current Roadmap.

    Keep old power/evidence ledgers unmodified for scientific reference and
    historical power curves; gate ONLY the UI projection labeled RECENT PROGRESS.
    In the absence of a trusted snapshot date or active block, fail closed.
    """
    if not isinstance(snapshot, dict):
        return []
    today = _v189_recent_progress_reference(snapshot)
    td = snapshot.get('training_direction') or {}
    qs = snapshot.get('question_state') or {}
    if not isinstance(td, dict) or not isinstance(qs, dict) or today is None:
        return []
    plan_start = str(td.get('started_on') or '').strip()[:10]
    scope_start = str(qs.get('scope_start') or '').strip()[:10]
    pe_scope = snapshot.get('performance_evidence') or {}
    evidence_start = str(pe_scope.get('effective_cycle_start') or '').strip()[:10] if isinstance(pe_scope, dict) else ''
    try:
        starts = [date.fromisoformat(d) for d in (plan_start, scope_start, evidence_start) if d]
    except (TypeError, ValueError):
        return []
    if not starts:
        return []
    earliest = max([today-timedelta(days=R189_RECENT_PROGRESS_LOOKBACK_DAYS)] + starts)
    if earliest > today:
        return []

    def in_window(value):
        raw = str(value or '').strip()
        try:
            return bool(re.fullmatch(r'\d{4}-\d{2}-\d{2}', raw)) and earliest <= date.fromisoformat(raw) <= today
        except ValueError:
            return False

    pe = snapshot.get('performance_evidence') or {}
    if not isinstance(pe, dict):
        pe = {}
    events = []
    for ev in pe.get('events') or []:
        if not isinstance(ev, dict) or not in_window(ev.get('date')):
            continue
        kind = str(ev.get('kind') or '').upper()
        if kind == 'VALIDATION':
            eligible = ev.get('formal_validation') is True and str(ev.get('outcome') or '').upper() == 'VALIDATED_GAIN'
        elif kind == 'PB':
            eligible = ev.get('the_lab_recognized') is True and ev.get('timing_proven') is True
        else:
            eligible = False
        if eligible:
            events.append(ev)

    dims = (qs.get('dimensions') or {}) if isinstance(qs.get('dimensions'), dict) else {}
    recent_dims = {
        name: row for name, row in dims.items()
        if isinstance(row, dict) and row.get('exit_satisfied') is True and in_window(row.get('last_evidence_date'))
    }
    # Isolated shallow containers: do not mutate persisted per-athlete history.
    scoped = dict(snapshot)
    scoped['performance_evidence'] = {**pe, 'events': events}
    scoped['question_state'] = {**qs, 'dimensions': recent_dims}
    return _v163_recent_progress_badges_r188(scoped, limit=limit)

R190_SCHEMA = 'V4.9.66-R190-1'


def _v190_block_dose_evidence(ledger, activities, direction, now=None):
    """Close foundation from verified COMPLETE comparative weeks in active block.

    The in-progress week is still the live dose-budget; this evidence concerns
    whether a developmental foundation was EVER earned in the active block.
    A new Monday is NOT counterevidence to a completed earlier progression.
    """
    if not isinstance(ledger, dict) or not ledger.get('available'):
        return ledger
    direction = direction if isinstance(direction, dict) else {}
    now = now or get_rome_now()
    day = now.date() if isinstance(now, datetime) else now
    start = str(direction.get('started_on') or '')[:10]
    goal = str(direction.get('primary_goal') or '').upper()
    try:
        block_start = date.fromisoformat(start)
    except (TypeError, ValueError):
        ledger['block_foundation_evidence'] = {'schema': R190_SCHEMA, 'status': 'UNAVAILABLE', 'reason_code': 'BLOCK_START_UNKNOWN'}
        return ledger
    key = 'SPECIFIC_DOSE' if goal in {'POWER_5','VO2MAX','POWER_1','SPRINTER'} else ('TIME_AT_PRESSURE' if goal in {'POWER_20','TIME_TRIAL'} else None)
    if not key:
        return ledger
    # Consecutive completed Mon-Sun windows. Comparisons use same native zone
    # evidence and same thresholds as _v82_question_record, but never conflate
    # in-progress zero minutes with a revoked achievement.
    latest_completed_sun = day - timedelta(days=day.weekday()+1)
    first_monday = block_start - timedelta(days=block_start.weekday())
    if first_monday < block_start:
        first_monday += timedelta(days=7)
    cursor = max(first_monday, latest_completed_sun-timedelta(days=7*25+6))
    reviewed = []
    max_weeks = 26  # time/CPU bounded, multi-athlete safe
    while cursor+timedelta(days=6) <= latest_completed_sun and len(reviewed)<max_weeks:
        sunday = cursor+timedelta(days=6)
        dose = _v4932_intensity_dose(activities, cursor, now=sunday)
        curr = dose.get('current') or {}; prev = dose.get('previous_comparable') or {}
        prop = 'high_minutes' if key == 'SPECIFIC_DOSE' else 'moderate_minutes'
        value = _rhythm_num(curr.get(prop)); base = _rhythm_num(prev.get(prop))
        enough = bool(curr.get('zoned_sessions') and prev.get('zoned_sessions') and value is not None and base is not None)
        threshold = max(3., 0.10*max(base or 0.,1.)) if key == 'SPECIFIC_DOSE' else max(5.,0.10*max(base or 0.,1.))
        row={'window_start': cursor.isoformat(),'window_end':sunday.isoformat(),
             'metric':prop,'current_min':value,'previous_min':base,
             'current_zoned_sessions':curr.get('zoned_sessions'), 'previous_zoned_sessions':prev.get('zoned_sessions'),
             'progression_threshold_min':round(threshold,2),
             'evidence_sufficient':enough,
             'progressed':bool(enough and value>=base+threshold)}
        reviewed.append(row)
        cursor += timedelta(days=7)
    earned = [r for r in reviewed if r['progressed']]
    ledger['block_foundation_evidence'] = {
        'schema':R190_SCHEMA,'goal':goal,'dimension':key,'block_start':start,
        'latest_completed_sunday':latest_completed_sun.isoformat(),
        'completed_weeks_checked':len(reviewed), 'completed_weeks':reviewed,
        'earned':bool(earned),'earned_windows':earned,
        'reason_code': 'COMPLETED_MICROCYCLE_DOSE_PROGRESS_CONFIRMED' if earned else 'NO_COMPLETED_MICROCYCLE_PROGRESS_CONFIRMED',
        'rule': 'The week-local load budget resets; within-block earned progression is not revoked merely by Monday rollover. Only complete windows supported by native cycling power-zone data count.'}
    return ledger

def build_question_state(goal_key, evidence_ledger=None, performance_evidence=None, roadmap=None, microcycle_ledger=None, aerobic_metabolic_range=None, now=None):
    """Preserve completed-block dose only when R111 breadth is satisfied.

    R191 repairs the R190 unordered append that caused a false
    QUESTION_STATE_CLOSED_SET_MISMATCH when later dimensions had closed before
    SPECIFIC_DOSE/TIME_AT_PRESSURE was carried forward. The canonical graph
    order remains the sole source for open/closed lists and cannot be bypassed.
    """
    out = _build_question_state_r189_final(goal_key, evidence_ledger, performance_evidence, roadmap, microcycle_ledger, aerobic_metabolic_range, now=now)
    if not isinstance(out, dict):
        return out
    led = microcycle_ledger if isinstance(microcycle_ledger, dict) else {}
    hist = led.get('block_foundation_evidence') or {}
    hist = hist if isinstance(hist, dict) else {}
    dims = out.get('dimensions') or {}
    dims = dims if isinstance(dims, dict) else {}
    goal = str(goal_key or 'ENDURANCE_BASE').upper()
    graph = _V503_GOAL_GRAPH.get(goal) or _V503_GOAL_GRAPH['ENDURANCE_BASE']
    seq = list(graph.get('sequence') or ())
    key = hist.get('dimension')
    current = dims.get(key) if isinstance(key, str) else None
    facet = out.get('facet_profile') or {}
    facet = facet if isinstance(facet, dict) else {}
    required = int(_rhythm_num(facet.get('required_count')) or 0)
    covered = int(_rhythm_num(facet.get('coverage_count')) or 0)
    # R111 requires goal-duration facet coverage as well as earned dose.
    # An earned completed week is evidence, never permission to skip breadth.
    breadth_satisfied = not required or covered >= required
    valid = (key in seq and isinstance(current, dict)
             and hist.get('schema') == R190_SCHEMA
             and hist.get('goal') == goal
             and hist.get('block_start') == out.get('scope_start')
             and hist.get('earned') is True and bool(hist.get('earned_windows')))
    if valid and not current.get('exit_satisfied') and breadth_satisfied:
        previous = {'state': current.get('state'), 'exit_satisfied': current.get('exit_satisfied'),
                    'reason': current.get('reason')}
        witness = hist['earned_windows'][-1]
        dims[key] = {
            **current, 'state': 'SUPPORTED', 'exit_satisfied': True,
            'reason': f"Completed active-block dose progression was established in {witness['window_start']}–{witness['window_end']}; Monday's new incomplete dose window does not revoke it.",
            'last_evidence_date': witness['window_end'],
            'source': 'COMPLETED_ACTIVE_BLOCK_NATIVE_POWER_ZONE_WINDOWS',
            'foundation_continuity': {'schema': R190_SCHEMA, 'reason_code': 'WEEK_ROLLOVER_NOT_BLOCK_REGRESSION',
                                      'live_week_record': previous, 'completed_witness': copy.deepcopy(witness)}}

    # Do not append; doing so fails whenever a subsequent question is closed.
    out['open_dimensions'] = [d for d in seq if d not in {'REVALIDATION','EXTENSION_OR_REVALIDATION'}
                              and not bool((dims.get(d) or {}).get('exit_satisfied'))]
    out['closed_dimensions'] = [d for d in seq if bool((dims.get(d) or {}).get('exit_satisfied'))]
    out['foundation_dose_accountability'] = copy.deepcopy(hist)
    return out

def _v190_transition(previous,current,now=None):
    old=_v190_summary_state(previous);new=_v190_summary_state(current)
    if not old['goal'] or not new['goal'] or old['goal']!=new['goal'] or old['block_start']!=new['block_start']:
        code='TRAINING_PLAN_SCOPE_CHANGED'
        if old==new:return None
    else:
        changes=[]
        if old['stage']!=new['stage']:changes.append('ROADMAP_STAGE_CHANGED')
        for k in sorted(set(old['questions'])|set(new['questions'])):
            b=old['questions'].get(k) or {};a=new['questions'].get(k) or {}
            if b.get('closed')!=a.get('closed'):changes.append('QUESTION_EXIT_CHANGED:'+k)
        if old['science_focus']!=new['science_focus']:changes.append('SCIENCE_FOCUS_CHANGED')
        if (old['hard_plan'] or {}).get('workout_id')!=(new['hard_plan'] or {}).get('workout_id'):changes.append('WORKOUT_ID_CHANGED')
        if old['quality_state']!=new['quality_state']:changes.append('QUALITY_GATE_CHANGED')
        if old['quality_at']!=new['quality_at']:changes.append('QUALITY_TIME_CHANGED')
        if not changes:return None
        code=';'.join(changes)
    doseprev=((previous.get('microcycle_ledger') or {}).get('dose_context') or {})
    dosenew=((current.get('microcycle_ledger') or {}).get('dose_context') or {})
    reason='TRAINING_DIRECTION_OR_BLOCK_SCOPE_CHANGED' if code=='TRAINING_PLAN_SCOPE_CHANGED' else 'DECISION_CHANGED_WITH_OBSERVED_EVIDENCE'
    if 'ROADMAP_STAGE_CHANGED' in code and doseprev.get('window_start')!=dosenew.get('window_start'):
        reason='MICROCYCLE_ROLLOVER_CHECK_FOUNDATION_CONTINUITY'
    if 'QUALITY_GATE_CHANGED' in code:
        reason='QUALITY_GATE_OR_RECOVERY_REEVALUATED'
    record={'schema':R190_SCHEMA,'time_local':(now or get_rome_now()).isoformat(timespec='seconds'),
            'reason_code':reason, 'change_codes':code.split(';'), 'previous':old,'current':new,
            'delta_evidence':{'prior_dose_context':doseprev,'current_dose_context':dosenew},
            'no_invented_cause':True}
    record['transition_id']=hashlib.sha256(json.dumps([old,new],sort_keys=True,default=str).encode()).hexdigest()[:20]
    return record

def save_latest_report(data, now=None, owner_user_id=None, qa_replay_bundle=None):
    owner_id=_data_owner_user_id(owner_user_id)
    before=load_latest_report(include_payload=True,owner_user_id=owner_id)
    prior=before.get('data') if isinstance(before,dict) else None
    incoming=data if isinstance(data,dict) else {}
    if 'nova_reasoning_trace' not in incoming:
        incoming['nova_reasoning_trace']=_v190_reasoning_trace(incoming)
    if 'nova_explanation' not in incoming:
        incoming['nova_explanation']=_v190_human_explanation(incoming)
    oldjournal=(prior or {}).get('nova_transition_history') or []
    journal=[copy.deepcopy(row) for row in oldjournal if isinstance(row,dict)][-31:]
    transition=_v190_transition(prior,incoming,now=now) if isinstance(prior,dict) else None
    if transition and (not journal or journal[-1].get('transition_id')!=transition['transition_id']):
        journal.append(transition)
    incoming['nova_transition_history']=journal[-32:]
    incoming['nova_transition_latest']=transition if transition else (journal[-1] if journal else None)
    # Retain the full curated decision state of multiple consecutive snapshots,
    # including decisions which did NOT change the workout. No recursion or raw
    # athlete identifiers; neither passwords nor complete wellness inputs enter.
    old_traces=(prior or {}).get('nova_reasoning_history') or []
    trace_history=[copy.deepcopy(v) for v in old_traces if isinstance(v,dict)][-23:]
    current_trace=incoming['nova_reasoning_trace']
    if not trace_history or trace_history[-1].get('trace_id')!=current_trace.get('trace_id'):
        trace_history.append(copy.deepcopy(current_trace))
    incoming['nova_reasoning_history']=trace_history[-24:]
    incoming['nova_observability_policy']={'schema':R190_SCHEMA,'per_athlete':True,
        'bounded_transitions':32,'bounded_decision_traces':24,
        'current_snapshot_trace_always_available':True,
        'no_ai_calls_required':True,'source':'ACTUAL_SNAPSHOT_DECISION_STATES',
        'historical_before_r190_incomplete':True}
    # R195.0 shadow integrity compares with this athlete's previous report.
    # No veto/replan, no additional DB query, no external service, no athlete UI.
    # If diagnostics unexpectedly fail, preserve the R194.3 save path.
    try:
        incoming['decision_integrity_audit'] = _v195_analyze(prior, incoming, now=now or get_rome_now())
    except Exception as exc:
        incoming['decision_integrity_audit'] = {'schema':R195_SCHEMA,
            'mode':'SHADOW_READ_ONLY', 'state':'UNASSESSED',
            'issues':[{'code':'SHADOW_VALIDATOR_UNAVAILABLE','severity':'WATCH','facts':{'error_class':type(exc).__name__}}],
            'policy':'R195.0 never interrupts normal Snapshot persistence.'}
    return _save_latest_report_r189_final(incoming,now=now,owner_user_id=owner_id,qa_replay_bundle=qa_replay_bundle)

def _r192_clean_details(value):
    """Explicit allowlist: exclude user IDs, IPs, prompts, tokens and raw stack traces."""
    try:
        raw = value if isinstance(value, dict) else json.loads(value or '{}')
    except (ValueError, TypeError):
        raw = {}
    if not isinstance(raw, dict):
        return {}
    clean = {}
    for key in _R192_SAFE_FIELDS:
        val = raw.get(key)
        if isinstance(val, str):
            clean[key] = val[:220]
        elif isinstance(val, bool) or val is None:
            if val is not None:
                clean[key] = val
        elif key == 'issue_codes' and isinstance(val, list):
            clean[key] = [str(v)[:90] for v in val[:16] if isinstance(v, str)]
    return clean

def _r192_diagnostic_events(owner_id, query=''):
    """Owner's own Snapshot audits; no other athlete's data without QA consent."""
    q = str(query or '').strip()[:90]
    placeholders = ', '.join('?' for _ in _R192_DIAGNOSTIC_ACTIONS)
    sql = (
        'SELECT created_at_utc, action, details_json FROM audit_log '
        'WHERE (actor_user_id=? OR target_user_id=?) '
        'AND action IN (' + placeholders + ')'
    )
    params = [str(owner_id), str(owner_id), *_R192_DIAGNOSTIC_ACTIONS]
    if q:
        sql += ' AND (details_json LIKE ? OR action LIKE ?)'
        params.extend(['%' + q.replace('\\', '\\\\').replace('%', '\\%').replace('_', '\\_') + '%',
                       '%' + q.replace('\\', '\\\\').replace('%', '\\%').replace('_', '\\_') + '%'])
        # Explicit SQL ESCAPE keeps diagnostic searches literal and bounded.
        sql = sql.replace('details_json LIKE ? OR action LIKE ?', "details_json LIKE ? ESCAPE '\\' OR action LIKE ? ESCAPE '\\'")
    sql += ' ORDER BY created_at_utc DESC LIMIT 100'
    rows = _db_execute(sql, tuple(params), fetch=True)
    events = []
    for row in rows:
        details = _r192_clean_details(row.get('details_json'))
        event = {
            'created_at_utc': str(row.get('created_at_utc') or '')[:40],
            'action': str(row.get('action') or '')[:80],
            'details': details,
        }
        event['diagnostic_json'] = json.dumps(event, ensure_ascii=False, indent=2)
        events.append(event)
    return events

R194_ENABLE_ON_VISIT_CLEANUP = False  # Deliberately no opt-in: legacy path bypasses verified backups.



def _r193_limits(now=None):
    now = now or datetime.now(timezone.utc)
    if now.tzinfo is None:
        now = now.replace(tzinfo=timezone.utc)
    today_rome = now.astimezone(ROME_TZ).date()
    def stamp(days):
        return (now - timedelta(days=days)).replace(microsecond=0).isoformat().replace('+00:00', 'Z')
    return {
        'today': today_rome.isoformat(),
        'logs': (today_rome - timedelta(days=R193_DAILY_LOG_DAYS)).isoformat(),
        'notes': (today_rome - timedelta(days=R193_DAILY_NOTE_DAYS)).isoformat(),
        'audit_routine': stamp(R193_AUDIT_ROUTINE_DAYS),
        'audit_security': stamp(R193_AUDIT_SECURITY_DAYS),
        'usage': stamp(R193_USAGE_DAYS),
        'usage_profile': stamp(R193_USAGE_PROFILE_DAYS),
        'session': stamp(R193_SESSION_DAYS),
        'run_history': stamp(R193_RUN_HISTORY_DAYS),
        'retry_after': stamp(0),
        'now': stamp(0),
    }


def _r193_targets(kind, dates):
    # ALL table names and predicates are hardcoded below. Never accept SQL, table names,
    # filters or retention intervals from a request or an admin input.
    local = (('daily_logs', 'local_date < ?', (dates['logs'],), None),) if R194_ENABLE_DAILY_LOG_RETENTION else ()
    # Free-text notes are not interchangeable with throwaway technical events.
    # Disable the 21-day note redaction by default until a Nova memory policy exists.
    if R194_ENABLE_NOTE_REDACTION:
        local += (('daily_logs', "local_date < ? AND note IS NOT NULL AND note <> ''", (dates['notes'],), 'note'),)

    if kind == 'personal':
        return local
    return local + (
        ('audit_log', 'created_at_utc < ? AND action IN (' + ','.join(['?'] * len(R193_ROUTINE_ACTIONS)) + ')',
         (dates['audit_routine'],) + R193_ROUTINE_ACTIONS, None),
        ('audit_log', 'created_at_utc < ? AND action NOT IN (' + ','.join(['?'] * len(R193_ROUTINE_ACTIONS)) + ')',
         (dates['audit_security'],) + R193_ROUTINE_ACTIONS, None),
        ('auth_sessions', '(revoked_at_utc IS NOT NULL AND revoked_at_utc < ?) OR last_seen_at_utc < ?',
         (dates['session'], dates['session']), None),
        ('ai_usage_events', 'created_at_utc < ?', (dates['usage'],), None),
        ('ai_usage_events', "created_at_utc < ? AND prompt_profile_json IS NOT NULL AND prompt_profile_json <> ''",
         (dates['usage_profile'],), 'prompt_profile_json'),
        ('storage_maintenance_runs', 'updated_at_utc < ?', (dates['run_history'],), None),
    )


def _r193_prune_target(conn, table, where, params, null_field=None, dry_run=False):
    # Targeted 200-row transactions with hard upper bound; no VACUUM FULL on Neon.
    # This avoids long-running locks, high write amplification and oversized requests.
    is_pg = _db_is_postgres()
    def sql_placeholders(query):
        return query.replace('?', '%s') if is_pg else query
    cur = conn.cursor()
    if dry_run:
        try:
            cur.execute(sql_placeholders('SELECT COUNT(*) FROM ' + table + ' WHERE ' + where), params)
            row = cur.fetchone()
            return {'matched': int(row[0]), 'changed': 0, 'more': False}
        finally:
            cur.close()
    n = 0
    try:
        while n < R193_MAX_ROWS_PER_TARGET:
            limit = min(R193_BATCH_SIZE, R193_MAX_ROWS_PER_TARGET - n)
            # DB stores text UUID ids; select by primary key, then execute a bounded mutation.
            predicate = 'id IN (SELECT id FROM ' + table + ' WHERE ' + where + ' LIMIT ?)'
            if table == 'storage_maintenance_runs':
                predicate = 'run_key IN (SELECT run_key FROM ' + table + ' WHERE ' + where + ' LIMIT ?)'
            cmd = ('UPDATE ' + table + ' SET ' + null_field + '=NULL WHERE ' + predicate
                   if null_field else 'DELETE FROM ' + table + ' WHERE ' + predicate)
            cur.execute(sql_placeholders(cmd), (*params, limit))
            changed = max(0, int(cur.rowcount or 0))
            conn.commit()
            n += changed
            if changed < limit:
                break
        more = False
        if n >= R193_MAX_ROWS_PER_TARGET:
            cur.execute(sql_placeholders('SELECT 1 FROM ' + table + ' WHERE ' + where + ' LIMIT 1'), params)
            more = cur.fetchone() is not None
        return {'matched': n, 'changed': n, 'more': more}
    except Exception:
        conn.rollback()
        raise
    finally:
        cur.close()


def _r193_prune_database(kind, dates, dry_run=False):
    if kind not in ('product','personal'):
        raise ValueError('Unsupported storage backend')
    if kind == 'personal' and (not OWNER_PERSONAL_DATABASE_URL or OWNER_PERSONAL_DATABASE_URL == DATABASE_URL):
        raise RuntimeError('Personal Neon must be configured and physically separate from Product Neon')
    if kind == 'product' and not _db_is_postgres():
        raise RuntimeError('Production maintenance requires Product Neon PostgreSQL; SQLite fallback refused')
    if kind == 'personal' and psycopg2 is None:
        raise RuntimeError('Personal Neon PostgreSQL driver missing')
    if kind == 'personal':
        _ensure_owner_personal_v4857_extensions()
    conn = _db_connect() if kind == 'product' else _owner_personal_db_connect()
    try:
        results = {}
        for table, where, params, null_field in _r193_targets(kind, dates):
            key = table + ('.'+null_field if null_field else '.rows')
            if key in results:
                key += '.security'
            results[key] = _r193_prune_target(conn, table, where, params, null_field, dry_run)
        return results
    finally:
        conn.close()


def _r194_check_both_backends():
    """Fail closed before the first DELETE; not a distributed transaction guarantee."""
    if not OWNER_PERSONAL_DATABASE_URL or OWNER_PERSONAL_DATABASE_URL == DATABASE_URL:
        return 'PERSONAL_DATABASE_MISSING_OR_SHARED'
    if not _db_is_postgres() or psycopg2 is None:
        return 'POSTGRES_BACKEND_UNAVAILABLE'
    for kind, connect in (('product', _db_connect), ('personal', _owner_personal_db_connect)):
        conn = None
        try:
            conn = connect()
            cur = conn.cursor()
            try:
                cur.execute('SELECT 1')
                if cur.fetchone()[0] != 1:
                    return 'DATABASE_PREFLIGHT_FAILED_' + kind.upper()
            finally:
                cur.close()
        except Exception:
            app.logger.exception('R194 %s preflight failed', kind)
            return 'DATABASE_PREFLIGHT_FAILED_' + kind.upper()
        finally:
            if conn is not None:
                conn.close()
    return None


def _r194_backup_status(dates=None, now=None):
    """Machine-attested encrypted export status; NOT a successful isolated restore drill."""
    now = now or datetime.now(timezone.utc)
    dates = dates or _r193_limits(now)
    try:
        rows = _db_execute('SELECT updated_at_utc,status,details_json FROM storage_maintenance_runs WHERE run_key=?',
                           ('backup:' + dates['today'],), fetch=True)
        if not rows or rows[0].get('status') != 'verified':
            return {'state':'missing', 'date':dates['today']}
        row = rows[0]
        attested = datetime.fromisoformat(str(row['updated_at_utc']).replace('Z','+00:00'))
        age = now - attested
        if age < timedelta(seconds=-300) or age > R194_BACKUP_MAX_AGE:
            return {'state':'stale', 'date':dates['today']}
        detail = json.loads(row.get('details_json') or '{}')
        if detail.get('backends') != ['product','personal'] or not str(detail.get('github_run_id','')).isdigit():
            return {'state':'invalid', 'date':dates['today']}
        return {'state':'verified_export', 'date':dates['today'], 'at_utc':row['updated_at_utc'],
                'github_run_id':str(detail['github_run_id'])}
    except Exception:
        app.logger.exception('R194 backup proof lookup failed')
        return {'state':'unavailable', 'date':dates['today']}


def _r193_acquire_daily_run(dates, dry_run):
    # One persisted run per mode per Rome day, independent of app process restarts.
    # If interrupted, allow retry only after 30 minutes; completed runs never replay.
    init_memory_db()
    key = ('preview:' if dry_run else 'live:') + dates['today']
    stamp = dates['now']
    stale = (datetime.fromisoformat(stamp.replace('Z','+00:00')) - timedelta(minutes=30)).isoformat().replace('+00:00','Z')
    conn = _db_connect()
    try:
        cur = conn.cursor()
        stmt = 'INSERT INTO storage_maintenance_runs(run_key,started_at_utc,updated_at_utc,status,details_json) VALUES(?,?,?,?,?) ON CONFLICT(run_key) DO NOTHING'
        cur.execute(_db_sql(stmt), (key, stamp, stamp, 'running', '{}'))
        inserted = cur.rowcount == 1
        conn.commit()
        if inserted:
            return key, True
        cur.execute(_db_sql('UPDATE storage_maintenance_runs SET started_at_utc=?,updated_at_utc=?,status=?,details_json=? WHERE run_key=? AND ((status=? AND updated_at_utc < ?) OR (status=? AND started_at_utc < ?))'),
                    (stamp,stamp,'running','{}',key,'failed',stale,'running',stale))
        claimed = cur.rowcount == 1
        conn.commit()
        return key, claimed
    finally:
        conn.close()


def _r193_finalize_run(key, status, payload, when):
    body = json.dumps(payload, separators=(',', ':'), ensure_ascii=True)[:8000]
    _db_execute('UPDATE storage_maintenance_runs SET status=?,updated_at_utc=?,details_json=? WHERE run_key=?',
                (status, when, body, key))


def _r193_run_retention(now=None, dry_run=None):
    dates = _r193_limits(now)
    if dry_run is None:
        dry_run = R193_DRY_RUN
    # Never return an unconditional 200 for a previous run of unknown status.
    key, claimed = _r193_acquire_daily_run(dates, dry_run)
    if not claimed:
        rows = _db_execute('SELECT status FROM storage_maintenance_runs WHERE run_key=?', (key,), fetch=True)
        previous = rows[0].get('status') if rows else 'unknown'
        return {'ok': previous in ('completed','preview'), 'status': 'already_processed_or_in_progress',
                'previous_status': previous, 'run_key': key, 'dry_run': dry_run}
    results = {}
    errors = []
    incomplete = False
    try:
        problem = _r194_check_both_backends()
        backup_status = _r194_backup_status(dates, now or datetime.now(timezone.utc)) if not dry_run else {'state':'preview_not_required'}
        if not problem and not dry_run and backup_status['state'] != 'verified_export':
            problem = 'BACKUP_EXPORT_NOT_VERIFIED'
        if problem:
            errors = [{'code':problem, 'backup_state':backup_status['state']}]
            _r193_finalize_run(key, 'failed', {'state':'failed','errors':errors,'change_count':0}, _now_utc_text())
            return {'ok':False,'status':'failed','run_key':key,'dry_run':dry_run,'errors':errors,'change_count':0}
        for kind in ('product','personal'):
            try:
                summary = _r193_prune_database(kind, dates, dry_run)
                results[kind] = summary
                incomplete |= any(bool(row.get('more')) for row in summary.values())
            except Exception as exc:
                errors.append({'database': kind, 'error_class': type(exc).__name__})
                results[kind] = {'state':'error','error_class':type(exc).__name__}
                app.logger.exception('R193 %s retention failed', kind)
        state = 'failed' if errors else ('partial' if incomplete else ('preview' if dry_run else 'completed'))
        payload = {'state': state,'dry_run': dry_run, 'databases': results, 'errors': errors,
                   'change_count':sum(int(r.get('changed',0)) for db in results.values() if isinstance(db,dict) for r in db.values() if isinstance(r,dict)),
                   'would_change_count':sum(int(r.get('matched',0)) for db in results.values() if isinstance(db,dict) for r in db.values() if isinstance(r,dict)) if dry_run else None,
                   'retention': {'daily_logs_days':R193_DAILY_LOG_DAYS,'daily_notes_days':R193_DAILY_NOTE_DAYS,
                                 'routine_audit_days':R193_AUDIT_ROUTINE_DAYS,'security_audit_days':R193_AUDIT_SECURITY_DAYS}}
        _r193_finalize_run(key, state, payload, _now_utc_text())
        return {'ok': not bool(errors), 'status':state,'run_key':key,'dry_run':dry_run,
                'databases':results,'errors':errors}
    except Exception:
        app.logger.exception('R193 retention orchestration aborted')
        try:
            _r193_finalize_run(key,'failed',{'error_class':'OrchestrationError'},_now_utc_text())
        except Exception:
            app.logger.exception('R193 could not persist failed maintenance status')
        raise


def _r193_db_bytes(conn):
    cur = conn.cursor()
    try:
        cur.execute('SELECT pg_database_size(current_database())')
        return int(cur.fetchone()[0])
    finally:
        cur.close()


def _r193_storage_overview():
    # Admin only. Never return credentials, user rows or any athlete note.
    init_memory_db()
    out = {'product':{'state':'unavailable'}, 'personal':{'state':'not_configured'}, 'recent_runs':[],
           'backup_receipt':_r194_backup_status(), 'note_cleanup_enabled':R194_ENABLE_NOTE_REDACTION, 'diary_cleanup_enabled':R194_ENABLE_DAILY_LOG_RETENTION,
           'legacy_on_visit_cleanup':R194_ENABLE_ON_VISIT_CLEANUP}
    for kind in ('product','personal'):
        if kind == 'personal' and not OWNER_PERSONAL_DATABASE_URL:
            continue
        conn = None
        try:
            if kind == 'personal':
                if OWNER_PERSONAL_DATABASE_URL == DATABASE_URL:
                    raise RuntimeError('Product and Personal database URLs must differ')
                conn = _owner_personal_db_connect()
            else:
                conn = _db_connect()
            used = _r193_db_bytes(conn)
            quota = DATABASE_STORAGE_LIMIT_BYTES
            tables=[]
            try:
                cur = conn.cursor()
                cur.execute('SELECT relname,pg_total_relation_size(relid) AS bytes FROM pg_stat_user_tables ORDER BY pg_total_relation_size(relid) DESC LIMIT 8')
                tables=[{'name':row[0],'human':_human_bytes(int(row[1]))} for row in cur.fetchall()]
                cur.close()
            except Exception:
                conn.rollback()  # Optional table ranking must never hide database size metrics.
            out[kind]={'state':'available','bytes':used,'human':_human_bytes(used), 'tables':tables,
                       'configured_quota_bytes':quota,'quota_human':_human_bytes(quota),
                       'approx_percent':round(100*used/quota,1) if quota else None}
        except Exception as exc:
            out[kind]={'state':'unavailable','error_class':type(exc).__name__}
        finally:
            if conn is not None:
                conn.close()
    try:
        rows = _db_execute('SELECT run_key,started_at_utc,updated_at_utc,status,details_json FROM storage_maintenance_runs WHERE run_key LIKE ? OR run_key LIKE ? ORDER BY started_at_utc DESC LIMIT 8',('live:%','preview:%'),fetch=True)
        for row in rows:
            try:
                detail=json.loads(row.get('details_json') or '{}')
            except Exception:
                detail={}
            out['recent_runs'].append({'run_key':row.get('run_key'),'started_at_utc':row.get('started_at_utc'),
                                       'updated_at_utc':row.get('updated_at_utc'),'status':row.get('status'),
                                       'details':detail})
    except Exception as exc:
        out['history_error_class']=type(exc).__name__
    return out

@app.route('/internal/maintenance/retention', methods=['POST'])
def r193_maintenance_cron():
    # Explicit machine authentication, distinct from browser cookies/CSRF.
    # No secret in URL, JSON reply, logs or Product Neon. Missing config always denies.
    actual = request.headers.get('X-TheLab-Maintenance-Key') or ''
    if not R193_MAINTENANCE_ENABLED or len(R193_MAINTENANCE_KEY)<32 or not hmac.compare_digest(actual,R193_MAINTENANCE_KEY):
        return Response(status=404, headers={'Cache-Control':'no-store'})
    if not _db_is_postgres():
        return jsonify({'ok':False,'status':'product_database_not_postgres'}),503

@app.route('/internal/maintenance/backup-attest', methods=['POST'])
def r194_backup_attest():
    provided = request.headers.get('X-TheLab-Backup-Key') or ''
    if len(R194_BACKUP_ATTEST_KEY) < 32 or not hmac.compare_digest(provided, R194_BACKUP_ATTEST_KEY):
        return Response(status=404, headers={'Cache-Control':'no-store'})
    if not _db_is_postgres():
        return jsonify({'ok':False,'code':'DATABASE_UNAVAILABLE'}),503
    payload = request.get_json(silent=True)
    if not isinstance(payload,dict):
        return jsonify({'ok':False,'code':'INVALID_PROOF'}),400
    run_id = str(payload.get('github_run_id') or '')
    backends = payload.get('backends')
    digests = payload.get('digests')
    if (not run_id.isascii() or not run_id.isdigit() or not 1 <= len(run_id) <= 30
        or backends != ['product','personal'] or not isinstance(digests,dict)
        or set(digests) != {'product','personal'}
        or any(not isinstance(digests[k],str) or len(digests[k]) != 64
               or any(c not in '0123456789abcdef' for c in digests[k]) for k in backends)):
        return jsonify({'ok':False,'code':'INVALID_PROOF'}),400
    # A successful signature proves trusted GitHub workflow attested both archives.
    # Never implies successful isolated restoration or email receipt by end user.
    dates = _r193_limits()
    when = _now_utc_text()
    record = json.dumps({'backends':backends,'github_run_id':run_id,'sha256_encrypted':digests,
                         'verification':'encrypted_export_gh_artifact_and_email'},sort_keys=True,separators=(',',':'))
    try:
        _db_execute('INSERT INTO storage_maintenance_runs(run_key,started_at_utc,updated_at_utc,status,details_json) '
                    'VALUES(?,?,?,?,?) ON CONFLICT(run_key) DO UPDATE SET updated_at_utc=excluded.updated_at_utc,'
                    'status=excluded.status,details_json=excluded.details_json',
                    ('backup:'+dates['today'],when,when,'verified',record))
    except Exception:
        app.logger.exception('R194 backup attestation storage unavailable')
        return jsonify({'ok':False,'code':'RECEIPT_STORAGE_FAILED'}),503
    return jsonify({'ok':True,'status':'verified_export','date':dates['today']}),200

# R194.3 — explicit, provenance-backed power prescriptions for canonical repeat intervals.
# Calibrate from verified comparable repeated work first. Fresh, valid FTP can
# supply a conservative initial training range, but is NEVER labelled observed.
# Do not confuse an individual power-curve best with repeated-work evidence.
R1943_SCHEMA = "V4.9.73-R194.3-POWER-CONTRACT-1"


def _v1943_finite(v, low=0.0, high=1e7):
    try:
        f = float(v)
        return f if math.isfinite(f) and low <= f <= high else None
    except (TypeError, ValueError, OverflowError):
        return None


def _v1943_band(template, evidence_ledger=None, ftp_anchor=None, as_of_date=None, dimension=None):
    """Compute an *explicit training prescription* distinct from PB/reference watts.

    Only validated repeated efforts of comparable duration/reps are direct support.
    FTP-only targets are conservative starter estimates, NOT achieved/validated
    repeatability. No valid data => no fabricated numeric hard-work prescription.
    """
    t = template if isinstance(template, dict) else {}
    pattern = str(t.get('pattern') or '').upper()
    if pattern not in {'STANDARD', 'SUSTAINED', 'SHORT_POWER'}:
        return {'schema':R1943_SCHEMA,'state':'UNSUPPORTED_STRUCTURE'}
    work_s = _v1943_finite(t.get('work_s'), 60, 1800)
    reps = int(_v1943_finite(t.get('reps'), 1, 20) or 0)
    goal = str(t.get('goal') or '').upper()
    if not (work_s and reps >= 2):
        return {'schema':R1943_SCHEMA,'state':'UNSUPPORTED_STRUCTURE'}
    valid_ftp = None
    ftp = ftp_anchor if isinstance(ftp_anchor, dict) else {}
    if ftp.get('available') is True and str(ftp.get('status') or '').upper() == 'VALID':
        valid_ftp = _v1943_finite(ftp.get('prescription_watts'),80,600)
    candidates=[]
    events=((evidence_ledger or {}).get('repeatability') or {}).get('events') or []
    for e in events:
        if not isinstance(e,dict) or e.get('strong') is not True:
            continue
        d = str(e.get('date') or '')[:10]
        try:
            elapsed=(date.fromisoformat(str(as_of_date)[:10])-date.fromisoformat(d)).days
            if elapsed < 0 or elapsed > 42: continue
        except (ValueError, TypeError):
            continue
        secs=_v1943_finite(e.get('interval_secs'),60,1800)
        er=int(_v1943_finite(e.get('reps'),1,30) or 0)
        watts=_v1943_finite(e.get('avg_watts'),70,1200)
        cv=_v1943_finite(e.get('cv_pct'),0,100)
        if secs is None or watts is None or cv is None or cv>6 or er < reps:
            continue
        # Direct transfer allowed only from SAME or LONGER interval work.
        if not (work_s*0.8 <= secs <= work_s*1.4): continue
        truth=str(e.get('activity_truth_source') or e.get('source') or '').upper()
        if truth not in {'DEVICE_LAPS','ICU_INTERVALS_CANONICAL'}: continue
        if valid_ftp is not None and not (0.65*valid_ftp <= watts <= 1.8*valid_ftp):continue
        candidates.append((abs(secs-work_s),elapsed,e,watts,secs))
    candidates.sort(key=lambda c:(c[0],c[1],str(c[2].get('activity_id') or '')))
    if candidates:
        _,age,e,obs,source_seconds=candidates[0]
        # Conservative, explicit coaching policy: preserve demonstrated repeated
        # intensity while slightly easing after a new preload/variable-load lever.
        ratio = .98 if str(dimension or '').upper() in {'RACE_TRANSFER','DURABILITY'} else 1.0
        if source_seconds < work_s:ratio *= .98  # longer upcoming reps: conservative extrapolation
        target=round((obs*ratio)/5)*5
        return {
            'schema':R1943_SCHEMA,'state':'PRESCRIBED',
            'confidence':'HIGH' if source_seconds>=work_s else 'MODERATE',
            'source':'COMPARABLE_VERIFIED_REPEATED_WORK',
            'activity_id':str(e.get('activity_id') or ''),'activity_date':str(e.get('date') or ''),
            'observed_repeated_avg_watts':round(obs,1),
            'target_watts':int(target),'low_watts':int(target-5),'high_watts':int(target+5),
            'reps':reps,'work_seconds':int(work_s),'evidence_interval_seconds':int(source_seconds),'evidence_age_days':age,
        }
    if valid_ftp is not None:
        # Initial reference bands require a truly VALID FTP. These are declared
        # provisional training-zone assumptions; never imply they were validated
        # by an executed identical workout. Subsequent real repeats must replace them.
        multipliers = {'POWER_5':1.10,'VO2MAX':1.08,'POWER_20':0.97,
                       'TIME_TRIAL':0.97,'POWER_1':1.22}
        mult=multipliers.get(goal)
        if mult is not None:
            if work_s >= 600 and goal in {'POWER_5','VO2MAX'}: mult=min(mult,1.03)
            if str(dimension or '').upper() in {'RACE_TRANSFER','DURABILITY'}: mult*=.98
            target=round(valid_ftp*mult/5)*5
            return {
                'schema':R1943_SCHEMA,'state':'PRESCRIBED','confidence':'PROVISIONAL',
                'source':'VALID_FTP_STARTER_POLICY_NOT_OBSERVED_REPEATABILITY',
                'ftp_watts':valid_ftp,'target_watts':int(target),
                'low_watts':int(target-5),'high_watts':int(target+5),
                'reps':reps,'work_seconds':int(work_s),
            }
    return {'schema':R1943_SCHEMA,'state':'NO_DEFENSIBLE_POWER_TARGET',
            'reason':'NO_RECENT_COMPARABLE_COMPLETED_REPS_OR_VALID_FTP'}


_v1943_build_before = build_nova_prescription

def build_nova_prescription(nova_decision, coach_clock, adaptive_roadmap=None, microcycle_ledger=None, evidence_ledger=None, power_model=None, ftp_anchor=None, aerobic_metabolic_range=None, training_definitions=None):
    out=_v1943_build_before(nova_decision,coach_clock,adaptive_roadmap=adaptive_roadmap,
        microcycle_ledger=microcycle_ledger,evidence_ledger=evidence_ledger,power_model=power_model,
        ftp_anchor=ftp_anchor,aerobic_metabolic_range=aerobic_metabolic_range,training_definitions=training_definitions)
    if not isinstance(out,dict):return out
    slots=list((coach_clock or {}).get('next_slots') or [])
    for idx,raw in enumerate(out.get('raw_sessions') or []):
        if not isinstance(raw,dict) or not str(raw.get('intensity') or '').upper().startswith('HARD'):continue
        wid=raw.get('_tl_nova_workout_id') or raw.get('_tl_nova_contract_id')
        template=next((t for t in _NOVA_WORKOUT_LIBRARY if t.get('id')==wid),None)
        if not template:continue
        slot_date=(slots[idx].get('date') if idx<len(slots) and isinstance(slots[idx],dict) else None)
        as_of=slot_date or ((coach_clock or {}).get('date') or datetime.now(timezone.utc).date().isoformat())
        raw['_tl_v1943_power_band']=_v1943_band(template,evidence_ledger,ftp_anchor,as_of,raw.get('_tl_nova_dimension') or out.get('dimension'))
    future=out.get('future_quality_candidate')
