<script id="v19697-feedback-controls-js">
(function () {
  var panel = document.querySelector('[data-v19697-feedback]');
  if (!panel) return;
  var names = {
    rpe: ['VERY EASY','EASY','EASY','MODERATE','MODERATE','MODERATE','HARD','HARD','VERY HARD','MAXIMAL'],
    execution: ['POOR','POOR','LOW','MIXED','MIXED','SOLID','GOOD','GOOD','EXCELLENT','EXCELLENT'],
    after: ['EXHAUSTED','EXHAUSTED','VERY TIRED','TIRED','FATIGUED','OKAY','OKAY','GOOD','FRESH','FRESH']
  };
  panel.querySelectorAll('[data-v19697-metric]').forEach(function (metric) {
    var field = metric.querySelector('[data-v19697-input]');
    var reading = metric.querySelector('[data-v19697-reading]');
    var steps = Array.prototype.slice.call(metric.querySelectorAll('[data-score]'));
    var category = metric.getAttribute('data-metric');
    function setValue(value) {
      var number = Number(value);
      if (!Number.isInteger(number) || number < 1 || number > 10) number = 0;
      field.value = number ? String(number) : '';
      steps.forEach(function (step) {
        var n = Number(step.getAttribute('data-score'));
        step.setAttribute('aria-pressed', n === number ? 'true' : 'false');
        step.setAttribute('data-filled', number > 0 && n <= number ? 'true' : 'false');
      });
      reading.textContent = number ? (number + '/10 · ' + names[category][number - 1]) : 'NOT RATED';
    }
    steps.forEach(function (step, index) {
      step.addEventListener('click', function () { setValue(index + 1); });
      step.addEventListener('keydown', function (event) {
        var target = index;
        if (event.key === 'ArrowRight' || event.key === 'ArrowUp') target = Math.min(9, index + 1);
        else if (event.key === 'ArrowLeft' || event.key === 'ArrowDown') target = Math.max(0, index - 1);
        else if (event.key === 'Home') target = 0;
        else if (event.key === 'End') target = 9;
        else return;
        event.preventDefault();
        setValue(target + 1);
        steps[target].focus();
      });
    });
    var clear = metric.querySelector('[data-v19697-clear]');
    if (clear) clear.addEventListener('click', function () { setValue(0); });
    setValue(field.value);
  });
})();
</script>
