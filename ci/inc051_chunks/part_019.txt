def _v196931_measured_dose_ready(road,dec):
    sci=(road or {}).get('scientific_progression') or {}
    dose=sci.get('dose_contract') or {}
    if (dose.get('stage')!=_R196924_ASSESSMENT or dose.get('available') is not True
            or str((dec or {}).get('action') or '').upper()!='TRAIN'
            or (dec or {}).get('hard_session_allowed') is not True):
        return False
    ref=dose.get('reference') or {};band=dose.get('prescription') or {}
    if not isinstance(ref,dict) or not isinstance(band,dict) or not ref.get('activity_id'):
        return False
    try:
        work=float(ref.get('work_avg_watts'));recover=float(ref.get('recovery_avg_watts'))
        values=[float(band[k]) for k in ('work_watts_low','work_watts_high','recovery_watts_low','recovery_watts_high')]
        reps=int(band.get('reps'));dur=int(band.get('interval_secs'));rest=int(band.get('recovery_secs'))
        if (not all(math.isfinite(z) for z in [work,recover]+values)
             or not (work>recover>=0 and 0<values[0]<values[1]<work
                 and 0<=values[2]<values[3]<values[0]
                 and 2<=reps<=20 and 30<=dur<=1800 and 20<=rest<=1800)):
            return False
    except (TypeError,ValueError,KeyError,OverflowError):
        return False
    return True

def _v196932_athlete_explicitly_unavailable(slot,window=None):
    """The coach may choose a slot, but may not override an actual 'no'."""
    for s in (slot,window):
        if not isinstance(s,dict):continue
        source=str(s.get('availability_source') or '').upper()
        if s.get('availability_confirmed') is False and (
                source.startswith('ATHLETE') or source.startswith('USER') or
                source.startswith('CALENDAR_EXPLICIT') or source.startswith('MANUAL_UNAVAILABLE')):
            return True
    return False

def _v196934_coached_dose_title(dose):
    """Use only attested existing work/recovery prescription, never infer watts."""
    if not isinstance(dose, dict) or dose.get('available') is not True:
        return None
    p = dose.get('prescription') or {}
    if not isinstance(p, dict):
        return None
    keys = ('reps', 'interval_secs', 'recovery_secs', 'work_watts_low',
            'work_watts_high', 'recovery_watts_low', 'recovery_watts_high')
    ints = {}
    for k in keys:
        try:
            v = p.get(k)
            if v is None or isinstance(v, bool):
                return None
            n = float(v)
            if not math.isfinite(n) or n != int(n):
                return None
            ints[k] = int(n)
        except (TypeError, ValueError, OverflowError):
            return None
    if (not 2 <= ints['reps'] <= 20 or not 60 <= ints['interval_secs'] <= 1800
            or not 20 <= ints['recovery_secs'] <= 1800
            or not 80 <= ints['work_watts_low'] < ints['work_watts_high'] <= 1200
            or not 40 <= ints['recovery_watts_low'] < ints['recovery_watts_high'] <= 600
            or ints['recovery_watts_high'] >= ints['work_watts_low']):
        return None
    # The structured dose (including recovery and watt bands) remains in
    # nova_workout_dose_identity / power_prescription / main_set. The athlete's
    # copied workout title is only the human-readable workout structure.
    return f"Repeatability {ints['reps']}×{ints['interval_secs']/60:g} min"

def _v117_select_quality_window(candidates, last_end, spacing, cadence, extra_delay, second_hard=False):
    """An under-evidenced weekday PRIOR cannot defer a safe candidate a week.

    Keep all upstream recovery/availability/Calendar restrictions and lower
    bounds. Preserve the native preferred physiological spacing, including any
    readiness penalty. Explicit athlete intent always outranks this heuristic.
    """
    native = _r196934_window_before(candidates, last_end, spacing, cadence,
                                    extra_delay, second_hard=second_hard)
    if not isinstance(native, tuple) or len(native) != 4:
        return native
    earliest, chosen, reason, selected_row = native
    # R196.9.48: a repeated, verified HARD DAY cadence is the coach's first
    # scheduling preference, not a small bonus outweighed by weekday scoring.
    # This operates ONLY on the upstream filtered slots and physiological
    # spacing lower bound; it cannot generate a slot or override explicit
    # athlete intent, race, injury, no-intensity or unavailability exclusions.
    try:
        cad = cadence if isinstance(cadence, dict) else {}
        gap = cad.get('dominant_gap_days')
        gap = int(gap) if gap is not None and not isinstance(gap, bool) else None
        samples = int(cad.get('sample_gaps') or 0)
        confidence = str(cad.get('gap_confidence') or '').upper()
        if (isinstance(chosen, datetime) and gap is not None and 1 <= gap <= 9
                and samples >= 3 and confidence in {'MODERATE', 'HIGH'}
                and not (isinstance(selected_row, dict) and selected_row.get('explicit'))):
            anchor = (date.fromisoformat(str(cad.get('last_hard_day'))[:10])
                      if cad.get('last_hard_day') else (last_end.date() if last_end else None))
            earliest_h = float((spacing or {}).get('earliest_hours') or 36)
            penalty_h = float(extra_delay or 0)
            if anchor and all(math.isfinite(v) and v >= 0 for v in (earliest_h, penalty_h)):
                lower = last_end + timedelta(hours=earliest_h + penalty_h) if last_end else None
                valid = [row for row in (candidates or []) if isinstance(row, dict)
                    and isinstance(row.get('dt'), datetime)
                    and (lower is None or row['dt'] >= lower)]
                # A separately preferred CALENDAR/athlete HARD remains first.
                if valid and not any(row.get('explicit') for row in valid):
                    target_day = anchor + timedelta(days=gap)
                    on_day = [row for row in valid if row['dt'].date() == target_day]
                    if on_day:
                        habituated = cad.get('hard_start_minute')
                        try:
                            habituated = float(habituated)
                            if not math.isfinite(habituated): habituated = None
                        except (TypeError, ValueError):
                            habituated = None
                        part = str(cad.get('hard_daypart') or '').upper()
                        def _rhythm_distance(row):
                            dt = row['dt']; minute = dt.hour*60 + dt.minute
                            mismatch = bool(part and ('AM' if minute < 720 else 'PM') != part)
                            return (int(mismatch), abs(minute-habituated) if habituated is not None else 0,
                                row.get('clock_index') is None, dt)
                        chosen_day = min(on_day, key=_rhythm_distance)
                        return earliest, chosen_day['dt'], 'NOVA_VERIFIED_HARD_CADENCE_PRIORITY', chosen_day
    except (TypeError, ValueError, KeyError, OverflowError):
        pass  # Preserve the tested native scheduling and all safety restrictions.
    if not (isinstance(earliest, datetime) and isinstance(chosen, datetime)
            and chosen - earliest >= timedelta(hours=72)
            and str(reason or '').startswith(('OBSERVED_HARD_', 'SPARSE_HARD_'))):
        return native
    cad = cadence or {}
    if (str(cad.get('gap_confidence') or 'LOW').upper() not in ('LOW', 'NONE', 'UNLEARNED')
            or str(cad.get('time_confidence') or 'LOW').upper() not in ('LOW', 'NONE', 'UNLEARNED')
            or str(cad.get('hard_week_confidence') or 'LOW').upper() not in ('LOW', 'NONE', 'UNLEARNED')):
        return native
    try:
        earliest_hours = float((spacing or {}).get('earliest_hours') or 36)
        preferred_hours = float((spacing or {}).get('preferred_hours') or 48)
        penalty = float(extra_delay)
        if not all(math.isfinite(n) and n >= 0 for n in (earliest_hours, preferred_hours, penalty)):
            return native
        lower = last_end + timedelta(hours=earliest_hours + penalty) if last_end else None
        preferred_lower = last_end + timedelta(hours=max(earliest_hours, preferred_hours) + penalty) if last_end else None
        rows = sorted((r for r in (candidates or []) if isinstance(r, dict)
                       and isinstance(r.get('dt'), datetime)
                       and (lower is None or r['dt'] >= lower)), key=lambda r: r['dt'])
        if not rows or any(r.get('explicit') for r in rows):
            return native
        # A statistically meaningful preferred weekday can still justify delay.
        if selected_row and isinstance(selected_row, dict):
            dt = selected_row.get('dt')
            if isinstance(dt, datetime) and _v19693_established_weekday_evidence(cad, dt.weekday()):
                return native
        target = next((r for r in rows if (preferred_lower is None or r['dt'] >= preferred_lower)
                       and r['dt'] < chosen), None)
        if target is None:
            return native
        # A scheduling prior never overrides a real unavailability, race or
        # recovery hold: candidates were filtered by the original selector.
        return earliest, target['dt'], 'R196934_WEAK_WEEKDAY_PRIOR_NO_LONG_DEFERRAL', target
    except (TypeError, ValueError, OverflowError, KeyError):
        return native

def _nova_public_workout_title(session):

# R196.9.36 · ONE ATHLETE-SCOPED STAGE CHECKPOINT + READ-ONLY OUTCOME.
# These functions DO NOT decide safety or mutate issued sessions / the ledger.
def _nova_carry_attained_stage(road, science, previous, direction):
    old = (previous or {}).get('adaptive_roadmap') or {}
    current = road if isinstance(road,dict) else {}
    goal = str((direction or {}).get('primary_goal') or '').upper()
    prior_goal = str(old.get('goal_key') or '').upper()
    prior_dir = (previous or {}).get('training_direction') or {}
    # Restarting a DISTINCT block is allowed; silent downgrade in the SAME
    # block is not. Neither a blank report nor another account is borrowed.
    if not goal or prior_goal != goal or not old:
        return {'state':'NO_COMPARABLE_PRIOR_STAGE'}
    before_start = str(prior_dir.get('started_on') or '')[:10]
    current_start = str((direction or {}).get('started_on') or '')[:10]
    if before_start != current_start:
        return {'state':'NEW_TRAINING_BLOCK','prior_start':before_start,
                'current_start':current_start}
    try:
        pi=int(old['stage_index']); ci=int(current['stage_index'])
    except (ValueError,TypeError,KeyError):
        return {'state':'STAGE_UNVERIFIED'}
    steps = current.get('steps') or []
    old_code = str((old.get('stage') or {}).get('code') or '')
    if not (0 <= pi < len(steps) and 0 <= ci < len(steps) and
            isinstance(steps[pi],dict) and old_code == str(steps[pi].get('code') or '')):
        return {'state':'STAGE_SCHEMA_CHANGED_REVIEW'}
    if ci >= pi:
        return {'state':'UNCHANGED_OR_ADVANCED','prior':pi,'current':ci}
    current['stage_index'] = pi
    current['steps'] = [dict(step,state=('BUILT' if i<pi else 'NOW' if i==pi else 'NEXT'))
                        for i,step in enumerate(steps)]
    current['stage'] = current['steps'][pi]
    if isinstance(science,dict):
        science['roadmap_stage_index'] = pi
        science['roadmap_stage_code'] = old_code
    # Stage is history, NEVER permission to bypass recovery, race or health.
    return {'state':'PRESERVED_ATTAINED_STAGE','prior':pi,
            'recomputed':ci,'hard_permission_granted':False}


def _nova_workout_execution_memory(ledger, direction, previous_blocks=None):
    """Read-only issued-vs-executed Nova ledger for future coaching/QA.

    Every row is causally linked by the persisted prescription fingerprint AND
    the exact executed activity ID; no matching by title, watts or calendar
    coincidence. Unlinked/missed workouts never receive artificial adherence.
    """
    led = ledger if isinstance(ledger,dict) else {}
    history = [r for r in led.get('prescription_history') or [] if isinstance(r,dict)]
    links = [r for r in led.get('prescription_execution_history') or [] if isinstance(r,dict)]
    blocks = {str(b.get('activity_id')):b for b in previous_blocks or []
              if isinstance(b,dict) and b.get('activity_id')}
    by_fp = {str(r.get('prescription_fingerprint')):r for r in history
             if r.get('prescription_fingerprint')}
    rows = []; seen = set()
    for link in links:
        fp = str(link.get('prescription_fingerprint') or '')
        activity_id = str(link.get('activity_id') or '')
        if not fp or not activity_id or (fp,activity_id) in seen:
            continue
        seen.add((fp,activity_id))
        plan=by_fp.get(fp)
        if not plan or str(plan.get('prescription_status') or '').upper()!='FULFILLED':
            continue
        if str(plan.get('executed_activity_id') or '') != activity_id:
            continue
        b=blocks.get(activity_id) or {}
        quality=(b.get('execution_quality') or {}).get('adherence') or {}
        structural = quality.get('score') if quality.get('available') is True else None
        rows.append({'workout_id':plan.get('nova_workout_id'),
            'planned_date':str(plan.get('date') or '')[:10],
            'executed_date':str(link.get('actual_date') or '')[:10],
            'activity_id':activity_id,'prescription_fingerprint':fp,
            'execution_match_basis':link.get('match_basis'),
            'execution_match_confidence':link.get('match_confidence'),
            'workout_structure_adherence_score':structural,
            'structure_available':structural is not None,
            'state':'MAIN_SET_VERIFIED' if structural is not None else 'ACTIVITY_LINKED_STRUCTURE_UNVERIFIED'})
    rows.sort(key=lambda x:(x['executed_date'],x['activity_id']))
    execution = (direction or {}).get('nova_execution') or {}
    return {'schema':'NOVA-R196948-EXECUTION-MEMORY-1',
        'source':'ATHLETE_SCOPED_PRESCRIPTION_AND_EXECUTION_LEDGER',
        'linked_workout_count':len(rows),
        'verified_execution':rows[-12:],
        'historical_unresolved':int(execution.get('unresolved') or 0),
        'historical_confirmed_missed':int(execution.get('confirmed_missed') or 0),
        'execution_adherence_score':execution.get('score') if execution.get('available') is True else None,
        'structure_adherence_count':sum(1 for r in rows if r['structure_available']),
        'linked_without_structure_count':sum(1 for r in rows if not r['structure_available']),
        'structure_adherence_score':None,  # Never grade from an unmatched/unknown execution.
        'scoring_policy':'EXACT_ISSUED_ACTIVITY_LINK_ONLY; POWER_ESTIMATES_NOT_PASS_MARKS',
        'no_retroactive_fulfillment':True}


def _nova_prescription_outcome(snapshot, previous=None):
    cur = snapshot if isinstance(snapshot,dict) else {}
    prev = previous if isinstance(previous,dict) else {}
    new = [s for s in cur.get('next_sessions') or [] if isinstance(s,dict)]
    earlier = [s for s in prev.get('next_sessions') or [] if isinstance(s,dict)]
    decision=cur.get('nova_decision') or {}
    pres=cur.get('nova_prescription') or {}
    quality=lambda s: bool(s.get('nova_workout_id') and s.get('hard_spacing_relevant')
                           and not s.get('provisional_quality')
                           and str(s.get('intensity') or '').upper() not in {'RECOVERY','EASY','DEFERRED'})
    explicit_restriction=lambda s: (str(s.get('restriction') or 'NONE').upper() not in ('NONE','')
                                    or s.get('is_race') is True)
    issues=[]
    current_by_slot={(str(s.get('date') or '')[:10],str(s.get('period') or '')):s for s in new}
    for s in earlier:
        if not quality(s) or str(s.get('prescription_status') or '').upper() in {'FULFILLED','CANCELLED','SUPERSEDED'}:
            continue
        try:
            if date.fromisoformat(str(s.get('date') or '')[:10]) < get_rome_now().date():
                continue
        except (TypeError,ValueError):
            continue
        replacement=current_by_slot.get((str(s.get('date') or '')[:10],str(s.get('period') or '')))
        moved = pres.get('postponed_quality') if isinstance(pres.get('postponed_quality'),dict) else {}
        bindings = [moved] + list(((cur.get('coach_clock') or {}).get('postponed_workouts') or []))
        deliberately_deferred = any(
            isinstance(b, dict)
            and str(b.get('workout_id') or '') == str(s.get('nova_workout_id') or '')
            and str(b.get('original_date') or '') == str(s.get('date') or '')[:10]
            and str(b.get('target_date') or '') != str(s.get('date') or '')[:10]
            for b in bindings)
        if deliberately_deferred:
            continue
        if replacement is None:
            # Slots may move when training availability / clock changes;
            # record an observation rather than a new technical veto.
            issues.append('PRIOR_QUALITY_SLOT_NOT_PRESENT_IN_NEW_CLOCK')
        elif (not quality(replacement) and not explicit_restriction(replacement)
              and str(decision.get('action') or '').upper() == 'TRAIN'
              and decision.get('hard_session_allowed') is True):
            issues.append('PRIOR_QUALITY_LOST_WITHOUT_EXPLICIT_REASON')
    has_quality = any(quality(s) for s in new)
    if (pres.get('requested_hard_prescription_available') is True and not has_quality
        and str(decision.get('action') or '').upper()=='TRAIN'
        and decision.get('hard_session_allowed') is True):
        issues.append('REQUESTED_QUALITY_LOST_AT_COMPILATION')
    unique_issues = sorted(set(issues))
    return {'schema':'NOVA-R196948-OUTCOME-OBSERVABILITY-2',
        'state':('QUALITY' if has_quality else 'UNEXPLAINED_QUALITY_LOSS' if issues
                 else 'EASY_OR_RECOVERY'),
        # A quality elsewhere in the horizon cannot erase a missing historical
        # promise. Keep the public state compatible and expose non-green truth.
        'integrity_state':'ALERT' if unique_issues else 'PASS',
        'issue_count':len(unique_issues), 'quality_preserved':has_quality,
        'issue_codes':unique_issues, 'source':pres.get('source'),
        'fail_reason':pres.get('fail_closed_reason'),
        'coach_action':decision.get('action'),
        'quality_is_not_a_wellness_color':True,
        'changes_any_prescription':False, 'another_blocking_gate':False}

# R196.9.45 — The saved report is the authoritative issued snapshot.
# Do NOT commit a future prescription to the ledger before its report is accepted.
# Product storage gets an atomic report+ledger transaction. On the owner's
# physically separate personal Neon, the report commits first and the ledger

# No new routes, tables, training gates, or athlete steps.
R196945_SCHEMA = 'R196.9.45-ISSUED-SNAPSHOT-LEDGER-1'
_v196945_db_execute_previous = _db_execute


def _v196945_pending():
    if not has_request_context():
        return None
    staged = getattr(g, 'nova_snapshot_ledger_pending', None)

def _db_execute(sql, params=(), fetch=False):
    pending = _v196945_pending()
    if (pending is not None and not fetch and
            re.match(r'^\s*UPDATE\s+microcycle_ledgers\s+SET\s+ledger_json\s*=', str(sql), re.I)):
        if len(params) != 4 or not str(params[2]) or not str(params[3]):
            raise ValueError('R196945_LEDGER_WRITE_SHAPE_NOT_SUPPORTED')
        row = json.loads(str(params[0]))
        if not isinstance(row, dict) or str(row.get('ledger_id') or '') != str(params[2]):
            raise ValueError('R196945_LEDGER_ID_MISMATCH')
        pending.append((str(params[2]), str(params[3]), str(sql), tuple(params)))
        return []
    return _v196945_db_execute_previous(sql, params=params, fetch=fetch)


def _v196945_finalize_pending(data, owner_id):
    pending = _v196945_pending()
    if pending is None or not pending:
        return None
    ids = {(x[0], x[1]) for x in pending}
    if len(ids) != 1 or next(iter(ids))[1] != str(owner_id):
        raise ValueError('R196945_CROSS_ACCOUNT_OR_MULTIPLE_LEDGER_WRITES')
    ledger = data.get('microcycle_ledger') if isinstance(data,dict) else None
    if not isinstance(ledger, dict) or str(ledger.get('ledger_id') or '') != pending[-1][0]:
        raise ValueError('R196945_FINAL_SNAPSHOT_LEDGER_MISMATCH')
    # Commit marker is deterministic for this finalized report, not a new
    # database schema. A later load can distinguish incomplete mirror writes.
    final = copy.deepcopy(ledger)
    final.pop('nova_snapshot_ledger_commit', None)
    canonical = json.dumps(final, ensure_ascii=False, sort_keys=True,
                           separators=(',', ':'), default=_report_json_default)
    marker = hashlib.sha256(canonical.encode('utf-8')).hexdigest()
    ledger['nova_snapshot_ledger_commit'] = marker
    ledger['nova_snapshot_ledger_schema'] = R196945_SCHEMA
    data['microcycle_ledger'] = ledger
    # Capture the exact finalized ledger used by the report; each earlier
    # issuance update concerns the same row, so only its last state matters.
    sql = pending[-1][2]
    params = (json.dumps(ledger, ensure_ascii=False, separators=(',', ':'),
                         default=_report_json_default), _now_utc_text(),
              pending[-1][0], pending[-1][1])
    pending[:] = [(pending[-1][0], pending[-1][1], sql, params)]
    return marker


def _v196945_same_store_ledger_commit(cur, owner_id):
    pending = _v196945_pending()
    if not pending or _owner_personal_data_enabled(owner_id):
        return
    ledger_id, user_id, sql, params = pending[-1]
    if str(owner_id) != user_id:
        raise ValueError('R196945_ATOMIC_OWNER_MISMATCH')
    cur.execute(_db_sql(sql), params)
    if cur.rowcount != 1:
        raise RuntimeError('R196945_ATOMIC_LEDGER_ROW_MISSING')
    pending.clear()


def _v196945_repair_ledger_from_saved_report(ledger, plan, user_id=None, read_only=False):
    if not isinstance(ledger, dict) or not ledger.get('ledger_id'):
        return ledger
    owner_id = _data_owner_user_id(user_id)
    report = load_latest_report(include_payload=True, owner_user_id=owner_id)
    saved = ((report or {}).get('data') or {}).get('microcycle_ledger')
    if not isinstance(saved, dict):
        return ledger
    token = str(saved.get('nova_snapshot_ledger_commit') or '')
    if (not re.fullmatch(r'[a-f0-9]{64}',token)
            or str(saved.get('ledger_id') or '') != str(ledger.get('ledger_id') or '')
            or str(saved.get('plan_id') or '') != str((plan or {}).get('id') or '')
            or str(saved.get('cycle_start') or '') != str(ledger.get('cycle_start') or '')):
        return ledger
    if token == str(ledger.get('nova_snapshot_ledger_commit') or ''):
        return ledger
    # A failed second-store write is repaired from the saved report, NEVER
    # from an unsaved work-in-progress. Re-run existing R146 evidence matching
    # against the latest actual device blocks: new rides still count.
    restored = copy.deepcopy(saved)
    for field in ('completed_hard_sessions', 'dose_context', 'block_foundation_evidence'):
        if field in ledger:
            restored[field] = copy.deepcopy(ledger[field])
    restored = _v146_reconcile_prescription_execution(
        restored, list(_V146_BLOCK_CONTEXT.get() or ()), now=get_rome_now())
    if not read_only:
        _v196945_db_execute_previous(
            'UPDATE microcycle_ledgers SET ledger_json=?, updated_at_utc=? WHERE id=? AND user_id=?',
            (json.dumps(restored, ensure_ascii=False, separators=(',', ':'),
                        default=_report_json_default), _now_utc_text(),
             restored['ledger_id'], owner_id))
    return restored


def _nova_apply_athlete_postponements(ledger, user_id=None):
    """Project explicit Calendar commitments onto the existing athlete ledger.

    No new table, stage reset, synthetic execution, or database write. The usual
    issued-Snapshot/ledger transaction persists this projection if the plan is
    actually saved. In read-only QA the exact same state is projected in memory.
    Only an existing, non-fulfilled SAME workout on its ORIGINAL date qualifies.
    """
    if not isinstance(ledger, dict) or not ledger.get('available'):
        return ledger
    items = load_planning_items(active_only=True, limit=100, owner_user_id=user_id)
    moves = [item.get('nova_postponement') for item in items
             if isinstance(item, dict) and isinstance(item.get('nova_postponement'), dict)]
    if not moves:
        return ledger
    out = copy.deepcopy(ledger)
    hist = out.get('prescription_history') or []
    applied = []
    for move in moves:
        wid = str(move.get('workout_id') or '')
        original = str(move.get('original_date') or '')[:10]
        target = str(move.get('target_date') or '')[:10]
        plan_id = str(move.get('original_plan_id') or '')
        try:
            valid = bool(wid and plan_id and date.fromisoformat(target) > date.fromisoformat(original))
        except (TypeError, ValueError):
            valid = False

def save_latest_report(data, now=None, owner_user_id=None, qa_replay_bundle=None):

                                            qa_replay_bundle=qa_replay_bundle)
    if pending:
        # In Product storage core save already cleared the staged writes in
        # the same SQL transaction. The split owner bridge cannot share a
        # transaction with Product Neon: sync the materialized ledger now.

                result['ledger_mirror_error_class'] = type(exc).__name__
            finally:

R196959_SCHEMA = 'NOVA-R196.9.59-VERIFIED-DOSE-ALTERNATION-1'
APP_VERSION = ('THE LAB · PRODUCT V4.9.159 R196.9.59 · VERIFIED QUALITY '
               'PROVENANCE + EVIDENCE-GATED ALTERNATION · NOVA R196.2 CORE')


def _nova_r196959_reference_match(dose, features):
    """Find one unique, completed, strong activity matching the WHOLE reference.

    R195.9 already computed work/recovery and the dose. We only attach the
    missing database/source ID if that same measured pair is demonstrably in
    the canonical, completed history. Never infer an ID from a date alone.
    """
    if not isinstance(dose, dict) or dose.get('available') is not True:
        return None
    ref = dose.get('reference') or {}
    if not isinstance(ref, dict):
        return None
    matches = []
    for event in (features or {}).get('recovery_dose_history') or []:
        if not isinstance(event, dict) or event.get('strong') is not True:
            continue
        aid = str(event.get('activity_id') or '').strip()
        if not aid or str(event.get('date') or '')[:10] != str(ref.get('date') or '')[:10]:
            continue
        try:
            exact = (int(event['reps']) == int(ref['reps'])
                     and abs(float(event['interval_secs']) - float(ref['interval_secs'])) <= 1
                     and abs(float(event['recovery_secs']) - float(ref['recovery_secs'])) <= 1
                     and abs(float(event['work_watts']) - float(ref['work_avg_watts'])) <= .11
                     and abs(float(event['recovery_watts']) - float(ref['recovery_avg_watts'])) <= .11
                     and abs(float(event['ftp']) - float(ref['ftp'])) <= .11)
        except (ValueError, TypeError, KeyError, OverflowError):
            exact = False
        if exact:
            matches.append(event)
    # Ambiguity is a proof failure, not a reason to take the first activity.
    if len(matches) != 1:
        return None
    return {'schema': R196959_SCHEMA,
            'activity_id': str(matches[0]['activity_id']),
            'date': str(matches[0]['date'])[:10],
            'reason': 'EXACT_VERIFIED_COMPLETED_COUPLED_REFERENCE'}

def _nova_r196960_number(value):
    try:
        if value is None or isinstance(value, bool):
            return None

    except (TypeError, ValueError, OverflowError):

def _nova_r196960_structure(workout):
    """Read only canonical workout mechanics, never title/text or target watts.

    The virtual coupled recovery contract has its own measured prescription;
    ordinary workouts use the versioned canonical library. Variable-intensity
    sets keep their set/float identity and must not be mistaken for steady reps.
    """
    if not isinstance(workout, dict):
        return None
    wid = str(workout.get('nova_workout_id') or workout.get('nova_contract_id') or '')
    if not wid:
        return None
    if wid == _R196933_VIRTUAL_ID:
        b = workout.get('power_prescription') or {}
        reps = _nova_r196960_number(b.get('reps'))
        sec = _nova_r196960_number(b.get('work_seconds'))
        rest = _nova_r196960_number(b.get('recovery_seconds'))
        if reps and reps >= 1 and sec and sec > 0 and (reps == 1 or (rest and rest > 0)):
            return ('REPEATED', int(reps), float(sec), float(rest or 0))
        return None
    template = (_R19696_LIBRARY_INDEX or {}).get(wid)
    if not isinstance(template, dict):
        return None
    pattern = str(template.get('pattern') or '').upper()
    sec = _nova_r196960_number(template.get('work_s'))
    if not sec or sec <= 0:
        return None
    if pattern in {'MICRO_30_15', 'SPRINT_CLUSTER'}:
        sets = _nova_r196960_number(template.get('sets'))
        per = _nova_r196960_number(template.get('reps_per_set'))
        flt = _nova_r196960_number(template.get('float_s'))
        group_rest = _nova_r196960_number(template.get('set_recovery_s'))
        if not sets or not per or not flt or sets < 1 or per < 1 or flt <= 0:
            return None
        return ('SETS_' + pattern, int(sets), int(per), float(sec),
                float(flt), float(group_rest or 0))
    reps = _nova_r196960_number(template.get('reps'))
    rest = _nova_r196960_number(template.get('recovery_s'))
    if not reps or reps < 1 or (reps > 1 and not (rest and rest > 0)):
        return None
    if pattern in {'SPRINT', 'OVER_UNDER'}:
        return (pattern, int(reps), float(sec), float(rest or 0))
    if pattern in _R19696_NUMERIC_PATTERNS | _R19696_EFFORT_PATTERNS:
        domain='REPEATED_TEMPO_Z3' if str(template.get('work_domain') or '').upper()=='TEMPO_Z3' else 'REPEATED'
        return (domain, int(reps), float(sec), float(rest or 0))
    return None


def _nova_r196960_structure_equivalent(a, b):
    """Mechanical near-identities only; 5×3 is NEVER the same as 3×5.

    A confirmed 3×300/296 can match a 3×300/300 due to lap rounding.
    Structurally different repetitions, set patterns and work/rest doses do
    not become the same merely because their total kilojoules are similar.
    """
    if not isinstance(a, tuple) or not isinstance(b, tuple) or a[0] != b[0] or len(a) != len(b):
        return False
    if a[0] in {'REPEATED', 'REPEATED_TEMPO_Z3', 'SPRINT', 'OVER_UNDER'}:
        if a[1] != b[1]:
            return False
        aw, bw, ar, br = a[2], b[2], a[3], b[3]
        return (abs(aw-bw) <= max(2.0, min(15.0, min(aw,bw)*.05))
                and abs(ar-br) <= max(4.0, min(20.0, min(ar,br)*.05)))
    if a[0].startswith('SETS_'):
        if a[1:3] != b[1:3]:
            return False
        return (abs(a[3]-b[3]) <= max(2.0, min(10.0, min(a[3],b[3])*.05))
                and abs(a[4]-b[4]) <= max(2.0, min(10.0, min(a[4],b[4])*.05))
                and abs(a[5]-b[5]) <= max(4.0, min(20.0, min(a[5],b[5])*.05)))
    return False


def _nova_r196960_recent_completed(workout, evidence_ledger, ledger, target_day):
    """Match RECENT completed independent quality by two verified routes.

    1. Reconciled fulfilled prescription: exact canonical workout ID, with
       linked real activity; compare its template mechanics for library IDs.
    2. Activity Truth RESOLVED repeated intervals: cross-family mechanics.
    Never match name, estimated zones, raw/unresolved intervals or fake IDs.
    """
    structure = _nova_r196960_structure(workout)
    if structure is None:
        return None
    try:
        candidate_day = date.fromisoformat(str(target_day or '')[:10])
    except (ValueError, TypeError):
        return None
    wid = str(workout.get('nova_workout_id') or workout.get('nova_contract_id') or '')
    candidates = []
    def within(day):
        try:
            ds = (candidate_day - date.fromisoformat(str(day or '')[:10])).days
            return ds if 0 <= ds <= 4 else None
        except (ValueError, TypeError, OverflowError):
            return None
    # FULFILLED is reconciliation authority. SAME library ID means the same
    # versioned mechanics; virtual dose IDs are variable, hence require proof
    # of matching mechanics (via the independent resolved interval route).
    for r in (ledger or {}).get('prescription_history') or []:
        if not isinstance(r, dict) or str(r.get('prescription_status') or '').upper() != 'FULFILLED':
            continue

        days = within(r.get('actual_date'))
        if days is None or not aid:
            continue
        proof = str(r.get('execution_match_basis') or '').upper()
        structure_attested = (proof == 'REPEATED_WORK_STRUCTURE' or
                              proof.startswith('VERIFIED_RAW_POWER_') or
                              proof in {'CANONICAL_INTERVAL_STRUCTURE',
                                        'VERIFIED_DEVICE_LAPS', 'VERIFIED_FIT_LAPS'})
        if not structure_attested:
            # An EXACT_DATE-only match or same-name ride is not a verified
            # quality execution. Require independent Activity Truth for this
            # linked activity before using its identity to suppress a repeat.
            structure_attested = any(
                isinstance(e, dict) and str(e.get('activity_id') or '') == aid
                and e.get('quality_relevant') is True
                and str(e.get('activity_truth_status') or '').upper() == 'RESOLVED'
                and str(e.get('activity_truth_source') or '').upper() in
                    {'ICU_INTERVALS_CANONICAL','DEVICE_LAPS',
                     'ORIGINAL_ACTIVITY_FIT_LAP','ORIGINAL_FIT_LAPS'}
                for e in (evidence_ledger or {}).get('events') or [])
        if not structure_attested:
            continue
        old_id = str(r.get('nova_workout_id') or r.get('nova_contract_id') or '')
        if (old_id == wid and wid != _R196933_VIRTUAL_ID
                and _nova_r196960_structure({'nova_workout_id': old_id}) == structure):
            candidates.append((days, {'source':'FULFILLED_CANONICAL_LEDGER',
                'activity_id':aid, 'date':str(r.get('actual_date'))[:10],
                'workout_id':old_id, 'days_before_candidate':days}))
        # Different library IDs can have identical actual mechanics: resolve
        # both from the same immutable template library, not approximate title.
        elif old_id != wid and _nova_r196960_structure_equivalent(
                structure, _nova_r196960_structure({'nova_workout_id':old_id})):
            candidates.append((days, {'source':'FULFILLED_COMPARABLE_LIBRARY_STRUCTURE',
                'activity_id':aid, 'date':str(r.get('actual_date'))[:10],
                'workout_id':old_id, 'days_before_candidate':days}))
    trusted = {'ICU_INTERVALS_CANONICAL', 'DEVICE_LAPS',
               'ORIGINAL_ACTIVITY_FIT_LAP', 'ORIGINAL_FIT_LAPS'}
    # Canonical repeatability evidence (including COMPLETED runs which faded)
    # can identify near-identical plain repeated workouts without using their
    # watts as a shortcut. Strong is a progression judgement, not completion.
    if structure[0] == 'REPEATED':
        for r in ((evidence_ledger or {}).get('repeatability') or {}).get('events') or []:
            if not isinstance(r, dict):
                continue
            aid = str(r.get('activity_id') or '').strip()
            days = within(r.get('date'))
            src = str(r.get('activity_truth_source') or '').upper()
            if (days is None or not aid or src not in trusted
                    or str(r.get('activity_truth_status') or '').upper() != 'RESOLVED'):
                continue
            count = _nova_r196960_number(r.get('reps'))
            sec = _nova_r196960_number(r.get('interval_secs'))
            rec = _nova_r196960_number(r.get('recovery_secs'))
            if not count or not sec or rec is None:
                continue
            if _nova_r196960_structure_equivalent(structure,
                    ('REPEATED',int(count),float(sec),float(rec))):
                candidates.append((days, {'source':'RESOLVED_CANONICAL_REPEATABILITY',
                    'activity_id':aid, 'date':str(r.get('date'))[:10],
                    'days_before_candidate':days}))
    if not candidates:
        return None
    return min(candidates, key=lambda p: (p[0],p[1]['activity_id']))[1]

def _r196962_number(value, low, high):
    if isinstance(value, bool) or value is None:
        return None
    try:
        result = float(value)
    except (ValueError, TypeError, OverflowError):
        return None
    return result if math.isfinite(result) and low <= result <= high else None


def _r196962_day(row):
    try:

    except (ValueError, TypeError, AttributeError):
        return None

def _r196962_optional_intervals_wellness(now):
    '''Read-only optional probe. Failure means unknown, NOT unhealthy.'''
    result = {'rows': [], 'fields': [], 'availability': 'NOT_CONFIRMED'}
    try:
        runtime = _resolve_intervals_runtime()
        headers = get_intervals_headers(runtime)
        base = 'https://intervals.icu/api/v1/athlete/' + str(runtime['athlete_id'])
        response = requests.get(base + '/custom-item', headers=headers, timeout=5)
        response.raise_for_status()
        definitions = response.json()
        if isinstance(definitions, dict):
            definitions = definitions.get('items') or definitions.get('customItems') or []
        available = set()
        if isinstance(definitions, list):
            for item in definitions:
                if not isinstance(item, dict):
                    continue
                content = item.get('content') if isinstance(item.get('content'), dict) else {}
                code = item.get('code') or content.get('code')
                if code in _R196962_CUSTOM_FIELDS:
                    available.add(code)
        codes = sorted(available)
        if not codes:
            result['availability'] = 'NO_OPTIONAL_FIELDS_CONFIGURED'
            return result
        # Documented custom wellness codes must be named in the fields query.
        # Do not touch existing engine wellness rows, nor create any fields.
        oldest = (now.date() - timedelta(days=60)).isoformat()
        params = {'oldest': oldest, 'newest': now.date().isoformat(),
                  'fields': 'id,' + ','.join(codes)}
        response = requests.get(base + '/wellness', headers=headers,
                                params=params, timeout=7)
        response.raise_for_status()
        rows = response.json()
        if not isinstance(rows, list):
            return result
        result.update(rows=[r for r in rows if isinstance(r, dict) and _r196962_day(r)],
                      fields=codes, availability='CUSTOM_FIELDS_CHECKED')
    except Exception:
        # Health enrichment may NEVER take down or withhold a Snapshot.
        result['availability'] = 'OPTIONAL_SOURCE_UNAVAILABLE'
    return result


def _r196962_series(rows, name, bounds, now):
    '''Deduplicate by day, reject bad/future/stale data; no artificial zeros.'''
    valid = {}
    for row in rows or []:
        if not isinstance(row, dict):
            continue
        day = _r196962_day(row)
        if day is None or day > now.date() or (now.date() - day).days > 60:
            continue
        if name == 'restingHR' and _intervals_temp_flag(row, 'tempRestingHR'):
            continue
        raw = _r196962_number(row.get(name), bounds[1], bounds[2])
        if raw is not None:
            valid[day] = raw
    return sorted(valid.items())


def _r196962_signal(series, name, now):
    '''Use robust personal comparisons, not generic clinical red zones.'''
    if not series:
        return None
    day, value = series[-1]
    age = (now.date() - day).days
    if age > 2:
        return {'name': name, 'available': False, 'reason': 'STALE', 'age_days': age}
    historical = [v for d, v in series[:-1] if (day - d).days <= 35 and d < day]
    samples = len(historical)
    baseline = statistics.median(historical) if samples >= 7 else None
    direction = 'UNKNOWN'
    delta = None
    if baseline is not None:
        delta = value - baseline
        if name == 'hrv':
            direction = 'WATCH' if baseline > 0 and value < baseline * 0.86 else 'USUAL'
        elif name == 'restingHR':
            direction = 'WATCH' if delta >= 4.0 else 'USUAL'
        elif name == 'respiration':
            direction = 'WATCH' if delta >= 1.8 else 'USUAL'
        elif name == 'GarminSkinTempDeviationC':
            direction = 'WATCH' if delta >= 0.45 else 'USUAL'
        elif name == 'sleepSecs':
            direction = 'WATCH' if delta < -2700 else 'USUAL'
        else:
            direction = 'DESCRIPTIVE'
    return {'name': name, 'value': round(value, 2), 'baseline': round(baseline, 2)
            if baseline is not None else None,
            'delta': round(delta, 2) if delta is not None else None,
            'samples': samples, 'date': day.isoformat(), 'age_days': age,
            'available': True, 'state': direction}


def _r196962_build_health_observatory(metrics, native_wellness, optional=None, now=None):
    '''Pure, descriptive sidecar. It never changes readiness/energy or workouts.'''
    now = now or get_rome_now()
    optional = optional if isinstance(optional, dict) else {}
    native_rows = _r196962_native_observations(native_wellness)
    custom_rows = [r for r in optional.get('rows') or [] if isinstance(r, dict)]
    supported = set(optional.get('fields') or []) & set(_R196962_CUSTOM_FIELDS)
    observations = {}
    for key, meta in _R196962_NATIVE_FIELDS.items():
        observations[key] = _r196962_signal(_r196962_series(native_rows, key, meta, now), key, now)
    for key in sorted(supported):
        observations[key] = _r196962_signal(_r196962_series(custom_rows, key,
                                       _R196962_CUSTOM_FIELDS[key], now), key, now)
    current = {k: v for k, v in observations.items() if v and v.get('available')}
    watched = [k for k in ('hrv', 'restingHR', 'respiration', 'GarminSkinTempDeviationC', 'sleepSecs')
               if (current.get(k) or {}).get('state') == 'WATCH']
    # At least TWO independently measured deviations are required for WATCH.
    status = 'WATCH' if len(watched) >= 2 else ('OBSERVING' if current else 'LEARNING')
    readable = {'hrv': 'HRV below its usual range',
                'restingHR': 'resting heart rate higher than usual',
                'respiration': 'respiratory rate higher than usual',
                'GarminSkinTempDeviationC': 'skin-temperature deviation above recent norm',
                'sleepSecs': 'sleep duration below recent habit'}
    if len(watched) >= 2:
        headline = 'Several recent physiological signals differ from your personal norm.'
        detail = ' · '.join(readable[k] for k in watched[:3])
    elif watched:
        headline = 'One signal differs from your recent norm; no multi-signal pattern established.'
        detail = readable[watched[0]]
    elif current:
        headline = 'No multi-signal deviation established from the available recent measurements.'
        detail = 'Observed signals only; missing measurements have no adverse interpretation.'
    else:
        headline = 'Wellness context is still learning from the data available.'
        detail = 'Missing or stale measurements are not interpreted as poor recovery.'
    contextual = []
    temp = current.get('GarminSkinTempDeviationC')
    if temp:
        unit = '°C'
        contextual.append('Skin temperature %+.2f %s vs device reference' % (temp['value'], unit))
        if temp['samples'] < 7:
            contextual.append('skin-temperature trend still learning')
    battery = current.get('BodyBatteryMax')
    if battery:
        contextual.append('Garmin Body Battery high %.0f/100 (separate from Nova Energy Bank)' % battery['value'])
    stress = current.get('GarminStressAvg')
    if stress:
        contextual.append('Garmin stress %.0f/100 (not Intervals subjective stress)' % stress['value'])
    stage_keys = ('GarminSleepDeepMinutes', 'GarminSleepREMMinutes', 'GarminSleepLightMinutes')
    if all(current.get(k) for k in stage_keys):
        durations = [current[k]['value'] for k in stage_keys]
        latest_dates = {current[k]['date'] for k in stage_keys}
        if len(latest_dates) == 1 and 60 <= sum(durations) <= 900:
            contextual.append('Sleep stages: deep %.0f · REM %.0f · light %.0f min' % tuple(durations))
    freshness = sum(1 for v in current.values() if v['age_days'] <= 2)
    # Explanations, not surrogate scoring formulas. Neither raw nor derived
    # optional values are ever passed back into compute_race_readiness/energy.
    readiness_line = 'Health context: ' + (detail if watched else headline)
    energy_line = 'Estimated reserve, not a measured Body Battery. ' + (
        'Additional wearable observations are shown below, not deducted from this score.'
        if contextual else 'Missing wearable metrics do not reduce this score.')
    return {
        'schema': R196962_HEALTH_SCHEMA, 'read_only': True,
        'affects_training_decision': False, 'affects_readiness_score': False,
        'affects_energy_score': False, 'state': status,
        'headline': headline, 'detail': detail,
        'readiness_line': readiness_line, 'energy_line': energy_line,
        'insights': contextual[:3], 'signals': observations,
        'available_measurements': freshness,
        'custom_fields_detected': sorted(supported),
        'optional_source': optional.get('availability', 'NOT_CHECKED'),
        'note': 'Descriptive, not diagnostic. Never overrides Nova workouts or explicit clinical restrictions.',
    }

def _nova_r196963_advisory_slots(clock, road, compiled_rows=None):
    """Actual learned future opportunities, not invented time or booked sessions.

    The preferred quality window is a lower bound for a DETACHED offer: moving
    an advisory preview earlier would bypass the hard-spacing decision. A
    later, distinct rhythm slot can be inspected, but is never confirmed.
    """
    preferred = _nova_r196956_datetime(clock, road)
    earliest = str(((road or {}).get('quality_window') or {}).get('earliest_iso') or '')[:16]
    if earliest:
        try:
            datetime.fromisoformat(earliest)
        except (TypeError, ValueError):
            earliest = ''
    boundary = max((x for x in (preferred, earliest) if x), default='')
    seen = set()
    result = []
    # Keep already-compiled, real planned QUALITY visible; never turn a planned
    # appointment into a blank slot for detached synthesis.
    compiled_days = {str(r.get('date') or '')[:10]
        for r in (compiled_rows or []) if _nova_r196956_structural_ok(r)}
    for raw in (clock or {}).get('next_slots') or []:
        if not isinstance(raw, dict):
            continue
        day = str(raw.get('date') or '')[:10]
        minute = _rhythm_num(raw.get('start_minute'))
        if minute is None:
            continue
        minute = int(minute)
        try:
            date.fromisoformat(day)
        except (TypeError, ValueError):
            continue
        if not 0 <= minute < 1440:
            continue
        start = f'{day}T{minute//60:02d}:{minute%60:02d}'
        if boundary and start < boundary:
            continue
        # An existing planned appointment is not a blank opportunity; never
        # erase its explicit duration, training demand or restriction.
        if (str(raw.get('restriction') or 'NONE').upper() != 'NONE'
                or raw.get('is_race') is True
                or (raw.get('is_planned_training') is True and day not in compiled_days)):
            continue
        limit = _nova_quality_slot_declared_seconds(raw)
        if limit is None or limit <= 0 or start in seen:
            continue
        seen.add(start)
        result.append((start, dict(raw), limit))
    result.sort(key=lambda item: item[0])
    return result

def _nova_r196965_verified_duplicate(workout, ledger, target_day, evidence_ledger=None):
    """Compare work/rest mechanics, not aliases or titles.

    verified = _nova_r196960_recent_completed(workout, evidence_ledger, ledger, target_day)

        return verified
    structure = _nova_r196960_structure(workout)

        return None
    try:
        day = date.fromisoformat(str(target_day)[:10])
    except (ValueError, TypeError):
        return None
    for entry in (ledger or {}).get('prescription_history') or []:

def _r196964_learned_duration(row, rhythm):
    if not isinstance(row, dict):
        return row
    out = dict(row)
    if (str(out.get('source') or '').upper() != 'RHYTHM'
            or out.get('is_planned_training') is True
            or out.get('is_race') is True
            or str(out.get('availability_source') or '').upper() in {'CALENDAR', 'ATHLETE_EXPLICIT'}):
        return out
    # Per-session evidence must not be replaced by the usually longer
    # total of all physical sessions recorded on the same training day.
    if not _v4895_rhythm_duration_reliable(rhythm):
        return out
    n = _rhythm_num(out.get('duration_hint_min'))
    median = _rhythm_num((rhythm or {}).get('median_session_duration_min'))
    if n is None or median is None or not 0 < n < median:
        return out
    count = int((rhythm or {}).get('training_day_duration_samples') or 0)
    if count < 5:
        return out

    # multi-session day. Use the proven session median as a descriptive
    # baseline, not a promised time window or permission to add intensity.
    out['original_weekday_duration_hint_min'] = n
    out['duration_hint_min'] = round(median, 1)
    out['duration_hint_label'] = _rhythm_format_duration(median)
    out['duration_estimate_source'] = 'PERSONAL_MEDIAN_SESSION_FROM_HISTORY'
    out['duration_estimate_not_availability'] = True
    out['duration_revision_schema'] = R196964_SCHEMA
    return out

    # Only identity-level exact matches after harmless formatting normalization;

def _r196966_recent_exposures(evidence_ledger, candidate_date):
    """Verified *exposure*, not FULFILLED adherence or completed main-set.

    Repeating an activity's exact canonical NAME when raw actual intervals
    differ is unsafe to label a fresh prescription. This evidence blocks
    re-issuance only; it does not grade adherence, invent repetitions or alter
    existing user-owned calendar commitments. The four-day window is bounded
    relative to each candidate slot, not system UTC today.
    """
    try:
        target = date.fromisoformat(str(candidate_date or '')[:10])
    except (TypeError, ValueError):
        return []
    found = []
    for event in (evidence_ledger or {}).get('events') or []:
        if not isinstance(event, dict) or event.get('quality_relevant') is not True:
            continue
        if str(event.get('activity_truth_status') or '').upper() != 'RESOLVED':
            continue
        source = str(event.get('activity_truth_source') or '').upper()
        if source not in _R196966_TRUSTED_ACTIVITY_SOURCES:
            continue
        aid = str(event.get('activity_id') or '').strip()
        name = _r196966_normalize_title(event.get('name'))
        if not aid or not name:
            continue
        try:
            day = date.fromisoformat(str(event.get('date') or '')[:10])
            days = (target - day).days
        except (TypeError, ValueError):
            continue
        if not 0 <= days <= 4:
            continue
        observed = event.get('repeatability') or {}
        # RESOLVED + a measured cadence / structure is required. Mere same
        # activity title or imported planned event is not sufficient evidence.
        reps = _nova_r196960_number(observed.get('reps'))
        work = _nova_r196960_number(observed.get('interval_secs'))
        if not reps or not work or reps < 1 or work <= 0:
            continue
        found.append({'activity_id':aid,'date':day.isoformat(),
                      'title':name,'source':source,'observed_reps':int(reps),
                      'observed_work_seconds':work,
                      'observed_float_seconds':_nova_r196960_number(observed.get('recovery_secs')),
                      'days_before_candidate':days})
    return found


def _r196966_named_exposure(workout, evidence_ledger, candidate_date):
    if not isinstance(workout, dict):
        return None
    wid = str(workout.get('nova_workout_id') or workout.get('nova_contract_id') or '')
    template = _R19696_LIBRARY_INDEX.get(wid) if wid else None
    title = _r196966_normalize_title((template or {}).get('title'))
    if not title:
        return None
    for exposure in _r196966_recent_exposures(evidence_ledger,candidate_date):
        if exposure['title'] == title:
            return dict(exposure,workout_id=wid,
                        type='OBSERVED_NAMED_WORKOUT_COMPLETION_UNVERIFIED',
                        canonical_fulfillment_asserted=False)
        # An independently resolved raw 30/15 exposure is a RECENT STIMULUS,
        # not just one catalogue name or a single total-repetition number.
        # Otherwise correcting a detector from 20 to 30 observed repeats
        # accidentally re-opens the shorter 2x10 variant the next day.
        # Scope is one verified 30/15 family and the existing 0..4-day window:
        # no unrelated VO2/sustained template, no title-only activity, no
        # unverified stream, and NO assertion of fulfilled adherence.
        if str((template or {}).get('pattern') or '').upper() != 'MICRO_30_15':
            continue
        try:
            candidate_work = float(template['work_s'])
            candidate_float = float(template['float_s'])
            candidate_dose = int(template['sets'])*int(template['reps_per_set'])
            observed_float = exposure.get('observed_float_seconds')
            same_cadence = (candidate_work > 0 and observed_float is not None and
                abs(candidate_work-exposure['observed_work_seconds']) <= 1.0 and
                abs(candidate_float-observed_float) <= 1.0)
            # Do not turn one noisy sprint into a whole 30/15 micro-session.
            reliable_micro_dose = exposure['observed_reps'] >= 16
            if (exposure['source'] == 'ICU_RAW_WATTS_30_15' and
                    same_cadence and reliable_micro_dose and candidate_dose > 0):
                return dict(exposure, workout_id=wid,
                    type='RECENT_VERIFIED_30_15_STIMULUS_FAMILY',
                    matched_family='MICRO_30_15',
                    candidate_total_reps=candidate_dose,
                    same_total_dose=candidate_dose == exposure['observed_reps'],
                    canonical_fulfillment_asserted=False,
                    grouping_equivalence_asserted=False,
                    cooldown_schema=R196968_SCHEMA)
        except (ValueError, TypeError, KeyError, OverflowError):
            continue
        # Other independently verified structural sources retain the older
        # exact-dose guard. Never assume a full family from canonical labels.
        try:
            total = int(template['sets'])*int(template['reps_per_set'])
            work = float(template['work_s'])
            flt = float(template['float_s'])
            observed_float = exposure.get('observed_float_seconds')
            if (total == exposure['observed_reps'] and
                    abs(work-exposure['observed_work_seconds']) <= 1 and
                    observed_float is not None and abs(flt-observed_float) <= 1):
                return dict(exposure,workout_id=wid,
                    type='RECENT_VERIFIED_RAW_EQUAL_TOTAL_MICRO_DOSE',
                    canonical_fulfillment_asserted=False,
                    grouping_equivalence_asserted=False)
        except (ValueError,TypeError,KeyError,OverflowError):
            continue
    return None


def _r196966_producer_exclusions(evidence_ledger, date_value, goal_key=None):
    recent = _r196966_recent_exposures(evidence_ledger, date_value)
    if not recent:
        return set()
    selected = set()
    for t in _NOVA_WORKOUT_LIBRARY:
        if goal_key and str(t.get('goal') or '').upper() != str(goal_key).upper():
            continue
        if _r196966_named_exposure({'nova_workout_id':t.get('id')},evidence_ledger,date_value):
            selected.add(str(t['id']))
    return selected
