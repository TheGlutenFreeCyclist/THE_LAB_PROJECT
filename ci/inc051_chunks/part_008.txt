def build_strength_pattern(activities, now=None):

    window_start = today - timedelta(days=STRENGTH_PATTERN_LOOKBACK_DAYS - 1)
    rows = [r for r in _rhythm_activity_records(activities, now) if _strength_type_key(r.get("type")) == "weighttraining" and r["date"] >= window_start]

        }

    weekday_occurrences = {i: 0 for i in range(7)}
    cursor = first_date
    while cursor <= today:
        weekday_occurrences[cursor.weekday()] += 1
        cursor += timedelta(days=1)
    weekday_counts = {i: 0 for i in range(7)}
    for d in dates:
        weekday_counts[d.weekday()] += 1
    weekday_support = {
        i: (weekday_counts[i] / weekday_occurrences[i]) if weekday_occurrences[i] else 0.0
        for i in range(7)
    }
    likely_weekdays = [i for i in range(7) if weekday_counts[i] >= 2 and weekday_support[i] >= 0.50]
    covered_sessions = sum(weekday_counts[i] for i in likely_weekdays)
    weekday_coverage = covered_sessions / max(1, len(dates))

    median_gap = statistics.median(gaps) if gaps else None
    if median_gap and len(gaps) >= 3:
        rel_dev = statistics.median(abs(g - median_gap) / max(1.0, float(median_gap)) for g in gaps)
        gap_consistency = max(0.0, min(1.0, 1.0 - rel_dev))
    else:
        gap_consistency = 0.0

        state, confidence = "EMERGING", "MODERATE"
    else:
        state, confidence = "OBSERVED", "LOW"
    starts = [(r.get("start_minute"), 1.0) for r in rows if r.get("start_minute") is not None]
    durations = [r.get("duration_min") for r in rows if r.get("duration_min")]
    typical_start = _rhythm_weighted_median(starts) if starts else None
    typical_duration = statistics.median(durations) if durations else None

def _v1964_weekday_evidence(recent, established, season, weekday_scores, state):
    """Training opportunities are supported per weekday, not capped by rounded weekly volume.

    A date has evidence when it repeats at least twice in the recent sample, or
    reliably in a longer established sample. Blended probability alone must not
    invent a day with no repeated observations. "Likely" is a candidate, never
    a mandate to train or a recovery clearance.
    """
    def occurrences(stats, weekday):
        try:
            lo = date.fromisoformat(str(stats.get("start_date")))
            hi = date.fromisoformat(str(stats.get("end_date")))
        except (TypeError, ValueError):
            return 0
        if hi < lo:
            return 0
        first = (weekday - lo.weekday()) % 7
        return max(0, ((hi - lo).days - first) // 7 + 1)

    result = {}
    for wd in range(7):
        r_occ = occurrences(recent, wd)
        e_occ = occurrences(established, wd)
        s_occ = occurrences(season, wd)
        rp = float((recent.get("weekday_probability") or {}).get(wd) or 0.0)
        ep = float((established.get("weekday_probability") or {}).get(wd) or 0.0)
        sp = float((season.get("weekday_probability") or {}).get(wd) or 0.0)
        r_days = int(round(r_occ * rp))
        e_days = int(round(e_occ * ep))
        s_days = int(round(s_occ * sp))
        score = float(weekday_scores.get(wd) or 0.0)
        recent_repeated = (r_occ >= 2 and r_days >= 2 and rp >= (2.0 / 3.0 - 0.001))
        established_repeated = (e_occ >= 4 and e_days >= 3 and ep >= (2.0 / 3.0 - 0.001)
                                and score >= 0.50)
        supported = (state != "LEARNING" and (recent_repeated or established_repeated))
        result[wd] = {
            "recent_observed": r_days, "recent_opportunities": r_occ,
            "established_observed": e_days, "established_opportunities": e_occ,
            "long_term_observed": s_days, "long_term_opportunities": s_occ,
            "blended_score": round(score, 3), "supported": supported,
            "reason": ("RECENT_REPEAT" if recent_repeated else
                       "ESTABLISHED_REPEAT" if established_repeated else
                       "LEARNING" if state == "LEARNING" else "INSUFFICIENT_WEEKDAY_EVIDENCE"),
        }
    return result

def build_training_rhythm(season_activities, now=None, scope="CYCLING"):
    now = now or get_rome_now()
    today = now.date()
    records = _rhythm_activity_records(season_activities, now)
    if str(scope or "CYCLING").upper() == "CYCLING":
        records = [r for r in records if _v4887_is_cycling_activity({"type": r.get("type"), "name": r.get("name")})]
    elif str(scope or "").upper() == "STRENGTH":
        records = [r for r in records if _v4890_is_strength_activity({"type": r.get("type"), "name": r.get("name")})]
    historical = [r for r in records if r["date"] < today]
    history_start = min((r["date"] for r in historical), default=None)
    end = today - timedelta(days=1)
    recent = _rhythm_window_stats(historical, end, RHYTHM_RECENT_DAYS, history_start)
    established = _rhythm_window_stats(historical, end, RHYTHM_ESTABLISHED_DAYS, history_start)
    season = _rhythm_window_stats(historical, end, RHYTHM_SEASON_DAYS, history_start)
    prior_end = end - timedelta(days=RHYTHM_RECENT_DAYS)
    prior_start = max(prior_end - timedelta(days=27), history_start) if history_start else prior_end - timedelta(days=27)
    prior = _rhythm_span_stats(historical, prior_start, prior_end) if prior_end >= prior_start else _rhythm_span_stats([], None, None)
    shift_signals = []
    if recent.get("days_observed", 0) >= 10 and prior.get("days_observed", 0) >= 14 and recent.get("training_days", 0) >= 3 and prior.get("training_days", 0) >= 3:
        if abs(recent["days_per_week"] - prior["days_per_week"]) >= 1.25:
            shift_signals.append("training-day frequency changed")
        if abs(recent["sessions_per_week"] - prior["sessions_per_week"]) >= 1.5:
            shift_signals.append("session frequency changed")
        prior_hours = prior.get("hours_per_week") or 0.0
        if prior_hours >= 2.0:
            hours_change = abs(recent.get("hours_per_week", 0.0) - prior_hours)
            if hours_change >= max(2.5, prior_hours * 0.30):
                shift_signals.append("weekly volume changed")
        if abs(recent.get("double_day_pct", 0.0) - prior.get("double_day_pct", 0.0)) >= 0.25:
            shift_signals.append("double-session pattern changed")
        rdur, pdur = recent.get("median_duration_min"), prior.get("median_duration_min")
        if rdur and pdur and abs(rdur - pdur) >= max(30, pdur * 0.35):
            shift_signals.append("typical session duration changed")
    if established.get("days_observed", 0) < 14 or established.get("training_days", 0) < 3:
        state = "LEARNING"
        confidence = "LOW"
    elif shift_signals:
        state = "ADAPTING"
        confidence = "MODERATE" if established.get("training_days", 0) >= 6 else "LOW"
    elif established.get("days_observed", 0) >= 35 and established.get("training_days", 0) >= 8:
        state = "STABLE"
        confidence = "HIGH"
    else:
        state = "TRACKING"
        confidence = "MODERATE"
    if state == "ADAPTING":
        raw_weights = ((recent, 0.30), (established, 0.52), (season, 0.18))
    elif recent.get("training_days", 0) >= 3:
        raw_weights = ((recent, 0.35), (established, 0.50), (season, 0.15))
    elif established.get("training_days", 0) >= 4:
        raw_weights = ((recent, 0.25), (established, 0.55), (season, 0.20))
    else:
        raw_weights = ((recent, 0.10), (established, 0.45), (season, 0.45))
    valid = [(stats, weight) for stats, weight in raw_weights if stats.get("days_observed", 0) > 0]
    total_weight = sum(weight for _, weight in valid) or 1.0
    valid = [(stats, weight / total_weight) for stats, weight in valid]
    def blend(key):
        return sum(float(stats.get(key) or 0.0) * weight for stats, weight in valid)
    def blend_optional(key):
        pairs = []
        for stats, weight in valid:
            value = _rhythm_num(stats.get(key))
            if value is not None:
                pairs.append((value, weight))
        if not pairs:
            return None
        denom = sum(weight for _, weight in pairs) or 1.0
        return sum(value * weight for value, weight in pairs) / denom
    weekday_scores = {}
    weekday_double = {}
    for wd in range(7):
        weekday_scores[wd] = round(sum(float(stats.get("weekday_probability", {}).get(wd) or 0.0) * weight for stats, weight in valid), 3)
        weekday_double[wd] = round(sum(float(stats.get("weekday_double_probability", {}).get(wd) or 0.0) * weight for stats, weight in valid), 3)
    est_days = max(0.0, min(7.0, blend("days_per_week")))
    est_sessions = max(est_days, blend("sessions_per_week"))
    est_hours = max(0.0, blend("hours_per_week"))
    double_propensity = max(0.0, min(1.0, blend("double_day_pct")))
    # Estimated days/week describes observed volume; it is NOT the number of
    # weekdays allowed to receive a candidate. An athlete can alternate days.
    weekday_evidence = _v1964_weekday_evidence(recent, established, season, weekday_scores, state)
    recent_start = end - timedelta(days=RHYTHM_RECENT_DAYS - 1)
    for wd in range(7):
        weekday_evidence[wd]["recent_recorded_dates"] = sorted({
            row["date"].isoformat() for row in historical
            if row["weekday"] == wd and row["date"] >= recent_start
        })
    ranked_days = sorted(range(7), key=lambda wd: (-weekday_scores[wd], wd))
    likely_weekdays = [wd for wd in ranked_days if weekday_evidence[wd]["supported"]]
    if len(likely_weekdays) == 7:
        # Uniform 4-5d/week athletes do not thereby train *every* weekday.
        # Only seven strong, nearly-daily weekday priors justify daily slots.
        pattern_strength = "HIGH" if min(weekday_scores.values()) >= 0.80 else "LOW"
    elif likely_weekdays:
        top = [weekday_scores[i] for i in likely_weekdays]
        rest = [weekday_scores[i] for i in ranked_days if i not in likely_weekdays]
        separation = (statistics.mean(top) if top else 0) - (statistics.mean(rest) if rest else 0)
        pattern_strength = "HIGH" if separation >= 0.28 else ("MODERATE" if separation >= 0.12 else "LOW")
    else:
        pattern_strength = "LOW"
    clock_profile = _rhythm_clock_profile(historical, today, history_start)
    median_duration = _rhythm_weighted_median([
        (stats.get("median_duration_min"), weight) for stats, weight in valid if stats.get("median_duration_min")
    ])
    if median_duration is None:
        median_duration = clock_profile.get("overall_duration_min")
    median_gap = _rhythm_training_gap(historical, today)
    typical_day_load = blend_optional("median_day_load")
    day_load_q25 = blend_optional("day_load_q25")
    day_load_q75 = blend_optional("day_load_q75")
    typical_day_minutes = blend_optional("median_day_minutes")
    day_minutes_q25 = blend_optional("day_minutes_q25")
    day_minutes_q75 = blend_optional("day_minutes_q75")
    typical_days = [_RHYTHM_WEEKDAYS[wd][:3] for wd in sorted(likely_weekdays)]
    recent_label = f'{recent.get("training_days",0)}d/{recent.get("sessions",0)}s · {recent.get("hours_per_week",0):.1f} h/wk'
    established_label = f'{established.get("training_days",0)}d/{established.get("sessions",0)}s · {established.get("hours_per_week",0):.1f} h/wk'
    short_label = f'{est_days:.1f} d/wk · {est_hours:.1f} h/wk' if state != "LEARNING" else "Learning from Intervals"
    detail_parts = [state]
    if typical_days and pattern_strength != "LOW":
        detail_parts.append("likely " + " · ".join(typical_days))
    if double_propensity >= 0.35:
        detail_parts.append(f'doubles {round(double_propensity*100)}%')
